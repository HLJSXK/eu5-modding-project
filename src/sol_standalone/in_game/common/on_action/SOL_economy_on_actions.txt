### SECTION: Monthly Country Pulse
### ============================================================
# EU5 1.3 population demand is driven by a location modifier, refreshed monthly.

monthly_country_pulse = {
	on_actions = {
		sol_substitution_cleanup_monthly_pulse
		sol_local_pop_demand_monthly_pulse
	}
}

sol_substitution_cleanup_monthly_pulse = {
	trigger = { OR = { num_locations = 0 total_population = 0 } }
	effect = { sol_clear_country_substitution = yes }
}

sol_local_pop_demand_monthly_pulse = {
	trigger = {
		total_population > 0
		num_locations > 0
	}
	effect = {
		# Initialize the yearly estate-building maintenance cache if missing.
		if = {
			limit = {
				current_date >= 1337.5.1
				NOT = { has_variable = sol_estate_building_maintenance_initialized }
			}
			sol_update_estate_building_maintenance = yes
			set_variable = { name = sol_estate_building_maintenance_initialized value = 1 }
		}
		# Humans respond monthly; AI economics may remain stale until its anniversary.
		if = {
			limit = { is_human = yes }
			sol_update_local_pop_demand_modifiers = yes
		}
		else = {
			# New countries initialize once; individual missing location coefficients
			# receive a neutral fallback inside the existing monthly apply pass.
			if = {
				limit = { NOT = { has_variable = sol_country_demand_solve_status } }
				sol_solve_country_pop_demand = yes
			}
			sol_apply_local_pop_demand_modifiers = yes
		}
	}
}

### ============================================================
### SECTION: Yearly Country Pulse
### ============================================================

yearly_country_pulse = {
	on_actions = {
		sol_estate_building_maintenance_yearly_pulse
		sol_ai_yearly_pop_demand_pulse
	}
}

sol_ai_yearly_pop_demand_pulse = {
	trigger = {
		total_population > 0
		num_locations > 0
		is_ai = yes
	}
	effect = {
		sol_update_local_pop_demand_modifiers = yes
	}
}

sol_estate_building_maintenance_yearly_pulse = {
	trigger = {
		total_population > 0
		num_locations > 0
	}
	effect = {
		sol_update_estate_building_maintenance = yes
	}
}

### ============================================================
### SECTION: GLS Native Post-Lobby Initialisation
### ============================================================

# Initialize a new game after country selection; loads keep scheduled caches.
on_game_started_after_lobby = {
	on_actions = {
		gls_init_on_game_started_after_lobby
	}
}

gls_init_on_game_started_after_lobby = {
	effect = {
		gls_recalculate_global_living_standard = yes
	}
}
