### ============================================================
### SECTION: Monthly Country Pulse
### ============================================================
# EU5 1.3 population demand is driven by a location modifier, refreshed monthly.

monthly_country_pulse = {
	on_actions = {
		sol_local_pop_demand_monthly_pulse
	}
}

sol_local_pop_demand_monthly_pulse = {
	trigger = {
		total_population > 0
		num_locations > 0
	}
	effect = {
		if = {
			limit = {
				current_date >= 1337.5.1
				NOT = { has_variable = sol_country_poverty_compensation_initialized }
			}
			sol_update_estate_building_maintenance = yes
			sol_update_country_poverty_compensation = yes
		}
		sol_update_local_pop_demand_modifiers = yes
	}
}

### ============================================================
### SECTION: Yearly Country Pulse
### ============================================================

yearly_country_pulse = {
	on_actions = {
		sol_estate_building_maintenance_yearly_pulse
	}
}

sol_estate_building_maintenance_yearly_pulse = {
	trigger = {
		total_population > 0
		num_locations > 0
	}
	effect = {
		sol_update_estate_building_maintenance = yes
		sol_update_country_poverty_compensation = yes
	}
}

### ============================================================
### SECTION: GLS Situation Start
### ============================================================
# Match PP's pending-flag pattern so the situation starts on the next monthly
# situation check, after the first game month has initialized estate income.

on_game_start = {
	on_actions = {
		sol_global_living_standard_situation_game_start
	}
}

sol_global_living_standard_situation_game_start = {
	effect = {
		if = {
			limit = {
				situation:global_living_standard = {
					situation_is_active = no
				}
			}
			set_global_variable = {
				name = sol_global_living_standard_situation_pending
				value = yes
			}
		}
	}
}

### ============================================================
### SECTION: GLS After-Lobby Initialisation
### ============================================================

gls_init_after_lobby = {
	effect = {
		gls_recalculate_global_living_standard = yes
	}
}
