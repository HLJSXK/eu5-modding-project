sol_living_standard = {
	map_color = {
		if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = gls_location_actual_per_capita_spending
			}
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "gls_location_color_factor"
			}
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "GLS_MAP_LEGEND_HIGH"
		color = define:NMapColors|MAP_COLOR_HIGH
	}

	legend_key = {
		desc = "GLS_MAP_LEGEND_MID"
		color = define:NMapColors|MAP_COLOR_MID
	}

	legend_key = {
		desc = "GLS_MAP_LEGEND_LOW"
		color = define:NMapColors|MAP_COLOR_LOW
	}

	tooltip_key = {
		if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = gls_location_actual_per_capita_spending
				var:gls_location_actual_per_capita_spending >= 0.5
			}
			value = MAPMODE_SOL_LIVING_STANDARD_TT_HIGH
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = gls_location_actual_per_capita_spending
				var:gls_location_actual_per_capita_spending >= 0.1
			}
			value = MAPMODE_SOL_LIVING_STANDARD_TT_MID
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = gls_location_actual_per_capita_spending
				var:gls_location_actual_per_capita_spending >= 0.01
			}
			value = MAPMODE_SOL_LIVING_STANDARD_TT_LOW
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = gls_location_actual_per_capita_spending
			}
			value = MAPMODE_SOL_LIVING_STANDARD_TT_VERY_LOW
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
			}
			value = MAPMODE_SOL_LIVING_STANDARD_TT_UNCACHED
		}
		else = {
			value = MAPMODE_SOL_LIVING_STANDARD_TT_WATER
		}
	}

	small_map_names = location
	medium_map_names = location
	large_map_names = location

	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location

	fill_in_impassable = yes
	enable_snow = no

	flatmap_behaviour = always
	use_fow = no

	category = economy
	index = 0

	map_markers = {
		all = no
	}

	gradient_parameters = {
		zoom_step = 2

		gradient_alpha_inside = 1
		gradient_alpha_outside = 1
		gradient_width = 0.25
		gradient_color_mult = 0.9
		edge_width = 0
		edge_sharpness = 0.01
		edge_alpha = 0
		edge_color_mult = 0
		before_lighting_blend = 0.5
		after_lighting_blend = 0.5
	}

	color_refresh_counters = { Month }
	color_and_names_refresh_counters = { Month }
}
