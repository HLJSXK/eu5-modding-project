on_game_started_after_lobby = {
    on_actions = {
        sol14_new_game_after_lobby
    }
}

on_game_load_after_lobby = {
    on_actions = {
        sol14_load_after_lobby
    }
}

sol14_new_game_after_lobby = {
    effect = {
        set_global_variable = { name = sol_game_started_after_lobby value = yes }
        set_global_variable = SOL_migration_v3_seen
        remove_global_variable = sol_is_new_game
        every_country = {
            limit = { is_ai = no }
            sol_register_cmf_mod = yes
        }
        trigger_event_silently = { on_action = gls_init_after_lobby }
    }
}

sol14_load_after_lobby = {
    effect = {
        set_global_variable = { name = sol_game_started_after_lobby value = yes }
        remove_global_variable = sol_is_new_game
        if = {
            limit = {
                NOT = { has_global_variable = SOL_migration_v3_seen }
            }
            trigger_event_silently = { on_action = SOL_migration_fire_event }
        }
        every_country = {
            limit = { is_ai = no }
            sol_register_cmf_mod = yes
        }
        trigger_event_silently = { on_action = gls_init_after_lobby }
    }
}
