# Root scope: global.
sol_initialize_cmm_bool_alias_defaults = {
	set_global_variable = { name = sol_sol_on_enabled value = yes }
	set_global_variable = { name = sol_sol_map_enabled value = yes }
	set_global_variable = { name = sol_as_enabled value = yes }
	set_global_variable = { name = sol_ae_enabled value = yes }
	set_global_variable = { name = sol_difficulty_enabled value = yes }
	set_global_variable = { name = sol_gdp_dev_enabled value = yes }
	set_global_variable = { name = sol_hw_enabled value = yes }
	set_global_variable = { name = sol_iwe_enabled value = yes }
	set_global_variable = { name = sol_hwe_enabled value = yes }
	set_global_variable = { name = sol_rwe_enabled value = yes }
	# Performance (perf) tab defaults: every setting in both groups defaults ON.
	set_global_variable = { name = sol_ai_reduced_freq_enabled value = yes }
	set_global_variable = { name = sol_solver_human_exact_enabled value = yes }
	set_global_variable = { name = sol_solver_human_approx_enabled value = yes }
	set_global_variable = { name = sol_solver_ai_exact_enabled value = yes }
	set_global_variable = { name = sol_solver_ai_approx_enabled value = yes }
	# No default-OFF settings remain in this mod. If one is added later it must NOT
	# be initialized here: cmm_sync_bool_alias uses remove_global_variable for the
	# off state, so has_global_variable must stay false until CMM turns it on.
}

# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
sol_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Features (feat) Tab
		## SOL-based pop demands (sol) Group
		flag:sol__sol_on = {
			cmm_sync_bool_alias = {
				setting = sol__sol_on
				alias = sol_sol_on_enabled
			}
			if = {
				limit = { has_global_variable = sol_sol_on_enabled }
				gls_recalculate_global_living_standard = yes
			}
		}
		flag:sol__sol_map = {
			cmm_sync_bool_alias = {
				setting = sol__sol_map
				alias = sol_sol_map_enabled
			}
		}
		## Economic Balance (as) Group
		flag:sol__as_on = {
			cmm_sync_bool_alias = {
				setting = sol__as_on
				alias = sol_as_enabled
			}
		}
		flag:sol__bte = {
			cmm_sync_setting_alias = {
				setting = sol__bte
				alias = sol_bte
			}
		}
		flag:sol__ae = {
			cmm_sync_bool_alias = {
				setting = sol__ae
				alias = sol_ae_enabled
			}
		}
		flag:sol__cts = {
			cmm_sync_setting_alias = {
				setting = sol__cts
				alias = sol_cts
			}
		}
		flag:sol__ds = {
			cmm_sync_setting_alias = {
				setting = sol__ds
				alias = sol_ds
			}
		}
		flag:sol__difficulty = {
			cmm_sync_bool_alias = {
				setting = sol__difficulty
				alias = sol_difficulty_enabled
			}
		}
		flag:sol__gdp_dev = {
			cmm_sync_bool_alias = {
				setting = sol__gdp_dev
				alias = sol_gdp_dev_enabled
			}
		}
		flag:sol__ai_pr = {
			cmm_sync_setting_alias = {
				setting = sol__ai_pr
				alias = sol_ai_pr
			}
		}
		## War Acceleration (hw) Group
		flag:sol__hw_on = {
			cmm_sync_bool_alias = {
				setting = sol__hw_on
				alias = sol_hw_enabled
			}
		}
		flag:sol__iwe = {
			cmm_sync_bool_alias = {
				setting = sol__iwe
				alias = sol_iwe_enabled
			}
		}
		flag:sol__hwe = {
			cmm_sync_bool_alias = {
				setting = sol__hwe
				alias = sol_hwe_enabled
			}
		}
		flag:sol__rwe = {
			cmm_sync_bool_alias = {
				setting = sol__rwe
				alias = sol_rwe_enabled
			}
		}
		# Performance (perf) Tab
		## Calculation Frequency (calc_freq) Group
		flag:sol__ai_reduced_freq = {
			cmm_sync_bool_alias = {
				setting = sol__ai_reduced_freq
				alias = sol_ai_reduced_freq_enabled
			}
		}
		## Country-level Solver (country_solver) Group
		flag:sol__solver_human_exact = {
			cmm_sync_bool_alias = {
				setting = sol__solver_human_exact
				alias = sol_solver_human_exact_enabled
			}
		}
		flag:sol__solver_human_approx = {
			cmm_sync_bool_alias = {
				setting = sol__solver_human_approx
				alias = sol_solver_human_approx_enabled
			}
		}
		flag:sol__solver_ai_exact = {
			cmm_sync_bool_alias = {
				setting = sol__solver_ai_exact
				alias = sol_solver_ai_exact_enabled
			}
		}
		flag:sol__solver_ai_approx = {
			cmm_sync_bool_alias = {
				setting = sol__solver_ai_approx
				alias = sol_solver_ai_approx_enabled
			}
		}
	}
