# Taken from the 'Reasonable Prices' mod, but with adjusted value caps.
# R.P. capped these too low; raised caps to 10x lower than vanilla,
# which still feels like a large difference while remaining functional.

TRY_REPLACE:change_gold_effect = {
	add_gold = {
		value = 1
		multiply = $scale$

		add = {
			value = monthly_income_trade_and_tax
			multiply = $scale$
		}
		if = {
			limit = {
				$scale$ >  0
			}
			min = {
				value = $scale$
			}
			max = {
				value = 1000 # Vanilla: 10,000 (in R.P, this was 50)
				multiply = $scale$
			}

		}

		if = {
			limit = {
				$scale$ < 0
			}
			max = {
				value = $scale$
			}
			min = {
				value = 1000 # Vanilla: 10,000 (in R.P, this was 50)
				multiply = $scale$
			}

		}
	}
}

# SOL calculation effects: savings pressure, panel display, map coloring,
# and panel stat accumulation. Called from yearly_country_pulse (Pass A/B).

gls_compute_savings_pressure = {
	# 计算财富锚点和各阶层存款压力。
	# 地图着色依赖这些值（通过 local_*_savings_pressure → local_average_gdp_per_capita_scale）。
	# 调用者：Pass A（所有国家，每年）、Pass B（人类，每月）
	# root = country
	set_variable = {
		name = gls_country_wealth_anchor_cached
		value = {
			value = monthly_income_trade_and_tax
			min = 0
		}
	}

	set_variable = {
		name = gls_nobles_savings_pressure
		value = {
			value = 0
			if = {
				limit = { country_has_estate = estate_type:nobles_estate }
				value = "estate(estate_type:nobles_estate).estate_gold"
				divide = {
					value = 200
					add = {
						value = var:gls_country_wealth_anchor_cached
						multiply = 8
					}
					min = 80
				}
				subtract = 1
				multiply = 0.50
				min = -0.50
				max = 3
			}
		}
	}

	set_variable = {
		name = gls_clergy_savings_pressure
		value = {
			value = 0
			if = {
				limit = { country_has_estate = estate_type:clergy_estate }
				value = "estate(estate_type:clergy_estate).estate_gold"
				divide = {
					value = 100
					add = {
						value = var:gls_country_wealth_anchor_cached
						multiply = 3
					}
					min = 60
				}
				subtract = 1
				multiply = 0.50
				min = -0.50
				max = 2
			}
		}
	}

	set_variable = {
		name = gls_burghers_savings_pressure
		value = {
			value = 0
			if = {
				limit = { country_has_estate = estate_type:burghers_estate }
				value = "estate(estate_type:burghers_estate).estate_gold"
				divide = {
					value = 100
					add = {
						value = var:gls_country_wealth_anchor_cached
						multiply = 3
					}
					min = 60
				}
				subtract = 1
				multiply = 0.50
				min = -0.50
				max = 2
			}
		}
	}

	set_variable = {
		name = gls_commoner_savings_pressure
		value = {
			value = 0
			if = {
				limit = { country_has_estate = estate_type:peasants_estate }
				value = "estate(estate_type:peasants_estate).estate_gold"
				divide = {
					value = 0
					add = {
						value = var:gls_country_wealth_anchor_cached
						multiply = 2
					}
					min = 45
				}
				subtract = 1
				multiply = 0.50
				min = -0.50
				max = 1.5
			}
		}
	}

	set_variable = {
		name = gls_tribesmen_savings_pressure
		value = {
			value = 0
			if = {
				limit = { country_has_estate = estate_type:tribes_estate }
				value = "estate(estate_type:tribes_estate).estate_gold"
				divide = {
					value = 0
					add = {
						value = var:gls_country_wealth_anchor_cached
						multiply = 1
					}
					min = 35
				}
				subtract = 1
				multiply = 0.50
				min = -0.50
				max = 1.5
			}
		}
	}

	# Liquid funds = 1 + savings_pressure (consumption rate at country level)
	set_variable = { name = gls_nobles_liquid_funds    value = { add = 1  add = var:gls_nobles_savings_pressure } }
	set_variable = { name = gls_clergy_liquid_funds    value = { add = 1  add = var:gls_clergy_savings_pressure } }
	set_variable = { name = gls_burghers_liquid_funds  value = { add = 1  add = var:gls_burghers_savings_pressure } }
	set_variable = { name = gls_commoner_liquid_funds  value = { add = 1  add = var:gls_commoner_savings_pressure } }
	set_variable = { name = gls_tribesmen_liquid_funds value = { add = 1  add = var:gls_tribesmen_savings_pressure } }

}

gls_update_location_map = {
	# 设置地图着色所需的地点变量，对 AI 和人类国家均执行。
	# 调用者：Pass A（所有国家，每年）
	# root = country
	if = {
		limit = { num_locations > 0 }
		every_owned_location = {
			limit = { is_land = yes }
			set_variable = {
				name = gls_location_avg_scale
				value = { add = local_average_gdp_per_capita_scale }
			}
			set_variable = {
				name = gls_location_sol_weighted
				value = {
					value = var:gls_location_avg_scale
					multiply = development
				}
			}
		}
	}
	set_variable = { name = gls_map_cache_initialized value = 1 }
}
