# Taken from the 'Reasonable Prices' mod, but with adjusted value caps.
# R.P. capped these too low; raised caps to 10x lower than vanilla,
# which still feels like a large difference while remaining functional.

TRY_REPLACE:change_gold_effect = {
	add_gold = {
		value = 1
		multiply = $scale$

		add = {
			value = monthly_income_trade_and_tax
			multiply = $scale$
		}
		if = {
			limit = {
				$scale$ >  0
			}
			min = {
				value = $scale$
			}
			max = {
				value = 1000 # Vanilla: 10,000 (in R.P, this was 50)
				multiply = $scale$
			}

		}

		if = {
			limit = {
				$scale$ < 0
			}
			max = {
				value = $scale$
			}
			min = {
				value = 1000 # Vanilla: 10,000 (in R.P, this was 50)
				multiply = $scale$
			}

		}
	}
}

### SECTION: Standard of Living Computation

# EU5 1.3 demand path: calibrated demand_add remains in goods, while
# monthly location local_pop_demand is sized as liquid funds / base spending.

gls_compute_savings_pressure = {
	# root = country. Unified pressure: (total savings / total target - 1) * 0.1
	set_variable = { name = gls_country_wealth_anchor_cached value = { value = monthly_income_trade_and_tax min = 0 } }

	set_variable = { name = gls_nobles_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:nobles_estate }
		set_variable = { name = gls_nobles_gold value = "estate(estate_type:nobles_estate).estate_gold" }
	}
	set_variable = { name = gls_nobles_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:nobles_estate }
		set_variable = { name = gls_nobles_target value = { value = 200 add = { value = var:gls_country_wealth_anchor_cached multiply = 8 } min = 80 } }
	}

	set_variable = { name = gls_clergy_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:clergy_estate }
		set_variable = { name = gls_clergy_gold value = "estate(estate_type:clergy_estate).estate_gold" }
	}
	set_variable = { name = gls_clergy_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:clergy_estate }
		set_variable = { name = gls_clergy_target value = { value = 100 add = { value = var:gls_country_wealth_anchor_cached multiply = 3 } min = 60 } }
	}

	set_variable = { name = gls_burghers_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:burghers_estate }
		set_variable = { name = gls_burghers_gold value = "estate(estate_type:burghers_estate).estate_gold" }
	}
	set_variable = { name = gls_burghers_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:burghers_estate }
		set_variable = { name = gls_burghers_target value = { value = 100 add = { value = var:gls_country_wealth_anchor_cached multiply = 3 } min = 60 } }
	}

	set_variable = { name = gls_commoner_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:peasants_estate }
		set_variable = { name = gls_commoner_gold value = "estate(estate_type:peasants_estate).estate_gold" }
	}
	set_variable = { name = gls_commoner_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:peasants_estate }
		set_variable = { name = gls_commoner_target value = { value = 0 add = { value = var:gls_country_wealth_anchor_cached multiply = 2 } min = 45 } }
	}

	set_variable = { name = gls_tribesmen_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:tribes_estate }
		set_variable = { name = gls_tribesmen_gold value = "estate(estate_type:tribes_estate).estate_gold" }
	}
	set_variable = { name = gls_tribesmen_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:tribes_estate }
		set_variable = { name = gls_tribesmen_target value = { value = 0 add = { value = var:gls_country_wealth_anchor_cached multiply = 1 } min = 35 } }
	}

	set_variable = { name = gls_total_savings value = { add = var:gls_nobles_gold add = var:gls_clergy_gold add = var:gls_burghers_gold add = var:gls_commoner_gold add = var:gls_tribesmen_gold } }
	set_variable = { name = gls_total_savings_target value = { add = var:gls_nobles_target add = var:gls_clergy_target add = var:gls_burghers_target add = var:gls_commoner_target add = var:gls_tribesmen_target } }
	set_variable = { name = gls_savings_pressure_adjustment value = 0 }
	if = {
		limit = { var:gls_total_savings_target > 0 }
		set_variable = { name = gls_savings_pressure_adjustment value = { value = var:gls_total_savings divide = var:gls_total_savings_target subtract = 1 multiply = 0.1 } }
	}
}

	set_variable = { name = sol_location_tribesmen_base_spending value = { value = var:sol_unit_spending_tribesmen multiply = num_pop_type:tribesmen } }

	change_variable = { name = sol_location_commoners_base_spending add = var:sol_location_soldiers_base_spending }

	change_variable = { name = sol_location_base_total_spending add = var:sol_location_burghers_base_spending }
	change_variable = { name = sol_location_base_total_spending add = var:sol_location_laborers_base_spending }
	change_variable = { name = sol_location_base_total_spending add = var:sol_location_peasants_base_spending }
	change_variable = { name = sol_location_base_total_spending add = var:sol_location_soldiers_base_spending }
	change_variable = { name = sol_location_base_total_spending add = var:sol_location_tribesmen_base_spending }

	set_variable = { name = sol_location_savings_pressure_adjustment value = 0 }
	if = {
		limit = { exists = owner }
		if = { limit = { owner = { has_variable = gls_savings_pressure_adjustment } } set_variable = { name = sol_location_savings_pressure_adjustment value = owner.var:gls_savings_pressure_adjustment } }
	}
	set_variable = { name = sol_location_liquid_funds_multiplier value = { value = 1 add = var:sol_location_savings_pressure_adjustment min = 0 } }
	set_variable = { name = sol_location_liquid_funds value = { value = var:sol_location_total_income multiply = var:sol_location_liquid_funds_multiplier } }
	set_variable = { name = sol_location_final_demand_scale value = 1 }
	if = {
		limit = { var:sol_location_base_total_spending > 0 }
		set_variable = { name = sol_location_final_demand_scale value = { value = var:sol_location_liquid_funds divide = var:sol_location_base_total_spending min = 0 } }
	}
	set_variable = { name = sol_location_pop_demand_modifier_size value = { value = var:sol_location_final_demand_scale subtract = 1 } }
	set_variable = { name = gls_location_avg_scale value = var:sol_location_final_demand_scale }
	set_variable = { name = gls_location_sol_weighted value = { value = var:gls_location_avg_scale multiply = development } }
}

	if = {

			add_location_modifier = {

				change_variable = { name = gls_income_commoners add = scope:gls_cached_location.local_sol_commoners_income_display }
				change_variable = { name = gls_income_tribesmen add = scope:gls_cached_location.local_sol_tribesmen_income_display }
				change_variable = { name = gls_total_income add = scope:gls_cached_location.local_sol_total_income }
				change_variable = { name = gls_country_base_spending add = scope:gls_cached_location.local_sol_base_spending_display }
				change_variable = { name = gls_country_liquid_funds add = scope:gls_cached_location.local_sol_liquid_funds_display }
			}
		}
	}
	if = { limit = { var:gls_location_count > 0 } set_variable = { name = gls_country_location_avg_scale value = { add = var:gls_location_scale_sum divide = var:gls_location_count } } }
	if = { limit = { var:gls_country_base_spending > 0 } set_variable = { name = gls_country_sol_all value = { value = var:gls_country_liquid_funds divide = var:gls_country_base_spending min = 0 } } }
}
