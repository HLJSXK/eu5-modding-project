# Taken from the 'Reasonable Prices' mod, but with adjusted value caps.
# R.P. capped these too low; raised caps to 10x lower than vanilla,
# which still feels like a large difference while remaining functional.

TRY_REPLACE:change_gold_effect = {
	add_gold = {
		value = 1
		multiply = $scale$

		add = {
			value = monthly_income_trade_and_tax
			multiply = $scale$
		}
		if = {
			limit = {
				$scale$ >  0
			}
			min = {
				value = $scale$
			}
			max = {
				value = 1000 # Vanilla: 10,000 (in R.P, this was 50)
				multiply = $scale$
			}

		}

		if = {
			limit = {
				$scale$ < 0
			}
			max = {
				value = $scale$
			}
			min = {
				value = 1000 # Vanilla: 10,000 (in R.P, this was 50)
				multiply = $scale$
			}

		}
	}
}

### SECTION: Standard of Living Computation

# EU5 1.3 demand path: calibrated demand_add remains in goods, while
# monthly location local_pop_demand is sized as liquid funds / base spending.

gls_compute_savings_pressure = {
	# root = country. Unified pressure: (total savings / total target - 1) * 0.25, capped at +50%.
	set_variable = { name = gls_country_wealth_anchor_cached value = { value = monthly_income_trade_and_tax min = 0 } }

	set_variable = { name = gls_nobles_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:nobles_estate }
		set_variable = { name = gls_nobles_gold value = "estate(estate_type:nobles_estate).estate_gold" }
	}
	set_variable = { name = gls_nobles_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:nobles_estate }
		set_variable = { name = gls_nobles_target value = { value = 200 add = { value = var:gls_country_wealth_anchor_cached multiply = 8 } min = 80 } }
	}

	set_variable = { name = gls_clergy_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:clergy_estate }
		set_variable = { name = gls_clergy_gold value = "estate(estate_type:clergy_estate).estate_gold" }
	}
	set_variable = { name = gls_clergy_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:clergy_estate }
		set_variable = { name = gls_clergy_target value = { value = 100 add = { value = var:gls_country_wealth_anchor_cached multiply = 3 } min = 60 } }
	}

	set_variable = { name = gls_burghers_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:burghers_estate }
		set_variable = { name = gls_burghers_gold value = "estate(estate_type:burghers_estate).estate_gold" }
	}
	set_variable = { name = gls_burghers_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:burghers_estate }
		set_variable = { name = gls_burghers_target value = { value = 100 add = { value = var:gls_country_wealth_anchor_cached multiply = 3 } min = 60 } }
	}

	set_variable = { name = gls_commoner_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:peasants_estate }
		set_variable = { name = gls_commoner_gold value = "estate(estate_type:peasants_estate).estate_gold" }
	}
	set_variable = { name = gls_commoner_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:peasants_estate }
		set_variable = { name = gls_commoner_target value = { value = 0 add = { value = var:gls_country_wealth_anchor_cached multiply = 2 } min = 45 } }
	}

	set_variable = { name = gls_tribesmen_gold value = 0 }
	if = {
		limit = { country_has_estate = estate_type:tribes_estate }
		set_variable = { name = gls_tribesmen_gold value = "estate(estate_type:tribes_estate).estate_gold" }
	}
	set_variable = { name = gls_tribesmen_target value = 0 }
	if = {
		limit = { country_has_estate = estate_type:tribes_estate }
		set_variable = { name = gls_tribesmen_target value = { value = 0 add = { value = var:gls_country_wealth_anchor_cached multiply = 1 } min = 35 } }
	}

	set_variable = { name = gls_total_savings value = { add = var:gls_nobles_gold add = var:gls_clergy_gold add = var:gls_burghers_gold add = var:gls_commoner_gold add = var:gls_tribesmen_gold } }
	set_variable = { name = gls_total_savings_target value = { add = var:gls_nobles_target add = var:gls_clergy_target add = var:gls_burghers_target add = var:gls_commoner_target add = var:gls_tribesmen_target } }
	set_variable = { name = gls_savings_pressure_adjustment value = 0 }
	if = {
		limit = { var:gls_total_savings_target > 0 }
		set_variable = { name = gls_savings_pressure_adjustment value = { value = var:gls_total_savings divide = var:gls_total_savings_target subtract = 1 multiply = 0.25 max = 0.5 } }
	}
}

sol_count_location_estate_buildings = {
	# root = location. Cached yearly; each estate building has 1 gold maintenance.
	set_variable = { name = sol_location_estate_buildings_nobles value = 0 }
	set_variable = { name = sol_location_estate_buildings_clergy value = 0 }
	set_variable = { name = sol_location_estate_buildings_burghers value = 0 }
	set_variable = { name = sol_location_estate_buildings_commoners value = 0 }
	set_variable = { name = sol_location_estate_buildings_tribesmen value = 0 }

	every_buildings_in_location = {
		if = {
			limit = { estate_type ?= estate_type:nobles_estate }
			location = { change_variable = { name = sol_location_estate_buildings_nobles add = 1 } }
		}
		if = {
			limit = { estate_type ?= estate_type:clergy_estate }
			location = { change_variable = { name = sol_location_estate_buildings_clergy add = 1 } }
		}
		if = {
			limit = { estate_type ?= estate_type:burghers_estate }
			location = { change_variable = { name = sol_location_estate_buildings_burghers add = 1 } }
		}
		if = {
			limit = { estate_type ?= estate_type:peasants_estate }
			location = { change_variable = { name = sol_location_estate_buildings_commoners add = 1 } }
		}
		if = {
			limit = { estate_type ?= estate_type:tribes_estate }
			location = { change_variable = { name = sol_location_estate_buildings_tribesmen add = 1 } }
		}
	}

	set_variable = { name = sol_location_estate_building_maintenance_nobles value = var:sol_location_estate_buildings_nobles }
	set_variable = { name = sol_location_estate_building_maintenance_clergy value = var:sol_location_estate_buildings_clergy }
	set_variable = { name = sol_location_estate_building_maintenance_burghers value = var:sol_location_estate_buildings_burghers }
	set_variable = { name = sol_location_estate_building_maintenance_commoners value = var:sol_location_estate_buildings_commoners }
	set_variable = { name = sol_location_estate_building_maintenance_tribesmen value = var:sol_location_estate_buildings_tribesmen }
}

sol_update_estate_building_maintenance = {
	# root = country. Yearly cache on each owned land location.
	if = {
		limit = { num_locations > 0 }
		every_owned_location = {
			limit = { is_land = yes }
			sol_count_location_estate_buildings = yes
		}
	}
}
