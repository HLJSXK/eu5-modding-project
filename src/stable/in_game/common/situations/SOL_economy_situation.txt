global_living_standard = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	can_start = {
		always = yes
	}

	can_end = {
		always = no
	}

	visible = {
		sol_sol_is_on = yes
	}

	on_monthly = {
		if = {
			limit = { sol_sol_is_on = yes }
			if = {
				limit = { OR = { current_month = 1 NOT = { has_global_variable = sol_market_pop_demand_maps_initialized } } }
				sol_refresh_market_pop_demand_maps = yes
			}
			every_country = {
				if = {
					limit = { total_population > 0 }
					if = {
						limit = { is_human = yes }
						gls_compute_panel_display = yes
						if = {
							limit = { OR = { current_month = 1 NOT = { has_variable = gls_country_final_demand_scale_avg } } }
							gls_accumulate_panel_stats = yes
						}
					}
				}
			}
			# Annual HUD ranking reads completed country caches only; no territory scan.
			if = { limit = { current_month = 1 } sol_ls_recalculate_world_ranking = yes }
			# Global scope: one CMF row summarising this month's solver census.
			sol_country_demand_census_report = yes
		}
	}

	tooltip = {
		# Living standard (annual funds per thousand + SOL modifier); thresholds match the sol_living_standard map mode.
		if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = sol_location_sol
				sol_map_living_standard_relative_value >= 2
			}
			custom_tooltip = GLS_LOCATION_SOL_HIGH_TT
			custom_tooltip = GLS_LOCATION_SOL_VALUE_TT
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = sol_location_sol
				sol_map_living_standard_relative_value >= 1
			}
			custom_tooltip = GLS_LOCATION_SOL_MID_TT
			custom_tooltip = GLS_LOCATION_SOL_VALUE_TT
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = sol_location_sol
				sol_map_living_standard_relative_value >= 0.5
			}
			custom_tooltip = GLS_LOCATION_SOL_LOW_TT
			custom_tooltip = GLS_LOCATION_SOL_VALUE_TT
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
				has_variable = sol_location_sol
			}
			custom_tooltip = GLS_LOCATION_SOL_VERY_LOW_TT
			custom_tooltip = GLS_LOCATION_SOL_VALUE_TT
		}
		else_if = {
			limit = {
				is_land = yes
				has_owner = yes
			}
			custom_tooltip = GLS_LOCATION_SOL_MAP_TT
		}
	}

	is_data_map = yes
	map_color = {
		if = {
			limit = {
				sol_map_is_on = yes
				is_land = yes
				has_owner = yes
				has_variable = sol_location_sol
			}
			if = {
				limit = { sol_map_living_standard_relative_value < 1 }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_LOW
					max_color = define:NMapColors|MAP_COLOR_MID
					factor = "sol_map_living_standard_below_factor"
				}
			}
			else = {
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MID
					max_color = define:NMapColors|MAP_COLOR_HIGH
					factor = "sol_map_living_standard_above_factor"
				}
			}
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}

	legend_key = {
		desc = "GLS_MAP_LEGEND_HIGH"
		color = define:NMapColors|MAP_COLOR_HIGH
	}

	legend_key = {
		desc = "GLS_MAP_LEGEND_MID"
		color = define:NMapColors|MAP_COLOR_MID
	}

	legend_key = {
		desc = "GLS_MAP_LEGEND_LOW"
		color = define:NMapColors|MAP_COLOR_LOW
	}

}
