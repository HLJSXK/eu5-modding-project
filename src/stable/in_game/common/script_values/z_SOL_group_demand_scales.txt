# Pop-scope demand scale routing wrappers.
# Each value delegates to the corresponding location-scope value in
# z_SOL_group_demand_scales_location.txt based on pop_type.
# Canonical calculations live there (single source of truth).

sol_staple_demand_scale = {
	value = 0
	if = {
		limit = { pop_type = pop_type:nobles }
		add = location.local_nobles_staple_demand_scale
	}
	else_if = {
		limit = { pop_type = pop_type:clergy }
		add = location.local_clergy_staple_demand_scale
	}
	else_if = {
		limit = { pop_type = pop_type:burghers }
		add = location.local_burghers_staple_demand_scale
	}
	else_if = {
		limit = {
			OR = {
				pop_type = pop_type:laborers
				pop_type = pop_type:soldiers
				pop_type = pop_type:peasants
			}
		}
		add = location.local_commoners_staple_demand_scale
	}
	else_if = {
		limit = { pop_type = pop_type:tribesmen }
		add = location.local_tribesmen_staple_demand_scale
	}
}
