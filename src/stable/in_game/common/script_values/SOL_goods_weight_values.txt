##############################################################
# SOL goods weight script values
# Scope: pop
# Part 1 — Substitution: scarcity multiplies weight by 0.5 (50% reduction)
# Part 2 — Religion / culture / location: multiplicative modifiers
# Final demand (single-group good) = (weight / group_total_weight) × N × demand_scale
# Final demand (multi-group good)  = Σ_g (group_weight_g / group_total_g) × N_g × scale_g
##############################################################

###############################################################
# GROUP 1: BASIC_CLOTHING (N=2): cloth, leather
#   cloth also participates in standard_clothing
###############################################################

SOL_cloth_weight = {
	value = 1
	# Part 1 — Substitution
	if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_cloth_severe_markets target = this } } }
		multiply = { value = 0.25 desc = "SOL_WEIGHT_SCARCE_3" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_cloth_moderate_markets target = this } } }
		multiply = { value = 0.5 desc = "SOL_WEIGHT_SCARCE_2" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_cloth_mild_markets target = this } } }
		multiply = { value = 0.75 desc = "SOL_WEIGHT_SCARCE_1" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_cloth_vcheap_markets target = this } } }
		multiply = { value = 2.0 desc = "SOL_WEIGHT_CHEAP_3" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_cloth_cheap_markets target = this } } }
		multiply = { value = 1.5 desc = "SOL_WEIGHT_CHEAP_2" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_cloth_affordable_markets target = this } } }
		multiply = { value = 1.25 desc = "SOL_WEIGHT_CHEAP_1" }
	}
	# Part 2 — Religion / culture / location
	location = {
		if = {
			limit = { location_max_winter_level = none }
			multiply = { value = 0.5 desc = "POP_DEMAND_CLIMATE_MAX_WINTER_LEVEL" }
		}
		else_if = {
			limit = { location_max_winter_level = mild }
			multiply = { value = 1.0 desc = "POP_DEMAND_CLIMATE_MAX_WINTER_LEVEL" }
		}
		else_if = {
			limit = { location_max_winter_level = normal }
			multiply = { value = 1.5 desc = "POP_DEMAND_CLIMATE_MAX_WINTER_LEVEL" }
		}
		else_if = {
			limit = { location_max_winter_level = severe }
			multiply = { value = 2.0 desc = "POP_DEMAND_CLIMATE_MAX_WINTER_LEVEL" }
		}
	}
}

	# Part 1 — Substitution
	if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_wine_severe_markets target = this } } }
		multiply = { value = 0.25 desc = "SOL_WEIGHT_SCARCE_3" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_wine_moderate_markets target = this } } }
		multiply = { value = 0.5 desc = "SOL_WEIGHT_SCARCE_2" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_wine_mild_markets target = this } } }
		multiply = { value = 0.75 desc = "SOL_WEIGHT_SCARCE_1" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_wine_vcheap_markets target = this } } }
		multiply = { value = 2.0 desc = "SOL_WEIGHT_CHEAP_3" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_wine_cheap_markets target = this } } }
		multiply = { value = 1.5 desc = "SOL_WEIGHT_CHEAP_2" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_wine_affordable_markets target = this } } }
		multiply = { value = 1.25 desc = "SOL_WEIGHT_CHEAP_1" }
	}
	# Part 2 — Religion / culture / location
	if = {
		limit = {
			religion = {
				OR = {
					this = religion:jain
					this = religion:sikhism
					group = religion_group:muslim
				}
			}
		}
		multiply = { value = 0 desc = "POP_DEMAND_RELIGION_FORBIDS" }
	}
	if = {
		limit = { culture = { has_culture_group = culture_group:andean_group } }
		multiply = { desc = "POP_DEMAND_CULTURAL_PREFERENCES" value = 0.5 }
	}
	if = {
		limit = { culture = { has_culture_group = culture_group:japanese_group } }
		multiply = { desc = "POP_DEMAND_CULTURAL_PREFERENCES" value = 0.5 }
	}
	if = {
		limit = {
			location.region = {
				OR = {
					this = region:iberia_region
					this = region:italy_region
					this = region:balkan_region
				}
			}
		}
		multiply = { value = 2 desc = "POP_DEMAND_MEDITERRANEAN" }
	}
	if = {
		limit = { location.sub_continent = sub_continent:south_asia }
		multiply = { value = 0.6 desc = "POP_DEMAND_INDIA" }
	}
	if = {
		limit = {
			pop_type = pop_type:clergy
			religion = {
				OR = {
					this = religion:shinto
					group = {
						OR = {
							this = religion_group:christian
							this = religion_group:israelite_group
						}
					}
				}
			}
		}
		multiply = { value = 10 desc = "POP_DEMAND_RITUAL_WINE" }
	}
	if = {
		limit = { culture = { has_culture_group = culture_group:baltic_group } }
		multiply = { value = 1.2 desc = "POP_DEMAND_BALTIC" }
	}
}

		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_books_cheap_markets target = this } } }
		multiply = { value = 1.5 desc = "SOL_WEIGHT_CHEAP_2" }
	}
	else_if = {
		limit = { location.market ?= { is_target_in_global_variable_list = { name = SOL_books_affordable_markets target = this } } }
		multiply = { value = 1.25 desc = "SOL_WEIGHT_CHEAP_1" }
	}
	# Part 2 — Religion / culture / location
	if = {
		limit = {
			location.owner ?= {
				has_embraced_institution = institution:printing_press
			}
		}
		multiply = { value = 1.2 desc = "POP_DEMAND_PRINTING_PRESS" }
	}
	multiply = {
		value = pop_literacy
		divide = 100
		desc = "POP_DEMAND_LITERACY"
	}
}

	add = SOL_books_weight
