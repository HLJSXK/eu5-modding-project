# Era-progression coefficient — guarded read of global_var:sol_era_coeff.
# Why a wrapper: the EU5 engine evaluates script_values during file-load validation,
# BEFORE any on_action (including on_game_start) fires. Reading global_var:sol_era_coeff
# directly inside z_SOL_group_demand_scales_location.txt at that point emits
# "Failed to fetch variable for 'sol_era_coeff'" / "Got value of type none" errors.
# This wrapper falls back to 1 (neutral, no era correction) if the global is unset,
# making the demand-scale evaluation safe at any load stage.
# Real value (1.5 → 1.16067 across ages) is set by sol_init_export_adj at on_game_start.
local_sol_era_coeff = {
    value = 1
    if = {
        limit = { has_global_variable = sol_era_coeff }
        value = global_var:sol_era_coeff
    }
}

# Per-stratum market price correction layer.
# Purpose: the simulator assumes pops buy at group-average prices; in practice, substitute
# groups force pops toward the cheapest available good.  Each stratum gets its own
# correction weighted by that stratum's budget shares — commoner demand reacts strongly
# to staple food prices, noble demand to luxury goods prices, etc.
# Storage: var:sol_market_scarcity_adj_<strata> on each market's hub location
# (market.location), written yearly by SOL_update_market_scarcity_scores.
# All locations sharing a market read the same cached value via market.location.var:*,
# eliminating O(locations × 336) per-stratum redundant computation.
# Defaults to 1 (neutral) before the first yearly cache write or when the hub
# is not a land location.
local_sol_scarcity_adj_nobles = {
    value = 1
    if = {
        limit = { market ?= { location = { has_variable = sol_market_scarcity_adj_nobles } } }
        value = market.location.var:sol_market_scarcity_adj_nobles
    }
}

local_average_gdp_per_capita_scale = {
    # Average savings-pressure-based demand signal across all strata, population-weighted.
    # Used by map-coloring in gls_compute_savings_pressure.
    value = 0
    add = {
        value = local_nobles_savings_pressure
        multiply = num_pop_type:nobles
    }
    add = {
        value = local_clergy_savings_pressure
        multiply = num_pop_type:clergy
    }
    add = {
        value = local_burghers_savings_pressure
        multiply = num_pop_type:burghers
    }
    add = {
        value = local_commoner_savings_pressure
        multiply = local_commoner_population
    }
    add = {
        value = local_tribesmen_savings_pressure
        multiply = num_pop_type:tribesmen
    }
    divide = {
        add = num_pop_type:nobles
        add = num_pop_type:clergy
        add = num_pop_type:burghers
        add = local_commoner_population
        add = num_pop_type:tribesmen
        min = 0.001
    }
    add = 1
}
