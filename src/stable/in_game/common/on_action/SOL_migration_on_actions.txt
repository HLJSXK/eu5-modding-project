# SOL Migration Notice
# Native lifecycle only; no GUI or CMF load bridge.
# The release-date marker is stamped with metadata by scripts/update_mod_version.py.
# SOL_migration_261109_seen persists in saves and suppresses this release's notice.
# Legacy sol_game_started_after_lobby / sol_is_new_game values are ignored.

on_game_start = {
	on_actions = {
		sol_migration_stamp_new_game
	}
}

# A real new-game hook replaces the old year-based guess.
sol_migration_stamp_new_game = {
	effect = {
		set_global_variable = SOL_migration_261109_seen
	}
}

monthly_country_pulse = {
	on_actions = {
		sol_migration_check_monthly
	}
}

# Country scope. A loaded, unmarked save is checked when simulation advances.
# AI pulses cannot consume the notice before a human country is available.
sol_migration_check_monthly = {
	trigger = {
		is_human = yes
		NOT = { has_global_variable = SOL_migration_261109_seen }
	}
	effect = {
		trigger_event_silently = { on_action = SOL_migration_fire_event }
	}
}

# Stamp before notifying every human, so later pulses/loads cannot repeat it.
# The dispatcher works from any caller scope.
SOL_migration_fire_event = {
	trigger = {
		NOT = { has_global_variable = SOL_migration_261109_seen }
	}
	effect = {
		set_global_variable = SOL_migration_261109_seen
		every_country = {
			limit = { is_ai = no }
			trigger_event_non_silently = sol_migration.1
		}
	}
}
