namespace = htc_inca_events

htc_inca_events.1 = {
	type = country_event
	title = htc_inca_events.1.title
	desc = htc_inca_events.1.desc

	fire_only_once = yes
	
	dynamic_historical_event = {
        tag = INC
		from = 1500.1.1
		to = 1836.1.1
		monthly_chance = 25
	}

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	trigger = {
		"estate_power(estate_type:crown_estate)" >= 0.5
		any_neighbor_country = {
			has_embraced_institution = institution:new_world
			num_embraced_institutions > from.num_embraced_institutions
			or = {
				original_capital ?= { continent = continent:europe }
				original_capital ?= { sub_continent = sub_continent:east_asia }
			}
		}
	}
	
	immediate = {
		random_neighbor_country = {
			limit = {
				has_embraced_institution = institution:new_world
				num_embraced_institutions > root.num_embraced_institutions
				or = {
					original_capital ?= { continent = continent:europe }
					original_capital ?= { sub_continent = sub_continent:east_asia }
				}
			}
			save_scope_as = source_country
		}
	}

	option = {
		name = htc_inca_events.1.a
		historical_option = yes
		
		remove_reform = government_reform:tawantinsuyu_monarchy
		add_reform = government_reform:reformed_tawantinsuyu_government

		add_legitimacy = legitimacy_weak_penalty
		change_societal_value = { type = traditionalist_vs_innovative value = societal_value_move_to_right }
		every_area = {
			limit = { is_discovered_by = scope:source_country }
			root = { discover_area = prev }
		}
	}
	option = {
		name = htc_inca_events.1.b

		add_legitimacy = legitimacy_mild_bonus
		change_societal_value = { type = traditionalist_vs_innovative value = societal_value_move_to_left }
	}
}
