namespace = htc_exploration_crutches

# north atlantic
htc_exploration_crutches.1 = {
	type = country_event
	title = htc_exploration_crutches.title
	desc = htc_exploration_crutches.desc
	fire_only_once = no

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	trigger = {
		is_ai = yes
		has_advance = open_sea_exploration
		monthly_balance > 1

		any_area = {
			sub_continent = sub_continent:north_atlantic_ocean_sub_continent
			not = { root = { has_discovered_area = prev } }
			any_neighbor_area = {
				root = { has_discovered_area = prev }
			}
		}
	}
	
	immediate = {
		random_area = {
			limit = {
				sub_continent = sub_continent:north_atlantic_ocean_sub_continent
				not = { root = { has_discovered_area = prev } }
				any_neighbor_area = {
					root = { has_discovered_area = prev }
				}
			}
			save_scope_as = target
		}
	}

	option = {
		name = htc.options.take
		historical_option = yes

		root = { discover_area = scope:target }
	}
	option = {
		name = htc.options.no
	}
}
