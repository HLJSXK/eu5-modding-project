namespace = htc_colonize_in_region_canada

htc_colonize_in_region_canada.1 = {
	type = country_event
	title = htc_colonize_in_region.title
	desc = htc_colonize_in_region.desc

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	trigger = {
		OR = {
            has_or_had_tag = CAN
		}
		current_date > 1580.1.1
		current_date < 1836.1.1
		is_ai = yes
		monthly_balance > 50
		num_colonial_charters <= 5
		any_province_definition = {
			region = region:canada_region
			not = { has_colonial_charter = root }
			not = {
				any_location_in_province_definition = {
					or = {
						# prevent colonization if either:
						# owner is a player, owner is a colonial nation, owner is european, doesn't have owner
						owner ?= { is_ai = no }
						owner ?= { is_colonial_subject = yes }
						owner ?= { capital ?= { continent = continent:europe } }
					}
				}
			}
			any_location_in_province_definition = {
				or = {
					within_colonial_range_of = root
					any_neighbor_location = {
						OR = {
							owner ?= root
							owner ?= {
								exists = overlord
								overlord = root
							}
						}
					}
				}
				has_owner = no
				is_ownable = yes
				is_discovered_by = root
			}
		}
	}
	
	immediate = {
		region:canada_region = {
			save_scope_as = source
		}
		random_province_definition = {
			limit = {
				region = region:canada_region
				not = { has_colonial_charter = root }
				not = {
					any_location_in_province_definition = {
						or = {
							# prevent colonization if either:
							# owner is a player, owner is a colonial nation, owner is european, doesn't have owner
							owner ?= { is_ai = no }
							owner ?= { is_colonial_subject = yes }
							owner ?= { capital ?= { continent = continent:europe } }
						}
					}
				}
				any_location_in_province_definition = {
					or = {
						within_colonial_range_of = root
						any_neighbor_location = {
							OR = {
								owner ?= root
								owner ?= {
									exists = overlord
									overlord = root
								}
							}
						}
					}
					has_owner = no
					is_ownable = yes
					is_discovered_by = root
				}
			}
			save_scope_as = geo
		}
	}

	option = {
		name = htc.options.take
		historical_option = yes
		ai_will_do = 1000

		create_colonial_charter = {
			target = scope:geo
		}
	}
	option = {
		name = htc.options.no
	}
}
