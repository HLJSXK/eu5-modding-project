namespace = htc_colonize_location_portugal

htc_colonize_location_portugal.1 = {
	type = country_event
	title = htc_colonize_location.title
	desc = htc_colonize_location.desc

	illustration_tags = {
		10 = happy
		10 = exterior
	}

	trigger = {
		OR = {
            has_or_had_tag = POR
		}
		current_date > 1380.1.1
		current_date < 1836.1.1
		is_ai = yes
		monthly_balance > 50
		location:bojador = {
			has_owner = no
			is_ownable = yes
			within_colonial_range_of = root
			is_discovered_by = root
		}
	}

	immediate = {
		location:bojador = {
			save_scope_as = geo
		}
	}
	

	option = {
		name = htc.options.take
		historical_option = yes
		ai_will_do = 1000
		
		add_gold = {
			value = -50
			add = {
				value = location:bojador.development
				multiply = -10
			}
		}
		
		location:bojador = {
			change_location_owner = root
			change_integration_level = integrated
		}
	}
	option = {
		name = htc.options.no
	}
}
