namespace = htc_map_sharing

# maps from trade - discover from `to_market`
htc_map_sharing.1 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	trigger = {
		any_trade = {
			to_market = {
				any_location_in_market = {
					not = { is_discovered_by = root }
					any_neighbor_location = { is_discovered_by = root }
				}
			}
		}
	}
	
	immediate = {
		random_trade = {
			limit = {
				to_market = {
					any_location_in_market = {
						not = { is_discovered_by = root }
						any_neighbor_location = { is_discovered_by = root }
					}
				}
			}
			to_market = {
				random_location_in_market = {
					limit = {
						not = { is_discovered_by = root }
						any_neighbor_location = { is_discovered_by = root }
					}
					area = { root = { discover_area = prev } }
				}
			}
		}
	}
}
