is_possible_colonial_empire = {
	custom_tooltip = {
		text = is_possible_colonial_empire_tt
		has_ports = yes
		or = {
			is_ai = no
			has_or_had_tag = POR
			has_or_had_tag = CAS
			has_or_had_tag = SPA
			has_or_had_tag = FRA
			has_or_had_tag = ENG
			has_or_had_tag = GBR
			has_or_had_tag = NED
			is_colonial_top_overlord = yes
			and = {
				is_great_power = yes
				current_year >= 1650
			}
		}
	}
}

is_possible_indian_colonizer = {
	custom_tooltip = {
		text = is_possible_indian_colonizer_tt
		not = {
			capital ?= { continent = continent:asia }
			any_owned_location = {
				sub_continent = sub_continent:south_asia
				is_connected_to = root.capital
			}
		}
	}
}
