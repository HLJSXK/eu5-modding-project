unification_of_inca = {
    monthly_spawn_chance = monthly_spawn_chance_unique
    hint_tag = hint_unification_of_inca
    
    can_start = {
		current_year >= 1337
        any_country = {
            capital ?= { region = region:andes_region }
			religion = religion:inti
        }
    }
    
    can_end = {
		or = {
            custom_tooltip = {
                text = uoi_inca_exists_tt
			    country_exists = c:INC
            }
			current_year > 1550
		}
    }
    
    visible = {
		or = {
			capital ?= { region = region:andes_region }
			religion = religion:inti
		}
    }

    on_start = {
        unification_of_inca_recalculate_strongest = yes
    }

    on_monthly = {
        unification_of_inca_recalculate_strongest = yes

        random_list = {
            5 = { 
                every_country = {
                    limit = {
						religion = religion:inti
					}
                    # trigger_event_non_silently = htc_unification_of_inca.1
                }
            }
            95 = {}
        }
    }

    on_ended = {}

    tooltip = {
        # incan
        if = {
            limit = {
                owner ?= {
					capital ?= { region = region:andes_region }
					religion = religion:inti
                }
            }
            custom_tooltip = htc_uoi_incan_tt
        }
        else = {
            custom_tooltip = htc_uoi_other_tt
        }
    }

    map_color = {
        if = {
            limit = {
                owner ?= {
                    top_overlord_or_this ?= { religion = religion:inti }
                }
            }
            value = owner.top_overlord_or_this.country_color
        }
        else_if = {
            limit = {
				region = region:andes_region
				has_owner = yes
			}
            value = rtr_rebel_color
        }
    }

    secondary_map_color = {
		if = {
			limit = {
				owner ?= {
					situation:unification_of_inca.var:strongest_incan ?= this
				}
			}
		}
	}
}
