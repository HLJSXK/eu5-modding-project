REPLACE_OR_CREATE:KON_f = {
	level = 3
	required_locations_fraction = 0.18
	rule = historical

	potential = {
		capital.region = region:kongo_region
	}

	allow = {
		culture.language = language:kongo_language
	}

	name = KON
	flag = KON
	adjective = KON_ADJ
	tag = KON
	color = map_KON

	areas = {
		west_kongo_area
		kongo_area		
	}

	form_effect = {
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank_effect = { rank = country_rank:rank_kingdom }
		}
	}
}

REPLACE_OR_CREATE:INC_f = {
	level = 3
	required_locations_fraction = 0.35 # total is 375
	rule = historical
	potential = {
		religion = religion:inti
		capital ?= { region = region:andes_region }
	}

	allow = {
		region:andes_region = {
			any_owner_in_region = {
				filter = { not = { this = root } }
				count = all
				is_subject_of = root
			}
		}
	}
	
	name = INC
	flag = INC
	adjective = INC_ADJ
	tag = INC
	color = map_INC

	regions = {
		andes_region
	}

	form_effect = {
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		if = { 
			limit = { has_reform = government_reform:andean_monarchy }
			remove_reform = government_reform:andean_monarchy
		}
		if = { 
			limit = {
				NOT = { has_reform = government_reform:reformed_tawantinsuyu_government }
			}
			if = {
				limit = { NOT = { government_type = government_type:monarchy } }
				change_government_type = government_type:monarchy
				add_legitimacy = legitimacy_radical_bonus
			}
			add_reform = government_reform:tawantinsuyu_monarchy
		}
		every_subject = {
			root = {
				annex_country = {
					country = prev
					reason = Diplomatic
					transfer = yes
				}
			}
		}
	}
}
