htc_prevent_all_institutions = {
	potential_trigger = {
		is_ai = yes
		is_colonial_subject = no
		not = { current_age = age_1_traditions }

		or = {
			# africa
			capital ?= { sub_continent = sub_continent:west_africa }
			capital ?= { sub_continent = sub_continent:central_africa }
			capital ?= { sub_continent = sub_continent:east_africa }
			capital ?= { sub_continent = sub_continent:southern_africa }

			# middle east
			capital ?= { region = region:arabia_region }

			# central asia
			capital ?= { sub_continent = sub_continent:central_asia }

			# southeast asia
			capital ?= { sub_continent = sub_continent:south_east_asia }
			capital ?= { sub_continent = sub_continent:pacific_islands }
		}

		# tags
		not = {
			or = {
				# east africa
				has_or_had_tag = OMA # oman
				has_or_had_tag = ORM # hormuz
			}
			ruler ?= { total_abilities >= 220 }
		}
	}
	alert = yes
	hide_effects = no
	global_institution_growth_modifier = -0.98
}
