htc_force_subjugation = {
	potential_trigger = {
		is_ai = yes
		current_year < 1700
		
		or = {
			capital ?= { sub_continent = sub_continent:south_asia }
			capital ?= { sub_continent = sub_continent:east_africa }
			capital ?= { sub_continent = sub_continent:west_africa }
			capital ?= { region = region:arabia_region }
			capital ?= { region = region:indonesia_region }
			capital ?= { region = region:melanesia_region }
		}
		# tags
		not = {
			or = {
				# india
				has_or_had_tag = DLH # delhi
				has_or_had_tag = MUG # mughals
				# west africa
				and = {
					has_or_had_tag = MAL # mali only loses it after 1425
					current_year >= 1425
				}
				# east africa
				has_or_had_tag = KIT # kitara
				has_or_had_tag = ETH # ethiopia
				has_or_had_tag = SOI # somalia
				has_or_had_tag = ZAN # kilwa
				# arabia
				has_or_had_tag = OMA # oman
				has_or_had_tag = ORM # hormuz
				# indonesia
				has_or_had_tag = MAJ # majapahit
			}
			ruler ?= { total_abilities >= 220 }
		}
	}
	alert = yes
	hide_effects = no
	global_war_score_cost = -0.25
	may_not_take_land_in_peace_treaties = yes
	subjugation_preference_modifier = 1
	annexation_speed_modifier = -1
}
