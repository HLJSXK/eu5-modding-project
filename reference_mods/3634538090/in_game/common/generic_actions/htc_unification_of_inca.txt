htc_uoi_settle_province = {
    type = situation
    show_message = no
    potential = {
        scope:actor.religion = religion:inti
        # situation:unification_of_inca.var:strongest_incan ?= scope:actor
    }

    allow = {
        scope:actor = { at_war = no }
        scope:actor = { is_subject = no }
    }

    automation_tick = never
    automation_tick_frequency = 12
    ai_tick = monthly
    ai_tick_frequency = 6

    cooldown = {
        type = htc_uoi_settle_province_key
        years = 10
    }

    select_trigger = {
        looking_for_a = situation
        interaction_source_list = {
            situation:unification_of_inca = {
                add_to_list = source
            }
        }
        target_flag = recipient
        name = "choose_situation"
        column = {
            data = name
        }
        visible = {
            situation:unification_of_inca = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = province_definition
        interaction_source_list = {
            area:quitu_area = { every_province_definition_in_area = { add_to_list = source } }
            area:chala_area = { every_province_definition_in_area = { add_to_list = source } }
            area:suni_area = { every_province_definition_in_area = { add_to_list = source } }
            area:puna_area = { every_province_definition_in_area = { add_to_list = source } }
            area:rupa_rupa_area = { every_province_definition_in_area = { add_to_list = source } }
            area:atacama_area = { every_province_definition_in_area = { add_to_list = source } }
            area:qullaw_area = { every_province_definition_in_area = { add_to_list = source } }
		}
        cache_interaction_source_list = yes
        target_flag = target
        name = "settle_province"
        none_available_msg_key = "no_province_available"
        column = { data = name }
        column = { data = population }
        visible = {
            any_location_in_province_definition = {
                is_ownable = yes
                has_owner = no
			}
        }
        enabled = {
            any_location_in_province_definition = {
                is_ownable = yes
                has_owner = no
                adjacent_to_owned_or_owned_by_subject = scope:actor
            }
        }
    }
    effect = {
        if = {
            limit = { exists = scope:target }
            scope:actor = {
                add_gold = {
                    value = 10
                    scope:target = {
                        every_location_in_province_definition = {
                            limit = {
                                is_ownable = yes
                                has_owner = no
                            }
                            add = population
                        }
                    }
                    max = 500
                    multiply = -1
                }
            }
        }
        else = {
            custom_tooltip = {
                text = htc_uoi_settle_province_effect_tt
            }
        }
        scope:target ?= {
            every_location_in_province_definition = {
                limit = {
                    is_ownable = yes
                    has_owner = no
                }
                change_location_owner = scope:actor
                change_integration_level = integrated
            }
            # custom_tooltip = {
            #     text = htc_uoi_purchase_province_tt
            #     scope:target = {
            #         trigger_event_non_silently = htc_unification_of_inca.4
            #     }
            # }
        }
    }

    ai_will_do = {
        value = -1
        if = {
            limit = {
                scope:actor = {    
                    gold > {
                        value = 10
                        scope:target = {
                            every_location_in_province_definition = {
                                limit = {
                                    is_ownable = yes
                                    has_owner = no
                                }
                                add = population
                            }
                        }
                        max = 600
                    }
                }
            }
            scope:target = {
                every_location_in_province_definition = {
                    limit = {
                        is_ownable = yes
                        has_owner = no
                    }
                    add = population
                }
            }
            if = {
                limit = { scope:target.area = area:quitu_area }
                multiply = 2
            }
            else_if = {
                limit = { scope:target.area = area:chala_area }
                multiply = 3
            }
            else_if = {
                limit = { scope:target.area = area:suni_area }
                multiply = 6
            }
            else_if = {
                limit = { scope:target.area = area:puna_area }
                multiply = 6
            }
            else_if = {
                limit = { scope:target.area = area:rupa_rupa_area }
                multiply = 4
            }
            else_if = {
                limit = { scope:target.area = area:qullaw_area }
                multiply = 2
            }
            else_if = {
                limit = { scope:target.area = area:atacama_area }
                multiply = 1
            }
        }
    }
}

            data = name
        }

        }
    }

    select_trigger = {
        looking_for_a = location
        interaction_source_list = {
            region:andes_region = {
                every_location_in_region = {
                    limit = {
                        is_ownable = yes
                        nor = {
                            area = area:cuyo_area
                            area = area:ngulumapu_area
                        }
                    }
                    add_to_list = source
                }
			}
		}
        cache_interaction_source_list = yes
        target_flag = target
        name = "settle_location"
        none_available_msg_key = "no_locations_available"
        column = { data = name }
        column = { data = population }
        visible = {
            is_ownable = yes
            has_owner = no
        }
        enabled = {
            adjacent_to_owned_or_owned_by_subject = scope:actor
        }
    }
    effect = {
        if = {
            limit = { exists = scope:target }
            scope:actor = {
                add_gold = {
                    value = 10
                    add = scope:target.population
                    max = 100
                    multiply = -1
                }
            }
        }
        else = {
            custom_tooltip = {
                text = htc_uoi_settle_location_effect_tt
            }
        }
        scope:target ?= {
            change_location_owner = scope:actor
            change_integration_level = integrated
        }
    }

    ai_will_do = {
        value = -1
        if = {
            limit = {
                scope:actor = {    
                    gold > {
                        value = 10
                        add = scope:target.population
                        max = 150
                    }
                }
            }
            add = scope:target.population
        }
    }
}

htc_uoi_conquer_neighbor = {
    type = situation
    show_message = no
    potential = {
        scope:actor.religion = religion:inti
    }

    allow = {
        scope:actor = { at_war = no }
        scope:actor = { is_subject = no }
    }

    automation_tick = never
    automation_tick_frequency = 12
    ai_tick = monthly
    ai_tick_frequency = 6

    cooldown = {
        type = htc_uoi_conquer_neighbor_key
        years = 2
    }

    select_trigger = {
        looking_for_a = situation
        interaction_source_list = {
            situation:unification_of_inca = {
                add_to_list = source
            }
        }
        target_flag = recipient
        name = "choose_situation"
        column = {
            data = name
        }
        visible = {
            situation:unification_of_inca = this
            situation_is_active = yes
        }
    }

    select_trigger = {
        looking_for_a = country
        target_flag = target
        name = "choose_country"
        column = { data = name }
        column = { data = population }
        column = { data = expected_army_size }
        visible = {
            any_owned_location = { region = region:andes_region }
        }
        enabled = {
			is_neighbor_of = scope:actor
            nor = {
                is_allied_with = { target = scope:actor }
                is_subject_or_below_of = scope:actor
            }
        }
    }
	
    effect = {
        scope:actor = {
            declare_war_with_cb = {
                target = scope:target
                type = casus_belli:cb_htc_unify_inca
            }
        }
    }

    ai_will_do = {
        value = 0
        if = {
            limit = { current_year > 1340 }
            scope:actor = {
                add = country_strength
                every_country_that_can_be_called_offensively = {
                    add = country_strength
                }
            }
            scope:target = {
                subtract = country_strength
                every_country_that_can_be_called_defensively = {
                    subtract = country_strength
                }
                # scope:actor = {
                #     subtract = {
                #         value = this_opinion_of_prev
                #         # multiply = 0.1
                #     }
                # }
            }
        }
        if = {
            limit = { scope:target = { is_rival_of = scope:actor } }
            add = 30
        }
        if = {
            limit = {
                scope:actor = {
                    any_owned_location = {
                        integration_level = conquered
                    }
                }
            }
            add = -20
        }
    }
}
