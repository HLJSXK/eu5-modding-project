on_siege_won = {
	on_actions = {
		htc_annex_native_on_capital_occupation
	}
}

on_location_occupied = {
	on_actions = {
		htc_annex_native_on_capital_occupation
	}
}

htc_annex_native_on_capital_occupation = {
	trigger = {
		or = {
			is_colonial_overlord = yes
			is_colonial_subject = yes
		}
		scope:target = {
			is_capital = yes
		}
		any_current_war = {
			defender_leader = scope:target.owner
			has_casus_belli = yes
			casus_belli ?= casus_belli:cb_htc_conquer_native
		}
	}
	effect = {
		if = {
			limit = {
				is_colonial_subject = yes
				is_neighbor_of = scope:target.owner
			}
			annex_country = {
				country = scope:target.owner
				reason = MilitaryConquest
			}
		}
		else_if = {
			limit = {
				is_colonial_overlord = yes
				any_subject = {
					is_colonial_subject = yes
					is_neighbor_of = scope:target.owner
				}
			}
			ordered_subject = {
				limit = { 
					is_colonial_subject = yes
					is_neighbor_of = scope:target.owner
				}
				order_by = {
					value = "capital.distance_to_squared(scope:target)"
					multiply = -1
				}
				max = 1

				annex_country = {
					country = scope:loser
					reason = MilitaryConquest
				}
			}
		}
	}
}
