iv2_on_game_start = {
	effect = {		
		every_country = {
			iv2_add_vars_effect = yes
			#iv2_add_bcs_effect = yes # Idea Variation 2 Research System
		}
	}
}

iv2_on_game_start_after_lobby_CMF = {
	effect = {		
		every_country = {
			iv2_add_bcs_effect = yes # Idea Variation 2 Research System
		}
	}
}

iv2_innovation_point_calc = {
	effect = {
		if = {
			limit = {
				exists = root
			}
			iv2_innovation_calc_effect = yes
		}	
	}
}

iv2_calc_alerts = {
	effect = {
		iv2_recalc_alerts = yes
	}
}

iv2_on_character_death = {
	effect = {
		scope:target = {
			if = {
				limit = {
					scope:target = { has_variable = Iv2_researcher_adm }
				}
				remove_variable = Iv2_researcher_adm
				root = { remove_variable = Iv2_researcher_adm }
			}
			if = {
				limit = {
					scope:target = { has_variable = Iv2_researcher_dip }
				}
				remove_variable = Iv2_researcher_dip
				root = { remove_variable = Iv2_researcher_dip }
			}
			if = {
				limit = {
					scope:target = { has_variable = Iv2_researcher_army }
				}
				remove_variable = Iv2_researcher_army
				root = { remove_variable = Iv2_researcher_army }
			}
			if = {
				limit = {
					scope:target = { has_variable = Iv2_researcher_navy }
				}
				remove_variable = Iv2_researcher_navy
				root = { remove_variable = Iv2_researcher_navy }
			}
		}
	}
}

iv_2_on_annex = {
	effect = {
		scope:target = {
			every_character = {

				remove_variable = Iv2_researcher_army
				remove_variable = Iv2_researcher_navy
			}
		}
	}
}
