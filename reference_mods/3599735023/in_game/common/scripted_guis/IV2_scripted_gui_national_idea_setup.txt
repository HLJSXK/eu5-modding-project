########################################################
####### Open Dynamic National Ideas Window
########################################################

iv2_sgui_open_dynamic_national_ideas = {
	scope = country 	
	
	is_shown = {  
		always = yes
	}		
	is_valid = {  
		iv2_scripted_trigger_use_dynamic_ideas = yes
	}	
	
	effect = {
		iv2_dynamic_national_ideas_list_setup_effect = yes
	}
}
