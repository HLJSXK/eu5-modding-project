		}

########################################################
####### Show all Ideagroups
########################################################

iv2_gui_show_all_ig_statistics = {
	scope = country 	
	
	is_shown = {  
		always = yes
	}		
	is_valid = {  
		has_variable = iv2_var_show_all_ig_statistics_no
	}	
	
	effect = {
		set_variable = {
			name = iv2_var_show_all_ig_statistics_yes
			value = 0
		}
		remove_variable = iv2_var_show_all_ig_statistics_no
		
		iv2_add_statistics_list_adm_effect = yes
		iv2_add_statistics_list_dip_effect = yes
		iv2_add_statistics_list_army_effect = yes
		iv2_add_statistics_list_navy_effect = yes
	}
	
}

########################################################
####### Hide all Ideagroups
########################################################

iv2_gui_hide_all_ig_statistics = {
	scope = country 	
	
	is_shown = {  
		always = yes
	}

		}

		iv2_remove_statistics_list_dip_effect = yes
		iv2_remove_statistics_list_army_effect = yes
		iv2_remove_statistics_list_navy_effect = yes
	}
	
}
