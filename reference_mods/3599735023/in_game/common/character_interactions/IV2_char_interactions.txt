###################################################
###### Appoint Adm Researcher
###################################################

iv2_appoint_researcher_adm = { 
	message = yes
	is_consort_action = no
	on_own_nation = yes


	potential = {
		scope:actor = {
			NOT = {
				has_variable = Iv2_researcher_adm
			}
		}
	}

	ai_tick = daily
	ai_tick_frequency = 45

	allow = {
		custom_tooltip = {
		text = iv2_tt_not_researcher_already
			scope:recipient = {
				NOT = {
					OR = {
					has_variable = Iv2_researcher_adm
					has_variable = Iv2_researcher_dip
					has_variable = Iv2_researcher_army
					has_variable = Iv2_researcher_navy
					}
				}
			}
		}	
	}
	
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
			is_alive = yes
		}
		enabled = {
		}
	}

	effect = {
		scope:actor = {
			every_character = {
				remove_variable = Iv2_researcher_adm # remove variable if there are old chars with it present
			}
			set_variable = {
			name = Iv2_researcher_adm
			value = scope:recipient
			}
		}
		scope:recipient = {
			set_variable = {
			name = Iv2_researcher_adm
			}
			add_character_modifier = {
				modifier = banned_from_cabinet
				years = -1
			}
			add_character_modifier = {
				modifier = block_leading_armies_or_navies
				years = -1
			}
		}

	}

	ai_will_do = {
		add = {
			value = scope:recipient.total_abilities
			subtract = 250
		}
	}
}
