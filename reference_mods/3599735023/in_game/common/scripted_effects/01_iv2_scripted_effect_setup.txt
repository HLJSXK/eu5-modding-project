iv2_add_bcs_effect = {
		if = {
			limit = {
				has_game_rule = iv2_research_progress_1
			}
			add_bureaucracy = bureaucracy_type:iv2_bc_research_75	
		}
		if = {
			limit = {
				has_game_rule = iv2_research_progress_2
			}
			add_bureaucracy = bureaucracy_type:iv2_bc_research_100
		}
		if = {
			limit = {
				has_game_rule = iv2_research_progress_3
			}
			add_bureaucracy = bureaucracy_type:iv2_bc_research_125	
		}
		if = {
			limit = {

			add_bureaucracy = bureaucracy_type:iv2_bc_research_200	
		}
