###############################################################
#### Idea Variation 2 National Idea Setup
###############################################################


iv2_scripted_trigger_has_unique_national_idea_set = {
	OR = {
		has_or_had_tag = CAS
		has_or_had_tag = BYZ
		has_or_had_tag = FLO
	}
}


iv2_scripted_trigger_use_historic_ideas = {
	iv2_scripted_trigger_has_unique_national_idea_set = yes
	has_game_rule = iv2_rule_national_ideas_1
	NOT = { 
		OR = {
		has_game_rule = iv2_rule_national_ideas_2
		has_game_rule = iv2_rule_national_ideas_3 
		}
	}
}

iv2_scripted_trigger_use_dynamic_ideas = {
	OR = {
		has_game_rule = iv2_rule_national_ideas_2
		AND = {
			has_game_rule = iv2_rule_national_ideas_1
			iv2_scripted_trigger_has_unique_national_idea_set = no
		}
	}
	NOT = { has_game_rule = iv2_rule_national_ideas_3 }
}
