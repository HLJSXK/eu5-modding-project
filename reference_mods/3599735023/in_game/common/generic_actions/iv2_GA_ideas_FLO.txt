###############################################################
#### Select National Idea FLO_1
###############################################################

iv2_ga_select_historic_FLO_1 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 36

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform

	allow = {
		scope:actor = {
			hidden_trigger = {
				NOT = { has_country_modifier = national_idea_modifier_FLO_1 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_1_1 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_1_2 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_1_3 }
			}
			custom_tooltip = {
			text = iv2_tt_no_points_for_nat_idea_adm
			var:iv2_var_progress_points_adm >= var:iv2_var_national_idea_cost_adm
			}
			custom_tooltip = {
			text = iv2_tt_not_reached_national_idea_limit
			var:iv2_var_current_national_idea_limit > var:iv2_var_national_idea_current
			}
			iv2_sct_nat_idea_FLO_1 = yes
		}
	}

	ai_will_do = {
		add = {
			value = 1000
		}
	}
	
	effect = {
		scope:actor = {								
			hidden_effect = {
				add_country_modifier = { 
					modifier = national_idea_modifier_FLO_1 
					years = -1 
				}
				set_variable = { 
					name = iv2_var_nat_adm_1
					value = flag:iv2_picked_historic_idea_FLO_1
				}
				add_to_variable_list = {
						name = iv2_picked_national_ideas_list
						target = flag:iv2_picked_historic_idea_FLO_1
				}
				change_variable = {
					name = iv2_var_national_idea_current
					add = 1
				}	
			}
			custom_tooltip = {
			text = iv2_tt_sub_points_adm
				change_variable = {
					name = iv2_var_progress_points_adm
					subtract = var:iv2_var_national_idea_cost_adm
				}
			}
			custom_tooltip = {
			text = iv2_tt_increase_points_adm
				change_variable = {
					name = iv2_var_national_idea_cost_adm
					add = iv2_progress_cost_increase
				}
			}
		}
	}
}

#### Deselect National Idea adm_1
###############################################################


iv2_ga_deselect_historic_FLO_1 = {
	type = owncountry	

	ai_tick = never

	allow = {
		scope:actor = {
			OR = {
				has_country_modifier = national_idea_modifier_FLO_1
				has_country_modifier = national_idea_modifier_FLO_1_1
				has_country_modifier = national_idea_modifier_FLO_1_2
				has_country_modifier = national_idea_modifier_FLO_1_3
			}		
		}
	}
	
	effect = {
		scope:actor = {								
			custom_tooltip = {
				text = iv2_tt_give_up_nat_idea_adm
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_1
					}
					remove_country_modifier = national_idea_modifier_FLO_1
					change_variable = {
						name = iv2_var_national_idea_cost_adm
						subtract = iv2_progress_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_1_1
					}
					remove_country_modifier = national_idea_modifier_FLO_1_1
					change_variable = {
						name = iv2_var_national_idea_cost_adm
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_adm
						subtract = iv2_innovation_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_1_2
					}
					remove_country_modifier = national_idea_modifier_FLO_1_2
					change_variable = {
						name = iv2_var_national_idea_cost_adm
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_adm
						subtract = iv2_innovation_cost_decrease_x2
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_1_3
					}
					remove_country_modifier = national_idea_modifier_FLO_1_3
					change_variable = {
						name = iv2_var_national_idea_cost_adm
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_adm
						subtract = iv2_innovation_cost_decrease_x3
					}
				}			
			}
			hidden_effect = {
				remove_list_variable = {
					name = iv2_picked_national_ideas_list
					target = flag:iv2_picked_historic_idea_FLO_1
				}
				change_variable = {
					name = iv2_var_national_idea_current
					subtract = 1
				}
			}
		}
	}
}

					}

					limit = {
						has_country_modifier = national_idea_modifier_FLO_3_1
					}
					remove_country_modifier = national_idea_modifier_FLO_3_1

						name = iv2_var_national_idea_cost_adm

						name = iv2_var_national_idea_innovation_cost_adm
						subtract = iv2_innovation_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_3_2
					}
					remove_country_modifier = national_idea_modifier_FLO_3_2
					change_variable = {
						name = iv2_var_national_idea_cost_adm
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_adm
						subtract = iv2_innovation_cost_decrease_x2
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_3_3
					}
					remove_country_modifier = national_idea_modifier_FLO_3_3
					change_variable = {
						name = iv2_var_national_idea_cost_adm
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_adm
						subtract = iv2_innovation_cost_decrease_x3
					}
				}			
			}
			hidden_effect = {
				remove_list_variable = {
					name = iv2_picked_national_ideas_list
					target = flag:iv2_picked_historic_idea_FLO_3
				}
				change_variable = {
					name = iv2_var_national_idea_current

			}

					}

				remove_list_variable = {
					name = iv2_picked_national_ideas_list

iv2_ga_select_historic_FLO_25 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 36

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform

	allow = {
		scope:actor = {
			hidden_trigger = {
				NOT = { has_country_modifier = national_idea_modifier_FLO_25 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_25_1 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_25_2 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_25_3 }
			}
			custom_tooltip = {
			text = iv2_tt_no_points_for_nat_idea_army
			var:iv2_var_progress_points_army >= var:iv2_var_national_idea_cost_army
			}
			custom_tooltip = {
			text = iv2_tt_not_reached_national_idea_limit
			var:iv2_var_current_national_idea_limit > var:iv2_var_national_idea_current
			}
			iv2_sct_dnat_idea_army_1 = yes
		}
	}

	ai_will_do = {
		add = {
			value = -1000
		}
	}
	
	effect = {
		scope:actor = {								
			hidden_effect = {
				add_country_modifier = { 
					modifier = national_idea_modifier_FLO_25 
					years = -1 
				}
				set_variable = { 
					name = iv2_var_nat_army_1
					value = flag:iv2_picked_historic_idea_FLO_25
				}
				add_to_variable_list = {
						name = iv2_picked_national_ideas_list
						target = flag:iv2_picked_historic_idea_FLO_25
				}
				change_variable = {
					name = iv2_var_national_idea_current
					add = 1
				}	
			}
			custom_tooltip = {
			text = iv2_tt_sub_points_army
				change_variable = {
					name = iv2_var_progress_points_army
					subtract = var:iv2_var_national_idea_cost_army
				}
			}
			custom_tooltip = {
			text = iv2_tt_increase_points_army
				change_variable = {
					name = iv2_var_national_idea_cost_army
					add = iv2_progress_cost_increase
				}
			}
		}
	}
}

iv2_ga_deselect_historic_FLO_25 = {
	type = owncountry	

	ai_tick = never

	allow = {
		scope:actor = {
			OR = {
				has_country_modifier = national_idea_modifier_FLO_25
				has_country_modifier = national_idea_modifier_FLO_25_1
				has_country_modifier = national_idea_modifier_FLO_25_2
				has_country_modifier = national_idea_modifier_FLO_25_3
			}		
		}
	}
	
	effect = {
		scope:actor = {								
			custom_tooltip = {
				text = iv2_tt_give_up_nat_idea_army
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_25
					}
					remove_country_modifier = national_idea_modifier_FLO_25
					change_variable = {
						name = iv2_var_national_idea_cost_army
						subtract = iv2_progress_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_25_1
					}
					remove_country_modifier = national_idea_modifier_FLO_25_1
					change_variable = {
						name = iv2_var_national_idea_cost_army
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_army
						subtract = iv2_innovation_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_25_2
					}
					remove_country_modifier = national_idea_modifier_FLO_25_2
					change_variable = {
						name = iv2_var_national_idea_cost_army
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_army
						subtract = iv2_innovation_cost_decrease_x2
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_25_3
					}
					remove_country_modifier = national_idea_modifier_FLO_25_3
					change_variable = {
						name = iv2_var_national_idea_cost_army
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_army
						subtract = iv2_innovation_cost_decrease_x3
					}
				}			
			}
			hidden_effect = {
				remove_list_variable = {
					name = iv2_picked_national_ideas_list
					target = flag:iv2_picked_historic_idea_FLO_25
				}
				change_variable = {
					name = iv2_var_national_idea_current
					subtract = 1
				}
			}
		}
	}
}

					limit = {
						has_country_modifier = national_idea_modifier_FLO_32
					}
					remove_country_modifier = national_idea_modifier_FLO_32
					change_variable = {
						name = iv2_var_national_idea_cost_navy
						subtract = iv2_progress_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_32_1
					}
					remove_country_modifier = national_idea_modifier_FLO_32_1
					change_variable = {
						name = iv2_var_national_idea_cost_navy
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_navy
						subtract = iv2_innovation_cost_increase
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_32_2
					}
					remove_country_modifier = national_idea_modifier_FLO_32_2
					change_variable = {
						name = iv2_var_national_idea_cost_navy
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_navy
						subtract = iv2_innovation_cost_decrease_x2
					}
				}
				
				if = {
					limit = {
						has_country_modifier = national_idea_modifier_FLO_32_3
					}
					remove_country_modifier = national_idea_modifier_FLO_32_3
					change_variable = {
						name = iv2_var_national_idea_cost_navy
						subtract = iv2_progress_cost_increase
					}
					change_variable = {
						name = iv2_var_national_idea_innovation_cost_navy
						subtract = iv2_innovation_cost_decrease_x3
					}
				}			
			}
			hidden_effect = {
				remove_list_variable = {
					name = iv2_picked_national_ideas_list
					target = flag:iv2_picked_historic_idea_FLO_32
				}
				change_variable = {
					name = iv2_var_national_idea_current
					subtract = 1
				}
			}
		}
	}
}

iv2_ga_select_historic_FLO_35 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 36

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform

	allow = {
		scope:actor = {
			hidden_trigger = {
				NOT = { has_country_modifier = national_idea_modifier_FLO_35 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_35_1 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_35_2 }
				NOT = { has_country_modifier = national_idea_modifier_FLO_35_3 }
			}
			custom_tooltip = {
			text = iv2_tt_no_points_for_nat_idea_navy
			var:iv2_var_progress_points_navy >= var:iv2_var_national_idea_cost_navy
			}
			custom_tooltip = {
			text = iv2_tt_not_reached_national_idea_limit
			var:iv2_var_current_national_idea_limit > var:iv2_var_national_idea_current
			}
			iv2_sct_dnat_idea_navy_1 = yes
		}
	}

	ai_will_do = {
		add = {
			value = -1000
		}
	}
	
	effect = {
		scope:actor = {								
			hidden_effect = {
				add_country_modifier = { 
					modifier = national_idea_modifier_FLO_35 
					years = -1 
				}
				set_variable = { 
					name = iv2_var_nat_navy_1
					value = flag:iv2_picked_historic_idea_FLO_35
				}
				add_to_variable_list = {
						name = iv2_picked_national_ideas_list
						target = flag:iv2_picked_historic_idea_FLO_35
				}
				change_variable = {
					name = iv2_var_national_idea_current
					add = 1
				}	
			}
			custom_tooltip = {
			text = iv2_tt_sub_points_navy
				change_variable = {
					name = iv2_var_progress_points_navy
					subtract = var:iv2_var_national_idea_cost_navy
				}
			}
			custom_tooltip = {
			text = iv2_tt_increase_points_navy
				change_variable = {
					name = iv2_var_national_idea_cost_navy
					add = iv2_progress_cost_increase
				}
			}
		}
	}
}
