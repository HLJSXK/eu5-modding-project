iv2_ga_select_ideagroup_dip_1 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 36

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform


	potential = {
		
	}

	allow = {
		scope:actor = {
			hidden_trigger = {
				NOT = { has_variable = iv2_ideagroup_select_dip_1_1 }
			}
			custom_tooltip = {
				text = iv2_tt_no_available_idea_dip
				var:iv2_var_current_idea_group_dip > var:iv2_var_current_picked_idea_group_dip
			}
		}
	}

	ai_will_do = {
		add = {
			value = 1000
		}
	}
	
	effect = {
		scope:actor = {
			set_variable = iv2_ideagroup_select_dip_1_1 
			set_variable = { 
				name = iv2_var_dip_1
				value = flag:iv2_dip_1
			}
			add_to_variable_list = {
					name = iv2_picked_ideagroups_dip
					target = flag:iv2_dip_1
			}			
			set_variable = { 
				name = iv2_var_stat_dip_1
				value = flag:iv2_stat_dip_1
			}
			add_to_variable_list = {
					name = iv2_statistics_list_dip
					target = flag:iv2_stat_dip_1
			}
			change_variable = {
				name = iv2_var_current_picked_idea_group_dip
				add = 1
			}
				
		}
	}
}

iv2_ga_deselect_ideagroup_dip_1 = {
	type = owncountry	

	ai_tick = never
	#ai_tick_frequency = 36

	potential = {
		
	}

	allow = {
		
	}
	
	effect = {
		scope:actor = {
			remove_variable = iv2_ideagroup_select_dip_1_1 
			remove_list_variable = {
					name = iv2_picked_ideagroups_dip
					target = flag:iv2_dip_1
			}				
			remove_list_variable = {
					name = iv2_statistics_list_dip
					target = flag:iv2_stat_dip_1
			}
			iv2_scripted_effect_ig_revoke_dip_1 = yes
			change_variable = {
				name = iv2_var_current_picked_idea_group_dip
				subtract = 1
			}
		}	
	}
}
