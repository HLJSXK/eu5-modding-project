iv2_ga_select_ideagroup_army_1 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 36

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform

	

	allow = {
		scope:actor = {
			hidden_trigger = {
				NOT = { has_variable = iv2_ideagroup_select_army_1_1 }
			}
			custom_tooltip = {
				text = iv2_tt_no_available_idea_army				
				var:iv2_var_current_idea_group_army > var:iv2_var_current_picked_idea_group_army
			}
		}
	}

	ai_will_do = {
		add = {
			value = 1000
		}
	}
	
	effect = {
		scope:actor = {
			set_variable = iv2_ideagroup_select_army_1_1 
			set_variable = { 
				name = iv2_var_army_1
				value = flag:iv2_army_1
			}
			add_to_variable_list = {
					name = iv2_picked_ideagroups_army
					target = flag:iv2_army_1
			}			
			set_variable = { 
				name = iv2_var_stat_army_1
				value = flag:iv2_stat_army_1
			}
			add_to_variable_list = {
					name = iv2_statistics_list_army
					target = flag:iv2_stat_army_1
			}
			change_variable = {
				name = iv2_var_current_picked_idea_group_army
				add = 1
			}
				
		}
	}
}

iv2_ga_deselect_ideagroup_army_1 = {
	type = owncountry	

	ai_tick = never
	#ai_tick_frequency = 36

	potential = {
		
	}

	allow = {
		
	}
	
	effect = {
		scope:actor = {
			remove_variable = iv2_ideagroup_select_army_1_1 
			remove_list_variable = {
					name = iv2_picked_ideagroups_army
					target = flag:iv2_army_1
			}				
			remove_list_variable = {
					name = iv2_statistics_list_army
					target = flag:iv2_stat_army_1
			}
			iv2_scripted_effect_ig_revoke_army_1 = yes
			change_variable = {
				name = iv2_var_current_picked_idea_group_army
				subtract = 1
			}
		}	
	}
}

			custom_tooltip = {
			text = iv2_tt_id_sub_points_army
				change_variable = {
						name = iv2_var_innovation_points_army

			}
		}
	}

iv2_ga_select_ideagroup_army_5 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 30

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform


	allow = {
		scope:actor = {
			custom_tooltip = {
				text = iv2_tt_no_available_idea_army
				var:iv2_var_current_idea_group_army > var:iv2_var_current_picked_idea_group_army
			}
			NOT = {
				OR = {
					has_variable = iv2_ideagroup_select_army_5_1 
					has_variable = iv2_ideagroup_select_army_6_1 
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = 1000
		}
	}
	
	effect = {
		scope:actor = {
			set_variable = iv2_ideagroup_select_army_5_1 
			set_variable = { 
				name = iv2_var_army_5
				value = flag:iv2_army_5
			}
			add_to_variable_list = {
					name = iv2_picked_ideagroups_army
					target = flag:iv2_army_5
			}			
			set_variable = { 
				name = iv2_var_stat_army_5
				value = flag:iv2_stat_army_5
			}
			add_to_variable_list = {
					name = iv2_statistics_list_army
					target = flag:iv2_stat_army_5
			}
			change_variable = {
				name = iv2_var_current_picked_idea_group_army
				add = 1
			}
				
		}
	}
}

iv2_ga_deselect_ideagroup_army_5 = {
	type = owncountry	

	ai_tick = never
	#ai_tick_frequency = 36

	potential = {
		
	}

	allow = {
		
	}
	
	effect = {
		scope:actor = {
			remove_variable = iv2_ideagroup_select_army_5_1 
			remove_list_variable = {
					name = iv2_picked_ideagroups_army
					target = flag:iv2_army_5
			}				
			remove_list_variable = {
					name = iv2_statistics_list_army
					target = flag:iv2_stat_army_5
			}
			iv2_scripted_effect_ig_revoke_army_5 = yes
			change_variable = {
				name = iv2_var_current_picked_idea_group_army
				subtract = 1
			}
		}	
	}
}

iv2_ga_select_ideagroup_army_7 = {
	type = owncountry	

	ai_tick = monthly
	ai_tick_frequency = 37

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreform


	allow = {
		scope:actor = {
			custom_tooltip = {
				text = iv2_tt_no_available_idea_army
				var:iv2_var_current_idea_group_army > var:iv2_var_current_picked_idea_group_army
			}
			NOT = {
				OR = {
					has_variable = iv2_ideagroup_select_army_7_1 
					has_variable = iv2_ideagroup_select_army_8_1 
					has_variable = iv2_ideagroup_select_army_9_1 
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = 1000
		}
	}
	
	effect = {
		scope:actor = {
			set_variable = iv2_ideagroup_select_army_7_1 
			set_variable = { 
				name = iv2_var_army_7
				value = flag:iv2_army_7
			}
			add_to_variable_list = {
					name = iv2_picked_ideagroups_army
					target = flag:iv2_army_7
			}			
			set_variable = { 
				name = iv2_var_stat_army_7
				value = flag:iv2_stat_army_7
			}
			add_to_variable_list = {
					name = iv2_statistics_list_army
					target = flag:iv2_stat_army_7
			}
			change_variable = {
				name = iv2_var_current_picked_idea_group_army
				add = 1
			}
				
		}
	}
}

iv2_ga_deselect_ideagroup_army_7 = {
	type = owncountry	

	ai_tick = never
	#ai_tick_frequency = 36

	potential = {
		
	}

	allow = {
		
	}
	
	effect = {
		scope:actor = {
			remove_variable = iv2_ideagroup_select_army_7_1 
			remove_list_variable = {
					name = iv2_picked_ideagroups_army
					target = flag:iv2_army_7
			}				
			remove_list_variable = {
					name = iv2_statistics_list_army
					target = flag:iv2_stat_army_7
			}
			iv2_scripted_effect_ig_revoke_army_7 = yes
			change_variable = {
				name = iv2_var_current_picked_idea_group_army
				subtract = 1
			}
		}	
	}
}
