# ==========================================================================
# DANUBIAN MONARCHY -- timed modifiers.
# Level 4: 35-50 yr duration, <=8 modifier lines, signature values strong
# but within vanilla static-modifier distribution.
#
# The founding dilemma is NOT military-vs-economy. Both choices are about
# what holds a multi-ethnic Danube empire together: the external enemy
# (the Ottoman frontier, Path A) or the internal machine (the centralized
# Gesamtmonarchie, Path B). Each draws from several categories and carries
# an orthogonal cost.
# ==========================================================================

# ===== FORMATION CHOICES =====

# Path A : Antemurale Christianitatis -- the Bulwark of Christendom.
# The Militaergrenze, the Grenzer soldier-colonies, the perpetual Ottoman
# war that gave the Danube state its reason to exist. Christendom's shield
# wins prestige and standing; the cost is a court tied up on the frontier.
nd_dnm_antemurale_christianitatis = {
	game_data = {
		category = country
	}
	# Signature: the frontier manpower wall and the standing wall of forts
	global_manpower_modifier = 0.2
	global_levy_size_modifier = 0.25
	global_garrison_size_modifier = 0.25
	# The army of the long Turkish war
	land_morale_modifier = 0.15
	global_hostile_attrition = 0.15
	# The shield of Christendom -- standing among Catholic Europe
	diplomatic_reputation = 2
	tolerance_own = 1
	# Cost (orthogonal -- administrative, not military): the court and
	# treasury are bled by the perpetual frontier garrison.
	country_cabinet_efficiency = -0.1
}

# Path B : Gesamtmonarchie -- the Total Monarchy.
# The Hofkammer, the Hofkriegsrat, the Pragmatic Sanction: three crowns
# welded into one administrative body answerable to Vienna alone. The
# centralizing court is efficient and obeyed; the cost is the resentment
# of the estates and the diet, slowing dynastic diplomacy.
nd_dnm_gesamtmonarchie = {
	game_data = {
		category = country
	}
	# Signature: the centralized administrative machine
	global_monthly_control = 0.005
	country_cabinet_efficiency = 0.15
	legislative_efficiency = 0.1
	# The Pragmatic realm: one body, one treasury, fewer fault lines
	tax_income_efficiency = 0.05
	global_separatism = -0.1
	global_pop_assimilation_speed_modifier = 0.4
	cultures_capacity = 1
	# Cost (orthogonal -- diplomatic, not administrative): the crowns and
	# the imperial diet resent the centralizing court.
	diplomatic_reputation = -2
}

# ==========================================================================
# SUCCESSION CHAIN MODIFIERS (nd_dnm.30/.31/.32)
# Each event grants a perpetual modifier (years = -1) and explicitly removes
# the previous one. The chain narrative is "the open question -> a chosen
# stance -> a chosen action". All five are removed in the disaster's on_end
# AND the situation's on_ended, so they never outlive their narrative.
#
# Axis choice: chain modifiers hit INTERNAL COHESION (legitimacy, estates,
# separatism), not diplomacy. The Reich pariah already hits diplomacy heavily
# -- stacking more would just be "more of the same bad thing." Internal
# cohesion is the second pressure-shape, distinct from external.
# ==========================================================================
