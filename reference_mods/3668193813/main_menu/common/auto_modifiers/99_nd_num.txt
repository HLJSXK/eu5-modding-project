# Comancheria (NUM) bureaucracy slot grant.
# Activates after NUM is formed (nd_num_formed) so the player only
# gains the bureaucracy-system permission once Numunu Sookobitu has
# been proclaimed, and only when the bureaucracies game-rule is on.
# Capital adds 1 implicit slot; this grants 2 more = 3 effective
# slots vs 4 institutions, forcing the player to leave one council
# uninstitutionalized -- in keeping with the federation's preference
# for occasion-summoned councils over standing offices.

nd_num_bureaucracy_authority = {
	potential_trigger = {
		tag = NUM
		has_variable = nd_num_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
