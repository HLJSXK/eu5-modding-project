# Grants bureaucracy authority to BNO once the unified Bornean state has
# been formed and the player has the bureaucracies game-rule enabled.
# Disappears if the player forms a successor formable (uses tag, not
# has_or_had_tag).

nd_bno_bureaucracy_authority = {
	potential_trigger = {
		tag = BNO
has_variable = nd_bno_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
