# Anglo-French Dominion (CAN) bureaucracy slot grant.
# Activates after CAN is formed (nd_can_formed) so the player only gains
# the bureaucracy-system permission once the unified Canadian dominion has
# been declared, and only when the bureaucracies game-rule is enabled.
# Disappears if the player forms a successor formable (uses tag, not
# has_or_had_tag).

nd_can_bureaucracy_authority = {
	potential_trigger = {
		tag = CAN
has_variable = nd_can_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
