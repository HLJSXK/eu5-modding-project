# Grants bureaucracy authority to SOE once the Songhai imperial revival
# has been formed and the player has the bureaucracies game-rule enabled.
# 2 slots + capital's 1 = 3 total. Player picks 3 of 6 institutions.

nd_soe_bureaucracy_authority = {
	potential_trigger = {
		tag = SOE
has_variable = nd_soe_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
