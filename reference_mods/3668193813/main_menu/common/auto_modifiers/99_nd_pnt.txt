# National Destinies -- PNT (Pintados Confederation) bureaucracy authority
#
# Activates after the formation event nd_pnt.1 has set nd_pnt_formed.
# Grants allow_bureaucracy + 2 extra bureaucracy slots; the capital
# provides one more, so the player picks 3 of 5 institutions.
nd_pnt_bureaucracy_authority = {
    potential_trigger = {
        tag = PNT
        has_variable = nd_pnt_formed
        has_game_rule = nd_bureaucracies_enabled
    }
    allow_bureaucracy = yes
    global_max_bureaucracy_slots = 2
}
