# National Destinies -- AYU bureaucracy authority
#
# On formation (variable nd_ayu_formed set by nd_ayu.1), Ayutthaya
# gains allow_bureaucracy + 2 supplemental slots. Combined with the
# universal capital +1 slot, the player has 3 effective slots to
# spend across the five AYU bureaucracies.
nd_ayu_bureaucracy_authority = {
	potential_trigger = {
		tag = AYU
has_variable = nd_ayu_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
