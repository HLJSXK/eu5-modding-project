# Grants bureaucracy authority to GUJ once Gujarat has been formed and the
# player has the bureaucracies game-rule enabled. Disappears if the player
# forms a successor formable (uses tag, not has_or_had_tag).

nd_guj_bureaucracy_authority = {
	potential_trigger = {
		tag = GUJ
has_variable = nd_guj_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
