# Sokoto Caliphate bureaucracy slot grant.
# Activates after formation (nd_sko_formed) so the player only gains the
# bureaucracy-system permission once the caliphate has been declared,
# and only when the bureaucracies game-rule is enabled.

nd_sko_bureaucracy_authority = {
	potential_trigger = {
		tag = SKO
has_variable = nd_sko_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
