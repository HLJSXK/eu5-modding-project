# Imerina (Madagascar) bureaucracy slot grant.
# Activates after MIR is formed (nd_mir_formed) so the player only gains the
# bureaucracy-system permission once the unified Malagasy kingdom has been
# proclaimed, and only when the bureaucracies game-rule is enabled. Two
# slots + the capital slot = three effective: the player picks three of
# the five Malagasy institutions.

nd_mir_bureaucracy_authority = {
	potential_trigger = {
		tag = MIR
has_variable = nd_mir_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
