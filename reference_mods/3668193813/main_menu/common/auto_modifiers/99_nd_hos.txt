# Grants bureaucracy authority to HOS once the Order has been refounded as a
# sovereign state and the player has the bureaucracies game-rule enabled.
# Disappears if the player forms a successor formable (uses tag, not has_or_had_tag).

nd_hos_bureaucracy_authority = {
	potential_trigger = {
		tag = HOS
		has_variable = nd_hos_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
