# Al-Andalus bureaucracy slot grant.
# Activates after formation (nd_adu_formed) so the player only gains the
# bureaucracy-system permission once the caliphate has been declared,
# and only when the bureaucracies game-rule is enabled.

nd_adu_bureaucracy_authority = {
	potential_trigger = {
		tag = ADU
has_variable = nd_adu_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
