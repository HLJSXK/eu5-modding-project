# Hanoverian bureaucracy slot grant.
# Activates after HAN is formed (nd_han_formed) so the player only gains the
# bureaucracy-system permission once the Welf state has been declared,
# and only when the bureaucracies game-rule is enabled.

nd_han_bureaucracy_authority = {
	potential_trigger = {
		tag = HAN
has_variable = nd_han_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
