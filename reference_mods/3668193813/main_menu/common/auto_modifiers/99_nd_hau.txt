# National Destinies -- HAU bureaucracy authority
#
# Grants 2 bureaucracy slots once HAU has been formed (nd_hau_formed variable
# set by the formation event nd_hau.1) and the player has the bureaucracies
# game-rule enabled. Combined with the capital's universal +1 slot the player
# may keep 3 of the 6 Hausa institutions active.

nd_hau_bureaucracy_authority = {
	potential_trigger = {
		tag = HAU
has_variable = nd_hau_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
