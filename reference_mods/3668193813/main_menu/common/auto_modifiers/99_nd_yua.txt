# National Destinies -- YUA bureaucracy authority
#
# On formation (variable nd_yua_formed set by nd_yua.1), the Yuan
# gains allow_bureaucracy + 2 supplemental slots. Combined with the
# universal capital +1 slot, the player has 3 effective slots to
# spend across the six YUA bureaucracies (Zhongshu Sheng, Shumi
# Yuan, Yushi Tai, Semu caste, Phags-pa script, Yan Zheng salt).
nd_yua_bureaucracy_authority = {
	potential_trigger = {
		tag = YUA
has_variable = nd_yua_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
