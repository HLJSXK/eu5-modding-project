# Song dynasty bureaucracy slot grant.
# Activates after formation (nd_cso_formed) so the player only gains the
# bureaucracy-system permission once the Song restoration is declared,
# and only when the bureaucracies game-rule is enabled.
# Capital adds 1 implicit slot; this grants 2 more = 3 effective slots
# vs 7 institutions, forcing meaningful choice among the Song offices.

nd_cso_bureaucracy_authority = {
	potential_trigger = {
		tag = CSO
has_variable = nd_cso_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
