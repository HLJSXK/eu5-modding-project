# Maya bureaucracy slot grant.
# Activates after formation (nd_may_formed) so the player only gains the
# bureaucracy-system permission once the unified Maya state has been
# proclaimed, and only when the bureaucracies game-rule is enabled.

nd_may_bureaucracy_authority = {
	potential_trigger = {
		tag = MAY
has_variable = nd_may_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
