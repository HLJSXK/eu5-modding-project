# Mossi bureaucracy slot grant.
# Activates after MSI is formed (nd_msi_formed) so the player only gains the
# bureaucracy-system permission once the unified Mossi state has been
# declared, and only when the bureaucracies game-rule is enabled.

nd_msi_bureaucracy_authority = {
	potential_trigger = {
		tag = MSI
has_variable = nd_msi_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
