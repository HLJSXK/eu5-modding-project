# Nusantara bureaucracy slot grant.
# Activates after NUS is formed (nd_nus_formed) so the player only gains the
# bureaucracy-system permission once the unified archipelagic state has been
# declared, and only when the bureaucracies game-rule is enabled.

nd_nus_bureaucracy_authority = {
	potential_trigger = {
		tag = NUS
has_variable = nd_nus_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
