# Grants bureaucracy authority to KHL once the universal Caliphate has been
# formed and the player has the bureaucracies game-rule enabled. Disappears
# if the player forms a successor formable (uses tag, not has_or_had_tag).

nd_khl_bureaucracy_authority = {
	potential_trigger = {
		tag = KHL
		has_variable = nd_khl_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
