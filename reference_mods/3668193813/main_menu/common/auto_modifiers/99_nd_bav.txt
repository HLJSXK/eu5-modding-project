# Bavarian bureaucracy slot grant.
# Activates after BAV is formed (nd_bav_formed) so the player only gains the
# bureaucracy-system permission once the Wittelsbach state has been declared,
# and only when the bureaucracies game-rule is enabled.

nd_bav_bureaucracy_authority = {
	potential_trigger = {
		tag = BAV
has_variable = nd_bav_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
