# Grants bureaucracy authority to OYE once the Oyo Empire has been formed and
# the player has the bureaucracies game-rule enabled. Disappears if the player
# forms a successor formable (uses tag, not has_or_had_tag). Two slots from
# this auto-modifier plus one from the capital location = three effective
# slots; the player chooses three of six historical institutions.

nd_oye_bureaucracy_authority = {
	potential_trigger = {
		tag = OYE
has_variable = nd_oye_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
