# Grants bureaucracy authority to EDO once the Kingdom of Benin has been
# formed and the player has the bureaucracies game-rule enabled. Disappears
# if the player forms a successor formable (uses tag, not has_or_had_tag).
# Two slots from this auto-modifier plus one from the capital location =
# three effective slots; the player chooses three of six historical
# institutions (Uzama Nihinron, Eghaevbo n'Ogbe, Igun-Eronmwon, Igue
# Festival, Iyase, Ologbo'na).

nd_edo_bureaucracy_authority = {
	potential_trigger = {
		tag = EDO
has_variable = nd_edo_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
