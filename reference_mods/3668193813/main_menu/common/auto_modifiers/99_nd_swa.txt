# Swabian bureaucracy slot grant.
# Activates after SWA is formed (nd_swa_formed) so the player only gains the
# bureaucracy-system permission once unified Swabia has been declared, and
# only when the bureaucracies game-rule is enabled. Two slots + the capital
# slot = three effective: the player picks three of the five Swabian
# institutions.

nd_swa_bureaucracy_authority = {
	potential_trigger = {
		tag = SWA
has_variable = nd_swa_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
