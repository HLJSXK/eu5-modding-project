# Grants bureaucracy authority to TBT once the Yarlung restoration has been
# proclaimed and the player has the bureaucracies game-rule enabled.
# Disappears if the player forms a successor formable (uses tag, not has_or_had_tag).

nd_tbt_bureaucracy_authority = {
	potential_trigger = {
		tag = TBT
		has_variable = nd_tbt_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
