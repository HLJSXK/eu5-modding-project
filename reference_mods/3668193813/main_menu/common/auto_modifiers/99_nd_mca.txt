# Grants bureaucracy authority to MCA once the Muisca Confederation has been
# formed and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).

nd_mca_bureaucracy_authority = {
	potential_trigger = {
		tag = MCA
has_variable = nd_mca_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
