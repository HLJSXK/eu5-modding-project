# National Destinies -- KAZ bureaucracy authority
#
# Vanilla EU5 grants KAZ no bureaucracy slots. Once the player has
# gone through the KAZ formation event (nd_kaz.1, sets nd_kaz_formed),
# this auto-modifier unlocks the bureaucracy system and grants two
# slots. With the universal capital +1, that is three effective
# slots out of five KAZ bureaucracies -- the player picks the
# institutional shape of the steppe confederation.
nd_kaz_bureaucracy_authority = {
	potential_trigger = {
		tag = KAZ
has_variable = nd_kaz_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
