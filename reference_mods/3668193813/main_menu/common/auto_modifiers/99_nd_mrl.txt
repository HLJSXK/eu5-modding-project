# Bureaucratic authority for Maharlika
# Activates on formation, gated on the bureaucracies game rule.
# 5 bureaucracies + 1 capital slot + 2 here = 3 effective slots, pick 3 of 5.
nd_mrl_bureaucracy_authority = {
    potential_trigger = {
        tag = MRL
        has_variable = nd_mrl_formed
        has_game_rule = nd_bureaucracies_enabled
    }
    allow_bureaucracy = yes
    global_max_bureaucracy_slots = 2
}
