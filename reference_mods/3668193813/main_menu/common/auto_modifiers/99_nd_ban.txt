# Banten Sultanate bureaucracy slot grant.
# Activates after BAN is formed (nd_ban_formed) so the player only gains the
# bureaucracy-system permission once the unified Banten sultanate has been
# declared, and only when the bureaucracies game-rule is enabled. Disappears
# if the player forms a successor formable (uses tag, not has_or_had_tag).

nd_ban_bureaucracy_authority = {
	potential_trigger = {
		tag = BAN
has_variable = nd_ban_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
