# Grants bureaucracy authority to TIB once Tibet has been formed and the
# player has the bureaucracies game-rule enabled. Disappears if the player
# forms a successor formable (uses tag, not has_or_had_tag).
#
# Two slots from this auto_modifier + one capital slot = three effective.
# Player picks 3 of the 6 Ganden Phodrang bureaucracies.

nd_tib_bureaucracy_authority = {
	potential_trigger = {
		tag = TIB
has_variable = nd_tib_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
