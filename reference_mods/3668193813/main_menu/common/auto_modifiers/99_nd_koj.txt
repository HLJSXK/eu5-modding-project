# Kingdom of Jerusalem bureaucracy slot grant.
# Activates after KOJ is formed (nd_koj_formed) so the player only gains the
# bureaucracy-system permission once the unified Crusader state has been
# declared, and only when the bureaucracies game-rule is enabled.

nd_koj_bureaucracy_authority = {
	potential_trigger = {
		tag = KOJ
has_variable = nd_koj_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
