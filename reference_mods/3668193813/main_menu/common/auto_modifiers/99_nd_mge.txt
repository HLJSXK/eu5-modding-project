# National Destinies -- MGE bureaucracy authority
#
# Vanilla EU5 grants MGE no bureaucracy slots (it is a non-formed
# revival tag). Once the player has gone through the MGE formation
# event (nd_mge.1, sets nd_mge_formed), this auto-modifier unlocks
# the bureaucracy system and grants three slots. With the universal
# capital +1, that is four effective slots out of six MGE bureaucracies
# -- the player picks the institutional shape of their Mongol Empire.
nd_mge_bureaucracy_authority = {
	potential_trigger = {
		tag = MGE
has_variable = nd_mge_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
