# National Destinies -- AIR bureaucracy authority
#
# Grants 1 bureaucracy slot once AIR has been formed (nd_air_formed variable
# set by formation event nd_air.1) and the player has the bureaucracies
# game-rule enabled. Combined with the capital's universal +1 slot the player
# may keep 2 of the 4 Aïr institutions active. The choice forces the player
# to commit to either confederation legitimacy (Sultanate plus Cure Salée)
# or merchant-clerical legitimacy (Caravan Compact plus Ineslemen Tariqat),
# the two routes by which the historical Sultanate of Agadez held itself
# together against its constituent drum-chiefs.

nd_air_bureaucracy_authority = {
	potential_trigger = {
		tag = AIR
has_variable = nd_air_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
