# Ryukyu bureaucracy slot grant.
# Activates after formation (nd_ryu_formed) so the player gains the
# bureaucracy-system permission once the Sho-dynasty kingdom has been
# affirmed, and only when the bureaucracies game-rule is enabled.
# 7 bureaucracies, 3 slots granted here, +1 from capital = 4 total.

nd_ryu_bureaucracy_authority = {
	potential_trigger = {
		tag = RYU
has_variable = nd_ryu_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
