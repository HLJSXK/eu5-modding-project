# Tabor Republic bureaucracy slot grant.
# Activates after TAB is formed (nd_tab_formed) so the player only gains the
# bureaucracy-system permission once the chalice republic has been declared,
# and only when the bureaucracies game-rule is enabled. Grants 2 slots; the
# capital adds 1 = 3 total. The player picks 3 of 4 institutions.

nd_tab_bureaucracy_authority = {
	potential_trigger = {
		tag = TAB
		has_variable = nd_tab_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
