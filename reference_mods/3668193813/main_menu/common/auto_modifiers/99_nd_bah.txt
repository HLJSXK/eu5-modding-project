# Grants bureaucracy authority to BAH once the sultanate has been formed
# and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).
# 1 slot; capital adds 1 = 2 effective. Player picks 2 of 4 from 99_nd_bah.txt.

nd_bah_bureaucracy_authority = {
	potential_trigger = {
		tag = BAH
has_variable = nd_bah_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
