# National Destinies -- NPL bureaucracy authority
#
# Grants Nepal the ability to establish bureaucracies plus 2
# additional slots (the capital city already grants 1, so the
# player has 3 effective slots and chooses 3 of 5 institutions).
# Gated on the formation event having fired (nd_npl_formed) and
# on the player's nd_bureaucracies_enabled game rule.
nd_npl_bureaucracy_authority = {
	potential_trigger = {
		tag = NPL
has_variable = nd_npl_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
