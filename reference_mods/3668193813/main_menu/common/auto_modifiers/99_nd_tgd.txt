# Great Tang bureaucracy slot grant.
# Activates after formation (nd_tgd_formed) so the player only gains the
# bureaucracy-system permission once the dynasty is reconstituted, and
# only when the bureaucracies game-rule is enabled.
# Capital adds 1 implicit slot; this grants 2 more = 3 effective slots
# vs 6 institutions, forcing meaningful choice among the Tang offices.

nd_tgd_bureaucracy_authority = {
	potential_trigger = {
		tag = TGD
has_variable = nd_tgd_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
