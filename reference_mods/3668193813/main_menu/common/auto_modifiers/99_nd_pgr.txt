# Pagarruyung bureaucracy slot grant.
# Activates after formation (nd_pgr_formed) so the player only gains the
# bureaucracy-system permission once the kingdom has been declared, and
# only when the bureaucracies game-rule is enabled.

nd_pgr_bureaucracy_authority = {
	potential_trigger = {
		tag = PGR
has_variable = nd_pgr_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
