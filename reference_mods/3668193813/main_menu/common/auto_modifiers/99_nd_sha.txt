# Grants bureaucracy authority to SHA once the Shan States have been formed
# and the player has the bureaucracies game-rule enabled. Disappears if
# the player forms a successor formable (uses tag, not has_or_had_tag).

nd_sha_bureaucracy_authority = {
	potential_trigger = {
		tag = SHA
has_variable = nd_sha_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
