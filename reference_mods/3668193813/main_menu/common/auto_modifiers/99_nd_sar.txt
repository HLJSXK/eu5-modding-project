# Grants bureaucracy authority to SAR once the Kingdom of Sardinia has been
# formed (or the player started the game as SAR) and the bureaucracies game-
# rule is enabled. 2 slots + 1 from capital = 3 total, picked from 5
# historically-grounded institutions.

nd_sar_bureaucracy_authority = {
	potential_trigger = {
		tag = SAR
has_variable = nd_sar_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
