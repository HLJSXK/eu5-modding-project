# Grants bureaucracy authority to TKT once the Templar sovereign state has
# been refounded and the player has the bureaucracies game-rule enabled.
# Disappears if the player forms a successor formable (uses tag, not has_or_had_tag).

nd_tkt_bureaucracy_authority = {
	potential_trigger = {
		tag = TKT
		has_variable = nd_tkt_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
