# Grants bureaucracy authority to ROM_BYZ-formed countries (the BYZ-Italian
# Roman restoration). Vanilla BYZ already grants 2 slots via byz_bureaucracy_advance,
# and ROM_f's auto_modifier grants 3 slots if that path was taken instead.
# This auto_modifier adds 3 slots specifically when the ROM_BYZ formation
# event has fired (nd_rom_byz_formed). With the capital +1, the player has
# 4 effective slots for these 6 ROM_BYZ-specific bureaucracies (on top of
# whatever stacking from BYZ heritage applies).
#
# Uses tag = ROM rather than has_or_had_tag so slots do NOT stack across
# any future successor formable; bureaucracies themselves persist via the
# has_or_had_tag pattern in their own potential blocks.

nd_rom_byz_bureaucracy_authority = {
	potential_trigger = {
		tag = ROM
		has_variable = nd_rom_byz_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 3
}
