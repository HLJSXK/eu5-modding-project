# Franconian bureaucracy slot grant.
# Activates after FKN is formed (nd_fkn_formed) so the player only gains the
# bureaucracy-system permission once the unified Franconian state has been
# declared, and only when the bureaucracies game-rule is enabled.

nd_fkn_bureaucracy_authority = {
	potential_trigger = {
		tag = FKN
has_variable = nd_fkn_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
