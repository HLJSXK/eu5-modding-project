# National Destinies -- BYZ bureaucracy authority
#
# Vanilla EU5 grants BYZ 2 bureaucracy slots via byz_bureaucracy_advance
# (age 1, requires medieval_administration). This auto-modifier adds ONE
# further slot once the player has gone through the BYZ formation event,
# making room for the National Destinies supplemental bureaucracies in
# addition to the eleven vanilla ones.
#
# The has_variable gate fires after nd_byz.1 (formation event triggered
# either on country formation or on game-start by the on_action handler
# for game-start BYZ).
nd_byz_bureaucracy_authority = {
	potential_trigger = {
		tag = BYZ
has_variable = nd_byz_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 1
}
