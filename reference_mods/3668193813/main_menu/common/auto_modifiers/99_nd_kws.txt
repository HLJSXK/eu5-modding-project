# Kilwa Sultanate bureaucracy slot grant.
# Activates after formation (nd_kws_formed) so the player only gains the
# bureaucracy-system permission once the sultanate has been declared,
# and only when the bureaucracies game-rule is enabled.

nd_kws_bureaucracy_authority = {
	potential_trigger = {
		tag = KWS
has_variable = nd_kws_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
