# Schleswig-Holstein bureaucracy slot grant.
# Activates after formation (nd_shl_formed) so the player only gains
# the bureaucracy-system permission once the duchy has been declared,
# and only when the bureaucracies game-rule is enabled.

nd_shl_bureaucracy_authority = {
	potential_trigger = {
		tag = SHL
has_variable = nd_shl_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
