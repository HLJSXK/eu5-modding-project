# National Destinies -- SIA bureaucracy authority
#
# On formation (variable nd_sia_formed set by nd_sia.1), Siam gains
# allow_bureaucracy + 2 supplemental slots. Combined with the universal
# capital +1 slot, the player has 3 effective slots to spend across
# the six SIA bureaucracies.
nd_sia_bureaucracy_authority = {
	potential_trigger = {
		tag = SIA
has_variable = nd_sia_formed
		has_game_rule = nd_bureaucracies_enabled
	}
	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
