# and the player has the bureaucracies game-rule enabled. 2 slots + 1 from

		tag = CEL
has_variable = nd_cel_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow_bureaucracy = yes
	global_max_bureaucracy_slots = 2
}
