namespace = nd_tbt

# === FORMATION EVENT ===
# When the Tsenpo's banner rises again — the player chooses between
# the pre-Buddhist Bon imperial cult of the warrior-tsenpos
# or the Buddhist imperial synthesis of Trisong Detsen and the Samye debate.
nd_tbt.1 = {
	type = country_event
	title = nd_tbt.1.t
	desc = nd_tbt.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_tbt.1.a
		# The Bon-restorationist path: the player's tsenpo throws over
		# Trisong Detsen's Buddhist synthesis and restores the pre-imperial
		# Yungdrung Bon cult as state religion. This is the alt-history
		# "what if Langdarma had won" -- the Buddhist persecution of 838-842
		# made permanent. Mechanical conversion to bon makes the choice
		# substantive rather than purely flavour.
		change_religion = religion:bon
		add_country_modifier = {
			modifier = nd_tbt_yungdrung_imperial_cult
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tbt_formed
			set_variable = nd_tbt_path_a
		}
	}
	option = {
		name = nd_tbt.1.b
		add_country_modifier = {
			modifier = nd_tbt_dharma_kings_synthesis
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tbt_formed
			set_variable = nd_tbt_path_b
		}
	}
}
