# === Formation Event: Council of Mosul ===
# Beth Nahrain has just been declared. The patriarchate's seat is restored,
# the Tur Abdin monasteries swear fealty, the Crusader-era libraries of
# Edessa are inventoried. The new state must choose its founding character:
# a sacred patriarchate-state, a militarized monastic frontier, or a
# Latin-engaged cosmopolitan hybrid.
nd_bnh.1 = {
	type = country_event
	title = nd_bnh.1.t
	desc = nd_bnh.1.d
	outcome = neutral
	fire_only_once = yes

	# Path A: The Patriarch's Restoration
	option = {
		name = nd_bnh.1.a
		add_country_modifier = {
			modifier = nd_bnh_patriarchs_restoration
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_bnh_formed
			set_variable = nd_bnh_path_a
		}
	}

	# Path B: The Tur Abdin Defenders
	option = {
		name = nd_bnh.1.b
		add_country_modifier = {
			modifier = nd_bnh_tur_abdin_defenders
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_bnh_formed
			set_variable = nd_bnh_path_b
		}
	}

	# Path C: The Edessene Renaissance
	option = {
		name = nd_bnh.1.c
		add_country_modifier = {
			modifier = nd_bnh_edessene_renaissance
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_bnh_formed
			set_variable = nd_bnh_path_c
		}
	}
}
