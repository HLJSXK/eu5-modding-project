# ============================================================
# Formation event — Wallmapu Endurance vs Pillan's Spear
# ============================================================

nd_map.1 = {
	type = country_event
	title = nd_map.1.t
	desc = nd_map.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_map.1.a
		add_country_modifier = {
			modifier = nd_map_wallmapu_endurance
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_map_formed
			set_variable = nd_map_path_a
		}
	}
	option = {
		name = nd_map.1.b
		add_country_modifier = {
			modifier = nd_map_pillan_spear
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_map_formed
			set_variable = nd_map_path_b
		}
	}
}
