# ==========================================================================
# nd_pon.1 -- Formation event for the Pontic restoration.
# Two paths reflecting the central tension of any Pontic-Greek polity:
#   A) Komnenian Restoration -- the Imperial Komnenoi-of-Trebizond model:
#      court ceremonial, late-Byzantine purple, Sumela as imperial chapel,
#      aristocracy and silver-mining oligarchs. Centralised, Imperial.
#   B) Pontic Republic -- the urban-mercantile commonwealth of Sinope,
#      Trebizond, and Kerasounta as constitutional partners, an
#      autocephalous Pontic Orthodox Church as political pole. Mercantile,
#      conciliar, decentralised.
# ==========================================================================
nd_pon.1 = {
	type = country_event
	title = nd_pon.1.t
	desc = nd_pon.1.d
	outcome = neutral
	fire_only_once = yes

	# OPTION A: Komnenian Restoration
	option = {
		name = nd_pon.1.a
		add_country_modifier = {
			modifier = nd_pon_komnenian_restoration
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pon_formed
			set_variable = nd_pon_path_a
		}
	}

	# OPTION B: Pontic Republic
	option = {
		name = nd_pon.1.b
		add_country_modifier = {
			modifier = nd_pon_pontic_republic
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pon_formed
			set_variable = nd_pon_path_b
		}
	}
}
