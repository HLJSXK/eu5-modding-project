# Formation event — Continental Empire vs Sovereign Nation
# ============================================================

nd_num.1 = {
	type = country_event
	title = nd_num.1.t
	desc = nd_num.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_num.1.a
		add_country_modifier = {
			modifier = nd_num_continental_empire
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = offensive_vs_defensive
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_num_formed
			set_variable = nd_num_path_a
		}
	}
	option = {
		name = nd_num.1.b
		add_country_modifier = {
			modifier = nd_num_sovereign_nation
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_num_formed
			set_variable = nd_num_path_b
		}
	}
}
