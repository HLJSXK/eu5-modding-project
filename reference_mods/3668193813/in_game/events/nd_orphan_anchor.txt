namespace = nd_orphan_anchor

# 1.2 added a stricter check that flags variables set but only read in
# localization tooltips (custom_tooltip text, [Concept('GetVariable', ...)]).
# This anchor event reads each tooltip-only variable in a real trigger
# context so the engine's reachability check sees them as used. The event
# never fires (impossible always-false trigger), so it has no gameplay effect.
nd_orphan_anchor.1 = {
	type = country_event
	hidden = yes
	fire_only_once = yes
	orphan = yes

	# Always-false outer guard so the event never fires, but the engine still
	# parses the trigger block and counts every nested has_variable read as
	# real script usage of that variable. `always = no` is the canonical
	# Paradox idiom for an unconditional false trigger.
	trigger = {
		always = no
		OR = {
			has_variable = nd_ayu_formed
			has_variable = nd_ayu_path_a
			has_variable = nd_ayu_path_b
			has_variable = nd_bah_formed

			has_variable = nd_bah_path_b

			has_variable = nd_mlc_path_a
			has_variable = nd_mlc_path_b
			has_variable = nd_mtm_formed
			has_variable = nd_mtm_path_a
			has_variable = nd_mtm_path_b
			has_variable = nd_pgr_formed
			has_variable = nd_pgr_path_a
			has_variable = nd_pgr_path_b

	}

		name = nd_orphan_anchor.1.a
	}
}
