# Formation event: Suvarnabhumi -- the Land of Gold rises again under one dharma-wheel
nd_svb.1 = {
	type = country_event
	title = nd_svb.1.t
	desc = nd_svb.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_svb.1.a
		add_country_modifier = {
			modifier = nd_svb_cakkavatti
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_svb_formed
			set_variable = nd_svb_path_a
		}
	}
	option = {
		name = nd_svb.1.b
		add_country_modifier = {
			modifier = nd_svb_sangha_federation
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_svb_formed
			set_variable = nd_svb_path_b
		}
	}
	option = {
		name = nd_svb.1.c
		add_country_modifier = {
			modifier = nd_svb_three_worlds
			years = 40
			mode = add_and_extend
		}

			type = traditionalist_vs_innovative

		}
		hidden_effect = {
			set_variable = nd_formed
