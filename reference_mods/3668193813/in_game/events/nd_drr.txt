# ===== FORMATION EVENT =====
nd_drr.1 = {
	type = country_event
	title = nd_drr.1.t
	desc = nd_drr.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_drr.1.a
		add_country_modifier = {
			modifier = nd_drr_loya_jirga
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_drr_formed
			set_variable = nd_drr_path_a
		}
	}
	option = {
		name = nd_drr.1.b
		add_country_modifier = {
			modifier = nd_drr_sword_of_khorasan
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_drr_formed
			set_variable = nd_drr_path_b
		}
	}
}
