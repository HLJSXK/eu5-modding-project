# Aragonese Empire (CDA) — Formation event + advance DHE events
# National Destinies mod
namespace = nd_cda

# ============================================================================
# Formation Event
# ============================================================================
# Fires from form_effect via trigger_event_non_silently. The choice represents
# Alfonso V the Magnanimous's real strategic fork in 1442: having seized
# Naples after a 1435-1442 war, would the Aragonese crown anchor itself in
# the Mediterranean as Catalan-Italian thalassocracy, or consolidate westward
# as the dominant Iberian crown that absorbs rather than is absorbed?
nd_cda.1 = {
	type = country_event
	title = nd_cda.1.t
	desc = nd_cda.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_cda.1.a
		add_country_modifier = {
			modifier = nd_cda_mediterraneum_aragonense
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_large_move_to_right
		}
		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cda_formed
			set_variable = nd_cda_path_a
		}
	}

	option = {
		name = nd_cda.1.b
		add_country_modifier = {
			modifier = nd_cda_hispano_aragonese
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_cda_formed
			set_variable = nd_cda_path_b
		}
	}
}
