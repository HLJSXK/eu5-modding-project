namespace = nd_tkt

# ============================================================
# Knights of the Temple (TKT) - Formation event + DHE events
# Alt-history Iberian-Atlantic Templar successor that refused
# the dissolution of 1312. The Tomar convent is preserved with
# the original eight-pointed cross rather than transferred to
# the Order of Christ; the Iberian banking houses, the secret
# chapter, and the brotherhood's heretical reputation pass
# from suppressed Order to sovereign state.
# ============================================================

# nd_tkt.1 - Formation event
# The papal bull Vox in excelso is lifted; the Pauperes
# Commilitones Christi are refounded as a sovereign Iberian
# realm. The brothers must choose what kind of state to be.
nd_tkt.1 = {
	type = country_event
	title = nd_tkt.1.t
	desc = nd_tkt.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_tkt.1.a
		add_country_modifier = {
			modifier = nd_tkt_old_order_restored
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tkt_formed
			set_variable = nd_tkt_path_a
		}
	}
	option = {
		name = nd_tkt.1.b
		add_country_modifier = {
			modifier = nd_tkt_order_of_christ
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tkt_formed
			set_variable = nd_tkt_path_b
		}
	}
}
