namespace = nd_tdr

# ========================================================================
# nd_tdr.1 -- Formation event: choose between Roman/Greek "Regnum Bospori"
# and Crimean Gothic "Kingdom of the Cimmerian Bosporus" naming.
# Each path renames the country (change_country_name) and grants a thematic
# 25-year timed modifier reflecting the chosen heritage.
# ========================================================================
nd_tdr.1 = {
	type = country_event
	title = nd_tdr.1.t
	desc = nd_tdr.1.d
	outcome = neutral
	fire_only_once = yes

	# OPTION A: Regnum Bospori -- the Hellenic-Roman revival
	option = {
		name = nd_tdr.1.a
		change_country_name = TDR_REGNUM
		add_country_modifier = {
			modifier = nd_tdr_regnum_bospori
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tdr_formed
			set_variable = nd_tdr_path_a
		}
	}

	# OPTION B: Kingdom of the Cimmerian Bosporus -- Crimean Gothic
	option = {
		name = nd_tdr.1.b
		change_country_name = TDR_CIMMERIAN
		add_country_modifier = {
			modifier = nd_tdr_cimmerian_bosporus
			years = 25
			mode = add_and_extend
		}
		change_societal_value = {
			type = quality_vs_quantity
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tdr_formed
			set_variable = nd_tdr_path_b
		}
	}
}
