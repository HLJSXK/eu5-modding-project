# Formation event: Maharlika
nd_mrl.1 = {
	type = country_event
	title = nd_mrl.1.t
	desc = nd_mrl.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_mrl.1.a
		add_country_modifier = {
			modifier = nd_mrl_tagalog_datu_court
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mrl_formed
			set_variable = nd_mrl_path_a
		}
	}
	option = {
		name = nd_mrl.1.b
		add_country_modifier = {
			modifier = nd_mrl_pan_filipino_federation
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mrl_formed
			set_variable = nd_mrl_path_b
		}
	}
	option = {
		name = nd_mrl.1.c
		add_country_modifier = {
			modifier = nd_mrl_kris_and_bagani
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mrl_formed
			set_variable = nd_mrl_path_c
		}
	}
}
