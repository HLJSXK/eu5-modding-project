# === Formation Event: the Burgundian crown is placed ===
nd_gbu.1 = {
	type = country_event
	title = nd_gbu.1.t
	desc = nd_gbu.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_gbu.1.a
		add_country_modifier = {
			modifier = nd_gbu_frankish_imperial_kingdom
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_gbu_formed
			set_variable = nd_gbu_path_a
		}
	}
	option = {
		name = nd_gbu.1.b
		add_country_modifier = {
			modifier = nd_gbu_estates_kingdom
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_gbu_formed
			set_variable = nd_gbu_path_b
		}
	}
}
