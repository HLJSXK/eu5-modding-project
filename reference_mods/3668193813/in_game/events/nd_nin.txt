# === Formation Event: The Crowning at Nineveh ===
# The crown of Aššur is placed upon the formed empire's emperor at the
# restored Sargonid capital. The Patriarchate of the East, the Tur Abdin
# monasteries, and the surviving Aramaic civic council stand witness.
# The empire must declare its founding doctrine: the warrior-state of
# Sargon and Esarhaddon, the administrative-imperial Pax Assyriaca, or
# the civilizational synthesis where ancient stelae and the Christian
# cross stand together.
nd_nin.1 = {
	type = country_event
	title = nd_nin.1.t
	desc = nd_nin.1.d
	outcome = neutral
	fire_only_once = yes

	# Path A: Sargonid Imperium -- the warrior-emperor template
	option = {
		name = nd_nin.1.a
		add_country_modifier = {
			modifier = nd_nin_sargonid_imperium
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_nin_formed
			set_variable = nd_nin_path_a
		}
	}

	# Path B: Pax Assyriaca -- the imperial-administrative template
	option = {
		name = nd_nin.1.b
		add_country_modifier = {
			modifier = nd_nin_pax_assyriaca
			years = 40
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_nin_formed
			set_variable = nd_nin_path_b
		}
	}

	# Path C: Civilizational Synthesis -- the cross-and-Lamassu fusion
	option = {
		name = nd_nin.1.c
		add_country_modifier = {
			modifier = nd_nin_civilizational_synthesis

			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed

			set_variable = nd_nin_path_c
		}
	}
