namespace = nd_rom_byz

# ===== FORMATION EVENT =====
# ROM_BYZ -- the Italian Roman Restoration. Vanilla form_effect already
# pushed latinization_vs_hellenization hard left. The choice is now: do we
# accept that drift and embrace a Latin imperial restoration, or do we
# resist it and synthesize Greek and Italian into a hybrid Romaiosyne?
nd_rom_byz.1 = {
	type = country_event
	title = nd_rom_byz.1.t
	desc = nd_rom_byz.1.d
	outcome = neutral
	fire_only_once = yes

	# Path A : Romanitas Vincit -- accept latinization, restore the Latin imperium
	# Papal-imperial reconciliation, return of Latin liturgy, the West reclaimed.
	option = {
		name = nd_rom_byz.1.a
		add_country_modifier = {
			modifier = nd_rom_byz_romanitas_vincit
			years = 50
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_rom_byz_formed
			set_variable = nd_rom_byz_path_a
		}
	}
	# Path B : Hellenikos Romaios -- resist full latinization, fuse Greek and Italian
	# Greek liturgy in Latin churches, Eastern court ceremony in Rome, Plethon's heirs.
	option = {
		name = nd_rom_byz.1.b
		add_country_modifier = {
			modifier = nd_rom_byz_hellenikos_romaios
			years = 50
			mode = add_and_extend
		}
		# Counter half of vanilla's huge-left latinization push, locking in synthesis
		change_societal_value = { type = latinization_vs_hellenization value = societal_value_large_move_to_right }
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_rom_byz_formed
			set_variable = nd_rom_byz_path_b
		}
	}
}
