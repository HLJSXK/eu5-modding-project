# ============================================================================
# Formation Event: The Mount of Tabor
# ============================================================================

nd_tab.1 = {
	type = country_event
	title = nd_tab.1.t
	desc = nd_tab.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_tab.1.a
		# Tabor Commune -- peasant-egalitarian millenarian republic.
		# Communal property, married priesthood, women combatants, full Sirotci radicalism.
		add_country_modifier = {
			modifier = nd_tab_commune_of_brethren
			years = 20
			mode = add_and_extend
		}
		change_government_type = government_type:republic
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_large_move_to_right
		}
		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tab_formed
			set_variable = nd_tab_path_a
		}
	}

	option = {
		name = nd_tab.1.b
		# Calixtine Realm -- Utraquist-radical noble-led Hussite republic.
		# Prokop the Great's institutional vision: chalice for all, but Czech
		# nobles lead and the war-wagon army stays disciplined.
		add_country_modifier = {
			modifier = nd_tab_calixtine_compact
			years = 20
			mode = add_and_extend
		}
		change_government_type = government_type:republic
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_move_to_right
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_tab_formed
			set_variable = nd_tab_path_b
		}
	}
}
