# Formation event : fires from form_effect (trigger_event_non_silently)
# ============================================================

# nd_hos.1 - Formation event
nd_hos.1 = {
	type = country_event
	title = nd_hos.1.t
	desc = nd_hos.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_hos.1.a
		add_country_modifier = {
			modifier = nd_hos_crusading_order
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_hos_formed
			set_variable = nd_hos_path_a
		}
	}
	option = {
		name = nd_hos.1.b
		add_country_modifier = {
			modifier = nd_hos_hospitalier_realm
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_large_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_hos_formed
			set_variable = nd_hos_path_b
		}
	}
}
