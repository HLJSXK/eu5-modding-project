# Formation event: Three founding visions for the unified Moro polity
nd_mro.1 = {
	type = country_event
	title = nd_mro.1.t
	desc = nd_mro.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_mro.1.a
		add_country_modifier = {
			modifier = nd_mro_sultan_caliph
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}
		change_societal_value = {
			type = mysticism_vs_jurisprudence
			value = societal_value_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mro_formed
			set_variable = nd_mro_path_a
		}
	}
	option = {
		name = nd_mro.1.b
		add_country_modifier = {
			modifier = nd_mro_datu_confederation
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mro_formed
			set_variable = nd_mro_path_b
		}
	}
	option = {
		name = nd_mro.1.c
		add_country_modifier = {
			modifier = nd_mro_sword_and_sail
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_large_move_to_right
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_mro_formed
			set_variable = nd_mro_path_c
		}
	}
}
