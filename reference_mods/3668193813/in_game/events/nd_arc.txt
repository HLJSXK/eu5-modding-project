namespace = nd_arc

# === Formation Event: The Council of Sis ===
# Hethumid Cilicia at its moment of survival: the church question forced upon Sis
# is the same one the historical Council of Sis (1307) attempted to answer --
# remain Apostolic and miaphysite, or pursue formal Latin union with Rome to
# secure Crusader and Italian aid. Both were existential strategies.
nd_arc.1 = {
	type = country_event
	title = nd_arc.1.t
	desc = nd_arc.1.d
	outcome = neutral
	fire_only_once = yes

	# Path A: Apostolic Sanctuary -- preserve the national church
	option = {
		name = nd_arc.1.a
		add_country_modifier = {
			modifier = nd_arc_apostolic_sanctuary
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_arc_formed
			set_variable = nd_arc_path_a
		}
	}
	# Path B: Latin Union of Sis -- formal union with Rome, Crusader court
	option = {
		name = nd_arc.1.b
		add_country_modifier = {
			modifier = nd_arc_latin_union
			years = 30
			mode = add_and_extend
		}
		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_arc_formed
			set_variable = nd_arc_path_b
		}
	}
}
