# ===== FORMATION EVENT =====
# You have just walked Austria, Bohemia and Hungary out of the Reich and
# surrendered the imperial crown. The Emperor's electors and the Pope are
# not pleased -- but that reckoning is a separate Situation. Here the
# question is what kind of empire the Danube state will be: the sword of
# Christendom on the Ottoman line, or a single obedient Gesamtmonarchie.
nd_dnm.1 = {
	type = country_event
	title = nd_dnm.1.t
	desc = nd_dnm.1.d
	outcome = neutral
	fire_only_once = yes

	# Path A : Antemurale Christianitatis -- the bulwark of Christendom.
	# The Militaergrenze, the Grenzer soldier-colonists, the perpetual
	# Ottoman war. The Danube line is the empire's reason to exist.
	option = {
		name = nd_dnm.1.a
		add_country_modifier = {
			modifier = nd_dnm_antemurale_christianitatis
			years = 45
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_dnm_formed
			set_variable = nd_dnm_path_a
			# Grace: 10-year window before disasters can fire (was 25).
			set_variable = { name = nd_dnm_grace value = yes years = 10 }
			# 50-year backup escape timer for the Reich situation; the
			# situation's can_end checks for this variable's absence
			# together with legitimacy/prestige thresholds.
			set_variable = { name = nd_dnm_pariah_timer value = yes years = 50 }
			# Succession Question chain: 3 narrative events that fire at
			# fixed intervals via timed-variable expiry. Each `_pending` var
			# decays at the year shown; an on_country_yearly_pulse handler
			# then fires the matching event and sets a `_fired` guard.
			set_variable = { name = nd_dnm_chain_30_pending value = yes years = 2 }
			set_variable = { name = nd_dnm_chain_31_pending value = yes years = 5 }
			set_variable = { name = nd_dnm_chain_32_pending value = yes years = 8 }
			activate_situation = situation:nd_dnm_reaction_of_the_reich
		}
	}
	# Path B : Gesamtmonarchie -- the total monarchy. The Hofkammer, the
	# Hofkriegsrat, the Pragmatic Sanction: three crowns welded into one
	# administrative body answerable to the Vienna court alone.
	option = {
		name = nd_dnm.1.b
		add_country_modifier = {
			modifier = nd_dnm_gesamtmonarchie
			years = 45
			mode = add_and_extend
		}
		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_large_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_dnm_formed
			set_variable = nd_dnm_path_b
			set_variable = { name = nd_dnm_grace value = yes years = 10 }
			set_variable = { name = nd_dnm_pariah_timer value = yes years = 50 }
			set_variable = { name = nd_dnm_chain_30_pending value = yes years = 2 }
			set_variable = { name = nd_dnm_chain_31_pending value = yes years = 5 }
			set_variable = { name = nd_dnm_chain_32_pending value = yes years = 8 }
			activate_situation = situation:nd_dnm_reaction_of_the_reich
		}
	}
}

nd_dnm.19 = {                                   # disaster on_end: fragmentation
	# Real territory loss, not just a flat debuff. Models a constituent
	# crown actually breaking away as a sovereign kingdom: try Hungarian
	# transdanubia first, fall back to Bohemia if Hungary is already gone,
	# fall back to the modifier alone if neither is owned. Pattern follows
	# the vanilla flavor_HAB Galicia-Lodomeria release (create_country_from_
	# location + every_owned_location transfer with core + integration).
	type = country_event
	title = nd_dnm.19.t
	desc = nd_dnm.19.d
	outcome = negative
	fire_only_once = yes
	option = {
		name = nd_dnm.19.a
		hidden_effect = {
			if = {
				limit = {
					any_owned_location = { area = area:transdanubia_area }
				}
				random_owned_location = {
					limit = { area = area:transdanubia_area }
					save_scope_as = nd_dnm_breakaway_location
				}
				scope:nd_dnm_breakaway_location = {
					create_country_from_location = {
						save_scope_as = nd_dnm_breakaway_country
						set_country_rank = country_rank:rank_kingdom
					}
				}
				every_owned_location = {
					limit = { area = area:transdanubia_area }
					change_location_owner = scope:nd_dnm_breakaway_country
					add_core = scope:nd_dnm_breakaway_country
					change_integration_level = core
				}
			}
			else_if = {
				limit = {
					any_owned_location = { area = area:bohemia_area }
				}
				random_owned_location = {
					limit = { area = area:bohemia_area }
					save_scope_as = nd_dnm_breakaway_location
				}
				scope:nd_dnm_breakaway_location = {
					create_country_from_location = {
						save_scope_as = nd_dnm_breakaway_country
						set_country_rank = country_rank:rank_kingdom
					}
				}
				every_owned_location = {
					limit = { area = area:bohemia_area }
					change_location_owner = scope:nd_dnm_breakaway_country
					add_core = scope:nd_dnm_breakaway_country
					change_integration_level = core
				}
			}
		}
		add_stability = stability_severe_penalty
		add_country_modifier = {
			modifier = nd_dnm_fractured_realm
			years = 30
			mode = add_and_extend
		}
		hidden_effect = { remove_variable = nd_dnm_nq_resolved }
	}
}

# ==========================================================================
# THE SUCCESSION QUESTION CHAIN (nd_dnm.30 / .31 / .32)
# A 3-event narrative arc fired by the Reich situation's on_monthly when
# the corresponding _pending timer expires (set in nd_dnm.1 with years =
# 2 / 5 / 8). Each event sets country flags that feed the Pragmatic
# Sanction disaster's counter on disaster start: a player who fully prepped
# enters the disaster with up to +40 of the 80-target Crown Confidence.
# The chain runs DURING grace; the disaster fires shortly AFTER grace
# (year ~10).
#
# Why country variables (not globals): if DNM is conquered and re-formed by
# a new player, country variables die with the country. Globals would
# carry over as a free head start they didn't earn.
# ==========================================================================

# nd_dnm.30 -- The Question of Inheritance.
# Year 2 post-formation, no choice. Sets nd_dnm_succession_step = 1 and
# grants the perpetual nd_dnm_succession_anxiety modifier (a quiet drag on
# legitimacy + nobles satisfaction). This is the "the realm has not yet
# answered" beat -- it tells the player the chain has begun and the
# disaster is on its way.
nd_dnm.30 = {
	type = country_event
	title = nd_dnm.30.t
	desc = nd_dnm.30.d
	outcome = negative
	fire_only_once = yes
	option = {
		name = nd_dnm.30.a
		add_country_modifier = {
			modifier = nd_dnm_succession_anxiety
			years = -1
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = { name = nd_dnm_succession_step value = 1 }
		}
	}
}

# nd_dnm.31 -- The Diets Murmur. Year 5 post-formation, 2 choices.
# Removes the previous chain modifier (anxiety) so the chain is one
# narrative thread, not a stack. Sets nd_dnm_succession_step = 2.
# Choice A (rally) sets the country flag nd_dnm_succession_groundwork
# which seeds the disaster counter with +15 and is read by nd_dnm.24's
# Upheld trigger as a third OR-clause.
nd_dnm.31 = {
	type = country_event

	desc = nd_dnm.31.d
	outcome = neutral
	fire_only_once = yes

	}
	# A : Rally the estates around the dynasty.
	option = {
		name = nd_dnm.31.a
		add_country_modifier = {
			modifier = nd_dnm_rallying_the_diets
			years = -1
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = { name = nd_dnm_succession_step value = 2 }
			set_variable = nd_dnm_succession_groundwork
		}
	}
	# B : The court will not be hurried.
	option = {
		name = nd_dnm.31.b
		add_country_modifier = {
			modifier = nd_dnm_royal_silence
			years = -1
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = { name = nd_dnm_succession_step value = 2 }
		}
	}
}
