namespace = nd_alm

# Formation Event: fires from form_effect
nd_alm.1 = {
	type = country_event
	title = nd_alm.1.t
	desc = nd_alm.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_alm.1.a
		add_country_modifier = {
			modifier = nd_alm_mahdist_revolution
			years = 30
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_alm_formed
			set_variable = nd_alm_path_a
		}
	}
	option = {
		name = nd_alm.1.b
		add_country_modifier = {
			modifier = nd_alm_caliphal_civilization
			years = 30
			mode = add_and_extend
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_alm_formed
			set_variable = nd_alm_path_b
		}
	}
}
