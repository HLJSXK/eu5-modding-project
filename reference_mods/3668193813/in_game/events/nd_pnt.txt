# Formation Event: The Painted Banner Raised
# ============================================================
nd_pnt.1 = {
	type = country_event
	title = nd_pnt.1.t
	desc = nd_pnt.1.d
	outcome = neutral
	fire_only_once = yes

	option = {
		name = nd_pnt.1.a
		# Federation of the Datus -- governance, cohesion, council legitimacy
		add_country_modifier = {
			modifier = nd_pnt_federation_of_datus
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = aristocracy_vs_plutocracy
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pnt_formed
			set_variable = nd_pnt_path_a
		}
	}
	option = {
		name = nd_pnt.1.b
		# Kakarlong Brotherhood -- mangayaw raiders, tattoo warrior code
		add_country_modifier = {
			modifier = nd_pnt_kakarlong_brotherhood
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_minor_move_to_left
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pnt_formed
			set_variable = nd_pnt_path_b
		}
	}
	option = {
		name = nd_pnt.1.c
		# Maritime Confederation -- balangay merchants, inner-sea trade
		add_country_modifier = {
			modifier = nd_pnt_maritime_confederation
			years = 20
			mode = add_and_extend
		}
		change_societal_value = {
			type = land_vs_naval
			value = societal_value_minor_move_to_right
		}
		hidden_effect = {
			set_variable = nd_formed
			set_variable = nd_pnt_formed
			set_variable = nd_pnt_path_c
		}
	}
}
