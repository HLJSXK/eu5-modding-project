# ==========================================================================
# DANUBIAN MONARCHY -- the Strength-path peace treaty.
#
# Replaces the fragile on_monthly war_score_versus poll in the Reaction of
# the Reich situation. Instead of the engine silently checking 80% war score
# every month (which depended on a brittle save_temporary_scope_as + nested
# is_emperor lookup), the player now demands an explicit peace term against
# the Emperor. The cost = 80 IS the "80% war score" requirement: peace treaty
# cost is measured directly in war-score points (see peace_treaties/readme.txt
# "cost = script value for war score cost").
#
# When DNM enforces this term, scope:winner (DNM) gets nd_dnm_reich_humbled
# and the strength-path victory event nd_dnm.33; the Reich situation's can_end
# then closes the situation on its next tick.
# ==========================================================================
peace_nd_dnm_humble_reich = {

	# 80 war-score points to enforce -- the "humble the Reich to 80%" gate.
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 80
		}
	}

	# Only available while the Reaction of the Reich is active on DNM
	# (pariah modifier present) and the loser is the sitting Emperor.
	# scope:winner = the side imposing terms (DNM, whether it declared the
	# war or is winning as defender against the imperial-ban CB).
	potential = {
		scope:winner = {
			tag = DNM
			has_country_modifier = nd_dnm_reich_pariah
			NOT = { has_variable = nd_dnm_reich_humbled }
		}
		scope:loser = {
			is_emperor = yes
		}
	}

	allow = {
		always = yes
	}

	effect = {
		scope:winner = {
			hidden_effect = { set_variable = nd_dnm_reich_humbled }
			trigger_event_non_silently = nd_dnm.33
		}
		scope:loser = {
			add_prestige = prestige_extreme_penalty
		}
	}

	# The Emperor strongly resists conceding the imperial verdict; the AI
	# should only accept under heavy war-score pressure.
	ai_desire = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = -100
		}
	}
}
