# ===== PONTIC CURING WORKS =====
# Coastal curing yards and grain distilleries on the Tauric shore.
# The Genoese factor at Caffa wired the Pontic peninsula into the
# Mediterranean luxury and victualling trades: salted Black Sea
# sturgeon roe shipped to Florence and Venice, and grain-mash
# spirits brewed for the steppe trade. This is the production-side
# complement to the trade-side Tauric Emporion -- a coastal workshop
# rather than a customs house.
#
# PM 1: Pontic Caviar Works -- gated on nd_tdr_caffa_factor (age 2).
# PM 2: Pontic Grain Spirits -- gated on nd_tdr_pontic_revival (age 5).
nd_tdr_pontic_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = laborers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = guild_build_time

	country_potential = {
		OR = {
			has_advance = nd_tdr_caffa_factor
			has_advance = nd_tdr_pontic_revival
		}
	}

	location_potential = {
		is_coastal = yes
	}

	modifier = {
		local_burghers_estate_power = 0.3
		local_burghers_pop_growth = 0.001
		local_trades_per_burgher = 0.05
	}

	unique_production_methods = {
		nd_tdr_caviar_works = {
			potential = { has_advance = nd_tdr_caffa_factor }
			fish = 0.3
			salt = 0.1
			produced = jewelry
			output = 0.1679
			category = guild_input
			debug_max_profit = guild_profit_margin

		nd_tdr_grain_spirits = {
			potential = { has_advance = nd_tdr_pontic_revival }
			wheat = 0.3
			lumber = 0.1
			produced = liquor
			output = 0.216
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
	}
	construction_demand = guild_construction
}
