# Minor governor buildings ported from the eu5fixes mod.
# Weaker alternative to vanilla local_governor / naval_governor
# (proximity 50 vs 80) used by National Destinies bureaucracies in
# place of num_local_governors. The capital-region requirement of the
# original eu5fixes provincial_estate has been removed by design.

nd_provincial_estate = {
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes

	employment_size = large_noble_employment

	category = government_category

	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes

	allow = {
		custom_tooltip = {
			text = nd_provincial_estate_tt
			scope:actor = { total_building_levels_including_construction:nd_provincial_estate < modifier:num_nd_provincial_estates }
		}
		owner = scope:actor
	}

	location_potential = {
		has_road_to_capital = yes
		owner.capital != this
	}

	remove_if = {
		location = { has_road_to_capital = no }
	}

	expensive = yes

	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes

	build_time = government_build_time

	unique_production_methods = {
		nd_provincial_estate_maintenance = {
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 3
			books = 1
			goods_gold = 1
			furniture = 2
			fine_cloth = 1

			category = building_maintenance
		}
	}

	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 50
		local_nobles_estate_power = 0.33
	}
}

nd_port_commandery = {
	is_foreign = no

			goods_gold = 1
			furniture = 2
			naval_supplies = 0.5
			fine_cloth = 0.5

			category = building_maintenance
		}
	}

	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 50

	}
