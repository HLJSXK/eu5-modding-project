# Cilician Workshop -- production-side complement to Mopsuestia.
# Italian factors at the Pyramus warehouses processed Cilician iron and
# Taurus timber into shipyard-grade naval supplies and forwarded Silk Road
# raw silk to Genoese-licensed weavers turning out fine cloth for Italian
# markets. This is the workshop where the export trade becomes a production
# trade -- a consumer-goods manufactory rather than a customs entrepot.
#
# PM 1: Iron and Timber Export -- gated on nd_arc_pyramus_warehouses (age 2).
# PM 2: Silk Caravan Trade -- gated on nd_arc_silk_road_terminus (age 3).
nd_arc_cilician_workshop = {
	is_special = yes
	is_foreign = no
	pop_type = burghers
	max_levels = 1
	employment_size = guild_employment
	category = consumer_goods_category
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = guild_build_time

	country_potential = {
		OR = {
			has_advance = nd_arc_pyramus_warehouses
			has_advance = nd_arc_silk_road_terminus
		}
	}

	modifier = {
		local_burghers_estate_power = 0.3
		local_burghers_pop_growth = 0.001
		local_trades_per_burgher = 0.05
	}

	unique_production_methods = {
		nd_arc_export_pm = {
			potential = { has_advance = nd_arc_pyramus_warehouses }
			iron = 0.2
			lumber = 0.2
			produced = naval_supplies
			output = 0.36
			category = guild_input
			debug_max_profit = guild_profit_margin
		}
		nd_arc_silk_caravan_pm = {
