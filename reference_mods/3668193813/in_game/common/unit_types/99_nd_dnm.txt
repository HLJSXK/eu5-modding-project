# ===== DANUBIAN MONARCHY -- Custom Units =====
# All are levy units (buildable = no, levy = yes): they are raised through
# the levy system and unlocked via unlock_levy on the advance tree, matching
# the Habsburg reality of estate- and frontier-raised troops rather than
# crown-recruited regiments.

# Hungarian Huszar -- the light horse of the Hungarian crown. Fast,
# far-ranging raiders born of the Ottoman border war; copied by every
# army in Europe by 1700. (Age 2)
a_nd_dnm_huszar_levy = {
	is_special = yes
	category = army_light_cavalry
	copy_from = a_age_2_renaissance_light_cavalry
	upgrades_to = a_light_lancers
	buildable = no
	levy = yes

	# Base age 2 light cavalry: max_strength=0.4, combat_power=4
	movement_speed = 0.15
	combat_speed = 0.1
	attrition_loss = -0.1

	country_potential = {
		has_or_had_tag = DNM
	}

	gfx_tags = { medium_tag }
}

# Tschaike -- the Danube war flotilla galley (Tschaikisten). Shallow-draft
# river craft that fought the Ottoman fleet from Komorn to Belgrade. Fast
# and well-gunned for its size; the shallow draft (a smaller hull than
# the seagoing galley) is its identity and its tradeoff. (Age 3)
n_nd_dnm_tschaike_levy = {
	is_special = yes
	category = navy_galley
	copy_from = n_age_3_discovery_galley
	upgrades_to = n_galleass
	buildable = no
	levy = yes

	# Base age 3 galley: hull_size=8, cannons=12, movement_speed=4
	# Shallow draft is the tradeoff: a smaller hull than the base galley
	# in exchange for river speed and firepower.
	hull_size = 7
	cannons = 14
	movement_speed = 5
	blockade_capacity = 0.4

	country_potential = {
		has_or_had_tag = DNM
	}

	gfx_tags = { mediterranean_galley_tag }
}
