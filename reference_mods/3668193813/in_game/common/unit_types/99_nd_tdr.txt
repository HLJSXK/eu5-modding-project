# Bosporan Kingdom (TDR) -- custom units
# Mangup Drungaroi: heavy infantry levy drawn from the cliff garrisons of
#   Theodoro. Defensive-leaning, attrition-resistant -- thrived in the
#   Tauric uplands and rock-cut casemates.
# Tauric Stratiotes: late-Komnenian light cavalry recruited from Pontic
#   coast Greeks and Goth horsemen. Skirmishers, scouts, raiders.
# Bosporan Liburnian: small-but-fast galley patterned on the late Roman
#   Black Sea liburnae used for grain convoys and coastal patrols.
# Pontic Galley: age-4 buildable war galley descended from the Pontic
#   liburnae crews -- heavier than the liburnian, faster than vanilla.
# ==========================================================================

# ===== MANGUP DRUNGAROI -- age 2 levy heavy infantry =====
# Garrison troops of the cliff fortresses. Slow but sturdy, accustomed
# to fortified defence and rocky ground. Levied; never buildable.
a_nd_tdr_mangup_drungaroi = {
	is_special = yes
	category = army_heavy_infantry
	copy_from = a_age_2_renaissance_heavy_infantry
	upgrades_to = a_halberdiers
	buildable = no
	levy = yes

	# Base template: max_strength = 1.0, combat_power = 1
	# Defensive cliff-garrison niche: takes less damage on rough ground
	strength_damage_taken = -0.1
	morale_damage_taken = -0.05

	country_potential = {
		has_or_had_tag = TDR
	}

	combat = {
		hills = 0.1
		mountains = 0.1
	}

	gfx_tags = { medium_tag }
}

# ===== BOSPORAN LIBURNIAN -- age 1 buildable light ship =====
# Heir to the late Roman/Byzantine liburna of the Pontus Euxinus. Small,
# fast escort galley used to shepherd grain ships along the Tauric coast
# and run despatches between Mangup, Cherson, and Trebizond.
n_nd_tdr_bosporan_liburnian = {
	is_special = yes
	category = navy_light_ship
	copy_from = n_age_1_traditions_light_ship
	upgrades_to = n_caravel
	buildable = yes

	# Base template: speed=8, hull=4, cannons=4, crew=0.030, blockade=0.5
	movement_speed = 9
	hull_size = 5
	blockade_capacity = 0.7

	country_potential = {
		has_or_had_tag = TDR
	}

	gfx_tags = { medium_tag }
}
