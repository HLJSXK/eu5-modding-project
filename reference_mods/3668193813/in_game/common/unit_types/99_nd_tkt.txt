# - Brother-Knight: heavy cavalry levy (clergy/nobles), distinct

# - Templar Frigate: age 5 buildable light ship.
# Stats are absolute, not deltas (per gotcha #10).
# ============================================================

	is_special = yes

	upgrades_to = a_heavy_cavalrymen

	max_strength = 0.25
	combat_power = 5
	morale_damage_done = 0.15

	maintenance_demand = heavy_cavalry_maintenance
	construction_demand = heavy_cavalry_construction

	country_potential = {
		has_or_had_tag = TKT

	gfx_tags = { crusader_tag sword_tag heavy_tag }
}
