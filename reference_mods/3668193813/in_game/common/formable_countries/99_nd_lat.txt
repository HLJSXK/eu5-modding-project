# 1.2 fix: vanilla LAT_f already declares a form_effect, and the engine
# rejects a second declaration injected via INJECT (other formables work
# because their vanilla form_effect is empty). Switching to REPLACE: the
# vanilla form_effect content is preserved verbatim and the mod's effect
# (formation event + stability bonus) is appended.
REPLACE:LAT_f = {
	level = 3
	required_locations_fraction = 0.75
	rule = plausible

	potential = {
		NOT = { country_exists = c:LAT }
		NOT = { has_variable = lat_reduced_requirements_to_form }
		religion = religion:catholic
		is_frankokratia_state = yes
		NOT = { current_age_or_later = { age = age_4_reformation } }
	}

	allow = {
		NOT = { country_exists = c:LAT }
		owns = location:constantinople
		religion = religion:catholic
		OR = {
			culture = { has_culture_group = culture_group:french_group }
			culture = { has_culture_group = culture_group:iberian_group }
			culture = { has_culture_group = culture_group:italian_group }
		}
		OR = {
			current_age = age_1_traditions
			current_age = age_2_renaissance
			current_age = age_3_discovery
			current_age = age_4_reformation
		}
	}

	name = LAT
	flag = LAT
	adjective = LAT_ADJ
	tag = LAT
	color = map_LAT

	areas = {
		northern_greece_area
		morea_area
	}
	provinces = {
		north_epirus_province
		constantinople_province
		edirne_province
		evros_province
		dardanelles_province
		central_macedonia_province
		lower_macedonia_province
		upper_macedonia_province
		chalkidiki_province
		naxos_province
		euboea_province
		lemnos_province
	}
	form_effect = {
		# Mod content: formation event + small stability bonus
		trigger_event_non_silently = nd_lat.1
		add_stability = stability_mild_bonus

		# Vanilla content (verbatim from 00_formable_countries.txt)
		if = {
			limit = { country_rank_level < 4 }
			set_country_rank_effect = { rank = country_rank:rank_empire }
		}
		if = {
			limit = { has_reform = government_reform:partitio_reform }
			remove_reform = government_reform:partitio_reform
		}
		unlock_government_reform_effect = { type = lat_restored_rule_of_the_latins }
		add_reform = government_reform:lat_restored_rule_of_the_latins
		if = {
			limit = {
				culture = culture:catalan
				NOT = { has_variable = de_brienne_rule_of_athens }
				NOT = { has_reform = government_reform:ach_heirs_of_the_latin_empire }
			}
			hidden_effect = {
				change_country_color = map_LAT_catalan_restoration
			}
		}
		if = {
			limit = {
				has_reform = government_reform:ach_heirs_of_the_latin_empire
				NOT = { has_variable = de_brienne_rule_of_athens }
			}
			hidden_effect = {
				change_country_color = map_LAT_titular_restoration
			}
		}
		else_if = {
			limit = {
				NOT = { has_reform = government_reform:ach_heirs_of_the_latin_empire }
				NOT = { culture = culture:catalan }
			}
			hidden_effect = {
				change_country_color = map_LAT
			}
		}
	}
}
