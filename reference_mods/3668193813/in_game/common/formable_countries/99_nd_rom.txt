# ROM_f is the non-BYZ Roman restoration path -- the "I conquered half of
# Europe and North Africa" fantasy that nd_rom.1 and the destiny tree are
# flavored for. Vanilla 1.2 added ROM_BYZ_f as a separate, smaller-scope
# Italian-Roman path for BYZ/TRE players (italy_region only, 25% fraction,
# latinization societal-value shift). We respect that split -- vanilla's
# NOR-block on BYZ/TRE is preserved; BYZ players get our ROM_BYZ_f content
# instead via REPLACE:ROM_BYZ_f below.
REPLACE:ROM_f = {
	level = 5
	required_locations_fraction = 0.75
	rule = fantasy

	potential = {
		NOR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
		}
		OR = {
			country_rank = country_rank:rank_empire
			owns = location:constantinople
			AND = {
				owns = location:rome
				NOT = { tag = PAP }
			}
		}
	}

	allow = {
		owns = location:rome
		owns = location:constantinople
		country_rank = country_rank:rank_empire
		is_hegemon = yes
		NOT = { country_exists = c:BYZ }

		OR = {
			religion.group = religion_group:christian
			religion = religion:hellenism_religion
		}
	}

	name = ROM
	flag = ROM
	adjective = ROM_ADJ
	tag = ROM
	color = map_ROM

	regions = {
		italy_region
		iberia_region
		france_region
		balkan_region
		great_britain_region
		anatolia_region
		south_german_region
		crescent_region
		egypt_region
		maghreb_region
		carpathia_region
		caucasus_region
	}
	form_effect = {
		if = { 
			limit = { NOT = { court_language = language:latin_language } }
			set_court_language = dialect:roman_dialect
		}
		if = {
			limit = { NOT = { capital = location:rome } }
			# 1.2: vanilla ROM_f switched to move_capital_event_effect, which
			# fires the standard "capital moved" notification + UI fanfare
			# instead of silently relocating.
			move_capital_event_effect = {
				from = root.capital
				to = location:rome
			}
		}
		trigger_event_non_silently = nd_rom.1
		add_stability = stability_extreme_bonus
		add_government_power = government_power_extreme_bonus
	}
}
