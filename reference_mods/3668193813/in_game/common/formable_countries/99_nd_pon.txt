# Pontus (PON) -- REPLACE the vanilla [p] form_effect.
# Vanilla promotes to kingdom rank only. We preserve that and add the
# formation event trigger so the player chooses between the Komnenian
# Restoration and the Pontic Republic at formation.
REPLACE:PON_f = {

	level = 3
	required_locations_fraction = 0.75
	rule = fantasy

	potential = {
		OR = {
			culture = culture:pontic_greek_culture
			owns = location:trebizond
		}
	}

	allow = {
		AND = {
			culture = culture:pontic_greek_culture
			owns = location:trebizond
		}
	}

	name = PON
	flag = PON
	adjective = PON_ADJ
	tag = PON
	color = map_pontus

	areas = {
		pontus_area
		crimea_area
	}

	provinces = {
		taman_province
		samegrelo_province
		abkhazia_province

		amasya_province
		tokat_province
		erzurum_province
	}

	form_effect = {
		set_country_rank_effect = { rank = country_rank:rank_kingdom }
		trigger_event_non_silently = nd_pon.1
	}
}
