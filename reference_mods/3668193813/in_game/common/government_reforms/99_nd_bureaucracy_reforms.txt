# National Destinies - Bureaucratic Doctrines
# Three mutually-exclusive minor reforms defining a country's maintenance posture
# for its bureaucratic institutions. Available only to formed countries that have
# unlocked bureaucracies. Excluded from BYZ and TRE (their vanilla advances already
# define their bureaucratic identity).

nd_patrimonial_doctrine = {
	age = age_1_traditions
	
	potential = {
		has_game_rule = nd_bureaucracies_enabled
		allowed_bureaucracies = yes
		NOR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
		}
	}
	
	allow = {
		NOT = { has_reform = government_reform:nd_oligarchic_doctrine }
		NOT = { has_reform = government_reform:nd_meritocratic_doctrine }
	}
	
	country_modifier = {
		global_max_bureaucracy_slots = 1
		global_bureaucracy_maintenance_efficiency = -0.1
		global_bureaucracy_removal_efficiency = 0.15
	}
	
	years = 2
}

nd_oligarchic_doctrine = {
	age = age_2_renaissance
	
	potential = {
		has_game_rule = nd_bureaucracies_enabled
		allowed_bureaucracies = yes
		NOR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
		}
	}
	
	allow = {
		NOT = { has_reform = government_reform:nd_patrimonial_doctrine }
		NOT = { has_reform = government_reform:nd_meritocratic_doctrine }
	}
	
	country_modifier = {
		global_max_bureaucracy_slots = 1
		global_bureaucracy_maintenance_efficiency = 0.1
		reshape_bureaucracy_cost_modifier = 0.05
	}
	
	years = 2
}

nd_meritocratic_doctrine = {
	age = age_3_discovery
	
	potential = {
		has_game_rule = nd_bureaucracies_enabled
		allowed_bureaucracies = yes
		NOR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
		}
	}
	
	allow = {
		NOT = { has_reform = government_reform:nd_patrimonial_doctrine }
		NOT = { has_reform = government_reform:nd_oligarchic_doctrine }
	}
	
	country_modifier = {
		global_max_bureaucracy_slots = 1
		global_bureaucracy_maintenance_efficiency = -0.05
		global_bureaucracy_entrenchment_speed_modifier = 0.15
	}
	
	years = 2
}
