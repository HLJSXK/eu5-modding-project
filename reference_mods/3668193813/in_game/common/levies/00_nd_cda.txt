# Aragonese Empire (CDA) — Custom levies
# National Destinies mod
#
# Vanilla provides Almogavar levies (levy_a_almogavars,
# levy_a_late_almogavars) to all aragonese/catalan-cultured countries via
# culture_iberia.txt — we do NOT recreate those.
#
# This file adds the maritime crew levy unique to the Crown's coastal
# trading cities: the Catalan Galley levy, drawing rowers and marines
# from the burgher and laborer pops of Barcelona, Valencia, Palma,
# Naples and Palermo. The 00_nd_ prefix loads before vanilla's generic
# levies so this specialized levy is evaluated first in coastal locations.

levy_nd_cda_catalan_galley = {
	size = levy_ship_size
	unit = n_nd_cda_levy_galley

	allow = {
		pop_type = pop_type:burghers
	}
	allow_as_crew = {
		OR = {
			pop_type = pop_type:peasants
			pop_type = pop_type:laborers
		}
	}

	country_allow = {
		has_or_had_tag = CDA
	}
}
