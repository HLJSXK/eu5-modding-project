# Balangay levy -- standard naval levy on the burgher pop, peasants/laborers
# crew. Uses normal levy_ship_size (NOT _special).
levy_nd_pnt_balangay = {
	size = levy_ship_size
	unit = n_nd_pnt_balangay

	allow = {
		pop_type = pop_type:burghers
	}
	allow_as_crew = {
		OR = {
			pop_type = pop_type:peasants
			pop_type = pop_type:laborers
		}
	}

	country_allow = {
		has_or_had_tag = PNT
	}
}
