# Maharlika Custom Levies
# Specialized levies first; load order ensures these evaluate before vanilla generics.

# Maharlika warrior class — the Tagalog warrior-noble freemen, raised from
# soldier and noble pops as a class privilege.
levy_nd_mrl_maharlika_warriors = {
	size = levy_generic_infantry_size
	unit = a_nd_mrl_maharlika_levy

	allowed_pop_type = soldiers
	allowed_pop_type = peasants

	country_allow = {
		has_or_had_tag = MRL
	}
}
