# National Destinies - Bureaucratic Funding Law
# A single administrative law with four funding policies. Defines the
# build-and-shape posture of a country's bureaucracy. Three policies available
# from formation; Treasury Bureau unlocks in age 5 as a peer modernization choice.
# No maintenance modifier appears here -- maintenance is the reform's domain.
# Available only to formed countries with bureaucracies allowed; BYZ and TRE
# excluded (their vanilla advances cover this ground).

nd_bureaucratic_funding_law = {
	law_category = administrative
	
	potential = {
		has_game_rule = nd_bureaucracies_enabled
		allowed_bureaucracies = yes
		NOR = {
			has_or_had_tag = BYZ
			has_or_had_tag = TRE
		}
	}
	
	nd_tax_farming_policy = {
		country_modifier = {
			global_bureaucracy_implementation_efficiency = 0.25
			global_bureaucracy_removal_efficiency = 0.15
		}
		estate_preferences = {
			nobles_estate
		}
		years = 2
	}
	
	nd_crown_coffers_policy = {
		country_modifier = {
			global_bureaucracy_implementation_efficiency = 0.15
			global_bureaucracy_entrenchment_speed_modifier = 0.20
		}
		estate_preferences = {
			crown_estate
		}
		years = 2
	}
	
	nd_estate_subvention_policy = {
		country_modifier = {
			reshape_bureaucracy_cost_modifier = -0.25
		}
		estate_preferences = {
			clergy_estate
		}
		years = 2
	}
	
	nd_treasury_bureau_policy = {
		allow = {
			OR = {
				current_age = age_5_absolutism
				current_age = age_6_revolutions
			}
		}
		country_modifier = {
			global_bureaucracy_implementation_efficiency = 0.15
			reshape_bureaucracy_cost_modifier = -0.15
			global_bureaucracy_entrenchment_speed_modifier = 0.10
		}
		estate_preferences = {
			burghers_estate
		}
		years = 2
	}
}
