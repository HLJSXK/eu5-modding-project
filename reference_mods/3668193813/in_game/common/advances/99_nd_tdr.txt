# ==========================================================================
# Bosporan Kingdom (TDR) -- advance tree
# Stepping-stone density (PON L3 in Anatolia is the loose successor; not
# a clean upward chain in Crimea). 20 advances across ages 1-5 (4 per age).

# Komnenoid dynastic claim, Crimean Gothic heritage, Pontic Orthodoxy.
# ==========================================================================

# ==========================================================================
# ===== DESTINY TREE =====
# Branches on the formation choice:
#   PATH A (Regnum Bospori) -- Pontic Hegemon. Mithridatic naval-trade power.
#     Recreate the ancient Bosporan Black Sea ring: Sinope, Trebizond,
#     Varna, Constantinople. Capstone unlocks the Mithridatic Admiralty.
#   PATH B (Cimmerian Bosporus) -- Tauric Imperium. Multiethnic land empire
#     reaching into the steppe and Ruthenia: Tana, Sarai al-Jadid, Kyiv.
#     Capstone unlocks the Mangup Imperial Chancery.
# ==========================================================================

nd_tdr_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = TDR has_variable = nd_tdr_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	monthly_army_tradition = 0.05
}

	requires = nd_tdr_destiny
	potential = { has_or_had_tag = TDR has_variable = nd_tdr_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_tdr_path_a_tt

		owns_or_non_sovereign_subject_owns = location:amasra
