# Aragonese Empire (CDA) — Advance tree
# National Destinies mod
#
# Vanilla already provides ara_almogavars (age 1, levy unlock), 
# ara_catalan_crossbowmen (age 1, unit unlock), and ara_late_almogavars
# (age 2, levy unlock) to all aragonese/catalan-cultured countries via
# culture_iberia.txt — we do NOT duplicate. Vanilla also covers the broad
# Catholic religion advance set across all six ages. Region-tagged Iberian
# advances grant conquistador (age 3) and galiot ship (age 4).
#
# This tree covers what vanilla does not: pactist constitutionalism, the
# Catalan maritime law tradition, the cartographic and humanist court
# legacy of Alfonso V the Magnanimous, the Drassanes Reials and Catalan
# Galley naval specialty, the Casa Llotja de Mar mercantile institution,
# the Tercio de Napoles military innovation, the Generalitat parliament,
# and the late-period Catalan industrial revival.
#
# Stepping-stone density (4-5 per age) — successor formables include
# ROM and ROM_BYZ at L5. Total: 20 advances, 12 unlocks (60%).

# ============================================================================
# AGE 1 — TRADITIONS (4 advances, 1 unlock)
# ============================================================================

# Pacta Conventa — the constitutional pact between the Aragonese crown
# and the kingdom's nobles, codified in the Privilegio General of 1283.
# The kingdom's "if not, no" clause was an unprecedented contractual
# limit on royal power.
nd_cda_pacta_conventa = {
	age = age_1_traditions
	potential = { has_or_had_tag = CDA }
	requires = legalism_advance
	icon = legalism_advance

	nobles_estate_target_satisfaction = 0.05
	burghers_estate_target_satisfaction = 0.05
	global_separatism = -0.03
}

# Llibre del Consolat de Mar — the Catalan maritime code (compiled c.1320,

	age = age_1_traditions

	requires = guilds
	icon = guilds

	trade_sea_efficiency = 0.04

	global_trade_center_power = 0.05

# ===== DESTINY TREE =====
# Post-formation mission objectives.
# Branches on the imperial project chosen at formation. Path A pushes the
# Mediterranean thalassocracy toward full eastern dominance (recover Catalan
# Greece, contest the Aegean, dominate the inner sea). Path B redirects
# Aragonese maritime expertise into an alt-history Atlantic colonial empire
# led by Catalan-Aragonese rather than Castilian sponsorship.
#
# Variable gate: nd_cda_path_a (Mediterraneum Aragonense) OR nd_cda_path_b
# (Hispano-Aragonese / Catalan New World). Set in formation event.

nd_cda_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = CDA has_variable = nd_cda_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance

	government_size = 0.1
}

# ----------------------------------------------------------------------------
# PATH A — LATIN MARE NOSTRUM (eastern Mediterranean dominance)
# ----------------------------------------------------------------------------

# Stage 1A — Recover Catalan Greece. The Catalan Company under Roger de
# Flor and Berenguer d'Entenca took Athens, Neopatras and Patras between
# 1303 and 1311. Reclaim those duchies for the crown.
nd_cda_dest_catalan_greece = {
	age = age_1_traditions
	requires = nd_cda_destiny
	potential = { has_or_had_tag = CDA has_variable = nd_cda_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_cda_path_a_tt
			has_variable = nd_cda_path_a
		}
		owns_or_non_sovereign_subject_owns = location:athens
		owns_or_non_sovereign_subject_owns = location:neopatras
		owns_or_non_sovereign_subject_owns = location:patras
		gold >= 1000
	}
	icon = imperial_ambitions

	naval_damage_done = 0.075
	army_movement_speed = 0.05
}

	age = age_1_traditions

		total_development >= 3000

}
