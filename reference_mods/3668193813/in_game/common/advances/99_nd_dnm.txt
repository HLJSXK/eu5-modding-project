# ==========================================================================
# DANUBIAN MONARCHY -- advance tree.
#
# Covers the Habsburg/Danubian identity vanilla leaves untold:
#  - the composite-monarchy administration (Hofkammer, Hofkriegsrat,
#    Pragmatic Sanction, Theresian-Josephine reforms)
#  - the Military Frontier / Grenzer soldier-colony system
#  - the Danube riverine trade spine (Vienna staple, salt and iron)
#  - dynastic diplomacy ("bella gerant alii, tu felix Austria nube")
#  - the multi-ethnic estate balance that defined the realm
#
# Vanilla already provides (and is NOT duplicated here): the Catholic
# religion advances (tolerance, conversion, devotion, prestige, literacy),
# the Balkan/Carpathia region advances (poklisars diplomats, balkan_hajduks
# = a_hajduk, cro_pandurs_recruitment = a_pandur, expansionist_ambitions
# integration), and the HUN/CRO country advances. Story begins in the
# Renaissance: the 1526 dual inheritance of Bohemia and Hungary is when
# the composite monarchy is born, so no age 1 advances.
#
# requires: same-age vanilla attachment points only (cross-age requires
# silently makes advances invisible). All attach to always-available
# 0_age_of_* trees, never the for=adm/dip/mil specialization trees.
# ==========================================================================

# ==========================================================================
# ===== DESTINY TREE =====
#
# Root + 2 paths x 3 stages. Gated on nd_dnm_formed (set in the formation
# event's hidden_effect) so the cluster is invisible until the Danubian
# Monarchy is actually forged. The two branches are the formation-event
# choices, NOT origin-based:
#   path_a = Antemurale Christianitatis -- the bulwark of Christendom,
#            the Militaergrenze identity, the Ottoman war pressed south.
#   path_b = Gesamtmonarchie -- the centralized total monarchy, the three
#            crowns welded into one administrative body from Vienna.
#
# Per the spec (resolution #2): the Gesamtmonarchie branch THEMATICALLY
# builds centralization/compromise capacity but does NOT create an
# "Ausgleich" / "Austria-Hungary" capstone -- that beat is reserved for a
# later custom Disaster's compromise exit.
# ==========================================================================

nd_dnm_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = DNM has_variable = nd_dnm_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	monthly_legitimacy = 0.05
}

# ----- PATH A : ANTEMURALE CHRISTIANITATIS -----------------------------

# A1 : The Long Turkish War -- recover the lower Danube line. The Sava-
# Danube confluence: Belgrade, Smederevo, the Vidin gate. The frontier
# pressed back to where Eugene of Savoy would draw it.
nd_dnm_dest_sava_line = {
	age = age_1_traditions
	requires = nd_dnm_destiny
	potential = { has_or_had_tag = DNM has_variable = nd_dnm_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_dnm_path_a_tt
			has_variable = nd_dnm_path_a
		}
		owns_or_non_sovereign_subject_owns = location:belgrad
		owns_or_non_sovereign_subject_owns = location:smederevo
		army_size >= 30
	}
	icon = feudalism_advance
	army_heavy_infantry_power = 0.1
	land_unit_attrition = -0.1
}

	age = age_1_traditions
	requires = nd_dnm_dest_cameral_foundation
	potential = { has_or_had_tag = DNM has_variable = nd_dnm_formed has_game_rule = nd_destinies_enabled }

		gold >= 2000
		total_development >= 6500
		any_market_present_in_country = {
