# === Beth Nahrain Heritage Advances ===
# Tier 1 of the Syriac restoration chain. Stepping stone to ASR (Assyrian
# Empire, L4). Focus heritage advances on the Mesopotamian-Christian
# foundation story; the imperial bombast belongs to ASR.

# Vanilla coverage: 31 miaphysite religion advances + 1 region advance.
# Most modifier types in those advances are documented in the blocklist
# at the bottom of creator_context. We avoid duplicating them here.

# ===== DESTINY TREE =====
# Post-formation mission tree. Two paths: A (Restoration of Aššur) prepares
# the player for the ASR formable; B (Light of the East) keeps Beth Nahrain
# as the Christian-Syriac civilizational pole. Mapping to formation choices:
# destiny path_a is gated on nd_bnh_path_b (the militarist Tur Abdin choice);
# destiny path_b is gated on nd_bnh_path_a (the scholarly Patriarchate
# choice). Path C (Edessene) has no destiny continuation by design.

nd_bnh_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = BNH has_variable = nd_bnh_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	power_projection = 5
}

# === Path A: Restoration of Aššur (Imperial) ===
# Stage 1: own the Tigris twin cities. Mosul is the kingdom's seat;
# Baghdad is the Abbasid capital, the next step on the road to Aššur.
nd_bnh_dest_a1 = {
	age = age_1_traditions
	requires = nd_bnh_destiny
	potential = { has_or_had_tag = BNH has_variable = nd_bnh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_bnh_path_a_tt
			has_variable = nd_bnh_path_b
		}
		owns_or_non_sovereign_subject_owns = location:mosul
		owns_or_non_sovereign_subject_owns = location:baghdad
	}
	icon = feudalism_advance
	army_heavy_infantry_power = 0.1
	expected_army_size_modifier = 0.1
}

# Stage 2: The Maphrian-General marches. The kingdom's writ reaches
# Tabriz (Ilkhanid capital, Christian-court memory) and Basra (the Shatt
# al-Arab, the gateway to the Gulf). Total dev and army size required.

	age = age_1_traditions
	requires = nd_bnh_dest_a1
	potential = { has_or_had_tag = BNH has_variable = nd_bnh_formed has_game_rule = nd_destinies_enabled }
	allow = {
		owns_or_non_sovereign_subject_owns = location:tabriz
		owns_or_non_sovereign_subject_owns = location:basra
		army_size >= 50
		total_development >= 1000
	}
	icon = imperial_ambitions
	global_supply_limit_modifier = 0.1
	army_heavy_cavalry_power = 0.1
}
