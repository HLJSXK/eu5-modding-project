# ==========================================================================
# Pontus (PON) -- advance tree.
# Destination tier: Level 3 standalone, no upward chain. Dense tree.
# 30 advances across ages 1-6. 18 unlock content (60%).
# Themes:
#   - Komnenian Imperial tradition (court, coronation, basileia)
#   - Sumela monastery + Pontic-Greek Orthodox autocephaly
#   - Argyria silver mines and the Trapezuntine aspron
#   - Pontic-Greek mercantile cities (Trebizond, Sinope, Kerasounta)
#   - Caucasian Christian frontier (Bagrationi, Lazica)
#   - Black Sea Pontic naval tradition
#   - Cappadocian rock-hewn monastic networks
#   - 1461 alt-history: empire that did not fall
#   - Pontic-Greek diaspora and Phanariot administrative class
#   - Late Pontic-Greek Megale Idea / national revival
# ==========================================================================

# ==========================================================================
# ===== DESTINY TREE =====
# Post-formation branching tree for the Pontic restoration.
# Path A -- Roman Restoration: the Komnenian basileia returns from Pontic
#   exile to reclaim the Roman inheritance, ending in Constantinople itself.
# Path B -- Black Sea Commonwealth: a federated Greek-Caucasian-Pontic
#   compact, an Orthodox Council of the Faithful around the Pontic shore.
# ==========================================================================

nd_pon_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = PON has_variable = nd_pon_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	prestige_decay = -0.001
}

# === PATH A: ROMAN RESTORATION (Komnenian Imperial Reconquista) ===

# Stage 1: The Anatolian reconquest. Reach back from the Pontic shore
# into the old Anatolian heartland of the Roman idea -- Bursa (Prousa)
# and Nicaea (Iznik), the cradle cities of late-Roman Asia Minor.
nd_pon_dest_anatolikon = {
	age = age_1_traditions
	requires = nd_pon_destiny
	potential = { has_or_had_tag = PON has_variable = nd_pon_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_pon_path_a_tt
			has_variable = nd_pon_path_a
		}
		owns_or_non_sovereign_subject_owns = location:bursa
		owns_or_non_sovereign_subject_owns = location:iznik
		prestige >= 50
	}
	icon = feudalism_advance
	army_heavy_infantry_power = 0.1
	siege_ability = 0.1
}

# Stage 2: The Aegean strategos. Push the Komnenian frontier into the
# Aegean and the European balkans -- Smyrna, Chios, Edirne. The
# Pontic empire now speaks as a Roman peer of all the western powers.
nd_pon_dest_aegean_strategos = {
	age = age_1_traditions

		total_development >= 1000

# Stage 3: New Rome restored. The Komnenian basileus returns from

		custom_tooltip = {
			text = nd_pon_path_b_tt
