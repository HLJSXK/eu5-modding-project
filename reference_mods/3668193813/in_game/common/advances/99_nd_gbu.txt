# === Greater Burgundy Advance Tree ===
# Vanilla coverage: Frisian/Lorrain culture advances + Catholic religion advances
# already cover Frisian trade, Lorrain mining/diversity/three-bishoprics,
# and the broad Catholic ladder. Our tree fills the Burgundian-state story
# vanilla does not tell: the urban Flemish wealth, the Estates-General
# tradition, the Order of the Golden Fleece, the Burgundian Wars and gendarme
# reform, the Trier coronation alt-history, and the Burgundian inheritance.

# ===== DESTINY TREE =====

# Root -- the post-formation project. The crown is placed; now choose between
# Imperial Restoration of the Lotharingian Middle Kingdom or the Atlantic
# Mercantile Kingdom of Antwerp and the bourse.
nd_gbu_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = GBU has_variable = nd_gbu_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	aggressiveness_modifier = -0.15
}

# === Path A: Imperial Restoration ===
# Stage 1 -- Rhineland Frontier: hold the Carolingian coronation cities of
# Trier and Aachen, the spine of Lothair's Middle Kingdom.
nd_gbu_dest_rhineland_frontier = {
	age = age_1_traditions
	requires = nd_gbu_destiny
	potential = { has_or_had_tag = GBU has_variable = nd_gbu_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_gbu_path_a_tt
			has_variable = nd_gbu_path_a
		}
		owns_or_non_sovereign_subject_owns = location:trier
		owns_or_non_sovereign_subject_owns = location:aachen
		gold >= 1000
	}
	icon = feudalism_advance
	army_movement_speed = 0.05
