# ===== DESTINY TREE =====
# Two natural directions for the revived Yarlung-imperial age:
#  Path A — Inner Asian Empire: the warrior-tsenpo's reach into the Tarim
#           Basin, the Hexi corridor, and the Mongol-steppe frontier.
#  Path B — Dharma Imperium: the Mahayana civilizational pole — Pala-era
#           Bengal, the Bengali sangha, and the Buddhist conversion of
#           Mongolia in the Altan Khan / Dalai Lama tradition.

# === ROOT ===
# Hidden until the empire is formed. Anchors both destiny paths.
nd_tbt_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = TBT has_variable = nd_tbt_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	monthly_towards_centralization = 0.025
}

# === PATH A: INNER ASIAN EMPIRE ===

# Stage 1 — Tarim Basin restored
# Reclaim the Anxi Four Garrisons: Khotan and Kashgar, the Tibetan Silk
# Road of the late 8th century. Light cavalry strikes sweep the passes.
nd_tbt_dest_tarim_recovery = {
	age = age_1_traditions
	requires = nd_tbt_destiny
	potential = { has_or_had_tag = TBT has_variable = nd_tbt_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_tbt_path_a_tt
			has_variable = nd_tbt_path_a
		}
		# Mongol steppe reach: Karakorum and Urga acknowledge the
		# Tsenpo's banner. The formation requirements (Tarim + Hexi)
		# already covered the historical Yarlung-imperial reach;
		# the destiny pushes north into Mongolia, repeating the
		# steppe-overlord role the empire briefly held in the late 8th c.
		owns_or_non_sovereign_subject_owns = location:karakorum
		owns_or_non_sovereign_subject_owns = location:urga
	}
	icon = imperial_ambitions
	regiment_recruit_speed = 0.10
	army_auxiliary_power = 0.10
}

# Stage 2 — Hexi corridor reclaimed
# The Tang frontier, Gansu marchlands. The chama trade flows back
# through the imperial customs-houses. Garrisons hold every oasis.
nd_tbt_dest_hexi_corridor = {
	age = age_1_traditions
	requires = nd_tbt_dest_tarim_recovery
	potential = { has_or_had_tag = TBT has_variable = nd_tbt_formed has_game_rule = nd_destinies_enabled }
	allow = {
		# Penetration into Tang heartland: past Hexi (already formation-
		# requirement) into the Wei-river/Sichuan corridor. Lanzhou
		# (the upper Yellow River prefectural seat), Xining (Qinghai
		# corridor) and Chengdu (the Sichuan basin's commercial capital)
		# mark the limits of the 763-829 Tibetan-Tang frontier wars; the
		# empire never quite held them in fact, but did campaign there.
		# The destiny makes that contested frontier permanent imperial
		# ground.
		owns_or_non_sovereign_subject_owns = location:lanzhou
		owns_or_non_sovereign_subject_owns = location:chengdu
		owns_or_non_sovereign_subject_owns = location:xining
		army_size >= 50
		total_development >= 5000
	}
	icon = feudalism_advance
	expected_army_size_modifier = 0.10
	army_maintenance_efficiency = 0.075
}

# Stage 3 — Steppe frontier dominion (capstone)
# The empire reaches the Mongol steppe and the Tian Shan oasis network.
# Karakorum and Urga acknowledge the imperial banner. Unlocks the
# Imperial Stone-Pillar Palace, a capital monument of the warrior-empire.
nd_tbt_dest_inner_asian_empire = {
	age = age_1_traditions
	requires = nd_tbt_dest_hexi_corridor
	potential = { has_or_had_tag = TBT has_variable = nd_tbt_formed has_game_rule = nd_destinies_enabled }
	allow = {
		# Permanise the 763 AD capture of Chang'an: Tibet briefly held
		# the Tang capital for fifteen days; the alt-history capstone
		# turns that single famous episode into sustained imperial rule
		# over the Wei valley. Combined with Kunming as the Yunnan-
		# Burma frontier (the Nanzhao tributary boundary the historical
		# empire briefly reached), the capstone marks the absolute
		# imperial peak: Tsenpo of all Inner Asia, from the Mongol
		# steppe through the Wei valley to the Yunnan plateau.
		owns_or_non_sovereign_subject_owns = location:jingzhao
		owns_or_non_sovereign_subject_owns = location:kunming
		army_size >= 75
		prestige >= 75
	}
	icon = crown_power_advance_renaissance
	unlock_building = nd_tbt_imperial_pillar_palace
	military_tactics = 0.05
	assault_ability = 0.10
	global_war_score_efficiency = 0.10
}
