# ===== DESTINY TREE =====
# Post-formation mission tree. Two paths, three stages each. Root fires for either.
# Path A: Empire of the Wheel — military-imperial, Cakkavatti expansion
# Path B: Civilization of the Sangha — institutional Buddhist civilization

# Root: Suvarnabhumi awakens
nd_svb_destiny = {
	age = age_1_traditions
	depth = 0
	potential = { has_or_had_tag = SVB has_variable = nd_svb_formed has_game_rule = nd_destinies_enabled }
	allow = { stability >= 75 }
	icon = crown_power_advance_renaissance
	monthly_diplomats = 0.05
}

# ===== PATH A: Empire of the Wheel =====

# Stage 1 — The Champa Conquest
nd_svb_dest_champa_conquest = {
	age = age_1_traditions
	requires = nd_svb_destiny
	potential = { has_or_had_tag = SVB has_variable = nd_svb_formed has_game_rule = nd_destinies_enabled }
	allow = {
		custom_tooltip = {
			text = nd_svb_path_a_tt
			has_variable = nd_svb_path_a
		}
		owns_or_non_sovereign_subject_owns = location:simhapura
		owns_or_non_sovereign_subject_owns = location:vijaya
		army_size >= 30
	}
	icon = imperial_ambitions
	siege_ability = 0.10
	land_morale_recovery = 0.010
}

# Stage 2 — The Red River Submits
