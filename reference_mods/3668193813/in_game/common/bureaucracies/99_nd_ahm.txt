# National Destinies -- Ahom Kingdom bureaucracies
# Four institutions of the Tai-Ahom polity (1228-1826) of the Brahmaputra
# valley. Auto-modifier (nd_ahm_bureaucracy_authority) grants 2 slots; the
# capital adds 1 = 3 effective. The player picks 3 of 4.
#
# These institutions deliberately use a Tai-Ahom and Assamese vocabulary
# distinct from the Sanskritic-Telugu palette of VJN: the Swargadeo
# (heavenly king) ruled through a council of seven Great Offices
# (Borgohain), a corvee paik administration, the Buranji court chronicle
# tradition, and -- in the Vaishnava reform period -- the Sattradhikar
# monastic network of Sankardev's Eka-Sarana faith.

# 1. Patro Mantri Sabha -- the standing council of the Seven Great Offices
#    (the Patra Mantris): Burhagohain, Borgohain, Borpatrogohain,
#    Borphukan, Borbarua, Bargohain, and the lesser offices. The council
#    confirmed succession to the Swargadeo's throne and could (and did)
#    depose unfit kings. Codified under Suhungmung (r. 1497-1539) and
#    enlarged under Pratap Singha (r. 1603-1641), the Sabha was the
#    constitutional check that distinguished the Ahom polity from its
#    Mughal and Bengali neighbors -- the Swargadeo ruled with the council,
#    not over it. Distinct from the static advance that codifies the
#    offices: the Sabha is the council in continuous session.
ahm_patro_mantri_sabha_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = AHM
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The Sabha's mere standing presence binds the Patra Mantri lineages
	# into the constitutional fabric of the realm.
	neutral_modifier = {
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Working: the council deliberates campaigns, succession, and revenue.
	# Patra Mantri lineages contribute their paik-quotas in good order; the
	# Borphukan and Bargohain coordinate Brahmaputra-valley defense; council
	# legislation moves with deliberate but legitimate weight.
	positive_modifier = {
		scale = { value = scope:maintenance }
		legislative_efficiency = 0.15
		nobles_estate_satisfaction_recovery = 0.001
		land_morale_modifier = 0.05
	}

	# Decayed: the Patra Mantris become an oligarchic faction. The
	# Burhagohain monopolizes appointments, the Borphukan's Guwahati
	# residence becomes a private court, and the Swargadeo's writ no longer
	# runs through the council -- an institutional hollowing that
	# historically presaged the Moamoria rebellions and the kingdom's
	# nineteenth-century collapse.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		crown_estate_satisfaction_decay = 0.020
	}
}

# 2. Paik-Khel Administration -- the standing corvee bureau under the
#    Likchau (paik-roll keepers). Every adult male was enrolled into a
#    got of three or four paiks, each paik owing rotating service to a
#    Khel office under one of the Patra Mantris: three paiks supported
#    the fourth in militia or labor service. Codified under Momai
#    Tamuli Borbarua (1609) and successively reformed, the system was
#    the kingdom's fiscal-military backbone -- the paiks built the
#    embankments, manned the bachari river fleet, and constituted the
#    standing levy that broke the Mughal armies seventeen times.
ahm_paik_khel_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = AHM
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# The paik-rolls give the Swargadeo a precise count of every man owed
	# to the realm -- a fiscal-military census without parallel among the
	# kingdom's neighbors.
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# Working: paiks rotate through militia, dyke labor, dockyard service,
	# and the Borphukan's bachari fleet. Embankments hold against the
	# monsoon, the standing levy musters on demand, and the kingdom's
	# carrying capacity for sustained campaigning is structural.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		tax_income_efficiency = 0.08
		global_garrison_size_modifier = 0.10
	}

	# Decayed: the rolls fall out of date as Khel officers under-report
	# their paiks, the Likchau take bribes to remove names, peasant got
	# break their rotations and abscond to the hill country, and the
	# Moamoria preachers find their first audience among paiks
	# overburdened by the rotting system.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_satisfaction_decay = 0.020
		monthly_complacency = 0.005
	}
}

# 3. Buranji Likchau -- the institutionalized court chronicle apparatus.
#    The Buranjis (the word means "store of unprecedented things") were
#    written first in Tai-Ahom script and later translated into Assamese.
#    Some forty-odd surviving Buranji manuscripts cover dynastic, military,
#    diplomatic, and frontier-survey matter; a literate scribal corps
#    (the Likchau) attended every court session, every embassy, and every
#    campaign, producing one of the great non-imperial chronicle traditions
#    of South Asia. The Buranjis were consulted as legal precedent and as
#    diplomatic record into the nineteenth century -- an institutional
#    memory the kingdom's Mughal and Bengali rivals lacked.
ahm_buranji_likchau_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = AHM
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		clergy_estate
		burghers_estate
	}
	estates_that_dislike = {}

	# The Likchau's standing record-keeping anchors diplomatic and
	# dynastic continuity to written precedent.
	neutral_modifier = {
		monthly_legitimacy = 0.03
	}

	# Working: the court chronicle records frontier surveys, embassy
	# protocols, and dynastic precedent. The literate scribal corps
	# trains the kingdom's reading class; cultural prestige radiates
	# from Garhgaon through Charaideo and Rangpur, and the kingdom's
	# diplomatic word is backed by six centuries of receipts.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_max_literacy = 5
		cultural_tradition = 0.10
		diplomatic_reputation = 1
	}

	# Decayed: the Likchau corps thins to a handful of underpaid copyists,
	# manuscripts rot in the Garhgaon archives, dynastic precedent is
	# lost or fabricated to suit factions, and Patra Mantri factions feed
	# rival genealogies into the chronicle for their own ends.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.015
		monthly_complacency = 0.005
	}
}

# 4. Sattradhikar Mandali -- the institutionalized network of Vaishnava
#    monastic authorities (the Sattradhikars), heads of the satra
#    monasteries founded under Srimanta Sankardev (1449-1568) and codified
#    by his disciple Madhabdev. By the seventeenth century, the great
#    satras (Auniati, Dakhinpat, Garhmur, Kuruwabahi) held substantial
#    crown grants of land (debottar and brahmottar tenures), administered
#    their own peasant cultivators, ran the namghar village congregations,
#    and arbitrated disputes outside the Patra Mantri courts. Eka-Sarana
#    devotion -- one God, no Brahmin gatekeeping -- offered a path of
#    integration to the Bodo, Kachari, and other valley peoples for whom
#    Brahmanical Hinduism had no place. The Sattradhikars eventually
#    became a parallel power that the Swargadeos at first patronized and
#    then, in the Moamoria era, found themselves at war with.
ahm_sattradhikar_mandali_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = AHM
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Sattradhikar network presupposes the Vaishnava reform actually
	# having taken root in the valley.
	allow = {
		owns = location:bordowa
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The namghar congregations are a permanent civic-religious presence
	# in every village they reach -- the Eka-Sarana faith expands by mere
	# institutional standing.
	neutral_modifier = {
		global_pop_assimilation_speed_modifier = 0.05
	}

	# Working: satras administer debottar lands, namghar village courts
	# arbitrate within Eka-Sarana norms, Bodo and Kachari peoples find a
	# devotional path that does not demand the abandonment of their gods,
	# and peasant cultivators on satra estates are insulated from the
	# heavier paik corvees.
	positive_modifier = {
		scale = { value = scope:maintenance }
		tolerance_heathen = 1
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		monthly_devotion = 0.05
	}

	# Decayed: the Sattradhikars become hereditary rentiers on their
	# crown grants, the namghar courts drift outside royal jurisdiction,
	# Patra Mantri nobles resent the satras' tax-free lands and
	# independent peasantries, and Moamoria preachers radicalize the
	# devotional movement into rebellion against the throne.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		stability_decay = 0.0005
	}
}
