# Nanzhao bureaucracies — historical institutions of the Yunnan polity
# spanning the Nanzhao kingdom (738–902), the Dali kingdom (937–1253),
# and the Yuan/Ming tusi era (1253 onward). Revived on formation.

# ---------------------------------------------------------------------------
# Qingping Council — six chief ministers of Nanzhao
# ---------------------------------------------------------------------------
# The qingping guan ("Pure and Just Officials") were the six chief
# ministers who advised the Nanzhao king. Seats were drawn from the
# great Wuman noble houses; the council was the venue where the warrior
# aristocracy bound the throne to consultation. Modelled on Tang
# chancellery practice but adapted to Yunnan's highland clan politics.

qingping_council_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = NZO
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# neutral: a permanent ministerial council enlarges the apparatus
	# the crown must fund and consult, but never closes the door on
	# noble counsel
	neutral_modifier = {
		government_size = 1
	
		country_cabinet_efficiency = -0.10
}

	# positive: aristocratic consensus legitimates royal decrees,
	# satisfies the great clans, and channels their energies into the
	# realm rather than against it
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_legitimacy = 0.05
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# negative: a neglected council degenerates into faction; merchants
	# and outsiders are walled out of decision-making and crown control
	# struggles to reach the lowlands and counting houses
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		global_max_urban_control = -0.05
	}
}

# ---------------------------------------------------------------------------
# Acharya Master-State — Azhali Tantric clerk-priests of Dali
# ---------------------------------------------------------------------------
# The Azhali (阿吒力) tantric Buddhist masters of Dali (937–1253) served
# not only as ritual officiants but as scribes, archivists, and chancery
# clerks staffing the royal administration. Their authority flowed
# through master–disciple lineages, breaking the monopoly of the old
# Wuman clans on literate office. Distinct from Han Buddhist clergy,
# the Acharya were a hereditary priestly caste with state functions.

# ---------------------------------------------------------------------------
# Tusi Chieftaincy System — hereditary native chieftains as administrators
# ---------------------------------------------------------------------------
# The tusi (土司) framework, formalized under the Yuan and intensified
# under the Ming, recognized hereditary chieftains of the Yi, Bai, Hmong,
# and Tai peoples as the realm's local administrators in exchange for
# tribute and levies. The system raised formidable highland armies and
# pacified frontiers cheaply, but each generation of confirmed
# inheritance hardened the chieftains into a parallel power.

tusi_chieftaincy_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = NZO
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		total_development >= 1000
	}

	estates_that_like = {
		nobles_estate
		tribes_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# neutral: each confirmed chieftain is a local governor of the
	# realm, projecting administrative reach into the highlands
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# positive: tusi levies cost the throne little and respond quickly
	# to highland threats; chieftain manpower is the kingdom's frontier
	# shield, and the tribes feel honoured to be confirmed
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.075
		army_food_gathering_modifier = 0.1
		tribes_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# negative: a neglected tusi system does not weaken the chieftains;
	# it merely starves the central registers that once tracked them.
	# Each confirmed lineage becomes a parallel state, separatism
	# climbs, and the highlands drift out of the throne's grasp
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.1
		global_max_rural_control = -0.05
	}
}

# ---------------------------------------------------------------------------
# Chamasi (Tea-Horse Office) — state monopoly over the tea-for-horses trade
# ---------------------------------------------------------------------------
# From late Tang into the Ming, the chamasi (茶馬司) was a state monopoly
# office regulating the cross-frontier tea-for-horses trade along the
# Tea Horse Road through Yunnan. Officials taxed caravans, fixed
# exchange ratios between brick tea and frontier mounts, and licensed
# approved merchant houses, capturing for the throne the wealth of one
# of medieval Asia's most lucrative trade circuits.
