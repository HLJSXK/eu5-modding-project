###############################################################
# National Destinies -- BAH (Bahmani Sultanate) bureaucracies
#
# Four institutions specific to the Bahmani period (1347-1527),
# pre-fragmentation. Distinct from DEC's post-Bahmani layer
# (Tarafdari, Gawan reforms, Pardesi factions, Vijayanagara
# Nayakas, Persianate Diwan, Deshmukh, Asaf Jahi) -- BAH covers
# the founding-dynasty institutional skeleton: the office of the
# Wakil, Mahmud Gawan's research madrasa, the Sufi-tariqa shrine
# patronage that anchored the sultanate's religious legitimacy,
# and the gold-output Daulatabad-Gulbarga-Bidar mint that paid
# its mercenary cavalry. Auto-modifier grants 1 slot; capital
# adds 1 = 2 effective. Player picks 2 of 4.
#
# 1. Wakil-us-Sultanat       -- the early Bahmani vicegerency
# 2. Madrasa Mahmud Gawan    -- the 1472 Bidar three-storey madrasa
# 3. Bahmani Sufi Patronage  -- Gesudaraz Chishti shrine network
# 4. Daulatabad Royal Mint   -- gold tanka and silver dirham strike
#
# All IDs are tag-prefixed (bah_) to avoid collisions with DEC.
###############################################################


# =====================================================================
# 1. Wakil-us-Sultanat
#
# The Wakil-us-Sultanat was the Bahmani sultan's vicegerent, a single
# great office that combined household management, the inspectorate
# of the cavalry rolls (Amir-i-Jumla functions), and command of the
# central army when the sultan did not march in person. Hasan Gangu
# (Ala-ud-din Bahman Shah, r. 1347-58) staffed the office with Iranian
# and Iraqi emigres rather than Tughluq carryovers; Muhammad Shah I
# (r. 1358-75) used it to lead the campaigns against Vijayanagara and
# Telingana. Concentrating war, household and inspectorate functions
# in one office gave the early Bahmanis their cohesion -- and made the
# Wakil's office the real prize of every faction fight in Gulbarga.
# =====================================================================
bah_wakil_us_sultanat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BAH
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		nobles_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# neutral: a permanent vicegerency keeps cavalry rolls audited and
	#         the household disciplined whether the sultan campaigns or not
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# positive: a competent Wakil leads the Vijayanagara campaigns,
	#         pays the cavalry on time, and keeps the great atraf in line
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_heavy_cavalry_power = 0.05
		monthly_legitimacy = 0.05
		global_max_control = 0.02
	}

	# negative: the office becomes the prize of faction war, the Wakil
	#         intrigues with the atraf against the throne, the cavalry
	#         rolls are padded with phantom horsemen
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.05
		army_heavy_cavalry_maintenance_cost_modifier = 0.10
	}
}


# =====================================================================
# 2. Madrasa Mahmud Gawan
#
# Completed at Bidar in 1472 by Khwaja Mahmud Gawan, the three-storey
# Madrasa Mahmud Gawan was the largest scholarly endowment in the Indian
# sultanates of the fifteenth century. Its plan -- a courtyard ringed by
# 100 student cells, two minarets, library, mosque, and lecture-halls --
# was modelled on the Khargird madrasa of Khurasan; its faculty included
# scholars Gawan recruited personally from Samarkand, Shiraz and Herat.
# The library, said to hold three thousand manuscripts, was the largest
# in southern India; an explosion of its powder magazine in 1696 ruined
# the western minaret, but the surviving structure remains the most
# impressive Persianate monument on the Deccan plateau.
# =====================================================================
bah_madrasa_mahmud_gawan_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = BAH
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		current_age_or_later = { age = age_2_renaissance }
	}

	estates_that_like = {
		clergy_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: a permanent Persianate research academy puts Bidar on the
	#         scholarly map of the Indian Ocean Sunni world
	neutral_modifier = {
		global_burghers_max_literacy = 5
	}

	# positive: faculty recruited from Khurasan, a 3,000-manuscript library,
	#         100 student cells turning out trained katibs and qadis
	positive_modifier = {
		scale = { value = scope:maintenance }
		research_speed_modifier = 0.05
		global_monthly_literacy = 0.025
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# negative: the Khurasani faculty drift home, the library's manuscripts
	#         are pawned for repairs, the locally-born nobility resents the
	#         remaining foreign scholars and their stipends
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.025
		global_pop_assimilation_speed_modifier = -0.10
	}
}


# =====================================================================
# 3. Bahmani Sufi Patronage
#
# The Chishti master Sayyid Muhammad al-Husayni Gesudaraz arrived at
# Gulbarga in 1397 at the invitation of Firuz Shah Bahmani; his
# dargah, built around his death-place in 1422, became the pilgrimage
# centre of the Deccan Sufi establishment. Around it the Bahmanis
# patronized Junaidi shrines at Daulatabad, Qadiri lodges at Bidar
# after 1430, and the Shattari order in Mandu. Endowing waqfs to
# these tariqas anchored the sultanate's legitimacy among Deccani
# Muslims who otherwise looked to the more numerous Hindu majority
# for kinship and patronage; Bahmani sultans were buried alongside
# their Sufi shaykhs at Khuldabad and Gulbarga, not in court mausolea.
# =====================================================================
bah_sufi_tariqa_patronage_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = BAH
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:sunni
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {}

	# neutral: state-endowed tariqas anchor a Muslim ruling stratum
	#         in a Hindu-majority plateau without coerced conversion
	neutral_modifier = {
		monthly_devotion = 0.05
	}

	# positive: pilgrim networks bind the realm religiously, the
	#         Gesudaraz dargah draws conversions to the urus festivals,
	#         peasant grievances reach the throne through the shaykhs
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_pop_conversion_speed_modifier = 0.10
		tolerance_heathen = 1
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# negative: tariqas grow into rival authorities; their waqfs eat
	#         tax-bearing land; the shaykhs at Gulbarga and Khuldabad
	#         issue verdicts the sultans cannot overrule, and the longer
	#         the patronage runs the more land is permanently alienated
	negative_modifier = {
		scale = {
			value = 1
			add = scope:entrenchment
			multiply = { value = 1 subtract = scope:maintenance }
		}
		global_clergy_estate_power = 0.05
		tax_income_efficiency = -0.05
	}
}


# =====================================================================
# 4. Daulatabad Royal Mint
#
# Bahmani gold tankas and silver dirhams were struck principally at
# Daulatabad (the old Tughluq Devagiri), with subsidiary mints at
# Gulbarga and, after 1430, Bidar. The mint-output paid the Arabian
# horse-trade through Goa, Chaul and Dabhol -- the indispensable
# pipeline for the cavalry the Bahmanis fielded against Vijayanagara
# and Khandesh -- and underwrote the Pardesi cavalry stipends that
# bound Iranian and Iraqi nobles to the throne. The standardized
# tanka, weight-controlled to the dirham of the Caliphate, gave
# Bahmani currency Indian Ocean acceptance; debased coin, by
# contrast, was always a sign that the Wakil's office was failing.
# =====================================================================
bah_daulatabad_mint_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BAH
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# neutral: a standardized tanka gives the realm Indian Ocean credit
	neutral_modifier = {
		global_horses_output_modifier = 0.10
	}

	# positive: weight-controlled coin pays the Arabian horse-trade,
	#         underwrites the Pardesi cavalry stipends, and lubricates
	#         the trans-plateau caravan trade
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_light_cavalry_power = 0.05
		trade_land_efficiency = 0.05
		bank_interest = -0.03
	}

	# negative: the mint debases its tanka to pay arrears; horse-traders
	#         demand more coin per stallion; the rural cultivators bear
	#         the brunt of monetized rents that no longer match real prices
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_inflation = 0.005
		global_horses_output_modifier = -0.10
	}
}
