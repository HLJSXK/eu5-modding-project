# National Destinies -- Kingdom of Kongo bureaucracies
# Five historical institutions of the Kongo state, spanning the pre-Christian
# kingdom (c.1390-1491), the Catholic transformation under Afonso I Mvemba a
# Nzinga (r.1509-1542), the late kingdom of Garcia II and Antonio I (1641-1665),
# and the Pedro IV restoration (1709-1718). Auto-modifier grants 2 slots; the
# capital adds 1 = 3 total. The player picks 3 of 5. Each bureaucracy reflects a
# distinct face of Kongolese governance: the mwene-province executive, the
# kanda-elector council, the nzimbu treasury monopoly, the literate royal
# chancery, and the São Salvador Catholic confraternities.

# ---------------------------------------------------------------------------
# Mwene Provinces -- the six royal-appointed mani governorships
# ---------------------------------------------------------------------------
# The kingdom was divided into six provinces -- Soyo, Mpemba, Mbamba, Mpangu,
# Mbata, and Nsundi -- each governed by a mani (lord) appointed by the
# Manikongo at Mbanza Kongo. Provincial governors held delegated military,
# fiscal, and judicial authority but served at the king's pleasure: Afonso I
# and his successors moved governors between provinces to break any single
# mani's local entrenchment. The system gave the throne unprecedented reach
# into the interior of west-central Africa; its weakness was that an unpaid
# or unrotated mani could turn his province into a rival court, as Soyo did
# from the late seventeenth century onward.

kon_mwene_provinces_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price

		multiply = 0.003
	}

	potential = {
		has_or_had_tag = KON
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: the six provincial governorships project royal authority into
	# the basin and give the king local agents in each region
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# positive: rotated mani-governors collect tribute, raise levies, and
	# enforce decrees on the throne's behalf, holding the realm together
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_max_rural_control = 0.03
		global_levy_size_modifier = 0.05
		global_crown_estate_power = 0.10
	}

	# negative: a neglected mani-system breeds parallel courts -- Soyo
	# secedes in all but name, Mbamba dabbles with Portuguese vassalage,
	# and the throne's reach contracts to the area around São Salvador
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.10
		global_max_urban_control = -0.05
	}
}

# ---------------------------------------------------------------------------
# Kanda Council -- matrilineal-aristocratic electors of the Manikongo
# ---------------------------------------------------------------------------
# The mwisikongo aristocracy was organized into matrilineal kanda lineages,
# the most senior of which (the Kimpanzu and Kinlaza after the seventeenth
# century, earlier the Mwene Vunda, Mwene Mbata, and Mwene Soyo) held the
# right to elect each new Manikongo. Succession in Kongo was elective, not
# strictly hereditary -- the king came from a royal kanda but had to be
# chosen by the council of elder lineage-heads. The arrangement bound the
# throne to its great clans; when the council fractured at Mbwila (1665),
# the kingdom fragmented into a forty-year civil war.

# ---------------------------------------------------------------------------
# Nzimbu Royal Monopoly -- shell-currency treasury and head tax
# ---------------------------------------------------------------------------
# The nzimbu were small marine snail shells (Olivancillaria nana) harvested
# from the island of Luanda Bay, where the Manikongo maintained a royal
# fishery monopoly. Sized and counted into pots of fixed denomination (40,
# 100, 250, 400, 500), nzimbu served as the kingdom's currency. The throne
# extracted an annual head tax in nzimbu from every adult male, redistributed
# the shells as titles, salaries, and tribute, and fined Portuguese
# interlopers who tried to harvest from Luanda. The Portuguese seizure of
# Luanda in 1576 and their flooding of the market with cowrie shells from
# Brazil destroyed the royal monopoly within a generation.

kon_nzimbu_monopoly_bureaucracy = {

# ---------------------------------------------------------------------------
# Royal Chancery of Mbanza Kongo -- Afonso's literate diplomatic apparatus
# ---------------------------------------------------------------------------
# Afonso I Mvemba a Nzinga (r.1509-1542) created a permanent literate
# chancery at Mbanza Kongo that produced over four hundred surviving letters
# in Portuguese and Latin, addressed to the kings of Portugal, the popes in
# Rome, and Kongo's own provincial governors. Staffed by mission-trained
# Kongolese scribes and a small corps of Portuguese clerics, the chancery
# kept the realm's diplomatic archive, drafted royal decrees, and sustained
# the embassies to Lisbon and the papal court. It made Kongo the only
# sub-Saharan African kingdom to participate in early-modern European
# diplomacy on its own terms; by Garcia II's reign (1641-1661) the chancery
# was negotiating directly with the Dutch and the Spanish Habsburgs.

	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		clergy_estate
		crown_estate

# ---------------------------------------------------------------------------
# São Salvador Confraternities -- Catholic lay brotherhoods of the capital
# ---------------------------------------------------------------------------
# After Afonso I's renaming of Mbanza Kongo as São Salvador in 1568 and the
# elevation of its cathedral to a bishopric in 1596, the capital developed a
# dense network of Catholic lay confraternities -- the Holy Spirit, Our Lady
# of the Rosary, and Saint Anthony among them -- that organized processions,
# funded the cathedral's upkeep, ransomed enslaved members, and ran charity
# for the urban poor. Mwisikongo nobles competed for officerships; the
# brotherhoods became a parallel civic institution where lineage affiliation
# mattered less than membership status. After the abandonment of São Salvador
# in 1678 the confraternities fragmented but never died; Pedro IV's
