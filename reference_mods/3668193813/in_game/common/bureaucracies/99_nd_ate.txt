# National Destinies -- Ashanti Confederacy bureaucracies
# Five historical institutions of the Asante kingdom that crystallized between
# the founding of the Confederacy under Osei Tutu and Okomfo Anokye (c.1670-
# 1701) and the Anglo-Ashanti collapse of 1900. The auto-modifier grants 2
# slots and Kumasi contributes 1 = 3 total; the player picks 3 of 5. Each
# bureaucracy reflects a distinct face of Ashanti governance: the federal
# Asantemanhyiamu of paramount chiefs, the centralizing Nhenkwaa palace
# functionaries, the Akwantufo road-warden toll administration, the Daban
# gold-dust treasury, and the Kotoko-Adamfo war-and-embassy apparatus.

# ---------------------------------------------------------------------------
# Asantemanhyiamu -- the federal council of paramount chiefs
# ---------------------------------------------------------------------------
# At Odwira each year the Asantehene convened the Asantemanhyiamu, the
# national assembly of the amanhene -- the paramount chiefs of the constituent
# Akan states (Mampong, Dwaben, Bekwai, Kokofu, Nsuta, Kumawu, and others
# added by conquest). The assembly ratified declarations of war, confirmed
# successions, arbitrated inter-state disputes, and approved the levy quotas
# each oman would supply. Its authority was rooted in the founding compact
# of c.1701 by which Osei Tutu's victorious confederates had pooled sovereignty
# under the Sika Dwa Kofi while retaining their hereditary stools. The
# assembly was the kingdom's federal constitution; without its consent no
# Asantehene could compel the chiefs, and through it the kingdom mobilized
# armies that no single oman could field.

ate_asantemanhyiamu_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = ATE
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# neutral: a federal compact under the Sika Dwa Kofi binds every oman's
	# stool into the throne's legitimacy, regardless of how active the
	# council currently is
	neutral_modifier = {
		global_estate_satisfaction_from_legitimacy = 0.05
	}

	# positive: when the assembly meets in good faith at Odwira, the
	# amanhene ratify levy quotas, the throne's prestige flows down the
	# stool hierarchy, and the kingdom mobilizes far beyond Kumasi's
	# direct reach
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		monthly_legitimacy = 0.05
		nobles_estate_target_satisfaction = 0.05
	}

	# negative: an assembly the Asantehene cannot afford to convene
	# loses its compact -- amanhene withdraw their levies, pre-conquest
	# tributaries reassert their independence, and the federation frays
	# into squabbling stools
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.05
		subject_loyalty = -2
	}
}

# ---------------------------------------------------------------------------
# Nhenkwaa -- the palace functionaries of the Asantehene's household
# ---------------------------------------------------------------------------
# The Nhenkwaa were a permanent caste of palace servants attached directly
# to the Asantehene at Manhyia: gun-bearers (atumtufo), drummers (akyeremade),
# horn-blowers (asokwafo), umbrella-carriers, executioners (abrafo), torch-
# bearers, hammock-carriers, and the dreaded sumankwafo who maintained the
# kingdom's protective talismans. Recruited largely from war captives and
# the unfree, they were legally outside the matrilineal clan structure and
# owed everything to the throne; many rose to be confidential agents and
# ambassadors despite their servile origins. By making the palace household
# a self-contained administrative apparatus loyal only to the Asantehene,
# successive kings -- especially Opoku Ware (r.1720-1750) and Osei Kwadwo
# (r.1764-1777) -- bypassed the amanhene and built an executive capable of
# running the kingdom without the chiefs' consent.

ate_nhenkwaa_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = ATE
		has_game_rule = nd_bureaucracies_enabled
	}

	# the Nhenkwaa are the household of the Asantehene at Manhyia --
	# they exist only where the throne sits
	allow = {
		owns_or_non_sovereign_subject_owns = location:kumasi
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: a permanent palace household run by men outside the
	# matrilineal clan order projects the throne's executive reach
	# into every corner of the kingdom
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# positive: gun-bearers, executioners, and confidential agents
	# loyal only to the Asantehene give the throne an instrument of
	# state that no chief can divert, and unfree palace servants
	# rising to high office crack the matrilineal closed shop
	positive_modifier = {

		crown_estate_target_satisfaction = 0.05
		monthly_legitimacy = 0.05
		global_peasant_enfranchisment = 0.05
	}

	# negative: an unpaid palace household becomes a parasitic court
	# -- the abrafo executioners take fees for sparing the condemned,
	# the amanhene seethe at servile upstarts, and the kingdom drifts
	# toward the dynastic instability that prefigured Mensa Bonsu's
	# 1883 deposition
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		stability_decay = 0.00025
	}
}

# ---------------------------------------------------------------------------
# Akwantufo -- the great-roads wardens and toll administration
# ---------------------------------------------------------------------------
# Eight great roads radiated from Kumasi to the frontier provinces, hewn
# through the rainforest and maintained as the arteries of the Confederacy.
# Each road was patrolled by akwanmofo (road wardens) under an akwansrafohene
# who reported to the throne; toll-keepers (nkwansrafo) at the frontier of
# every oman levied duties on cola nuts moving north, kente cloth and gold
# moving south, captives, salt, and European trade goods. The roads doubled
# as the levy-call infrastructure: when the Asantehene declared war the
# atumtufo standard was carried down each road and every district along it
# mustered. Bowdich's 1817 embassy described a pavement of beaten earth
# kept clear of vegetation -- a sophisticated public-works system unique
# among the Guinea Coast kingdoms. The roads were what made Kumasi a
# real capital rather than just a paramount chief's seat.

	allow = {}

# ---------------------------------------------------------------------------
# Daban -- the royal gold-dust treasury and weights administration
# ---------------------------------------------------------------------------
# Gold dust (sika futuro) was the Confederacy's currency, weighed against
# brass abrammuo weights cast in geometric and figurative forms. The Daban,
# the royal treasury at Manhyia, was administered by the Gyaasewahene as
# chief treasurer with the Sanaahene as his deputy; under them the fotosanfohene
# inspected weights and the sannaa kept the daily accounts. The throne
# enforced a national standard set of weights -- merchants caught with
# false abrammuo were liable to confiscation and worse -- which made
# long-distance trade between Akan markets possible without constant dispute.
# All gold mined in the kingdom owed a royal share to the Daban, all
# unclaimed estates escheated to it, and the great fines levied at the
# Asantehene's court flowed into its strongrooms. Bowdich saw piles of
# dust kept in elephant-skin sacks that he estimated at hundreds of pounds
# of gold per chamber -- a fiscal apparatus on par with any in early
# modern West Africa.

	allow = {}

	estates_that_like = {
		burghers_estate

# ---------------------------------------------------------------------------
# Kotoko Council and Adamfo -- the war council and royal-friend embassies
# ---------------------------------------------------------------------------
# The Kotoko was the inner war council of the Asantehene, named for the
# porcupine (kotoko: "if you kill a thousand, a thousand will come") that
# was the kingdom's military emblem. It comprised the great captains --
# the Bantamahene as senior war commander, the Adontenhene of the vanguard,
# the Nifahene of the right wing, the Benkumhene of the left, the Kyidomhene
# of the rear -- and met to plan campaigns under the Bogyawe tree. Parallel
# to the Kotoko ran the adamfo system: each tributary state and each
# major external power (Dagomba, Gyaman, the European castles at Cabo
# Corso and Elmina) was assigned a "royal friend" at Kumasi -- a senior
# courtier who handled all diplomatic traffic with that party, hosted
# its envoys, and was personally liable for any dispute. The combined
# institution made Ashanti foreign policy unusually professionalized for
# its time: every ambassador had a known counterpart at court, and every
# campaign was planned by a fixed roster of titled commanders rather than
# whichever chief happened to be at hand.
