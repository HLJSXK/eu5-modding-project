# National Destinies -- Danubian Monarchy bureaucracies
# 6 historically-grounded institutions of the Habsburg composite state,
# spanning the Ferdinandine settlement (1527-1556), the post-White-Mountain
# confessional reconquest (1620s), the Inner Austrian frontier (1578+), and
# the Theresian centralization (1749). The auto-modifier grants 2 slots; the
# capital adds 1 = 3 effective. Player picks 3 of 6.
#
# Identity: DNM is the multi-crown Danubian super-state. Its defining
# institutional question is the composite-monarchy tension -- a single
# Vienna fisc and privy council pulling against three crowns each with
# their own estates, diets and law. The bureaucracies put both poles of
# that struggle in the player's hands: the centralizing Vienna organs
# (Kameralhoheit, Geheime Konferenz, Directorium) against the
# estate-preserving constituent Landstaende, with the frontier and the
# Counter-Reformation as the two great special apparatuses.
#
# IDs use distinct institutional names where a stem is shared with an
# existing nd_dnm_ advance/building (Hofkammer, Militaergrenze,
# United Chancellery, Jesuit colleges) to avoid id/loc collisions.

# 1. Kameralhoheit -- the cameral sovereignty of the Vienna Hofkammer.
#    Ferdinand I made the Court Chamber independent of the Aulic Council
#    in 1527 and folded the Lower Austrian, Tyrolean, Bohemian and
#    Hungarian land-chambers under one fisc; Maria Theresa raised it to
#    the supreme authority for finance, mining, commerce and transport.
#    Revenue that the crown-land estates had voted and audited was now
#    assessed and collected from Vienna -- the single sharpest reduction
#    of estate fiscal autonomy in the monarchy's history.
nd_dnm_kameralhoheit_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		tag = DNM
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	neutral_modifier = {
		global_max_urban_control = 0.05
	}

	# Working: the fisc captures mining and commercial revenue and the
	# crown's hand reaches into the crown lands directly.
	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = 0.05
		trade_land_efficiency = 0.05
		global_crown_estate_power = 0.05
	}

	# Decaying: the chamber clerks slide into rent-seeking and the coinage
	# is debased to cover the gap -- the classic fate of a starved fisc.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_inflation = 0.005
		nobles_estate_satisfaction_decay = 0.015
	}
}

# 2. Geheime Konferenz -- the Privy Conference. Under Leopold I a handful

#    wider Privy Council entirely: a small aristocratic apex bound to the
#    dynasty, fast and secret, but a closed door to the towns.

		crown_estate
		nobles_estate

		burghers_estate

	neutral_modifier = {

#    tax-exempt and tolerated, in exchange for lifelong border service.
#    A self-feeding wall of free militiamen answerable to Vienna and not
#    to any seigneur -- the institutional spine of the Antemurale path.
