###############################################################
# National Destinies -- BYZ supplemental bureaucracies
#
# Vanilla EU5 1.2 (DLC d008_fate_of_the_phoenix) ships 11 BYZ
# bureaucracies in game/in_game/common/bureaucracies/byz.txt.
# These FIVE are non-overlapping ADDITIONS modeling Byzantine
# institutions vanilla doesn't cover:
#
#   1. Synodos Endemousa  -- Patriarchate's standing synod
#   2. Genikon Logothesion -- General Treasury / fiscal bureau
#   3. Mesazon            -- Late-Byzantine chief minister
#   4. Eparchate          -- Eparch of Constantinople (city prefect)
#   5. Mystikos & Asecretis -- Imperial chancery / secret secretariat
#
# IDs are prefixed byz_ but do NOT collide with vanilla IDs.
###############################################################


# =====================================================================
# Synodos Endemousa -- the Standing Synod of Constantinople
#
# The endemousa synodos was the permanent council of bishops resident
# in the imperial capital, presided over by the Ecumenical Patriarch.
# It ruled on canon law, dogma, marriage cases of the great families,
# and questions affecting the Orthodox world from the Balkans to
# Antioch. Distinct from the Patriarchal Academy (a building) and from
# vanilla's Romanitas/Court Eunuchs bureaucracies, the synod is the
# formal corporate body of the Orthodox episcopate -- a parallel
# legitimacy outside the throne.
# =====================================================================
byz_synodos_endemousa_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BYZ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:constantinople
	}

	estates_that_like = {
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	#neutral
	neutral_modifier = {
		clergy_estate_allowed_in_cabinet = yes
	}

	#positive: synod functioning -- ecumenical reach, Orthodox cohesion
	positive_modifier = {
		scale = { value = scope:maintenance }
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
		global_pop_conversion_speed_modifier = 0.10
		global_clergy_max_literacy = 5
	}

	#negative: synod neglected -- bishops drift, factionalism, schism
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.02
		stability_decay = 0.00025
	}
}


# =====================================================================
# Genikon Logothesion -- the General Treasury
#
# The genikon was the central fiscal bureau of the middle-Byzantine
# empire, headed by the logothetes tou genikou, with chartoularioi
# at the centre and epoptai auditing the themes. It assessed land,
# collected the kapnikon hearth-tax, audited arrears, and acted as
# a tax-court. Distinct from the Basilika (vanilla, law code) and
# from Allelengyon (vanilla, peasant collective burden); this is
