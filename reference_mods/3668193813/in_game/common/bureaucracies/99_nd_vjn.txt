# National Destinies -- Vijayanagara (Karnata Empire) bureaucracies
# Four institutional layers of the Hindu southern empire (1336-1646).
# Auto-modifier grants 2 slots; capital adds 1 = 3 total. Player picks 3 of 4.
#
# Each models a distinct apparatus of the Karnata-Vijayanagara state and
# is deliberately distinct from the pan-Indian HIN bureaucracies: this is
# the southern Dravidian empire's own institutional fingerprint, anchored
# at Hampi (Vijayanagara) and the Coromandel ports.

# 1. Pampa-Virupaksha Devasthanam -- the imperial Saiva temple complex at
#    Hampi, with the god Virupaksha (a form of Shiva) installed as the
#    tutelary deity and royal sign-manual of the Sangama dynasty. From
#    Harihara I onward, royal grants and edicts were issued "Sri-Virupaksha"
#    (in the god's name); the emperor ruled as the deity's deputy. The
#    Pampakshetra ritual centered the Tungabhadra valley as sacred geography.
vjn_pampa_virupaksha_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = VJN
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Sri-Virupaksha sign-manual presupposes possession of Hampi.
	allow = {
		owns = location:vijayanagar
		religion = religion:hindu
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# Royal authority is signed in the god's name -- legitimacy is
	# structurally underwritten by the temple's standing presence.
	neutral_modifier = {
		monthly_legitimacy = 0.05
	}

	# Working: Saiva court ritual binds Brahmin clergy and crown into a
	# single sacred-political compact; tolerance of own faith hardens
	# the social cement of the realm.
	positive_modifier = {
		scale = { value = scope:maintenance }
		tolerance_own = 1
		monthly_devotion = 0.05
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: the Saiva establishment becomes a closed donative
	# economy. Endowments calcify into rentier income, mercantile
	# guilds resent the temple's tax-free status, and pilgrim taxes
	# extract from urban traders.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}

# 2. Diwan-Kacheri -- the imperial chancery and revenue bureau formalized
#    under Krishnadeva Raya (r. 1509-1529) and codified in his Telugu
#    political treatise the Amuktamalyada. The Pradhani (chief minister)
#    headed eight ministerial portfolios; the Rayasam (royal scribes)
#    issued sasanas (copperplate edicts) across the realm. Distinct from
#    Mughal bureaucracy in retaining strong oral-Sanskritic legal tradition
#    alongside Telugu-language administration.
vjn_diwan_kacheri_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = VJN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Eight ministerial portfolios route business through the throne
	# regardless of upkeep -- the form of central administration is
	# always-on once instituted.
	neutral_modifier = {
		country_cabinet_efficiency = 0.10
	}

	# Working: the Rayasam scribal corps and Pradhani's court process
	# revenue, justice, and edict at imperial scale; literate Brahmin
	# and Komati merchant-officers staff the bureaus.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_burghers_max_literacy = 8
		legislative_efficiency = 0.10
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: Nayaka warrior-aristocrats bypass the chancery entirely,
	# issuing their own writs from amaram seats; central edicts are
	# ignored on the periphery, and the bureaucracy's own scribes
	# extract bribes from petitioners.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		num_nd_provincial_estates = 1
	}
}

# 3. Tirumala Devasthanam -- the great Vaishnava temple of Venkateswara
#    on the Tirumala hills above Tirupati. Krishnadeva Raya's seven
#    visits and lavish endowments (1513-1521) made the shrine the
#    Vaishnava counterpart to the Saiva Pampa-Virupaksha at Hampi.
#    Tirumala became the wealthiest temple in early-modern South India
#    through pilgrim hundi (cash offerings), grain reserves, and a
#    parallel temple-banking system that lent to merchants and nobles.
vjn_tirumala_devasthanam_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	# Maintaining the wealthiest pilgrimage shrine in the realm scales
	# with the size of the realm itself.
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = VJN
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Tirumala shrine itself must be owned for the devasthanam to operate.
	allow = {
		owns = location:tirupati
		religion = religion:hindu
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The pilgrim economy beneath the hill perpetually draws pop -- the
	# shrine's mere presence enlarges its temple-cities.
	neutral_modifier = {
		global_clergy_city_desired_pop_scaled = 0.10
	}

	# Working: hundi offerings circulate as low-interest temple credit;
	# pilgrim crowds redistribute prosperity across the Coromandel
	# hinterland; peasant cultivators benefit from temple grain reserves.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_monthly_prosperity = 0.005
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		trade_income = 0.10
	}

	# Decaying: temple wealth becomes a target. Without active royal
	# protection, Nayaka warlords skim hundi receipts, the great
	# endowment is dissipated into private retinues, and pilgrim routes
	# become unsafe.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		amount_looted_modifier = 0.10
		clergy_estate_satisfaction_decay = 0.020
	}
}

# 4. Coromandel Customs -- the eastern-coast revenue apparatus
#    administering Pulicat, Negapatnam, San Thome (Mylapore), and
#    Masulipatnam. The Vijayanagara state collected sungam (transit
#    duties) and licensed Tamil and Telugu merchant guilds (the
#    Ayyavole 500 and the Manigramam) to handle long-distance Indian
#    Ocean trade. Portuguese, Persian, and later Dutch factors paid
#    customs through the imperial Adhikari (port-governor) appointed
#    from Hampi; the eastern textile economy reached as far as Aceh,
#    Pegu, and the Spice Islands.
vjn_coromandel_customs_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = VJN
		has_game_rule = nd_bureaucracies_enabled
	}

	# Coromandel customs requires possession of the principal eastern port.
	allow = {
		owns = location:pazhaverkadu
	}

	estates_that_like = {
		burghers_estate
		crown_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# Imperial port-governors (Adhikari) collect sungam duties under
	# standing royal commission, irrespective of upkeep.
	neutral_modifier = {
		trade_income = 0.10
	}

	# Working: the Ayyavole 500 and Manigramam merchant guilds operate
	# under licensed protection; merchant capacity expands and the
	# imperial maritime presence radiates from Pulicat to Malacca.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_merchant_capacity_modifier = 0.10
		global_maritime_presence_modifier = 0.10
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: customs farming devolves to the Adhikaris as private
	# rent; foreign factors smuggle around the duties; Brahmin clerical
	# opinion turns against the cosmopolitan port economy and its ritual
	# pollution by sea-trading communities.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}
