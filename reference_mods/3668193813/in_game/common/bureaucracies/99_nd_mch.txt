###############################################################
# National Destinies -- MCH (Manchu / Qing dynasty) bureaucracies
#
# Six institutions of the Aisin-Gioro Qing state, distinct from
# the Han Chinese institutional palette (mandarinate, censorate,
# eunuch court). The Qing's defining feature was a DUAL apparatus:
# the inherited Ming Six Boards governed the Han core, while a
# parallel Manchu apparatus (Banners, Neiwufu, Lifan Yuan, Junji
# Chu) bound the dynasty's ruling caste, frontier subjects, and
# inner-court machinery to the Aisin-Gioro house directly.
#
# Auto-modifier (nd_mch_bureaucracy_authority) grants 2 slots;
# capital adds 1 = 3 total. Player picks 3 of 6.
###############################################################


# 1. Eight Banners (jakūn gūsa) -- the dual military-administrative
#    system Nurhaci codified in 1615. Eight Manchu banners (later
#    eight Mongol and eight Han banners, twenty-four total) enrolled
#    every banner subject as a hereditary soldier-household, with
#    pay, land allotments, and judicial separation from the Han
#    civilian population. The banners were the dynasty's caste
#    army and its census in one institution.
mch_eight_banners_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = MCH
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		tribes_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# neutral: banner households are exempt from civilian conscription
	neutral_modifier = {
		burghers_estate_blocked_from_leading_military = yes
	}

	# positive: banners on the rolls -- a standing caste army
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		tribes_estate_levy_size = 1
		land_morale_modifier = 0.05
	}

	# negative: bannermen draw stipends without service, the
	# rolls bloat with absentees and Han burghers buy in
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_inflation = 0.005
		levy_recovery_modifier = -0.20
	}
}
