###############################################################
# National Destinies -- YUA (Yuan dynasty) bureaucracies
#
# Six Yuan-era institutions, distinct from MGE (the unified
# Mongol Empire) and CHI (the Han-Chinese dynastic apparatus).
# YUA models what the Yuan invented or recombined uniquely:
# the dual civil/military secretariats, the elevated Censorate,
# the four-tier ethnic caste, the phonetic universal script
# commissioned from Phagpa, and the Yangzhou salt fisc.
#
# These do NOT duplicate MGE (Yasa, Kuriltai, Keshig, Yam,
# Darughachi, Yelu Chucai) nor CHI (Keju, Six Ministries,
# Censorate-Du-Cha, Hanlin, Grand Secretariat, Eunuch, Lifan
# Yuan). The Yuan Censorate (Yushi Tai) IS distinct from the
# Ming/Qing Du Cha Yuan: it was elevated to coequal status
# with the secretariats from 1268, not a mere ministry.
#
# Auto-modifier (nd_yua_bureaucracy_authority) grants 2 slots;
# capital adds 1 = 3 total. Player picks 3 of 6.
#
# IDs prefixed yua_; loc keys match the bare bureaucracy ID.
###############################################################


# =====================================================================
# Zhongshu Sheng -- the Central Secretariat
#
# Established by Khubilai Khan in 1260 and formalised in 1271, the
# Zhongshu Sheng was the apex civil organ of the Yuan state. It
# differed from Tang and Song precedent in two crucial respects: its
# Right and Left Chancellors (Cheng-xiang) were almost always
# Mongols or Semuren, never Han-ren; and its branch secretariats
# (Xing Zhongshu Sheng, the "moving" secretariats that became the
# direct ancestors of the Chinese province) institutionalised what
# had been ad hoc field administration. It absorbed the Liu Bu
# (Six Ministries) as its subordinates rather than allowing them
# the autonomy they enjoyed under earlier dynasties. The
# Zhongshu Sheng was the institutional bridge by which steppe
# conquest was transmuted into governable continental empire.
# =====================================================================
yua_zhongshu_sheng_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = YUA
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Central Secretariat operated from the imperial capital --
	# Dadu (Khanbaliq) was its seat from 1271, with Shangdu the
	# summer capital. Without one, there is no Zhongshu Sheng.
	allow = {
		OR = {
			owns = location:dadu
			owns = location:shangdu
		}
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# always-on: branch secretariats project the centre into the provinces
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# functioning: chancellors process memorials, Han clerks draft
	# decrees, the Six Ministries answer to a single coordinating organ
	positive_modifier = {
		scale = { value = scope:maintenance }
		country_cabinet_efficiency = 0.10
		global_monthly_control = 0.004
		global_max_control = 0.02
	}

	# neglected: chancellor factions, the secretariat freezes under
	# the weight of memorials, branch governors act as warlords
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_decay = 0.00025
		global_separatism = 0.10
	}
}


# =====================================================================
# Shumi Yuan -- the Bureau of Military Affairs
#
# The Shumi Yuan inherited its name from the Tang and Song privy
# councils but was rebuilt by Khubilai as a deliberately segregated
# military command. Where the Zhongshu Sheng handled civil affairs
# with Han participation, the Shumi Yuan was an exclusively
# Mongol-Semuren preserve: it commanded the keshig units, the
# tammachi border garrisons, and the imperial tumen that screened
# the steppe frontier. By 1287 it was directly co-equal with the
# civil secretariat, answering only to the khan. The institution
# expressed a structural Yuan principle: arms remain in the hands
# of the conquering people. No Han-ren held senior Shumi Yuan
# rank in the dynasty's first century.

# =====================================================================
# Yushi Tai -- the Yuan Censorate
#
# The Yushi Tai existed before the Yuan, but Khubilai's 1268 decree
# elevated it to a position no earlier dynasty had granted: full
# coequal of the Zhongshu Sheng and Shumi Yuan, the "third great
# administration" of the realm. Its Censor-in-Chief (Yushi Da-fu)
# could memorialise the throne against any official including
# the chancellors themselves; its provincial Branch Censorates
# (Xing Yushi Tai) at Shangdu and Jiangnan paralleled the branch
# secretariats and held coordinate impeachment authority. Mongol
# practice prized the censorate as a check by the khan on his own
# administration -- but in Yuan hands it also collided endlessly
# with the Mongol nobility, whose ortoqs and appanages the
# censors investigated with vigour. By the Toghon Temur reign
# (1333+), censorial impeachments had become a major theatre of
# court factional warfare.

# =====================================================================
# Semu Caste Hierarchy -- the four-tier ethnic order
#
# The defining political innovation -- and burden -- of Yuan rule
# was the formal four-tier ethnic classification: Mongols at the
# top, then Semuren ("various-eyed people": Central Asians,
# Persians, Tanguts, Tibetans, Uighurs, Turks, even some Europeans),
# then Han-ren (former Jin subjects of north China), then Nan-ren
# (former Southern Song subjects of the conquered south). The tiers
# determined access to senior office, judicial standing, weapon
# privileges, marriage rights, and tax assessment. Mongols and
# Semuren held all the chancellor seats and most provincial
# governorships; Han-ren and Nan-ren were structurally locked out
# of the upper bureaucracy. The system gave the throne a vast
# pool of foreign-born loyalists (Marco Polo, Rashid al-Din's
# correspondent Bolad Aqa, the Tibetan Phagpa, the Uighur
# Mengsus) who owed everything to the dynasty. It also seeded
# the Han-ren rebellions that ended Yuan rule.
# =====================================================================

# =====================================================================
# Phags-pa Script Apparatus -- the imperial universal alphabet
#
# In 1269 Khubilai Khan commissioned the young Tibetan lama Drogon
# Chogyal Phagpa to create a single phonetic script that could
# transcribe every language of the empire -- Mongolian, Chinese,
# Tibetan, Sanskrit, Uighur, Persian -- in one unified set of
# squared letters. Phagpa, working in Sanskrit grammatical
# tradition, produced a 41-letter alphabet derived from Tibetan
# but written vertically in the Mongol manner. Imperial decree
# made it the official script of the dynasty: state seals,
# coinage, the great paiza tablets, and the parallel-text edicts
# of the Bureau of National Languages all bore Phags-pa lettering.
# The script never displaced Chinese in everyday use, but its
# institutional function -- as a deliberate imperial-multicultural
# project, distinct from any Han precedent -- expressed Yuan
# universalism more clearly than any other policy. The Bureau
# also served as the institutional scaffold for Phagpa's role
# as Imperial Preceptor (Dishi), the Tibetan Buddhist office
# that linked Yuan throne to Sa-skya monastery.
# =====================================================================
yua_phags_pa_script_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price

		multiply = 0.003
	}

	potential = {
		has_or_had_tag = YUA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		clergy_estate
		crown_estate
	}

# =====================================================================
# Yan Zheng -- the Yangzhou Salt Monopoly
#
# Yuan fiscal policy depended -- to a degree no earlier dynasty
# matched -- on the salt monopoly administered from Yangzhou.
# Beginning under Khubilai and pressed to extreme heights by the
# minister Lu Shi-rong (executed 1285) and Sangha (executed 1291),
# the Yan Yun Si salt commission monopolised production at the
# coastal salterns of Liang-zhe and Liang-huai, distributed the
# product through a system of licensed merchants paid in
# certificate-warrants (yan-yin), and remitted the revenue
# directly to the Imperial Treasury (Tai-fu Si). Salt revenue
# accounted for as much as 80% of Yuan central government
# income at peak. The dependency was the dynasty's strength
# and its weakness: when the Yangzi rebellions disrupted
# southern salt extraction in the 1350s, central revenue
# collapsed within a decade.
# =====================================================================
yua_yan_zheng_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = YUA
		has_game_rule = nd_bureaucracies_enabled
	}

	# The salt monopoly required southern-Chinese coastal salterns:
	# Yangzhou (the commission's seat), Hangzhou (the Liang-zhe
	# salterns), or the Yangzi metropolis Jiangning (Nanjing).
	allow = {
		OR = {
			owns = location:yangzhou
			owns = location:hangzhou
			owns = location:jiangning
		}
	}

	estates_that_like = {

		peasants_estate

	neutral_modifier = {
