# National Destinies -- Malacca-specific bureaucracies
# These institutions belong to the Malaccan layer above and beyond the
# pan-Malay apparatus already covered by MSA. Where MSA codifies the
# four-pillar officialdom, the Hukum Kanun, the Shahbandaria, and the
# Orang Laut compact as the durable legacy of the Malay world, MLC
# isolates the institutions that were specifically Malaccan: the
# treasurer-customs-master Penghulu Bendahari, the Temenggung's bench,
# the Ming tributary concession at Bukit China, and the negeri-tribute
# apparatus that bound Pahang, Kelantan, Trengganu, Kedah, Indragiri,
# and Pasai to the throne at the Sungai Bertam.
#
# Auto-modifier grants 2 slots; capital adds 1 = 3 effective.
# Player picks 3 of 4. All IDs prefixed `mlc_` to avoid collision with MSA.

# 1. Penghulu Bendahari -- the third pillar of the Melaka officialdom,
#    treasurer of the realm and master of customs. The Bendahari
#    administered the royal treasury, the weights and measures (the
#    Melakan kati of 605 grams and the bahara of 200 kati became the
#    Indian Ocean reference standards), the syahbandar revenues from
#    the four foreign quarters, and the harbour-master's office that
#    weighed every cargo of pepper, cloves, tin, and gold-dust crossing
#    the wharf. He commanded a corps of literate clerks who kept ledgers
#    in Jawi script and reported the day's customs to the istana each
#    evening. Tome Pires (Suma Oriental, c. 1515) describes Melaka's
#    customs administration as the most rationally ordered in all the
#    Indies, with published rates of 3-6% ad valorem by goods class
#    and a single point of payment at the Bendahari's office.
mlc_penghulu_bendahari_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MLC
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Establishing the Bendahari's office gives the realm a single
	# customs-house, a single ledger, a single set of weights: every
	# cargo crossing the wharf is weighed on the royal kati, every
	# duty is paid at the published rate, and every coin of customs
	# revenue passes through one ledger before reaching the istana.
	neutral_modifier = {
		create_market_cost_modifier = -0.15
	}

	# Functioning -- the Bendahari at his desk, the harbour-master's
	# clerks weighing every bahara of pepper at the wharf, the customs
	# rate published and uniformly enforced, the daily ledger reaching
	# the sultan each evening -- the realm captures the entrepot's
	# transit revenue in a way no rival on the Straits can match. The
	# treasury is full, the merchant houses pay willingly knowing the
	# rate is fixed, and the tax-farms of the lesser ports are starved
	# of opportunity.
	positive_modifier = {
		scale = { value = scope:maintenance }
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
		burghers_estate_max_tax = 0.05
		global_max_urban_control = 0.05
	}

	# When the office decays into a private patrimony -- the Bendahari's
	# clerks taking commission from the same merchants they audit, the
	# weights at the wharf quietly adjusted in favour of the great
	# noble warehouses, the customs ledger rewritten to conceal what
	# the chief minister has skimmed -- the merchant capital flees to
	# Pasai, Pidie, or Aceh. Revenue collapses; the noble houses that
	# captured the office grow rich on what the throne loses.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.015
		nobles_estate_satisfaction_decay = -0.010
	}
}

# 2. Temenggung Bench -- the fourth pillar of the Melaka officialdom,
#    chief of palace security, magistrate of capital crimes, and
#    keeper of the royal gaol. Where the Bendahara directed civil
#    administration and the Bendahari handled fiscal affairs, the
#    Temenggung commanded the palace bodyguard, ran the istana's
#    prison, and presided over the high court for offences that the
#    qadi's shari'a court could not try (treason, lese-majeste, the
#    durhaka of disobedience to the throne, and the cluster of capital
#    matters reserved by adat to the sultan's pleasure). The
#    Temenggung's writ ran through the keris of his hulubalang
#    deputies, who policed the city, suppressed market disorders, and
#    enforced sumptuary rules against commoners wearing yellow or
#    riding sedan-chairs reserved for royalty. The post was hereditary
#    in the Bendahara faction's allied lineage.
mlc_temenggung_bench_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MLC
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Vesting the Temenggung with command of the istana's hulubalang and
	# magistracy of capital crimes places the throne's high justice in a
	# permanent office: durhaka and lese-majeste come before the sultan's
	# bench rather than the great noble councils, and the city's nightly
	# patrol answers to the istana rather than to the Bendahara's house.
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Honoured -- the Temenggung's hulubalang on every street-corner of
	# Bukit Larangan, the gaol kept in good order, the high court
	# meeting weekly to dispose of treason and durhaka cases, the
	# sumptuary rules enforced -- the realm enjoys an internal order
	# unusual for a great entrepot: the istana's writ reaches the
	# alleys of the river-quarter, the peasantry of the Bertam and
	# Kesang valleys feels the throne's protection, and the noble
	# houses learn that the sultan's high justice is not theirs to bend.
	positive_modifier = {
		scale = { value = scope:maintenance }
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		global_max_control = 0.02
		stability_investment = small_stability_investment
	}

	# When the office is captured by a Bendahara faction or sold to the
	# highest noble bidder -- the hulubalang turned to private intimidation,
	# the high court convicting the throne's critics on confected charges
	# of durhaka, the gaol filling with merchants who refused the chief
	# minister's price -- the apparatus that should defend the throne
	# becomes the noble faction's own keris, and no coast-trader's
	# household sleeps easy in Melaka.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_cost_efficiency = -0.1
		global_separatism = 0.05
	}
}

# 3. Bukit China Imperial Concession -- the Ming tributary nexus
#    centred on the hill of Bukit China north of the Sungai Melaka,
#    granted (according to the Sejarah Melayu) by Sultan Mansur Shah
#    to the Princess Hang Li Po and her retinue of five hundred upon
#    her marriage to the Sultan in 1459. The hill became the seat of
#    the Chinese tributary establishment: the residence of the Ming
#    envoys who carried the Yongle and Xuande emperors' patents to
#    successive sultans (Parameswara received the patent of kingship
#    in 1411, Mahmud Shah received the last embassy in 1481), the
#    quarter of the Sinkeh Chinese settlers, and the diplomatic-
#    commercial gateway through which Melaka secured Ming protection
#    against Siamese encroachment. Cheng Ho's seven voyages
#    (1405-1433) used Melaka as the Indian-Ocean staging-base,
#    leaving substantial fortifications and a permanent treaty
#    relationship that lasted until the Portuguese conquest.
mlc_bukit_china_concession_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = MLC
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:malacca
	}

	estates_that_like = {
		crown_estate
		dhimmi_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# Receiving the Ming envoy on Bukit China and lodging the Sinkeh
	# settlers in their chartered quarter binds the throne to an
	# Indian-Ocean diplomatic order in which the Son of Heaven's patent
	# is the senior protection: the Siamese vassal-claim is null where
	# Ming endorsement runs, the Champa and Brunei courts treat
	# Melaka's envoys as their senior, and the junk-trade flows in
	# protected convoys.
	neutral_modifier = {
		diplomatic_reputation = 1
	}

	# Functioning -- the Ming envoy received with proper ceremony each
	# decade, the Hang Li Po line maintained at court, the Sinkeh
	# quarter prosperous and the Chinese junks coming south in their
	# protected season -- the realm is shielded from its principal
	# threat (Siamese paramountcy from Ayutthaya) by the most powerful
	# court in the world, and no rival on the Straits can match the
	# diplomatic standing this confers.
	positive_modifier = {
		scale = { value = scope:maintenance }
		dhimmi_estate_target_satisfaction = small_permanent_target_satisfaction
		improve_relation_impact = 0.15
		diplomatic_range_modifier = 0.20
	}

	# When the concession decays -- the ulama at court denouncing the
	# tolerated kafir hill, the Sinkeh quarter forced to convert or
	# depart, the envoy received with insufficient ceremony, the
	# Sufi clergy pressing for a properly Islamic succession in place
	# of the Hang Li Po line -- the Ming protection lapses, the Siamese
	# vassal-claim revives, and the diplomatic standing that kept
	# Melaka inviolable for a century evaporates in a single embassy.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.020
		dhimmi_estate_satisfaction_decay = 0.015
	}
}

# 4. Negeri Tributary Apparatus -- the institutional network by which
#    Sultan Mansur Shah (r. 1459-1477) and Bendahara Tun Perak bound
#    the lesser Malay states to the Melaka throne under the negeri
#    (vassal-realm) status: Pahang from 1454, Kedah from c. 1470,
#    Kelantan and Trengganu from the 1460s, Indragiri and Kampar on
#    Sumatra from the 1450s-1470s, and Pasai from 1470 (under
#    indirect tribute after Tun Perak's 1474 expedition). Each
#    negeri's raja kept his throne in exchange for an annual bunga
#    mas dan perak tribute (a small tree of beaten gold and silver
#    flowers, of token weight but heavy ritual obligation), naval
#    levy in time of war, and acceptance of Melaka's Bendahara as
#    final arbiter in succession disputes. The apparatus was
#    administered by a roster of itinerant bentara (heralds) who
#    travelled the negeri annually to receive the bunga mas, and by
#    a permanent body of resident envoys at the Bertam istana
#    representing each tributary court.
mlc_negeri_tributary_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MLC
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# Receiving the bunga mas from the bentara each season and lodging
	# the negeri's resident envoys at court turns the loose Sumatran
	# and peninsular hierarchy into a permanent ritual order: each raja
	# is reminded annually of his place beneath the Melaka throne, and
	# the Bendahara's mediation in succession disputes is accepted as
	# the convention of the entire archipelago.
	neutral_modifier = {
		subject_pays_tributary_cost_modifier = -0.20
	}

	# Functioning -- the bentara despatched on schedule, the bunga mas
	# arriving from every negeri at the Bertam court, the resident
	# envoys housed in their separate compounds west of the istana,
	# the Bendahara's word taken as final in Pahang and Kedah and
	# Kampar succession quarrels -- the realm operates as a Straits-
	# wide hegemony at the cost of a single chancery's correspondence,
	# and the lesser rajas raise their levies for Melaka's wars without
	# coercion.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		diplomatic_capacity_modifier = 0.20
		annexation_speed_modifier = 0.20
	}

	# When the apparatus decays -- the bentara unpaid and travelling no
	# further than the Sungai Muar, the bunga mas withheld first by
	# Indragiri then by Kedah, the resident envoys recalled by their
	# rajas without replacement, the Bendahara's mediation refused in a
	# Pahang succession that the negeri-court settles by its own keris
	# -- the network unravels by stages, the merchant capital prices in
	# the loss of throughput, and Aceh and Siam hover at the unguarded
	# edges of what was once a Straits-wide order.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.015
		stability_cost_efficiency = -0.1
	}
}
