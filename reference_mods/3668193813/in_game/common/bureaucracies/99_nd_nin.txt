###############################################################
# National Destinies -- NIN (Assyrian Empire) bureaucracies
#
# Tier 2 of the Syriac restoration chain (BNH -> NIN). L4
# imperial formable channeling the Sargonid Neo-Assyrian peak
# (671 BC under Esarhaddon and Ashurbanipal) through medieval
# Syriac Christian institutions. Capital is Mosul (the seat of
# the Patriarchate of the East and the city facing Nineveh
# across the Tigris).
#
# Six bureaucracies model the institutional choices a restored
# Assyrian empire must make at imperial scale -- the Sargonid
# administrative apparatus that BNH's smaller patriarchal
# realm could not sustain. Slot pool = 3 of 6.
#
#   1. Kisir Sharruti       -- Royal Cohort standing army
#   2. Tab-Rabuti Bureau    -- Resettlement & integration corps
#   3. Saknu Temi System    -- Sargonid 70-province network
#   4. Aramaic Chancery     -- Imperial lingua franca scriptorium
#   5. Bit Tuppate          -- Library/scribal school network
#   6. Akitu Liturgy        -- Coronation festival as imperial cult
###############################################################


# =====================================================================
# Kisir Sharruti -- The Royal Cohort
#
# Tiglath-Pileser III's military reform (mid-8th century BC) created
# the kisir sharruti, the world's first professional standing army:
# career soldiers paid from the royal treasury, trained year-round,
# stationed in permanent garrisons at Kalhu and Nineveh. Sargon II
# and Sennacherib expanded it into the empire's iron core. Restored
# under Mosul, it draws conscripts from across the empire and pays
# them in silver from the imperial treasury, bypassing noble levies.
# =====================================================================
nd_nin_kisir_sharruti_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {

		multiply = 0.005
	}

	potential = {
		has_or_had_tag = NIN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		has_advance = nd_nin_kisir_sharruti
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
		peasants_estate
	}

	# neutral: career soldiers paid from the imperial treasury exist as
	# a standing institution; the empire can mobilize without bargaining
	# with provincial magnates
	neutral_modifier = {
		nobles_estate_blocked_from_leading_military = yes
	}

	# positive: the cohort drills year-round, marches at imperial speed,
	# and answers to the throne directly; commanders trained in royal schools
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_initiative = 0.10
		monthly_army_tradition = 0.05
		expected_army_size_modifier = 0.10
	}

	# negative: when the silver runs short, the cohort turns predatory --
	# garrisoned cities are squeezed for "soldier rations", peasants flee
	# the conscription rolls, and unpaid commanders bargain with nobles
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_satisfaction_decay = 0.02
		global_monthly_control = -0.10
	}
}

# =====================================================================
# Saknu Temi -- The Sargonid Province System
#
# Sargon II's reorganization (late 8th century BC) divided the empire
# into roughly seventy provinces, each under a saknu temi (provincial

# parallel accounts: a fiscal account to the treasury, a garrison
# account to the kisir sharruti, and a religious account to the
# Ekur-priesthood. The three reports cross-checked one another, making
# governor embezzlement difficult and noble entrenchment harder still.

# direct pipeline from Mosul to every district seat.
