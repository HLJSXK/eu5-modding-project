# National Destinies -- Kingdom of Jerusalem (KOJ) bureaucracies
# Five historically-grounded institutions of the Latin East, drawn from the
# constitutional, legal, and administrative apparatus of Outremer between the
# First Crusade (1099) and the fall of Acre (1291), with continuing form via
# the Lusignan kingdom of Cyprus and the Assises de Romanie. Auto-modifier
# grants 2 slots; the capital adds 1 = 3 effective. Picks 3 of 5.
#
# Scope note: KOJ is a Level 3 formable representing a revived Crusader
# polity. Its institutional life was unusually well-documented for a small
# state because the Frankish settlers wrote down their feudal customs (the
# Assises) and operated parallel jurisdictions for Latins, Eastern
# Christians, and military orders. A reborn Kingdom of Jerusalem inherits
# this layered apparatus.

# 1. Haute Cour -- the High Court of Jerusalem, the feudal tribunal of the
#    tenants-in-chief of the kingdom. Constituted under Godfrey of Bouillon
#    after 1099 and codified by John of Ibelin in the Livre des Assises de
#    la Haute Cour (c. 1265), it heard cases involving the king's vassals,
#    judged disputes over fiefs and homage, ratified successions, and -- by
#    the famous doctrine of conjuratio -- could compel a king to honour the
#    feudal contract on pain of release from homage. The Haute Cour was the
#    constitutional engine of the Latin East: kings ruled by its consent.
koj_haute_cour_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = KOJ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Convening the Haute Cour places the kingdom's tenants-in-chief in
	# permanent assembly: feudal succession, escheat, and homage are
	# adjudicated by peers, not by royal fiat.
	neutral_modifier = {
		nobles_estate_can_participate_in_parliament = yes
	}

	# Maintained, the Haute Cour delivers reliable feudal levies, settles
	# baronial feuds before they become rebellions, and binds the lordships
	# of Sidon, Galilee, Caesarea, and Jaffa to the unified state -- but
	# the conjuratio doctrine fetters royal taxation and statute.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.075
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		legislative_efficiency = -0.05
	}

	# Neglected, the conjuratio is invoked openly: barons withdraw homage,
	# claim disseisin against royal officers, and settle their disputes by
	# private war between fiefs -- the kingdom slides into the baronial
	# civil wars that dogged the historical Latin East from 1228 onwards.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.10
		stability_decay = 0.00025
	}
}

# 2. Cour des Bourgeois -- the Burgesses' Court, the urban tribunal for
#    Latin commoners (non-noble Franks) in the kingdom's cities. Codified
#    in the Livre des Assises de la Cour des Bourgeois (c. 1240, after a
#    lost original of c. 1150), it had jurisdiction over property, debt,
#    inheritance, and minor crimes among the burgensis class. A jury of
#    twelve burgesses sat under a viscount; commercial disputes followed
#    the Lex Mercatoria. Acre, Tyre, Jaffa, Caesarea, Nablus, and Jerusalem
#    each had their own Cour des Bourgeois.
koj_cour_des_bourgeois_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = KOJ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		OR = {
			owns = location:acre
			owns = location:jerusalem
			owns = location:jaffa
		}
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Granting the burgesses their viscount's court binds the Latin commoner
	# class -- artisans, merchants, sergeants -- to the unified state under
	# their own twelve-juror tribunal, distinct from the feudal Haute Cour.
	neutral_modifier = {
		burghers_estate_can_participate_in_parliament = yes
	}

	# Funded, the Cour des Bourgeois enforces the Lex Mercatoria across the
	# Frankish cities: the Acre, Tyre, and Jaffa markets thrive under a
	# predictable commercial law, and burgher migration from Italy and
	# Provence reinforces the kingdom's Latin demographic base.
	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_land_efficiency = 0.075
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
		global_pop_promotion_speed_modifier = 0.05
	}

	# Underfunded, the viscount's writs are ignored: noble lords seize
	# burgher property under feudal pretext, the cities' commercial law
	# fragments into Italian quarter privileges, and the Frankish artisan
	# class drifts back to Europe -- Acre's markets shrink to a fraction.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.020
		global_monthly_urban_control = -0.003
	}
}

# 3. Patriarchat Latin de Jerusalem -- the Latin Patriarchate of Jerusalem,
#    erected under Daimbert of Pisa in 1099 to replace the Greek hierarchy
#    in the conquered Holy Sepulchre. The Patriarch held the Quarter of the
#    Patriarch (a distinct urban jurisdiction), administered the tithe of
#    the kingdom, presided over ecclesiastical courts that drew jurisdiction
#    over Frankish marriage and testament, and after 1291 was preserved in
#    titular form at Acre and later Cyprus. The Latin Patriarchate was the
#    spiritual counterweight to the Haute Cour and the legal bridge to
#    Rome's canonical machinery.

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = KOJ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:catholic
		owns = location:jerusalem
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Recognising the Latin Patriarch's quarter at the Holy Sepulchre fixes
	# the spiritual capital of Christendom in the kingdom's hands -- the
	# tithe is collected, the ecclesiastical courts sit, and Rome's writ
	# runs through Outremer.
	neutral_modifier = {
		monthly_devotion = 0.05
	}

	# Maintained, the Patriarchate channels pilgrim alms and Roman
	# subventions into the royal fisc, the cathedral schools at Jerusalem
	# and Acre produce literate clerks for the chancery, and the Patriarch

	# Neglected, the tithe is diverted to private hands and the Patriarch

	# arrive bearing interdicts that paralyse the chancery.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_cost_efficiency = -0.1
		nobles_estate_satisfaction_decay = 0.020
	}

# 4. Charte des Ordres Militaires -- the royal charter governing the
#    Templar, Hospitaller, and (later) Teutonic Orders within the kingdom.
#    The military orders held vast estates (the Krak des Chevaliers, Margat,
#    Atlit, Belvoir) outside ordinary feudal obligation, answered only to
#    their Grand Masters and the Pope, and supplied the kingdom's most
#    reliable standing armies. Their parallel administrative apparatus --
#    commanderies, treasuries, infirmaries -- ran a state-within-a-state.
#    The charter defined what the orders owed the kingdom, and what they
#    were exempt from.

		nobles_estate

	# Recognising the orders' charter places the great Frankish castles --

	neutral_modifier = {

# 5. Assises de Jerusalem -- the codified feudal law of the kingdom,
#    originally inscribed on parchment-rolls (the Lettres du Sepulcre) kept
#    in the Holy Sepulchre and lost when Saladin took Jerusalem in 1187.
#    The corpus was reconstructed from memory by jurists in exile -- John
#    of Ibelin's Livre des Assises (c. 1265), Philip of Novara, James of
#    Ibelin -- and survived as the legal foundation of the kingdom of
#    Cyprus and, in adapted form, the Assises de Romanie of Frankish
#    Greece. The Assises codified the feudal contract, the criminal
#    pleas of the king, and the jurisdiction of the kingdom's courts.
koj_assises_de_jerusalem_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = KOJ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		clergy_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Promulgating the reconstructed Assises fixes a uniform feudal law
	# across the kingdom's courts: the Haute Cour, the Cour des Bourgeois,
	# and the merchants' Cour de la Fonde all judge by the same written
	# corpus rather than by remembered custom.
	neutral_modifier = {
		legislative_efficiency = 0.05
	}

	# Maintained, the Assises support a literate jurist class -- Ibelins,
	# Novarans, scribes of the chancery -- who train at the Holy Sepulchre's
	# scriptorium, draft royal charters with precision, and bind the
	# kingdom's far-flung lordships under a single jurisprudence; the
	# kingdom acquires a constitutional memory that survives reconquest.
	positive_modifier = {
		scale = { value = scope:maintenance }
		research_speed_modifier = 0.05
		global_max_literacy = 5
		stability_investment = small_stability_investment
	}

	# Neglected, the Assises decay into selective citation: barons quote
	# the chapters that limit royal taxation while ignoring those that
	# bind their own homage, the chancery's clerks lose precision, and
	# the kingdom's courts revert to the patchwork of remembered custom
	# that left Outremer's first century without a written law.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.03
		stability_decay = 0.00025
	}
}
