###############################################################
# National Destinies -- Mapuche Confederation (MAP) bureaucracies
# Four federal-tribal institutions; the player picks 3
# (capital implicit slot + 2 from the auto_modifier).
# The Mapuche federation was famously decentralized -- Lof to
# Rewe to Aillarewe to Wallmapu -- so these "bureaucracies" are
# council-and-federation forms, not centralizing imperial offices.
# Sources: Bengoa "Historia del pueblo mapuche", Boccara "Guerre
# et ethnogenese", Goicovich on the Parlamentos.
###############################################################

# Toki Apu Council -- the war-leader assembly. In ordinary times
# the lof and rewe answered to their lonkos; only when the
# Aillarewe convened to wage a campaign did they elect a Toki
# (axe-bearer), and only when several Aillarewe federated did
# they elect a Toki Apu of all war-bands. Lautaro held this rank
# against Valdivia, Pelantaru against Onez de Loyola. The council
# was an emergency machinery, not a standing court -- but once
# institutionalized it gave the federation a single war-fist.
map_toki_apu_council_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MAP
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		tribes_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: an axe-bearer presides over the war-host
	neutral_modifier = {
		monthly_tribal_cohesion = 0.05
	}

	# positive: the toki's mobilization order moves the rewe in unison
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.1
		land_morale_modifier = 0.1
		levy_combat_efficiency_modifier = 0.05
	}

	# negative: a war-leader without a war turns into a faction-chief,
	# and the ulmen of the great houses chafe under the axe-bearer
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_complacency = 0.05
	}
}

# Aillarewe Federation -- the regional confederacy of nine rewe.
# This is the institutional spine of Wallmapu. Each Aillarewe
# (Pikunche, Wenteche, Nagche, Lafkenche, Pewenche, Wulluche)
# was a permanent federation of nine councils that pooled
# disputes, organized winter ngillatun ceremonies, and kept
# the boundary-rolls between lineages. Confederation business
# was conducted by ulmen (great-house chiefs) seated at the
# rewe-stone -- not by any Toki or central court.
map_aillarewe_federation_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MAP
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		tribes_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# neutral: nine-rewe councils handle boundary disputes locally
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# positive: federated rewe administer the land without crown writ
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_max_rural_control = 0.03
		global_supply_limit_modifier = 0.1

# Machi Sisterhood -- the order of female (and sometimes machi-weye
# two-spirit) shamans who served as healers, judges, and chroniclers
# of the lineages. The machi presided over the ngillatun-rewe
# fertility rites, interrogated suspected wekufu (sorcerers) at
# the kalku-trials, and kept the genealogies that the lineages
# needed to argue land claims. Their authority sat outside the
# warrior hierarchy -- ulmen feared the kalku-trial, lonkos
# deferred to the trance-ceremony.
map_machi_sisterhood_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MAP
		has_game_rule = nd_bureaucracies_enabled
	}

	# The machi order is rooted in the tain-feyentun cosmology;
	# severing the federation's religion would dissolve their
	# institutional standing.
	allow = {
		religion = religion:tain_feyentun_religion
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: the kalku-trial recognizes peasant testimony equal
	# to ulmen testimony (a sociological inversion)
	neutral_modifier = {
		nobles_estate_blocked_from_leading_military = no
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# positive: ngillatun ritual cohesion, oral chronicles, and
	# trance-judgments knit the lineages
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_devotion = 0.05
		cultural_tradition = 0.05
		global_max_literacy = 5
	}

	# negative: kalku-witch hunts turn predatory; ulmen and
	# warrior-chiefs are denounced under trance-accusation
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		stability_cost_efficiency = -0.1
	}
}

# Parlamento General -- the formal council that the Spanish
# governors of Chile were forced to convene with the lonkos
# of the Aillarewe at Quilin (1641), Negrete, Tapihue, and
# elsewhere. After the Curalaba defeat the Crown of Castile
# acknowledged the federation as a sovereign counterparty;
# the Parlamentos issued bilingual treaty texts, set the
# Bio-Bio frontier, exchanged captives, and arbitrated trade.
# It is the only Indigenous-American institution recognized
# in the Atlantic diplomatic system before the eighteenth
# century -- a court of nations as much as a tribal council.
map_parlamento_general_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MAP
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Parlamento is the post-Quilin institutional form.
	# Without the recognition compact (the Treaty of Quilin
	# advance, 1641 in our tree) there is nothing to convene.
	allow = {
		has_advance = nd_map_treaty_of_quilin
	}

	estates_that_like = {
		nobles_estate
		burghers_estate
	}
	estates_that_dislike = {
		tribes_estate
	}

	# neutral: the federation negotiates as a treaty-bearing nation
	neutral_modifier = {
		diplomatic_reputation = 1
	}

	# positive: bilingual translators, captive-exchanges, frontier
	# trade hubs, and Atlantic recognition
	positive_modifier = {
		scale = { value = scope:maintenance }
		diplomatic_capacity = 1
		cultural_influence_modifier = 0.1
		trade_land_efficiency = 0.05
	}

	# negative: the rewe lineages resent Quilin-line concessions
	# and the lof feel sold to the huinca governors
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		tribes_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_tribal_cohesion = -0.05
	}
}
