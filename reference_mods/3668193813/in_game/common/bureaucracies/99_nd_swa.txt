# National Destinies -- Swabian bureaucracies
# 5 historically-grounded institutions of the unified Swabian polity, drawing
# on the Schwaebischer Bund (1488-1534), the Schwaebischer Reichskreis (1500),
# the Wuerttemberg Landtag and Tuebinger Vertrag of 1514, the Reichsritterschaft
# of the five Swabian cantons, and the Augsburger Stadtrat of the great
# imperial free city. Auto-modifier grants 2 slots; capital adds 1 = 3 effective.
# Player picks 3 of 5.
#
# Scope note: SWA is a Level 2 formable, but pre-mediatization Swabia was the
# institutionally most fragmented and yet best-organised region of the Reich.
# Five bureaucracies match the peer tier (BAV, SAX) and let the player decide
# whether to lean on the Bund's federal council, the Kreis's empire-wide police
# function, the constitutional Landtag of Wuerttemberg, the patchwork of
# Imperial Knights, or the patrician civic government of Augsburg.

# 1. Bundesrat des Schwaebischen Bundes -- the federal council of the Swabian
#    League, formed at the Diet of Esslingen on 14 February 1488. The Bundesrat
#    sat in three colleges -- princes, imperial cities, and the knights of
#    Sankt Joergenschild -- with two (later three) Hauptleute as chief captains
#    and equal numbers of councillors elected from each estate (initially
#    eighteen, expanded to twenty-one by the renewals of 1496 and 1500). It
#    fielded an army of 13,000 men, suppressed the Peasants' War of 1525, and
#    held southern Swabia together against feud, robber-baron, and confessional
#    schism until the Reformation broke the League in 1534.
swa_bundesrat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = SWA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# A standing tri-collegial council with chartered Hauptleute is a permanent
	# diplomatic stage on which the duke must negotiate with cities and knights
	# rather than simply command them.
	neutral_modifier = {
		diplomatic_reputation = 1
	}

	# Funded -- meaning the duke convenes the Bundesrat regularly, pays the
	# Hauptleute, and honours the league's collective-defence obligations --
	# the federal council suppresses feuds across Swabia, fields a unified
	# contingent against external threats, and holds the patchwork together.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_garrison_size_modifier = 0.10
		global_separatism = -0.05
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Neglected, the three colleges drift apart -- the cities suspect the
	# princes, the knights resent the merchants, and confessional fault-lines
	# split the assembly. The league's shared army cannot be raised without
	# cross-college consent, and the duke's diplomatic standing among the
	# Swabian estates erodes month by month.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.015
		stability_decay = 0.00025
	}
}

# 2. Schwaebischer Kreistag -- the deliberating assembly of the Schwaebischer
#    Reichskreis, one of the ten imperial circles created by Maximilian I in
#    1500. The Dukes of Wuerttemberg and the Bishops of Konstanz served as
#    kreisausschreibende Fuersten, summoning the Kreistag and chairing its
#    sessions. After 1512 the Kreis enforced the verdicts of the Reichskammer-
#    gericht; from 1530 it raised the imperial-army contingent (the
#    Reichsmatrikel); after 1555 it policed the Eternal Public Peace within
#    its territory. Of the ten circles, the Swabian was reckoned the best
#    organised, divided after 1563 into the four execution-quarters of
#    Augsburg, Konstanz, Wuerttemberg, and Baden.
swa_kreistag_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = SWA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:konstanz
	}

	estates_that_like = {
		clergy_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# A standing Kreistag at Konstanz with the Bishop and the Duke co-chairing
	# is a permanent imperial-court machinery: the Reichskammergericht's writs
	# arrive, the Kreismatrikel registers contingents, and Swabia speaks with
	# one voice at the Reichstag.
	neutral_modifier = {
		legislative_efficiency = 0.05
	}

	# Funded -- the Kreisobrist appointed, the Kreissekretaer paid, the
	# Muenzwardein vetting circle coinage, the four execution-quarters
	# patrolling -- the Kreis enforces the Landfrieden across Swabia, stamps
	# out feud and brigandage, and stands as a recognised intermediary with
	# the Emperor.
	positive_modifier = {
		scale = { value = scope:maintenance }
		stability_investment = small_stability_investment
		global_max_urban_control = 0.05
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Without funds the Kreistag becomes a debating club: the Reichskammer-
	# gericht's verdicts go unenforced, knights revert to private feud, and
	# the imperial-army contingent shows up half-strength to musters.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_cost_efficiency = -0.1
		global_garrison_size_modifier = -0.05
	}
}

# 3. Tuebinger Vertrag -- the constitutional contract concluded between Duke
#    Ulrich and the Wuerttemberg Landtag at Tuebingen on 8 July 1514, in the
#    wake of the "Armer Konrad" peasant rising against Ulrich's meat-and-wine
#    consumption tax. The treaty bound the duchy to consult the estates of
#    prelates, knights, and burgher Ehrbarkeit on extraordinary taxation,
#    war, and territorial alienation, and granted all subjects the right of
#    free emigration and lawful procedure in matters of life, limb, and
#    honour. Historians have called it the "Magna Carta" of Wuerttemberg --
#    it remained the territorial constitution, almost unchanged, until the
#    Napoleonic mediatisation of 1805.
swa_tuebinger_vertrag_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = SWA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		peasants_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the Vertrag as the territorial charter binds the duke to
	# consult the burgher Ehrbarkeit and the prelates on extraordinary
	# taxation, and grants every subject the right of free exit -- limiting
	# the crown's reach in exchange for a territorial constitutionalism.
	neutral_modifier = {
		burghers_estate_can_participate_in_parliament = yes
	}

	# Honoured, the Vertrag stabilises Wuerttemberg: estates assent to taxes,
	# burghers feel chartered, and rural rebellion is averted because peasants
	# can flee a bad lord and judges follow procedure. The duchy is calmer
	# than its absolutist neighbours.
	positive_modifier = {
		scale = { value = scope:maintenance }
		stability_investment = small_stability_investment
		global_separatism = -0.05
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Trampled, the Vertrag's broken word costs the duchy years of legitimacy:
	# the Ehrbarkeit petitions the Reichskammergericht, prelates withhold
	# grants, and the duke must rule by emergency edict against his own
	# constitution.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		stability_cost_efficiency = -0.1
	}
}

# 4. Reichsritterschaft Schwabens -- the Swabian Imperial Knights' Congregation,
#    formed between 1541 and 1545 from the dissolved Sankt Joergenschild and
#    organised into five cantons: Donau, Hegau-Allgaeu-Bodensee, Neckar-
#    Schwarzwald, Kocher, and Kraichgau. Each canton was governed by a
#    Direktorium under a Ritterhauptmann elected for life, with elected
#    Ritterraete; the cantons exercised taxation rights and military
#    sovereignty over their subjects, and sold knightly estates only to other
#    knights to keep the Ritterschaft territorially intact. The Swabian
#    knights were the densest knightly corporation in the Reich and survived,
#    immediate to the Emperor, until mediatisation in 1805/06.
swa_reichsritterschaft_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = SWA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
		burghers_estate
	}

	# Recognising the five cantons as immediate-to-the-Emperor knightly
	# corporations puts vast swathes of countryside under Ritterhauptmaenner
	# and Ritterraete instead of the duke's amtmann; the rural patchwork
	# follows knightly justice, not territorial law.
	neutral_modifier = {
		global_max_rural_control = 0.03
	}

	# Maintained -- canton-rolls accurate, Ritterraete sitting, Ritter-
	# hauptmaenner receiving the duke's diplomatic courtesies -- the
	# Ritterschaft musters its own contingents, polices its own villages,
	# and absorbs the cost of countryside government the duke would
	# otherwise pay himself.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		global_manpower_modifier = 0.05
	}

	# Decayed, the cantons descend into knightly extortion -- robber-knights
	# revive on the back-roads, burghers protest the lawless markets, and
	# peasants flee the worst Ritterherren.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.015
		peasants_estate_satisfaction_decay = 0.015
	}
}

# 5. Augsburger Stadtrat -- the patrician-and-guild civic government of
#    Augsburg, free imperial city since 1276 and the great Reichsstadt of
#    the south. Until 1548 the Rat sat under a Zunftverfassung guaranteeing
#    the guilds the majority over the patricians; the Welser, Hoechstetter,
#    Herwart, Baumgartner, and Fugger merchant houses sat alongside the
#    artisan guilds and the Stadtpfleger. After the Peace of Augsburg of 1555
#    a confessionally mixed Catholic-Lutheran council presided over a
#    majority-Protestant population. The Augsburger Stadtrat was the
#    institutional engine of the south-German banking trade -- the Fuggers'
#    private cash routinely lent to emperors -- until the Thirty Years' War
#    broke it.
swa_augsburger_stadtrat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = SWA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:augsburg
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A standing Stadtrat under the Stadtpfleger turns Augsburg into a
	# self-governing commercial commune that issues banking charters,
	# guarantees fair markets, and shelters Welser-Fugger credit on the
	# Reichsstadt's privileges.
	neutral_modifier = {
		create_market_cost_modifier = -0.15
	}

	# Funded -- the merchant guilds well-represented, the patrician houses
	# kept in check, the Stadtpfleger's office paid -- the council channels
	# south-German banking through Augsburg, where the Welser and the Fugger
	# lend to emperors and bankroll Swabian construction across the duchy.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_trades_per_burgher = 0.10
		global_build_buildings_efficiency = 0.1
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Neglected, the patrician families recapture the Rat (as they did after
	# 1548), and the council closes ranks against guildsmen and rival cities.
	# Augsburg's banking houses overextend on imperial loans, and confidence
	# in the Reichsstadt's credit slowly erodes.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.015
		monthly_inflation = 0.003
	}
}
