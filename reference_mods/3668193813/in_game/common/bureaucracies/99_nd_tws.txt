# National Destinies -- Two Sicilies bureaucracies
# Six historically-grounded institutions of the Norman, Hohenstaufen, Aragonese,

# 1. Constitutions of Melfi -- Frederick II's 1231 legal code (Liber
#    Augustalis), the founding administrative codification of the Regno.
#    Roman-law instrument of royal authority over feudatories; called
#    "the birth certificate of the modern administrative state" by
#    Kantorowicz. Basis of Sicilian law for six centuries.
tws_constitutions_of_melfi_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = TWS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A codified law-book -- the legal apparatus exists by simply being
	# promulgated, lending coherence to royal acts.
	neutral_modifier = {
		legislative_efficiency = 0.05
	}

	# At full investment, the curiae tenentium loca enforce the code:
	# baronial pretensions are bound, the king's writ runs throughout
	# the realm, and royal acts gain legitimacy from their codified form.
	positive_modifier = {
		scale = { value = scope:maintenance }
		stability_cost_efficiency = 0.1
		monthly_legitimacy = 0.05
		annexation_speed_modifier = 0.10
	}

	# When neglected, the code becomes a dead letter -- baronial
	# privileges multiply through neglected exemption clauses, and
	# the royal cancellaria loses the habit of consistent jurisprudence.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		country_cabinet_efficiency = -0.10
	}
}
