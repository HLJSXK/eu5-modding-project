# National Destinies -- Borneo bureaucracies
# Five institutional pillars of a unified pan-Bornean state, drawing on
# the Brunei Sultanate's four-wazir cabinet (Bendahara, Pengiran-
# Bendahara, Pengiran-Tumenggung, and Pengiran-Di-Gadong, formalised
# under Sultan Bolkiah c. 1485-1524 and surviving into the 19th c.),
# the hereditary pangeran-administered diamond and gold mining
# districts of Sambas, Landak, and Bau (the Chinese kongsi gold
# republics from c. 1740 sat under pangeran charters at Mempawah,
# Mandor, and Monterado), the Iban and Bidayuh Tuai Rumah longhouse
# compact governing the Sungai Rajang and Sungai Sarawak interior, the
# Iranun and Balanini sea-raider admiralty whose licensed cruises
# operated from Tempasuk, Marudu Bay, and the Sulu shores from c.
# 1768 onward, and the Bidayuh basanga adat-tribute of forest products
# (rattan, camphor, gaharu, beeswax, dragon's blood) by which the
# coastal sultans drew interior wealth without administering the
# uplands directly. Author-modifier grants 2 slots; capital adds 1 = 3
# effective. Player picks 3 of 5.
#
# All IDs prefixed with `bno_` to avoid cross-file collisions.

# 1. Wazir Council of Brunei -- the four-pillar civilian cabinet that
#    gave the Brunei Sultanate its administrative shape from the reign
#    of Sultan Bolkiah (r. 1485-1524) through the 19th-century cession
#    of Sarawak. The Pengiran Bendahara managed the chancery and was
#    senior of the four; the Pengiran Di-Gadong held the treasury and
#    the customs at the Brunei river-mouth; the Pengiran Pemancha
#    superintended the sultan's lands and ulu (river) jurisdictions;
#    and the Pengiran Temenggung commanded the war-prahus and the
#    palace guard. The wazirs were drawn from the noble Berunai clans
#    and held office by sultanic appointment, but in practice the
#    pillar offices ran in noble lineages and the council, sitting in
#    the istana, could outweigh any single sultan. The institution was
#    the working state of the Brunei polity in its golden age and the
#    template every Bornean sultanate copied in miniature.
bno_wazir_council_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BNO
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Constituting the four wazir-pillars in the istana means appoint-
	# ments, despatches, customs registers, and prahu mobilisation all
	# pass through the Bendahara, Di-Gadong, Pemancha, and Temenggung
	# before the sultan rules: the realm has a permanent civilian
	# chancery rather than a personal household around the throne.
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Functioning -- a Bolkiah-era Bendahara of weight, the Di-Gadong's
	# customs roll honestly kept, the Pemancha's writ running upriver
	# to the Tutong and Belait, the Temenggung's prahus drilled and
	# fed -- the council gives the sultanate its golden-age form: the
	# Berunai diaspora carries the office model from Sambas to Sulu,
	# and the throne is treated as senior partner in a working state
	# rather than a tribal first-among-equals.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		global_max_control = 0.02
		country_cabinet_efficiency = 0.075
	}

	# When the four pillars become four faction chiefs -- the Pengiran
	# clans hoarding offices in their own lineages, the Di-Gadong's
	# customs roll cooked for a kinsman's profit, the Temenggung's
	# prahus crewed by retainers who answer to him before the sultan
	# -- the council turns from servant of the istana into rival of
	# the throne, and the sultan reigns over wazirs he cannot dismiss.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		global_separatism = 0.05
	}
}

# 2. Pangeran Mining Districts -- the chartered hereditary governors
#    of the diamond fields of Landak (worked at Cempaka and at the
#    Kapuas tributaries since at least the 14th c.), the gold mines
#    of Bau and the Sambas hinterland, and the antimony seams of the
#    upper Sarawak. The pangerans of Sambas, Mempawah, and Landak
#    issued mining concessions, took a fixed proportion of yield (a
#    tenth was customary), and arbitrated disputes between the kongsi
#    fraternities of Hakka miners (the Lanfang Republic at Mandor
#    from 1777, the Heshun Zongting at Monterado, the Dagang at
#    Mempawah), the Dayak labour pools, and the coastal Malay
#    capital. The system financed the coastal sultanates -- diamond
#    revenue from Landak filled the Sambas treasury, and Sambas
#    pangerans were sometimes wealthier than their nominal overlords
#    at Brunei -- but it locked the throne into accepting kongsi
#    self-government and Hakka demographic predominance in the
#    mining districts.
bno_pangeran_mining_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BNO
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# Granting the pangerans of Sambas, Mempawah, and Landak hereditary
	# concessions over the diamond, gold, and antimony districts means
	# the kongsi fraternities and Dayak labour pools deal with a fixed
	# noble counterparty and pay a published tenth: the realm captures
	# mineral revenue without administering the mines directly.
	neutral_modifier = {
		global_raw_material_output = 0.05
	}

	# Honoured -- the pangerans collecting the customary tenth without
	# squeeze, the Lanfang and Heshun kongsi assemblies left to their
	# own internal courts, the Hakka miners working three shifts at
	# the Mandor and Monterado pits -- the diamond and gold yield of
	# the western coast is the wealthiest in the archipelago, and
	# Sambas pangerans furnish the throne with revenues no agrarian
	# sultanate can match.
	positive_modifier = {
		scale = { value = scope:maintenance }
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
		global_gems_output_modifier = 0.10
		global_iron_output_modifier = 0.10
	}

	# When the pangerans squeeze beyond the customary tenth, when
	# rival noble houses in Pontianak and Sambas fight over the same
	# concession, when the kongsi assemblies (the Lanfang at Mandor
	# in the 1850s, the Heshun at Monterado in the 1820s) take up
	# arms against extra-customary levies -- the mining districts
	# turn into a running revolt: Hakka militia entrench in the
	# pits, Dayak labourers desert the riverbeds, and the throne
	# loses both revenue and labour.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_satisfaction_decay = 0.020
		global_separatism = 0.05
	}
}

# 3. Tuai Rumah Longhouse Compact -- the customary recognition of the
#    Iban and Bidayuh longhouse-chief (the Tuai Rumah, with the
#    senior regional chiefs styled Penghulu and the paramount
#    Temenggong of the Sea Dayaks) as the throne's interlocutor for
#    the upriver communities of the Sungai Rajang, the Sungai
#    Sarawak, and the Sungai Saribas. Each Tuai Rumah was the elected
#    headman of his bilek longhouse (typically thirty to sixty
#    households under one ironwood roof), responsible for adat
#    rulings within the house, allocation of swidden plots, and the
#    sending of warriors to the sultan's call. The compact gave the
#    coastal throne what no direct administration ever achieved:
#    upland recruits for the war-prahus, mediated peace among feuding
#    longhouses, and a recognised channel for forest tribute -- in
#    exchange for non-interference in adat, headhunting trophies
#    (until the Brooke prohibition of 1841+), and the right of the
#    longhouse to name its own Tuai Rumah.
bno_tuai_rumah_compact_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = BNO
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		tribes_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Recognising the Tuai Rumah of every Iban and Bidayuh longhouse
	# as the throne's counterparty in the upriver country binds the
	# Sungai Rajang and Sungai Saribas communities by sworn compact
	# rather than by garrison: the bilek headmen mediate adat in their
	# own houses, the Penghulu chiefs answer for districts, and the
	# Temenggong of the Sea Dayaks brings warriors when the prahus
	# muster.
	neutral_modifier = {
		tribes_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Honoured -- the Tuai Rumah received in person at the istana, the
	# longhouses left to their own adat, the sultan's gifts of brass
	# gongs and trade-cloth flowing upriver -- the compact gives the
	# realm warriors no coastal levy can match: the Iban bangkong
	# crews, the Bidayuh blowpipe scouts, and the Kayan parang
	# infantry serve the throne by sworn allegiance, and the upland
	# country answers to summons rather than to garrison.
	positive_modifier = {
		scale = { value = scope:maintenance }
		tribes_estate_levy_size = 1
		global_levy_size_modifier = 0.05
		global_hostile_attrition = 0.10
	}

	# When the compact lapses -- coastal noble houses encroaching on
	# longhouse swidden grounds for personal estates, the Tuai Rumah
	# bypassed for sultanic agents who do not speak Iban, the customary
	# gifts withheld -- the longhouses withdraw their service, the
	# Penghulus quietly stop forwarding tribute, and (as repeatedly
	# happened to the Brunei pangerans in the 18th century) the
	# upland country becomes a land the throne taxes only on paper,
	# its institutional inertia hardening with every season the compact
	# is left in abeyance.
	negative_modifier = {
		scale = {
			value = 1
			add = scope:entrenchment
			multiply = { value = 1 subtract = scope:maintenance }
		}
		nobles_estate_satisfaction_decay = 0.015
		stability_cost_efficiency = -0.1
	}
}

# 4. Lanun Sea-Raider Admiralty -- the institutionalised piracy
#    regime of the Sulu and northern Bornean seas, by which the
#    sultans of Brunei, Sulu, and (after c. 1768) the petty datus of
#    Tempasuk, Marudu Bay, and Pandasan licensed Iranun and Balanini
#    cruises against Spanish, Dutch, and Bugis shipping in exchange
#    for a share of slaves, ransoms, and goods. The lanun fleets --
#    sometimes 200 prahus strong, with crews drawn from the Iranun
#    of southern Mindanao and the Balanini Samal of Tawi-Tawi -- ran
#    from April through October on the southwest monsoon, raided as
#    far as Java and Celebes, and brought their captures home for
#    sultanic adjudication and resale at Sulu's Jolo market. The
#    institution made the northern coast wealthy and feared, and
#    held off European intrusion for two generations beyond what
#    naked weakness would have permitted -- but it earned the realm
#    the unrelenting hostility of every legitimate trader and every
#    foreign power that touched the eastern seas.

	# admiralty gives the realm a fearsome cheap navy: foreign

		scale = { value = scope:maintenance }
		navy_initiative = 1
		ship_capture_chance = 0.05
		monthly_navy_tradition = 0.1
	}

	# When the licensing system collapses -- the throne's tenth
	# uncollected, the lanun captains raiding the realm's own coasts
	# for slaves rather than the rivals', the Spanish and Dutch
	# steam-gunboats finally arriving in numbers -- the prahu fleets
	# turn predator on Bornean ports themselves, every legitimate
	# merchant abandons the northern coast for Manila or Singapore,

# 5. Basanga Forest-Tribute -- the customary adat-tribute of forest
#    products by which the Bidayuh, Penan, and inland Dayak peoples
#    delivered rattan, camphor (the prized Borneo barus), gaharu (eagle
#    -wood), beeswax, dragon's blood resin, edible birds' nests, and
#    bezoar stones to coastal Malay and Brunei agents in exchange for
#    salt, iron tools, and trade-cloth. The basanga (a Bidayuh term
#    for the customary tribute-portion) was assessed by district
#    headmen against each longhouse, collected at riverside trade-
#    posts (the pasar) at the heads of the Sungai Sambas and Sungai
#    Sarawak, and forwarded to the sultanate as the throne's share
#    of upland produce. The system was the working economy of the
#    pre-modern interior -- Borneo barus camphor sold at Aleppo and
#    Cairo for its weight in silver, and Niah birds' nests still fund
#    Sarawak fortunes -- but it depended on the trade-post system
#    holding: when the sultanic agents squeezed the basanga or when
#    foreign trading-houses began bidding directly at the Niah and
#    Mukah river-mouths, the customary channels broke and the wealth
#    of the interior reached the throne only as smuggled remnant.

	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BNO
		has_game_rule = nd_bureaucracies_enabled

	allow = {}

	estates_that_like = {

	# the Bidayuh, Penan, and interior Dayak country flows through the

	neutral_modifier = {
		global_supply_limit_modifier = 0.10
	}

	# Honoured -- the headmen's assessments accepted at customary

	# the Niah and Mukah birds'-nest harvests channelled through the
	# sultanic pasar, the upland exchange of salt and iron flowing
	# back unsqueezed -- the realm draws upon the wealthiest forest
	# economy in the archipelago, and the interior peoples count
	# themselves better-served by the throne than by any open-trade
	# alternative.
	positive_modifier = {
		scale = { value = scope:maintenance }
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		global_incense_output_modifier = 0.10
		global_pop_assimilation_speed_modifier = 0.10
	}

	# When the trade-post system breaks -- coastal agents skimming
	# the basanga beyond customary measure, foreign trading-houses
	# bidding directly at the Niah and Mukah river-mouths,
	# Singapore and Manila firms drawing the Bidayuh away from the
	# pasar -- the headmen quietly stop forwarding the throne's
	# share, the camphor and gaharu reach foreign markets through
	# smugglers, and the urban customs-houses face a forest economy
	# they no longer mediate.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.015
		trade_sea_efficiency = -0.10
	}
