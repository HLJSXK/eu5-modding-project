# National Destinies -- Schleswig-Holstein bureaucracies
# Five historically-grounded institutions of the twin duchies of Schleswig
# and Holstein, held in personal union by Oldenburg dukes after the Treaty
# of Ribe (1460) under the constitutional doctrine "up ewig ungedeelt"
# (forever undivided). Auto-modifier grants 2 slots; capital adds 1 = 3
# total. Player picks 3 of 5.
#
# All IDs are tag-prefixed (shl_*) to avoid the id_collision check --
# generic names like "ritterschaft_bureaucracy" would clash with any
# future Mecklenburg / Holstein-Gottorp / Brandenburg formable that
# wanted the same institutional vocabulary.

# 1. Schleswig-Holsteinische Ritterschaft -- the united knighthood of
#    both duchies. From the Treaty of Ribe (1460) the noble corporation
#    of Schleswig and that of Holstein sat as one body, defending the
#    "up ewig ungedeelt" principle and the privilegia of the Equites
#    Originarii against any ruler who tried to partition them. Their
#    great seat was at Bordesholm; their tax-exempt landed estates
#    (adelige Güter) covered most of eastern Holstein.
shl_ritterschaft_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = SHL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
		peasants_estate
	}

	# The corporate body of the Equites Originarii is constitutionally
	# acknowledged: noble tax exemption is on the books while the
	# Ritterschaft sits.
	neutral_modifier = {
		nobles_estate_min_tax = -0.05
	}

	# Properly cultivated, the knighthood underwrites the duchies'
	# defence: the great Junker estates muster cavalry, hold the
	# eastern marshes, and bind the dukes to the Privilegium of Ribe.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		global_levy_size_modifier = 0.075
		global_separatism = -0.05
	}

	# Neglected, the Ritterschaft becomes a hereditary brake on
	# everything: peasants on adelige Güter sink into Leibeigenschaft,
	# the crown loses cabinet seats, and provincial control rots.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_satisfaction_decay = 0.025
		global_max_control = -0.03
	}
}

# 2. Landtag der Stände -- the tripartite estates assembly (nobility,
#    cathedral chapters, and the Vier Städte: Schleswig, Flensburg,
#    Kiel, and Rendsburg) constituted in 1460 to enforce the union.
#    The Landtag voted Bede taxes, registered ducal edicts, and
#    arbitrated between the Glückstadt and Gottorp lines after the
#    1544 partition. Its three benches met at Kiel and Rendsburg.
shl_landtag_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = SHL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		burghers_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The three benches sit by right -- legislation passes through them
	# rather than around them, and the duchies' parliamentary tradition
	# is permanently extended.
	neutral_modifier = {
		parliament_duration_modifier = 0.10
	}

	# When the Landtag is properly convened, its consent legitimises
	# Bede grants and edicts: tax revenue flows, parliamentary support
	# holds, and estate satisfaction tracks the throne's legitimacy.
	positive_modifier = {
		scale = { value = scope:maintenance }
		legislative_efficiency = 0.10
		parliament_base_support = 0.05
		global_estate_satisfaction_from_legitimacy = 0.05
	}

	# When the throne ignores the three benches and rules by
	# Verordnung, the estates retaliate: edicts are stalled,
	# burghers withdraw confidence, and the cabinet loses its grip on
	# the chancery.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.020
		country_cabinet_efficiency = -0.075
	}
}

# 3. Kieler Umschlag -- the annual financial fair held at Kiel in the
#    week after Epiphany. From the late 15th century onwards, this was
#    the principal credit market of northern Germany, the Duchies, and
#    the Kingdom of Denmark; Holstein Junkers lent at 6-7% to Danish
#    kings and Hanseatic merchants here, and most of the realm's
#    sovereign debt was rolled over each January.
shl_kieler_umschlag_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = SHL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:kiel
	}

	estates_that_like = {
		burghers_estate
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# The Umschlag fixes Kiel as the duchies' clearing house: every
	# January the realm's credit settles here, and trade taxation runs
	# permanently smoother.
	neutral_modifier = {
		tax_income_efficiency = 0.05
	}

	# Properly maintained, the Kieler Umschlag is the bond market of
	# the north: cheaper sovereign credit, a thicker burgher purse,
	# and steady rural-to-urban capital flow as the Junkers reinvest
	# their lending profits in commercial farming.
	positive_modifier = {
		scale = { value = scope:maintenance }
		bank_interest = -0.03
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
		global_monthly_urban_control = 0.003
	}

	# Neglected, the credit market thins: rates climb, the duchies'
	# debts compound at usurious terms, and Holstein peasant freeholds
	# fall into the hands of foreclosing creditors -- the late-16th
	# century pattern of Bauernlegen.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_inflation = 0.009
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
	}
}

# 4. Adelige Klöster -- the four Protestant noble convents (Itzehoe,
#    Preetz, Uetersen, and St. Johannis at Schleswig) that survived
#    the Reformation as collegiate foundations for unmarried daughters
#    of the Schleswig-Holstein nobility. Reorganised by the
#    Klosterordnung after 1542, they held vast church-estate lands,
#    managed by elected Pröpstinnen drawn from the Ritterschaft, and
#    served as both noble retirement-houses and a Lutheran-aristocratic
#    learned milieu.

		clergy_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# The Klosterordnung formally ties the four houses to the Lutheran

	neutral_modifier = {
		clergy_estate_min_tax = 0.05
	}

	# Sustained, the noble convents anchor a learned aristocratic

# 5. Deutsche Kanzlei -- the German Chancery, the duchies' core
#    administrative organ. Originally based at Gottorp under the
#    Holstein-Gottorp dukes, with a counterpart at Glückstadt for the
#    royal share after 1648, it conducted all official business of the
#    duchies in German (in pointed contrast to Copenhagen's Danske
#    Kancelli) and codified the Schleswigsches Landrecht, the
#    Holsteinisches Landrecht, and the Verordnungen that governed
#    them. From 1806 it was reorganised as the Schleswig-Holstein
