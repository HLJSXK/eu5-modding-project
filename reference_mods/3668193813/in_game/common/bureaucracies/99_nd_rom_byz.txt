###############################################################
# National Destinies -- ROM_BYZ supplemental bureaucracies
#
# ROM_BYZ_f is the BYZ/TRE-only Italian Roman restoration. The
# country IS legally ROM, so the player ALSO has ROM_f's seven
# bureaucracies and (via has_or_had_tag = BYZ) the eleven vanilla
# BYZ bureaucracies plus five National Destinies BYZ ones. These
# six bureaucracies are NEW institutional content specific to the
# Greek-Italian fusion the ROM_BYZ path produces -- post-1453
# Constantinopolitan refugees serving the restored basileus
# directly in Italy, Italograecia formalized, the Filioque
# reconciliation institutionalized, and a Mediterranean naval
# administration fusing Venetian-Genoese arsenale practice with
# the Byzantine dromon corps.
#
# Gating:
#   potential = { has_or_had_tag = ROM   has_variable = nd_rom_byz_formed }
# This keeps these bureaucracies invisible to the parallel ROM_f
# (Latin-only) path while persisting them across any future
# successor formation (has_or_had_tag, not tag).
#
# All IDs are rom_byz_-prefixed.
###############################################################


# =====================================================================
# Senatus Bilinguis Restitutus -- the Bilingual Senate Restored
#
# After 1453 the only court left that conducted real business in both
# Latin and Greek was the imperial chancery in exile -- George of
# Trebizond's Vatican translation bureau and Bessarion's Roman household
# trained an entire generation of clerics literate in both. Under the
# restored basileus this becomes the official imperial chancery: every
# decree issued in two languages, embassies received in either.  Where
# ROM_f's Senatus Romanus Renovatus is monolingual Latin (citizen
# commune of Rome, Pierleoni 1144), this is the bilingual successor: an
# imperial chancery whose dignity-of-office is its mastery of both
# tongues.
# =====================================================================
rom_byz_senatus_bilinguis_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = ROM
		has_variable = nd_rom_byz_formed
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: a permanent +1 to diplomatic capacity (the bilingual
	# chancery sustains more standing missions than a monolingual court),
	# paid for by a permanent overhead on cabinet decision-making
	# (every minute, every decree, has to be agreed in two tongues)
	neutral_modifier = {
		diplomatic_capacity = 1
		country_cabinet_efficiency = -0.05
	}

	# positive: the chancery functions -- legates fluent in both rites,
	# scribes who can edit a Greek synodal letter and a Latin papal bull
	# in the same afternoon
	positive_modifier = {
		scale = { value = scope:maintenance }
		diplomatic_reputation = 2
		cultures_capacity = 1
		global_burghers_max_literacy = 5
	}

	# negative: bilingualism degenerates into factionalism -- the Latin
	# scribes accuse the Greek of disloyalty to Rome and vice versa,
	# important correspondence sits in the wrong pile for months
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		country_cabinet_efficiency = -0.10
		global_estate_satisfaction_decay = 0.015
	}
}

# =====================================================================
# Commissio Filioque -- the Reconciliation Commission
#
# At the Council of Florence in 1439, after nine sessions on the Filioque,

# July. It was repudiated in Constantinople in 1484. Under the restored
# basileus, who personally embodies the union, a standing commission
# of theologians -- Latin canonists and Greek hieromonks -- works
# permanently to keep the doctrinal language reconciled, hear appeals
# of doctrinal disputes, and certify legitimate teaching across both
# rites.  This is the institutional version of the Romanitas Vincit /
# Hellenikos Romaios choice the formation event poses.
