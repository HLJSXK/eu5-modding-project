# Pagarruyung (Minangkabau) bureaucracies
# Each models a real institutional layer of the kingdom: the tripartite
# kingship, the council of four high ministers, the village-republic
# system, and the aristocratic political tradition (Lareh Koto Piliang).

rajo_tigo_selo_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = PGR
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		clergy_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Tripartite kingship is on the books -- legitimacy of the throne is
	# distributed across three seats by simply existing.
	neutral_modifier = {
		monthly_legitimacy = 0.05
	}

	# At full investment, the three royal courts coordinate: legitimacy
	# flows, factionalism is contained, custom and faith align.
	positive_modifier = {
		scale = { value = scope:maintenance }
		stability_cost_efficiency = 0.1
		global_estate_satisfaction_from_legitimacy = 0.075
		global_separatism = -0.05
	}

	# When neglected, the three thrones drift apart -- Raja Adat and Raja
	# Ibadat overshadow Raja Alam, and the executive is ignored. Custom
	# and clergy quarrel, and the kingdom slips toward institutional
	# paralysis.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		legislative_efficiency = -0.15
		stability_decay = 0.00025
	}
}

	}

	estates_that_like = {
		peasants_estate
		burghers_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the nagari as the unit of governance shifts authority
	# downward: the elected councils sit permanently above the throne in
	# matters of village life.
	neutral_modifier = {
		crown_estate_blocked_from_cabinet = yes
	}

	# When the throne respects nagari autonomy, the highland villages
	# return the favour with grain, manpower, and quiet loyalty.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_population_growth = 0.0008
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		global_monthly_rural_control = 0.005
	}

	# When the throne breaks faith with the nagari -- demanding levies
	# beyond custom, taxing the rumah gadang, overruling village courts --
	# every village becomes a potential rebellion centre.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.1
		stability_cost_efficiency = -0.15
	}
}

	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
		burghers_estate
	}

	# Adopting the Koto Piliang lareh codifies a hierarchy of penghulu
	# titles -- the great clans always sit above the rest.
	neutral_modifier = {
		nobles_estate_min_tax = 0.05
	}

	# At full strength, the aristocratic lareh disciplines the highland
	# nagari: clan chiefs answer to one another, militias muster reliably,
	# and feuds are quickly arbitrated by the senior clan.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		global_levy_size_modifier = 0.1
		global_max_control = 0.02
	}

	# Neglected, the Koto Piliang ranks calcify into hereditary privilege:
	# common clans see no advancement, merchant rantau communities are
	# squeezed by senior penghulu, and resentment builds in the lower
	# villages.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.025
		peasants_estate_max_tax = -0.05
	}
}
