# National Destinies -- Westphalian bureaucracies
# 5 historically-grounded institutions of the Westphalian state from the
# medieval Vehmic tribunals through the Imperial Circle and the prince-bishopric
# diets to the post-1648 settlement machinery.
# Auto-modifier grants 2 slots; capital adds 1 = 3 effective. Player picks 3 of 5.
#
# Scope note: WES is a Level 2 formable but Westphalia was institutionally
# distinctive even by Imperial standards -- a regional Reichskreis chaired
# jointly by the Bishop of Munster and the Duke of Cleves, the secret Vehmic
# courts whose Free Stools tried defendants from across the Empire, the
# closed cathedral chapters of Munster and Paderborn that elected prince-
# bishops from a quartered-nobility pool, the Westphalian Third of the
# Hansa anchored at Dortmund and Soest, and the Markgenossenschaft commons
# that managed forests and pastures by villager assembly.

# 1. Westfaelischer Reichskreis -- the Westphalian Imperial Circle established
#    by the 1500/1512 Reichstag reforms, one of ten regional Kreise that
#    coordinated defence, the execution of imperial law, and apportioned the
#    Roemermonate war-tax across territorial members. The circle was directed
#    by two co-presidents -- the Bishop of Munster (kreisausschreibende Stand
#    geistlich) and the Duke of Cleves-Juelich (weltlich) -- meeting at
#    Kreistage that decreed regional levies, regulated coinage, and policed
#    the Landfrieden. After 1648 the circle assumed the new Friedensvollzug
#    role, executing the territorial and confessional clauses of the treaty.
wes_reichskreis_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = WES
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A standing Kreistag chancery in Munster keeps the regional levy roll
	# current and coordinates fortress garrisons across the circle.
	neutral_modifier = {
		fort_maintenance_efficiency = 0.05
	}

	# Funded, the circle apportions the Roemermonate cleanly, executes
	# imperial mandates without local resistance, and keeps the Landfrieden
	# -- the regional treaty obligations are honoured in full.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_garrison_size_modifier = 0.10
		coalition_strength_tolerance = 0.10
		diplomatic_reputation = 1
	}

	# Underfunded, the circle's executive machinery seizes up: the co-
	# presidents quarrel over precedence, levy assessments fall into arrears,
	# and noble Standesherren refuse to muster their contingents.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_levy_size_modifier = -0.10
		nobles_estate_satisfaction_decay = 0.020
	}
}

# 2. Femgericht -- the Vehmic free courts whose secret tribunals sat at the
#    Freistuehle of Dortmund, Arnsberg, and Volmarstein and reached their
#    apogee between 1300 and 1500. Operating under the nominal authority of
#    the Archbishop of Cologne as Duke of Westphalia, the Femgericht summoned
#    defendants by sealed writ; verdicts -- almost always hanging -- were
#    executed by Frohnboten across the Empire. Free judges (Freischoeffen)
#    swore secrecy on pain of death and were drawn from sworn freemen rather
#    than territorial nobility, making the Veme a parallel justice that
#    bypassed the seigneurial courts of the Westphalian aristocracy.
wes_femgericht_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = WES
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		OR = {
			owns = location:dortmund
			owns = location:arnsberg
			owns = location:soest
		}
	}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A standing tribunal of sworn Freischoeffen issues sealed writs that
	# nobles cannot quash through their own seigneurial courts.
	neutral_modifier = {
		nobles_estate_blocked_from_cabinet = yes
	}

	# Funded, the Veme suppresses banditry on the Hellweg, breaks noble
	# private wars before they spread, and lets the crown reach defendants
	# the territorial courts will not touch.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_monthly_control = 0.003
		global_separatism = -0.05
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Neglected, the Femgericht decays into a rumour-court -- denunciations
	# multiply, the secret writs become tools of private vendetta, and the
	# territorial nobility rejoices at the institution's discredit.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_cost_efficiency = -0.1
		burghers_estate_satisfaction_decay = 0.015
	}
}

# 3. Domkapitel -- the cathedral chapters of Munster, Paderborn, and Osnabrueck
#    that elected the prince-bishops of Westphalia from among the imperial
#    knighthood. Canonries required the Ahnenprobe -- proof of sixteen noble
#    ancestors -- making each chapter a closed aristocratic preserve. The
#    chapters drafted the electoral capitulations (Wahlkapitulationen) that
#    constrained each new prince-bishop, controlled the cathedral chanceries,
#    and through the Stiftsfehde tradition could invalidate elections by force.
#    The Erbmaenner -- patrician lineages of Munster city -- bridged the
#    chapter and the urban diet, giving Westphalia its distinctive aristocratic-
#    burgher fusion at the territorial level.
wes_domkapitel_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = WES
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		clergy_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the cathedral chapters' electoral monopoly binds the
	# territorial church to the imperial knighthood and ties Munster's
	# patrician Erbmaenner into the prince-bishopric's governance.
	neutral_modifier = {
		clergy_estate_can_participate_in_parliament = yes
	}

	# Funded, the chapters draft careful electoral capitulations, the
	# canons supply learned councillors to the chancery, and the Erbmaenner
	# patriciate keeps Munster city aligned with the territorial estates.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction

	}

	# Underfunded, the Wahlkapitulationen become extortionate, vacant
	# canonries are auctioned to the highest-quartered bidder, and rival
	# noble factions ignite Stiftsfehden that paralyse the chancery.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		global_estate_satisfaction_decay = 0.010
	}
}

# 4. Westfaelisches Hansedrittel -- the Westphalian Third of the Hanseatic
#    League, one of three voting tertiae at the Hansetag, comprising Dortmund,
#    Soest, Munster, Lippstadt, Osnabrueck, and the smaller members of the
#    Maas-Niederrhein region. The Westphalian quarter coordinated tolls on the
#    Hellweg, ran the Hanseatic Kontore at Bergen and Brugge through their
#    appointed Aeltermaenner, defended urban liberties against territorial
#    princes, and disciplined member towns through Verhansung -- temporary
#    expulsion from the league. Under Soest's leadership it dominated the
#    inland salt and grain trade until the Thirty Years' War broke its
#    commercial network.

			owns = location:dortmund

	# Funded, the Aeltermaenner of the Westphalian quarter staff the Bergen

			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.020
		global_separatism = 0.05

# 5. Markgenossenschaft -- the Westphalian commons assemblies (Markverband)
#    that managed forest, pasture, peat, and waters by sworn villager councils
#    under codified Markenrechte. Combined with the distinctive Eigenbehoerige
#    tenure -- where peasants were personally bound to a manor but held
#    heritable plots and meaningful communal voice -- the Markgenossenschaft
#    gave Westphalian peasants more security than East-Elbian Gutsherrschaft
#    villagers ever enjoyed. The system survived every territorial reform
#    until the Markenteilungen of the early nineteenth century broke up the
#    commons. The Holzgrafen presided over the Markdinge; defaulters lost
#    grazing rights but not their tenancy, anchoring rural stability across
#    centuries of war and confessional division.
wes_markgenossenschaft_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = WES
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Codifying the Markenrechte gives villagers a sworn assembly under
	# their Holzgrafen and stabilises the rural fabric of the territory.
	neutral_modifier = {
		peasants_estate_can_participate_in_parliament = yes
	}

	# Funded -- meaning the Markdinge convene on schedule, the Holzgrafen
	# are paid, and grazing-right disputes are settled by communal record --
	# the countryside feeds itself, recruits steadily, and breeds none of
	# the unrest that East-Elbian villages produced in the same decades.
	positive_modifier = {
		scale = { value = scope:maintenance }
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		global_population_growth = 0.003
		pop_join_rebel_threshold = 0.02
	}

	# Neglected, the commons are encroached upon by noble Holzfrevel
	# (timber trespass), the Markdinge fall silent, and the Eigenbehoerige
	# tenure decays toward simple labour-rent -- the rural peace breaks.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_monthly_prosperity = -0.001
		peasants_estate_satisfaction_decay = 0.020
	}
}
