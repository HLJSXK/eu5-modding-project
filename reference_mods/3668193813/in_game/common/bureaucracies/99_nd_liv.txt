##############################
# Livonia - Historical Bureaucracies
##############################
# Four institutions of the Livonian Confederation and its Polish/Swedish/Russian
# successor administrations (c. 1237-1721). The auto_modifier grants 2 slots at
# formation; the capital adds 1 more; the player picks 3 of these 4 to maintain.
#
#  - Walk Landtag (Confederation diet, 1419-1559: Order, bishops, vassals, and
#    Hansa cities convening at Wolmar/Walk under Archbishop Ambundii's 1419
#    initiative; the working frame of medieval Livonia's collective rule)
#  - Livonian Order Konvent (Landmeister at Wenden/Riga + Großgebietiger
#    collegium of Landmarschall, Großkomtur, Drapier, Spittler, Tressler;
#    Komture commanding the Ordensburg network from Marienburg eastward)
#  - Hanseatic Stadtrat (Riga 1201, Reval 1248, Dorpat: governed under
#    Lübeck Law by elected Rat and four Bürgermeister; Brotherhood of
#    Blackheads as unmarried-merchant urban militia from 1407/1416)
#  - Ritterschaft Vasallenrecht (Knighthood corporations of Harju-Viru,
#    Ösel-Wiek, Estland, Livland: Mannrichter courts, Mannbuch enrollment,
#    Wackenbuch surveys, codified by the Privilegium Sigismundi Augusti
#    of 28 November 1561 into a Baltic German noble fiefdom that survived
#    until 1917-1920)
##############################

liv_walk_landtag_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = LIV
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# An active Landtag means decrees of any single ruler -- Master, archbishop,
	# bishop, or city council -- pass through a four-estate diet at Walk before
	# they bind the Confederation. The institution by its very existence
	# embeds collective ratification in Livonian governance.
	neutral_modifier = {
		legislative_efficiency = 0.05
	}

	# Working as designed: archbishop, Order, prince-bishops of Dorpat,
	# Ösel-Wiek, and Courland, the vassal corporations, and the Hanseatic
	# city deputies meet in regular session at Walk. Disputes that would
	# otherwise tip into private war are negotiated; the Confederation
	# coheres against external enemies (Pskov, Novgorod, Lithuania)
	# through a working forum the four estates accept as legitimate.
	positive_modifier = {
		scale = { value = scope:maintenance }
		stability_cost_efficiency = 0.1
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: Landtag sessions deadlock on the perennial Order-vs-archbishop
	# disputes, vassals refuse Master's summons, Hansa cities bargain
	# bilaterally with foreign princes, and the absence of a working forum
	# fragments the realm into faction. The crown -- whatever its source --
	# resents an institution that constrains it without producing decisions.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.15
		stability_decay = 0.00025
		crown_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
	}
}

liv_ordens_konvent_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = LIV
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Order is a military-religious congregation answering ultimately to
	# Rome; the Konvent system requires a Catholic confessional posture and
	# the institutional ties to the Teutonic Hochmeister and Marienburg.
	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		crown_estate
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The Order's celibate brethren staff the Großgebietiger collegium and
	# the commandry chapters; secular Baltic German lineages are
	# institutionally locked out of the realm's central command. The
	# Mannrichter and lay vassal corporations operate in parallel, but the
	# state's high offices are an order of monks-with-swords.
	neutral_modifier = {
		nobles_estate_blocked_from_cabinet = yes
	}

	# Working as designed: the Landmeister at Wenden (later Riga) presides
	# over the Großgebietiger -- Landmarschall, Großkomtur, Drapier, Spittler,
	# Tressler -- while Komture command the Ordensburg network from Riga to
	# Narva. Stone fortifications, professional heavy cavalry mustered from
	# the convents, and disciplined campaign-readiness give the realm a
	# military spine no neighboring polity can match brigade for brigade.
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_heavy_cavalry_power = 0.10
		global_defensive = 0.10
		army_initiative = 0.05
	}

	# Decaying: Konvent rosters thin as recruitment from the Empire dries up
	# (the Reformation will hollow the Order's Catholic recruitment base
	# entirely), Komture sell off Order lands to the secular nobility, and
	# the celibate-brethren monopoly on high office becomes a grievance the
	# blocked Baltic German lineages turn into open revolt. The fortress
	# network the Order maintained begins to crumble for lack of garrisons.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		global_garrison_size_modifier = -0.10
	}
}

liv_hanseatic_stadtrat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = LIV
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Stadtrat institution requires possession of a Hanseatic founding city
	# -- Riga (chartered 1201 under Lübeck Law), Reval/Tallinn (1248), or
	# Dorpat -- where the elected Rat, Bürgermeister, and Brotherhood of
	# Blackheads actually sit.
	allow = {
		OR = {
			owns = location:riga
			owns = location:tallinn
			owns = location:tartu
		}
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A working Hanseatic Rat means the wealthiest Low German merchant
	# lineages of Riga, Reval, and Dorpat hold elected office for life,
	# accumulate landed estates and shipping interests through the Brotherhood
	# of Blackheads, and present the realm a coherent commercial elite that
	# negotiates collectively with the Hochmeister, Lübeck, and Novgorod.
	neutral_modifier = {
		estate_enrichment = 0.05
	}

	# Working as designed: the Rat of each city, four Bürgermeister presiding
	# in rotation, sends magistrates to the Hansetag at Lübeck; the Riga
	# Peterhof manages the Novgorod fur and wax trade; the Brotherhood of
	# Blackheads (consecrated 1407 in Reval, 1416 in Riga) provides the
	# city militia and a permanent merchant-society that absorbs incoming
	# Low German immigrants into the urban patriciate. Riga's harbor revenue
	# alone rivals the Order's military budget.
	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_sea_efficiency = 0.10
		global_non_rural_monthly_prosperity = 0.05
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: the Hanseatic Kontor system fractures as Lübeck's primacy
	# falters and Dutch and English shippers route around it; Riga's Rat
	# turns into a closed oligarchy of three or four lineages; the Blackheads
	# devolve into a drinking society; and the rural population finds itself
	# squeezed between predatory urban factors and an unchecked grain market
	# that prices peasants out of their own crops.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_inflation = 0.002
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
	}
}

liv_ritterschaft_vasallenrecht_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = LIV
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# The Ritterschaft sits as the de facto provincial governor class
	# everywhere outside the Hansa city walls and the Order's commandries.
	# Each Knighthood -- Harju-Viru, Ösel-Wiek, Estland, Livland -- holds
	# its own Mannbuch enrollment and Mannrichter court, and through the
	# Wacken survey-and-corvée system administers the rural economy directly.
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Working as designed: the four Knighthoods muster their vassal cavalry
	# under indigenous corporate command, the Mannrichter courts adjudicate
	# disputes between vassals (later codified by the 1561 Privilegium
	# Sigismundi Augusti as the Indigenat right of self-administration),
	# and the Wackenbuch surveys collect predictable corvée labor from the
	# peasantry to outfit each knight's horse and arms. The Baltic German
	# lineage system that survives every change of sovereignty until 1917
	# is functioning at full strength.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_levy_size = 0.10
		army_light_cavalry_power = 0.05
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: the Wacken corvée hardens into outright Erbuntertänigkeit
	# (hereditary subjection -- the early-modern Baltic serfdom that grew
	# more onerous after 1561 even as Western Europe relaxed it), Mannrichter
	# justice becomes openly biased, peasants flee across the Russian border,
	# and the rural workforce that the realm's grain exports depend on
	# disintegrates into rebellion or simple flight.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		global_pop_promotion_speed = -0.05
		global_separatism = 0.10
	}
}
