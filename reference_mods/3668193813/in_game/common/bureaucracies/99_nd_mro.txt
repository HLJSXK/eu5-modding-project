mro_ruma_bichara_bureaucracy = {
    implementation_price = price:implement_bureaucracy_price
    maintenance_price = price:maintain_bureaucracy_price
    removal_price = price:remove_bureaucracy_price

    potential = {
        has_or_had_tag = MRO
        has_game_rule = nd_bureaucracies_enabled
    }

    allow = {}

    estates_that_like = {
        nobles_estate
        clergy_estate
    }
    estates_that_dislike = {
        crown_estate
    }

    # The council exists. Even idle, it gives the datus a venue and slows
    # autocratic drift toward the sultan.
    neutral_modifier = {
        global_nobles_estate_power = 0.025
    }

    # Functioning ruma bichara: deliberation supplies the sultan with elite
    # consensus and noble buy-in -- but every decision must be negotiated.

        scale = { value = scope:maintenance }
        nobles_estate_satisfaction_recovery = 0.001
        country_cabinet_efficiency = 0.05
        clergy_estate_target_satisfaction = 0.05
    }

    # Neglected council: meetings stop being called, factional datus harden
    # into rival blocs, the realm fractures from within.
    negative_modifier = {
        scale = {
            value = 1
            subtract = scope:maintenance
        }
        stability_decay = 0.0005
        nobles_estate_satisfaction_decay = 0.10
    }
}

    implementation_price = price:implement_bureaucracy_price
    maintenance_price = price:maintain_bureaucracy_price
    removal_price = price:remove_bureaucracy_price

    potential = {
        has_or_had_tag = MRO
        has_game_rule = nd_bureaucracies_enabled
    }

    # The pandita network presupposes Sunni Islam having taken root --
    # at game start MRO already qualifies, but a religion-flip would close it.
    allow = {
        religion = religion:sunni
    }

    estates_that_like = {
        clergy_estate
    }
    estates_that_dislike = {
        nobles_estate

    # An organised teaching ulama exists; every datu has his pandita.
    neutral_modifier = {
        tolerance_own = 0.5

    # Working madrasah: literate clerks, tighter doctrinal cohesion, and the
    # ulama's prestige underwrites the sultan's caliphal claim.

        scale = { value = scope:maintenance }
        global_clergy_max_literacy = 0.05
        global_pop_conversion_speed_modifier = 0.15
        clergy_estate_satisfaction_recovery = 0.001
    }

    # Decaying madrasah: pandita lineages quarrel, doctrinal fights become
    # political ones, and the noble datus stop deferring to scholar opinion.
    negative_modifier = {
        scale = {
            value = 1
            subtract = scope:maintenance
        }
        tolerance_heretic = -1.0
        global_nobles_estate_power = 0.025
    }
}

mro_tarsila_chancery_bureaucracy = {
    implementation_price = price:implement_bureaucracy_price
    maintenance_price = price:maintain_bureaucracy_price
    removal_price = price:remove_bureaucracy_price

    potential = {
        has_or_had_tag = MRO
        has_game_rule = nd_bureaucracies_enabled
    }

    # The tarsila chancery has to be situated where the dynastic archives
    # are kept. Maimbung was the Sulu seat; without it the lineage records
    # are just claims on paper.
    allow = {
        owns = location:maimbung
    }

    estates_that_like = {
        crown_estate
        clergy_estate
    }
    estates_that_dislike = {
        nobles_estate
    }

    # Maintained genealogies bolster the sultan's reputed descent from the
    # Prophet via Sayyid Abu Bakr, baseline legitimacy.
    neutral_modifier = {
        monthly_legitimacy = 0.05
    }

    # An active chancery validates new datu claims, polices marriage alliances,
    # and can deny ambitious regional lords their genealogical credentials.
    positive_modifier = {
        scale = { value = scope:maintenance }
        diplomatic_reputation = 1
        global_estate_satisfaction_from_legitimacy = 0.20
        antagonism_religion_influence = 0.15
    }

    # Unmaintained scrolls: forgeries circulate, claimants multiply,
    # rival sharifs press their own lines, and the sultan's uniqueness erodes.
    negative_modifier = {
        scale = {
            value = 1
            subtract = scope:maintenance
        }
        prestige_decay = 0.0025
        nobles_estate_satisfaction_decay = 0.10
    }
}
