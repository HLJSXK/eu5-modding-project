##############################
# Mutapa Empire - Historical Bureaucracies
##############################
# Four institutions of the Shona kingdom (c. 1430-1760). The auto_modifier
# grants 1 slot at formation; the capital adds 1 more; the player picks 2
# of these 4 institutions to maintain.
#
#  - Mhondoro Oracle (royal-ancestor mediums + Mwari shrines at Matonjeni)
#  - Kuronzera Cattle Clientship (royal cattle loaned for personal-bond service)
#  - Curva Annual Tribute (yearly vassal-chief renewal-of-submission ceremony)
#  - Mukomohasha Captaincy (appointed supreme military command outside lineage)
##############################

mhondoro_oracle_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MTP
		has_game_rule = nd_bureaucracies_enabled
	}

	# The Mwari shrine network was rooted in the Matobo Hills (Matonjeni). The
	# institution can also operate where royal devotional authority is the
	# explicit cultural-political stance of the realm.
	allow = {
		OR = {
			owns = location:abutua
			owns = location:hondosaka
			societal_value:spiritualist_vs_humanist > 0
		}
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Standing tolerance: an active oracle network keeps the indigenous religion
	# politically protected against missionary intrusion.
	neutral_modifier = {
		tolerance_own = 1
	}

	# Working as designed: Mhondoro mediums legitimate the king (no Mwenemutapa
	# without ancestral consent), the Matonjeni rain-cult sanctions decrees,
	# and the network reaches into every chiefdom binding peasant loyalty
	# around the throne rather than around the lineage chief.
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_devotion = 0.03
		global_pop_conversion_speed_modifier = 0.1
		monthly_legitimacy = 0.05
	}

	# Decaying: the same mediums begin to authorize rival claimants, withdraw
	# kingship legitimation, and shelter rebellions under sacred cover. The
	# clergy estate that ran the network turns against an unmaintained throne.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		pop_join_rebel_threshold = -0.10
		global_separatism = 0.10
	}
}

kuronzera_clientship_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = MTP
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The royal cattle herd is the institutional fabric: while kuronzera
	# operates, peasant clients are bound directly to the throne for service
	# rather than to their lineage chief.
	neutral_modifier = {
		nobles_estate_blocked_from_cabinet = yes
	}

	# Working as designed: royal cattle loaned to commoners and lesser chiefs
	# in exchange for labor, military service, and a share of calves. The
	# herds thread the realm with thousands of personal bonds running directly
	# to the Mwenemutapa, bypassing hereditary lineage authority.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_crown_estate_power = 0.10
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		global_monthly_rural_control = 0.005
	}

	# Decaying: drought, rinderpest, or royal greed collapses the network.
	# Clients drift back to their old lineage chiefs; the throne loses its
	# direct grip on the rural population.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.10
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		stability_decay = 0.00025
	}
}

curva_tribute_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = MTP
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		crown_estate
		nobles_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# The annual ceremony as institutional fact: an active curva is permanent
	# revenue extraction from the realm.
	neutral_modifier = {
		estate_enrichment = 0.05
	}

	# Working as designed: every vassal chief brings ivory, gold, hoes, and
	# cloth to renew submission; absence is rebellion. The king redistributes
	# part of the gifts to bind chiefs back into the hierarchy. Joao dos Santos
	# in 1609 described curva as the central event of the imperial calendar.
	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = 0.10
		monthly_legitimacy = 0.04
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Decaying: chiefs no longer attend; the redistribution lapses but the
	# extractive demand on cultivators below them does not. The throne's
	# annual binding ritual hollows into rural exploitation without political
	# cohesion.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_inflation = 0.002
		annexation_speed_modifier = -0.15
	}
}

mukomohasha_captaincy_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MTP
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
		peasants_estate
	}

	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Working as designed: the Mukomohasha is the supreme captain of the royal
	# host, an appointed office held distinct from the great lineage chiefs.
	# Subordinate captains drawn from across the realm muster, drill, and
	# campaign under royal -- not aristocratic -- direction.
	positive_modifier = {
		scale = { value = scope:maintenance }
		land_morale_modifier = 0.10
		discipline = 0.025
		global_levy_size_modifier = 0.10
	}

	# Decaying: provisioning the marching column without functional command
	# means the column lives off the cultivators it crosses. The lineage chiefs
	# who were displaced by the captaincy take the opportunity to reassert
	# control over their own warriors.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		army_initiative = -0.15
	}
}
