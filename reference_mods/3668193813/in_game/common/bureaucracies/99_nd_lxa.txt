# National Destinies -- Lan Xang bureaucracies
# Five historically-grounded institutions of the Tai-Theravada kingdom of
# the Million Elephants. The auto_modifier grants 2 slots; the capital
# adds a third. Player picks 3 of 5.

# 1. Senabodi -- the Cao's high council of senior ministers (the maha-sena
#    or thao phraya), drawn from a small circle of court nobility. Codified
#    under King Samsenthai (r. 1373-1416) as the central organ of royal
#    government in Muang Sua, the Senabodi was the closed aristocratic
#    apparatus through which the paramount king ruled, and its concentration
#    of power in great houses helped trigger the tripartite split of 1707.
lxa_senabodi_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = LXA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		crown_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	neutral_modifier = {
		peasants_estate_blocked_from_cabinet = yes
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		country_cabinet_efficiency = 0.10
		legislative_efficiency = 0.10
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_satisfaction_decay = 0.02
		monthly_complacency = 0.03
	}
}

# 2. Mueang-Long -- the sacred royal city as ritual-political center, built
#    around the Phra Bang Buddha image (palladium of the kingdom, gifted by
#    the Khmer court to Fa Ngum c. 1359) and from 1565 the Phra Keo Emerald
#    Buddha enshrined at Wat Pha Kaeo in Vientiane. The Bun Phra Wet annual
#    cycle, the That Luang stupa festival, and the royal ploughing rite
#    bound dynasty, sangha, and capital into a single ritual order. The
#    palladium's later sojourns -- Vientiane, Champasak, Bangkok -- traced

# 3. Sangha-Raja -- the Theravada Buddhist hierarchy, formalised under

# 5. Kha-Lao Tribute -- the extractive system by which lowland Lao
#    chao-mueang and Mekong-valley merchants drew annual tribute in
#    forest goods (cardamom, benzoin, sticklac, eaglewood, deer-skins,
#    elephant ivory) from the Khmu, Ksing-Mul, and other upland peoples
#    collectively termed kha (servant). The tribute fed the Mekong
#    long-distance trade with Ayutthaya and Hue while entrenching the
#    second-class legal status of the highland populations -- the
#    structural fault that repeatedly ignited upland revolts (the Khmu
#    risings of the seventeenth century, the later Hmong wars).
