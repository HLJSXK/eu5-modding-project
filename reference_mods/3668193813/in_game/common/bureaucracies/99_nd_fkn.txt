# National Destinies -- Franconian (FKN) bureaucracies
# Five historically-grounded institutions of the Franconian state, drawn from
# the corporate political life of the Imperial Circle established in 1500 and
# the bishoprics, free cities, and knightly cantons that gave it weight.
# Auto-modifier grants 2 slots; the capital adds 1 = 3 effective. Picks 3 of 5.
#
# Scope note: FKN is a Level 2 formable, but the historical Franconian
# political space was institutionally dense -- the Kreistag at Nuremberg, the
# Domkapitel of two prince-bishoprics, the Nuremberg patriciate, the Imperial
# Knights' canton, and the Landfrieden machinery were all running concurrently.
# A unified Franconia inherits this layered apparatus.

# 1. Fränkischer Kreistag -- the assembly of the Franconian Imperial Circle,
#    instituted under the 1500 reform and operative until 1806. Its
#    directorate (the Kreisausschreibungsamt) was a standing duumvirate of
#    the Bishop of Bamberg and the Margrave of Brandenburg-Ansbach; the
#    Kreistag itself convened at Nuremberg and matriculated each estate's
#    contribution to the imperial register, election of judges to the
#    Reichskammergericht, supervision of the coinage, and execution of the
#    Landfrieden. Of the ten Imperial Circles it was the most active and the
#    most successful at maintaining a standing Kreistroops levy.
fkn_kreistag_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = FKN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Convening the Kreistag binds the Franconian estates into a corporate
	# body; the directorate's clerks coordinate the matriculation of levy
	# quotas, court fees, and coinage assays across the territory.
	neutral_modifier = {
		nobles_estate_can_participate_in_parliament = yes
	}

	# Funded, the Kreistag delivers the Empire's most reliable Kreistroops,
	# moderates feuds among Franconian estates, and shields commerce by
	# enforcing the Landfrieden -- but the duumviral directorate keeps the
	# unified state from acting on its own initiative.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.075
		global_separatism = -0.05
		legislative_efficiency = 0.075
	}

	# Underfunded, the Kreistag fragments into camera-bound disputes between
	# the prince-bishoprics and the margraviates -- matriculation lapses,
	# feuding nobles disregard the directorate, and the unified state's
	# attempts to legislate stall in procedural objections.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_estate_satisfaction_decay = 0.020
		stability_decay = 0.00025
	}
}

# 2. Domkapitel -- the cathedral chapters of Bamberg and Würzburg, who
#    elected the prince-bishops via Wahlkapitulationen (electoral
#    capitulations) that constrained episcopal taxation, legislation, and
#    appointments. From the 15th century onward, the Domkapitel of Würzburg
#    routinely co-governed the Hochstift; the Bamberg chapter ran episcopal
#    justice through its Hofgericht and licensed the diocesan press. This
#    institution gave the Franconian clergy an entrenched aristocratic
#    chapter that no prince-bishop could circumvent.
fkn_domkapitel_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = FKN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:catholic
		OR = {
			owns = location:wurzburg
			owns = location:bamberg
		}
	}

	estates_that_like = {
		clergy_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the Wahlkapitulation as a binding charter places the
	# Hochstift's appointments and taxation under the canons' consent --
	# the chapter's seat is permanent, with or without an active bishop.
	neutral_modifier = {
		monthly_devotion = 0.05
	}

	# Maintained, the Domkapitel runs the diocesan press, the cathedral
	# schools at Würzburg and Bamberg, and the episcopal Hofgericht --
	# a literate clerical aristocracy with deep loyalty to the Hochstift,
	# but the chapter's veto fetters the ruler's revenue.
	positive_modifier = {
		scale = { value = scope:maintenance }
		research_speed_modifier = 0.05
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
		tax_income_efficiency = -0.05
	}

	# Neglected, the chapter's Wahlkapitulation slides into open
	# defiance: canons refuse extraordinary subsidies, episcopal nominees
	# are blocked, and the diocesan courts degenerate into protracted
	# canonical disputes between cathedral and palace.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		stability_cost_efficiency = -0.1
	}
}

# 3. Innerer Rat -- the Nuremberg patrician Lesser Council, formalised
#    around 1320 and closed to a roster of ~42 patrician families by the
#    1521 dance statute. Its septemvirate of Ältere Herren and triumvirate
#    of Oberste Hauptleute supervised every city office, financed the
#    Imperial Diet whenever it sat at Nuremberg, and from 1424 held the
#    custody of the Imperial Regalia at the Heilig-Geist-Spital under
#    Sigismund's charter. The Imperial Regalia were displayed annually at
#    the Heiltumsweisung, drawing the trade fair that anchored Nuremberg's
#    long-distance commerce.
fkn_innerer_rat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = FKN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:nuremberg
	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		clergy_estate
		peasants_estate
	}

	# Granting the Nuremberg patriciate its closed council and the custody
	# of the Imperial Regalia binds the city's commerce to the unified
	# state -- a permanent fiscal bridge between Franconia and the Empire.
	neutral_modifier = {
		burghers_estate_can_participate_in_parliament = yes
	}

	# Funded, the Innerer Rat handles fiscal subsidies in Nuremberg's silver
	# and the Heiltumsweisung trade fair pours merchant capital into the
	# city's workshops -- the burgher patriciate becomes an indispensable
	# revenue machine, even as the closed council excludes the guilds.
	positive_modifier = {
		scale = { value = scope:maintenance }
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
		trade_land_efficiency = 0.10
		global_monthly_urban_control = 0.003
	}

	# Underfunded, the patriciate hoards revenue, the closed council ignores
	# the rural Hinterland, and the imperial trade fair drifts to other
	# free cities -- artisanal guilds protest, peasants resent urban
	# privilege, and the city's faction disputes spill into civic unrest.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		peasants_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}

# 4. Ritterkanton -- the Franconian canton of the Imperial Knights,
#    organised under Maximilian I around 1500 and formalised in 1577 as
#    one of three Ritterkreise. The six cantons of Franconia (Odenwald,
#    Gebürg, Steigerwald, Altmühl, Baunach, Rhön-Werra) maintained their
#    own captains, archives, and matriculation lists; they paid voluntary
#    contributions (Charitativsubsidien) to the emperor in lieu of taxes
#    and were exempt from the territorial sovereignty of the prince-bishops
#    or margraves whose lands enclosed them. The Ritterschaft thus formed a
#    nobility-within-a-nobility, structurally hostile to any centralising
#    Franconian state.
fkn_ritterkanton_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = FKN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		burghers_estate
		crown_estate
	}

	# Recognising the cantonal matriculation lists places the Reichsritter
	# under their own captains and exempts their lands from territorial
	# levies -- a permanent enclave of imperial-immediate nobility within
	# the Franconian state.
	neutral_modifier = {
		burghers_estate_blocked_from_leading_military = yes
	}

	# Maintained, the Charitativsubsidien flow steadily into the war chest,
	# the cantonal cavalry musters quickly along the Main valley, and the
	# Reichsritter supply trained captains to the Franconian colours -- at
	# the cost of permanently excluding burgher officers from command.
	positive_modifier = {
		scale = { value = scope:maintenance }
		army_heavy_cavalry_power = 0.075
		monthly_army_tradition = 0.10
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Neglected, the cantons refuse Charitativsubsidien, the captains feud
	# over precedence in their archives, and the Knights revert to their
	# old habit of robber-baron raiding along the trade roads -- towns
	# arm themselves at their own expense and the rural countryside burns.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.020
		global_monthly_rural_control = -0.003
	}
}

# 5. Landfriede -- the regional public-peace ordinance executed through the
#    Imperial Circle. The Landfrieden tradition in Franconia ran from the
#    Würzburg Landfriede of 1287 through the Eternal Public Peace
#    (Ewiger Landfriede) of 1495 and was administered after 1500 by the
#    Kreistag's Landfriedenswahrung commissioners. It outlawed feuds
#    between estates, proclaimed peace on the great trade roads from
#    Nuremberg to Frankfurt and Würzburg to Erfurt, and pursued
#    Friedensbrecher through the Reichskammergericht. Where it held, the
#    Franconian markets thrived; where it failed, the Reichsritter resumed
#    their raiding.
fkn_landfrieden_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = FKN
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Proclaiming the Eternal Public Peace places the great roads from
	# Nuremberg to Frankfurt and Würzburg to Erfurt under imperial
	# protection; feud rights of the lesser nobility are formally extinct.
	neutral_modifier = {
		control_importance_modifier = 0.05
	}

	# Funded, Landfriedenswahrung commissioners patrol the trade roads,
	# the Reichskammergericht moves quickly on Friedensbrecher, and rural
	# peace lets the peasantry farm without quartering levies; the great
	# Franconian fairs at Nuremberg and Frankfurt knit closer together.
	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_land_efficiency = 0.075
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		stability_investment = small_stability_investment
	}

	# Neglected, the commissioners are unpaid and the Reichskammergericht's
	# writs are ignored; the lesser nobility resumes feuding, robber-knight
	# raiding closes the trade roads, and rural unrest follows the burning
	# convoys -- the Franconian peace becomes a parchment fiction.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_monthly_rural_control = -0.003
		pop_join_rebel_threshold = -0.02
	}
}
