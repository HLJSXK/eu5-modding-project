# National Destinies -- Bavarian (Wittelsbach) bureaucracies
# 5 historically-grounded institutions of the Bavarian state from the late
# medieval Duchy through the Electorate (1623) to the early Kingdom (1806).
# Auto-modifier grants 2 slots; capital adds 1 = 3 effective. Player picks 3 of 5.
#
# Scope note: BAV is a Level 2 formable but Wittelsbach Bavaria was an
# institutionally dense Catholic state with a Privy Council, a chartered
# estates Diet, a fiscal chamber running the salt regalia, an ecclesiastical
# council that ran the Counter-Reformation, and the Hofmark system of
# patrimonial seigneurial justice that survived until 1848.

# 1. Geheimer Rat -- the Privy Council established by Albert IV in 1488 and
#    reorganised under Albert V in 1570 as the central executive body of the
#    duchy. Its councillors -- a small body of trained jurists drawn from
#    territorial nobility and university-trained officials -- prepared decrees
#    for the duke, supervised the chancellery, and from 1620 ran wartime
#    government for Maximilian I. By the 17th century it had effectively
#    displaced the older Hofrat as the seat of executive power in Munich.
bav_geheimer_rat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = BAV
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# A standing council of trained jurists makes the duke's chancellery work
	# even when the duke himself is absent, ill, or a child.
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# Properly funded, the Geheimer Rat drafts decrees, manages dynastic
	# correspondence, and centralises decision-making in Munich at the cost
	# of the older feudal nobility's standing at court.
	positive_modifier = {
		scale = { value = scope:maintenance }
		legislative_efficiency = 0.10
		monthly_legitimacy = 0.05
		global_crown_estate_power = 0.05
	}

	# Underfunded, the council fragments into rival cliques of court favourites
	# and the chancery slows to a crawl -- decrees pile up, factional
	# secretaries leak documents, and the older nobility rejoices.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		country_cabinet_efficiency = -0.10
		stability_decay = 0.00025
	}
}

# 2. Hofkammer -- the Aulic Chamber, the duchy's central financial office
#    that ran the regalia revenues: the salt monopoly anchored on Reichenhall

#    at Munich, and the proceeds of ducal estates. Reorganised by Wilhelm IV
#    in the 1530s and again by Maximilian I after 1598, the Hofkammer made
#    the Bavarian salt monopoly one of the most lucrative regalian revenues
#    in the Empire and bankrolled the Catholic League.

# 3. Landstaende -- the Bavarian Estates Diet of prelates, nobles, and town
#    deputies that met irregularly at Munich (later Landshut) from the 14th
#    century. The 1577 Landtag produced the Landesfreiheitserklarung, the
#    territorial charter formalising estate rights to consent to extraordinary
#    taxation. The Diet effectively co-governed the duchy with the dukes
#    until Maximilian I's wartime emergencies sidelined it after 1620; it was
#    formally dissolved by Montgelas's reforms in 1808.

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price

	potential = {
		has_or_had_tag = BAV
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		clergy_estate
		burghers_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the Landtag as the chartered consenting body gives the
	# three estates a permanent seat in extraordinary taxation; it costs the
	# duke control of his own purse.
	neutral_modifier = {
		nobles_estate_can_participate_in_parliament = yes
	}

	# Funded -- meaning the duke convenes the Diet, honours its grievances,

# 5. Hofmarken -- the Bavarian system of patrimonial seigneurial jurisdiction
#    by which abbeys, knights, and noble landholders exercised lower courts,
#    police powers, and tax collection over their tenants. Codified in the
#    Landrecht of 1518 and successive Landgerichtsordnungen, the Hofmark was
#    the duchy's administrative substrate in the countryside: the rentmeister
#    coordinated, but the Hofmarksherren ran daily justice. The system
#    survived every reform until it was finally abolished by the Bavarian
#    constitution of 1818 and the Gemeindeedikt that followed.
