# Sansheng Liubu -- Three Departments (Zhongshu drafts, Menxia
# reviews, Shangshu executes) and the Six Boards under the
# Department of State Affairs. The Tang founded this checks-and-
# balances trinity; every later Chinese dynasty adapted it.
tgd_sansheng_liubu_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = TGD
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {}

	# neutral: standardized memorial flow processed at every level
	neutral_modifier = {
		country_cabinet_efficiency = 0.1
	}

	# positive: drafted edicts pass review and are executed across the realm
	positive_modifier = {
		scale = { value = scope:maintenance }
		legislative_efficiency = 0.1
		global_max_control = 0.02
		stability_cost_efficiency = 0.1
	}

	# negative: the three departments deadlock; reviewers veto drafters
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_complacency = 0.05
		global_pop_promotion_speed_modifier = -0.1
	}
}
