# National Destinies -- Mossi bureaucracies
# Five historically-grounded institutions of the unified Mossi state, drawing
# on the Mogho Naaba's court at Wagadugu (founded c. late 15th century by Naaba
# Niandfo, son of the Tenkodogo founder Naaba Wedraogo), the dual political-
# spiritual settlement between the conquering nakomse and the autochthonous
# tengsoba earth-priests, the patrilineal nakomse provincial system that fed
# the Mossi cavalry that turned back Mali and Songhai raids in 1328-1483, the
# weekly Friday Naam court at Wagadugu that bound chief to king through the
# rite of investiture, and the late-17th-century Yarse trade charter that
# settled Mande Muslim caravan-merchants under royal protection in the kola-
# salt-textile circuit between Salaga and Timbuktu. Auto-modifier grants 2
# slots; capital adds 1 = 3 effective. Player picks 3 of 5.
#
# All IDs prefixed with `msi_` to avoid cross-file collisions.

# 1. Nayiri Council of the Five Naabas -- the electoral and advisory council
#    of the Mogho Naaba, drawn since the founding of Wagadugu from five
#    palace ministers: the Baloum Naaba (head of the king's household), the
#    Gounga Naaba (chief of infantry), the Larle Naaba (keeper of the royal
#    tombs and customs), the Kamsaogo Naaba (manager of the palace eunuchs),
#    and the Ouidi Naaba (royal groom and chief of cavalry, dean of the
#    ministers). The council selected each new Mogho Naaba from the eligible
#    sons of the late king, and could pass over an heir judged unfit for
#    war or office. The same five offices governed the Yatenga court at
#    Ouahigouya. The council's veto over succession made it the institutional
#    counterweight to the throne for the entire pre-colonial era.
msi_nayiri_council_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MSI
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the Five Naabas as a chartered council means the Mogho Naaba
	# rules in consultation: succession runs through the Baloum and the Ouidi
	# Naaba's confirmation, and the throne speaks with the dignity of the
	# unanimous court rather than the mere voice of the king.
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Honoured -- the five offices held by men of stature, the Friday court
	# regularly assembled, the council's veto on succession respected -- the
	# kingdom enjoys orderly transitions of naam, the nakomse mobilise their
	# contingents on the Ouidi Naaba's call, and provincial chiefs accept
	# Wagadugu's edicts because the council guarantees them.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		global_levy_size_modifier = 0.10
		monthly_legitimacy = 0.05
	}

	# Sidelined -- the king ruling without his ministers, council seats left
	# vacant or filled with creatures of the throne -- the nakomse withhold
	# their cavalry contingents, succession disputes erupt on each death, and
	# the great provincial chiefs treat the Mogho Naaba as first among
	# equals rather than emperor.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.015
		global_separatism = 0.05
	}
}

# 2. Moro-Naba Friday Court -- the weekly Naam-investiture audience at
#    Wagadugu, attested in oral tradition since the foundation of the city
#    and still performed at dawn every Friday in the 21st century. The
#    Mogho Naaba emerges from the palace dressed in red and seated on a
#    horse as if for war; provincial chiefs, ministers, and lineage heads
#    arrange themselves by rank in the courtyard; oaths of loyalty are
#    renewed and grievances heard. The ritual is the institutional
#    mechanism by which naam -- the divine right to command -- flows from
#    the throne into every chieftaincy in the empire. No new village or
#    provincial chief can rule without his naam being conferred at
#    Wagadugu.

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002

	potential = {
		has_or_had_tag = MSI
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:ouagadougou
	}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The Friday court binds every village and provincial chief to the
	# throne by the ritual conferral of naam: no naba in the kingdom holds
	# legitimate office unless the Mogho Naaba has invested him, and the
	# weekly audience makes that bond visible to all.
	neutral_modifier = {
		monthly_legitimacy = 0.05
	}

	# Performed faithfully -- the Mogho Naaba mounted in red every dawn,
	# the chiefs travelling to Wagadugu by rotation, naam conferred in
	# proper form -- the ritual saturates the empire with royal authority:
	# provincial chiefs accept Wagadugu's edicts, peasants see the king as
	# the wellspring of order, and the kingdom holds together by ritual
	# repetition rather than by force alone.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_max_control = 0.02
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction

	}

	# When the Friday court lapses -- the king refuses audience, the
	# Ouidi Naaba's horse goes unsaddled, chiefs no longer travel to
	# renew their naam -- royal authority decays into mere local custom.
	# The nakomse, free of their ritual subordination, govern their
	# provinces by inheritance rather than by royal investiture.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		global_separatism = 0.05
	}
}

# 3. Tengsoba Earth-Priest Compact -- the dual settlement between the
#    Mossi conquerors descending from the marriage of Princess Yennenga
#    and the hunter Riale (c. early 13th century) and the autochthonous
#    Voltaic peoples (Nyonyose, Kibsi, Doghosie) who held the land before
#    them. Mossi political authority -- naam -- belongs to the nakomse;
#    spiritual authority over the soil -- tenga -- belongs to the
#    tengsoba earth-priest, who is always drawn from the autochthonous
#    lineage. No field can be cleared, no village founded, no rain
#    invoked, without the tengsoba's ritual sanction. The compact is the
#    institutional basis of Mossi rural stability: it lets a thin warrior
#    aristocracy govern a much larger conquered population without

	# Recognising the tengsoba's ritual sovereignty over land confines

# 4. Nakomse Provincial Governance -- the patrilineal system through which
#    the Mogho Naaba's sons (the Dimas) were assigned provinces, and
#    nakomse cousins were appointed to lesser chieftaincies down the
#    Mossi heartland. Each nakomse held his territory by hereditary right
#    of conquest, owed cavalry service on royal summons, collected
#    tribute through village chiefs, and forwarded a share to the throne.
#    The system gave the empire its famed long-range cavalry -- it was
#    Mossi horsemen who raided Mali in 1328-1338 and turned back a
#    Songhai expedition in 1483 -- but it also fed the centrifugal
#    tendency that produced the four-five rival kingdoms of Wagadugu,
