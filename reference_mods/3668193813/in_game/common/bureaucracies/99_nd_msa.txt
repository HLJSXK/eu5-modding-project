# National Destinies -- Malayan bureaucracies
# Five historically-grounded institutions of the unified Malay state, drawing
# on the four-pillar Melaka officialdom (Bendahara, Laksamana, Penghulu
# Bendahari, Temenggung), the Hukum Kanun Melaka legal code (codified under
# Sultan Muzaffar Shah, c. 1450) and its companion Undang-Undang Laut Melaka
# (Mahmud Shah, c. 1500), the Shahbandaria of the four foreign trading
# quarters (Tamil Hindu, Gujarati-Muslim, Chinese, and Javanese-Luzon) that
# made Melaka the entrepot of the Indian Ocean before 1511, and the Orang
# Laut compact between the Riau-Lingga sea-nomads and the Johor Sultanate
# that survived the 1511 Portuguese conquest and lasted into the Bugis
# ascendancy of the 18th century. Auto-modifier grants 2 slots; capital adds
# 1 = 3 effective. Player picks 3 of 5.
#
# All IDs prefixed with `msa_` to avoid cross-file collisions.

# 1. Bendahara Grand-Vizierate -- the highest civilian office of the Melaka
#    Sultanate, held in the kingdom's golden age by Tun Perak (c. 1456-1498),
#    who as Bendahara directed the realm through five sultans, defeated two
#    Siamese invasions (1456, 1488), and ran the chancery, foreign affairs,
#    and noble appointments from his residence at Bukit China. The office
#    was hereditary in the Tun lineage of Bendahara Seri Maharaja and his
#    descendants; in practice the Bendahara was the de facto first minister
#    while the sultan presided. The institution survived Melaka's fall:
#    Johor's Bendaharas eventually founded the breakaway state of Pahang
#    in 1770, and the Pahang Bendaharas became sultans in their own right.
msa_bendahara_grand_vizierate_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MSA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# Recognising the Bendahara as the chancery's permanent head means the
	# realm runs by the Tun lineage's hand: appointments, despatches, and
	# the sultan's correspondence with foreign courts all pass through the
	# Bendahara's residence before the istana sees them.
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# Honoured -- a Tun Perak in office, the Bendahara's seal trusted across
	# the Indian Ocean courts, the chancery filled with literate clerks of
	# the Bendahara's choosing -- the realm enjoys orderly administration
	# and the long noble peace that let Melaka grow from fishing village to
	# entrepot of the world in three generations.
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		global_max_control = 0.02
		improve_relation_impact = 0.10
	}

	# When the Bendahara becomes a faction chief rather than a minister --
	# Tun Mutahir's nepotism in the last years before 1511, the Pahang
	# Bendaharas' eventual secession -- the office turns from servant of the
	# realm into rival of the throne, monopolising patronage and starving
	# the sultan's own household of competent staff.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		global_separatism = 0.05
	}
}

# 2. Laksamana Admiralty -- the office of admiral-of-the-fleet, held in

# 3. Hukum Kanun Melaka -- the legal code compiled under Sultan Muzaffar
#    Shah (r. 1446-1459) and refined under Mahmud Shah (r. 1488-1511),
#    surviving in some 30 manuscript recensions across the Malay world.
#    The 44 chapters fused Islamic shari'a (in matters of marriage,
#    inheritance, and ritual offences) with Malay adat (in land tenure,
#    royal protocol, and capital crimes), producing a synthesis that the
#    later sultanates of Johor, Pahang, Kedah, and Patani all adopted as
#    their reference code. The companion Undang-Undang Laut Melaka --
#    Mahmud Shah's maritime ordinance -- governed shipboard authority,
#    cargo disputes, and the sharing of voyage profits between owner,
#    nakhoda (captain), and crew, and was binding on every vessel sailing
#    under Melaka registry from Mergui to Maluku.
msa_hukum_kanun_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}
	removal_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = MSA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:sunni
	}

	estates_that_like = {
		clergy_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Promulgating the 44 chapters of the Hukum Kanun and the Undang-Undang
	# Laut as binding law throughout the realm gives Malay courts a single
	# reference text: from Kedah to Pahang, the qadi cites the same chapter,
	# and the same maritime ordinance governs whether a Mergui or Maluku
	# port-of-call.
	neutral_modifier = {
		stability_investment = small_stability_investment
	}

	# Honoured -- the qadis of every coastal port reading from clean
	# manuscripts of the Hukum Kanun, the nakhodas swearing voyages on
	# the Undang-Undang Laut, the sultan's writ enforced uniformly --
	# the realm gains a prestige no rival can match: literate Islam fused
	# with Malay adat, the Tamil and Gujarati and Chinese merchants all
	# recognising the code that adjudicates their disputes.
	positive_modifier = {
		scale = { value = scope:maintenance }
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
		global_max_literacy = 5
		monthly_devotion = 0.10
	}

	# When the code becomes a lever rather than a measure -- the great
	# noble houses extracting custom rulings from compliant qadis,
	# Bendahara-faction judges reading the chapters as their patrons
	# require, the Undang-Undang Laut bent to favour the wealthy nakhoda
	# over the wronged crewman -- the synthesis decays into factional
	# instrument and the rural population learns that justice has a price.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.05
		stability_decay = 0.00025
	}
}

# 4. Shahbandaria of the Four Quarters -- the four port-officers of Melaka,
#    each governing a chartered foreign trading community of the entrepot
#    in Tome Pires's listing (Suma Oriental, c. 1515): the Shahbandar of
#    the Gujaratis (the wealthiest quarter, with its own Indian-built
#    warehouses west of the river); the Shahbandar of the Tamils, Pegu,
#    and Pasai; the Shahbandar of the Javanese, Maluku, Banda, Palembang,
#    Borneo, and Luzon; and the Shahbandar of the Chinese, Liu-Kiu, and
#    Champa. Each Shahbandar collected import-duties (3-6% by class of
#    goods), assigned warehouse plots, settled disputes within his
#    community by its own law, and represented his merchants to the
#    Bendahara's chancery. The institution made Melaka governable for
#    crown and merchant alike: foreigners traded under their own customs,
#    paid uniform duty, and never appeared before a Malay court except
#    on capital matters.

	# dragged before Malay courts in violation of charter -- the Gujarati
	# capital flees to Aceh, the Chinese junks redirect to Banten, and the

# 5. Orang Laut Tanah-Air Compact -- the institutional understanding
#    between the Malay throne and the Orang Laut sea-nomads of the Riau-
#    Lingga and Tudjuh archipelagos, by which the sea-people held
#    customary rights to specific territorial waters (tanah air -- "land
#    and water") in exchange for naval service to the sultan. The
#    compact, attested in the Sejarah Melayu's account of Sang Nila Utama
#    and his Orang Laut followers, gave the Sultanate of Melaka and then
#    Johor a body of pilots, irregular naval auxiliaries, and informants
#    whose knowledge of every reef and channel from Singapore to Lingga
#    was unmatched. The Orang Laut suku such as the Suku Bentan and Suku
#    Galang were the kingmakers of Johor: when they shifted allegiance
#    to the Bugis Yam Tuan Muda after 1722, the Bugis-Riau federation
#    became the dominant Malay polity, and when they later refused
#    service the Riau capital fell to Dutch and British incursions.
msa_orang_laut_compact_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = MSA
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		peasants_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# Recognising the Orang Laut suku in their tanah-air rights binds the
	# sea-people to the throne by sworn allegiance rather than by tribute:
	# their pilots accompany the royal kelulus, their lookouts report every
	# strange sail, and their knowledge of the reefs from Bentan to Lingga
	# becomes the throne's own.
	neutral_modifier = {
		naval_range_modifier = 0.10
	}

	# Honoured -- the suku Bentan and Galang chiefs received at the istana,
	# their territorial waters left unenclosed, the throne's gifts of cloth
	# and rice flowing to the sea-villages -- the compact gives the
	# Sultanate its eyes and its irregular fleet: pirates from outside the
	# compact are run down by the Orang Laut's own prahus, contraband
	# routes are mapped before they congeal, and no foreign squadron
	# threads the archipelago without the throne knowing of it.
	positive_modifier = {
		scale = { value = scope:maintenance }
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		naval_range_modifier = 0.1
		army_food_gathering = 0.10
	}

	# When the compact lapses -- noble factions enclosing reef-grounds for
	# private profit, the sultan's gifts withheld, the suku ignored at
	# court -- the Orang Laut withdraw their service and (as in 1722, and
	# again in the late 18th century) shift allegiance to whichever
	# prince or foreign power treats them with proper deference. The
	# kingdom loses its archipelagic eyes overnight.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.010
		stability_cost_efficiency = -0.1
	}
}
