# National Destinies -- Comancheria (NUM) bureaucracies
# Four Comanche institutions; the player picks 3
# (capital implicit slot + 2 from the auto_modifier).
# Comancheria was a horse-empire built on inter-band federation
# rather than centralized office-holding -- these "bureaucracies"
# are council, federation, license, and incorporation forms.
# Sources: Hamalainen "The Comanche Empire", Brooks "Captives and
# Cousins", Rivaya-Martinez on captive-kinship statistics, the
# Council Hill / Big Spring tradition.
###############################################################

# Council of the Comanche Nation -- the inter-band assembly
# under the dual paraibos / tekwuniwapi co-rulership. The
# paraibos (peace chiefs) and tekwuniwapi (war chiefs) presided
# jointly: the paraibos negotiated tribute, captive-ransom, and
# the Comanchero licenses; the tekwuniwapi led the raid-cycle.
# When the divisions converged at Council Hill or Big Spring the
# paired authority was raised over all five hordes at once,
# ratifying peace, war, and inter-band redistributions.
num_council_of_comanche_nation_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = NUM
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		tribes_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: the dual paraibos / tekwuniwapi office is recognized
	neutral_modifier = {
		monthly_tribal_cohesion = 0.05
	}

	# positive: paired peace/war voices ratify a single line for
	# the federation -- diplomacy and mobilization speak together
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_horde_unity = 0.05
		diplomatic_reputation = 1
		improve_relation_impact = 0.1
	}

	# negative: when the council does not convene, the paraibos
	# and tekwuniwapi pull in opposite directions -- raid in one
	# direction, treaty in the other -- and prestige bleeds out
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_complacency = 0.05
		monthly_prestige = -0.05
	}
}

# Multi-Horde Federation -- the five-division federal structure
# (Penateka honey-eaters, Yamparika root-eaters, Quahadi antelope,
# Nokoni who-turn-back, Kotsoteka buffalo-eaters). Each horde kept
# its own paraibos and its own raid-radius: Penateka south to
# Coahuila, Quahadi west across the Llano Estacado, Yamparika
# north to the Arkansas. The federation pooled members for war
# and ratified peace at the inter-band council, but never
# abolished the division-level autonomy.

# Comanchero Trade License -- the Comanche-administered system
# regulating Hispanic and Pueblo traders (the comancheros) who
# entered Comancheria to deal in horses, hides, captives, and
# silver. The Comanche granted licenses, fixed exchange rates at
# the rendezvous-fairs, and took a cut of every caravan. By the
# 1830s the comancheros operated under a recognized customs
# regime, returning yearly to fixed waterholes the paraibos had
# designated as licensed marts. Texan accounts describe a
# "non-state" with working tariff machinery.
num_comanchero_license_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = NUM
		has_game_rule = nd_bureaucracies_enabled
	}

	# The license system requires the rendezvous-trade institution
	# to exist (the Comanchero advance establishes the legal form).
	allow = {
		has_advance = nd_num_comanchero_trade
	}

	estates_that_like = {
		burghers_estate
		crown_estate
	}
	estates_that_dislike = {
		tribes_estate
	}

	# neutral: licensed marts at fixed waterholes give the
	# federation a recognized trading counterparty -- but
	# caravan-protocol consumes paraibos cabinet attention
	neutral_modifier = {
		diplomatic_capacity = 1
		country_cabinet_efficiency = -0.05
	}

	# positive: license fees, regulated exchange rates,
	# and the horse-hide-silver caravan economy
	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_land_efficiency = 0.05
		global_horses_output_modifier = 0.1
		cultural_influence_modifier = 0.1
	}

	# negative: the licensed marts decay into smuggling routes;
	# the older bands resent the paraibos' tariff-cut and accuse
	# the council of growing rich off Pueblo merchants
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		tribes_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_tribal_cohesion = -0.05
	}
}

# Captive-Incorporation System -- the institutionalized practice
# of absorbing captives (Mexican, Pueblo, Anglo) into Comanche
# bands as adopted kin. Captives were taken in raid, ransomed
# to relatives for horses or silver, or -- if young -- raised
# as Comanche children, marrying into the lineages. By the
# nineteenth century, sizeable percentages of Comanche
# population were captive-descended Mexican or Anglo. The
# institution was a multi-ethnic state-building engine that
# turned the raid-cycle into a demographic engine.
num_captive_incorporation_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price

	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = NUM
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		tribes_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: the band-mothers adopt taken children as kin
	neutral_modifier = {
		cultures_capacity = 1

	# positive: captives become Comanche, the war-band feeds
	# the lineages, the multi-ethnic federation grows
	positive_modifier = {

}
