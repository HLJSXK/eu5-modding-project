# National Destinies -- Pontus (PON) bureaucracies
# Four institutions of the Trapezuntine-Komnenoid imperial revival, drawn
# from the Empire of Trebizond (1204-1461) and the Pontic-Greek
# institutional inheritance preserved at Sumela, Argyria, and the Caffan
# emporia. Auto-modifier nd_pon_bureaucracy_authority grants 2 slots; the
# capital adds 1 = 3 effective. The player picks 3 of 4.
#
# Note on estate mapping: in the Pontic-Komnenoid constitution the
# "crown_estate" is the imperial Komnenoid household at Trebizond, the
# "clergy_estate" is the autocephalous Pontic synod at Sumela, the
# "nobles_estate" is the Tzanichite-Skouterides Pontic archontic class
# alongside the Caucasian frontier dynasts, and "burghers_estate" the
# Pontic-Italian-Caffan mercantile federation. This drives the alignments
# below.

# 1. Synodos of the Pontic Church -- the autocephalous Pontic Orthodox
#    synod, headed by the Metropolitan of Trebizond and centered on the
#    monastic complex of Sumela on the cliffs above the Pyxites valley.
#    The synod gathered the bishops of Trebizond, Kerasounta, Sinope, the
#    Hegumens of Sumela, Vazelon and Peristera, and the Caucasian Orthodox
#    sees of Lazica and Abkhazia. It conducted its own chrismation of
#    bishops, retained a distinct liturgical calendar honoring Saints
#    Eugenios and Aquila of Trebizond, and licensed the icon-painting
#    workshops of the Pontic school. The synod's autonomy from
#    Constantinople predated 1204 and outlived the empire by a generation.
pon_synodos_of_the_pontic_church_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = PON
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:orthodox
	}

	estates_that_like = {
		clergy_estate
		peasants_estate
	}
	estates_that_dislike = {}

	# The autocephalous synod's mere existence anchors Pontic religious
	# legitimacy independent of Constantinople.
	neutral_modifier = {
		tolerance_own = 1
	}

	# Working synod: chrismation of bishops at Sumela, Pontic-school icon
	# workshops, Caucasian Orthodox sees integrated under the Trapezuntine
	# metropolitan, monastic schools on the Pyxites teaching the upland
	# villagers to read.
	positive_modifier = {
		scale = { value = scope:maintenance }
		monthly_devotion = 0.10
		clergy_estate_satisfaction_recovery = 0.001
		global_burghers_max_literacy = 5
	}

	# Decayed synod: the Sumela hegumens quarrel with the Trebizond
	# metropolitan over chrismation rights, the Caucasian sees drift back
	# toward Mtskheta and the Constantinopolitan obedience, and rural
	# parishes lapse into syncretic practice with local Lazic and Tzan
	# folk-religion.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.020
		stability_decay = 0.0005
	}
}

# 2. Logothesion of Trebizond -- the imperial chancery of the Komnenoi-of-
#    Trebizond, modeled on the Constantinopolitan logothete tradition but
#    adapted to the empire's polyglot diplomatic environment. Under the
#    protonotarios and the megas logariastes it issued chrysobulls in the
#    Trapezuntine chancery hand, drafted correspondence in Greek, Persian,
#    Mongol Turkic and Italian Latin to suit the recipient (Ilkhan,
#    Aqquyunlu, Genoese podesta, Venetian bailo, Komnenian collaterals at
#    Mangup), and maintained the Imperial Court's ceremonial scroll-keepers
#    who recited the Komnenian polychronia at every audience. The chancery
#    bound a small empire to a vast diplomatic network -- but the dignity
#    inflated faster than the polity grew.
pon_logothesion_of_trebizond_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = PON
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:trebizond
	}

	estates_that_like = {
		crown_estate
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The Komnenian chancery anchors the imperial style and binds Pontic
	# diplomacy to an unmistakable Byzantine inheritance.
	neutral_modifier = {
		monthly_legitimacy = 0.05
	}

	# Working chancery: multilingual chrysobulls dispatched to Tabriz,
	# Genoa, Venice and the Caucasian courts; the protonotarios coordinates
	# the empire's marriage diplomacy across the Black Sea; the polychronia
	# at audience reinforces the Komnenian pretension.
	positive_modifier = {
		scale = { value = scope:maintenance }
		diplomatic_reputation = 1
		country_cabinet_efficiency = 0.10
		monthly_prestige = 0.10
	}

	# Decayed chancery: the protonotariate is sold to Pontic archontic
	# clans, chrysobulls inflate the Komnenian titulature beyond credibility,
	# and the great archons resent magisterial control of preferment.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		prestige_decay = 0.0025
	}
}

# 3. Argyria Mint and Mining Authority -- the coordinated silver-mining
#    and minting apparatus that underwrote Trapezuntine wealth. The
#    Argyria mines in the Pontic Alps south of Tripolis (modern Gumushane,
#    "the silver-house") supplied bullion to the imperial mint at
#    Trebizond, which struck the aspron-trachy and the silver aspron-
#    komnenaton in continuous high-fineness issues from the thirteenth
#    century onward. Trapezuntine silver circulated as the standard
#    medium of Black Sea trade, accepted at Caffa, Sinope, Tana and the
#    Anatolian beyliks. The mine-superintendent (despotes ton metallon)
#    reported directly to the Komnenian household; mining levies fell on
#    upland peasants but the bullion bought the empire its galleys, its
#    Persian alliances, and its Italian customs revenues.
pon_argyria_mint_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = PON
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	# The mint and the mines together formalize Pontic silver as a
	# state-coordinated fiscal-monetary instrument.
	neutral_modifier = {
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Working authority: the despotes ton metallon coordinates the Argyria
	# galleries and the Trebizond mint, the aspron-komnenaton circulates as
	# Black Sea standard, and Pontic bullion underwrites Komnenian
	# diplomacy across the eastern Mediterranean.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_silver_output_modifier = 0.15
		trade_income = 0.10
		estate_enrichment = 0.05
	}

	# Decayed authority: galleries flood for want of timbering, the mint
	# debases issues to cover household arrears, and the upland peasant
	# corvee for mine-labor curdles into open resentment.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_inflation = 0.005
		peasants_estate_satisfaction_decay = 0.020
	}
}

# 4. Caffa-Trebizond Mercantile Council -- the standing council of major
#    Pontic-Italian-Caffan mercantile families that regulated Black Sea
#    trade routes, prize and reprisal disputes, customs incidence, and
#    convoy escorts. Pontic archontic-mercantile lines (Tzanichites,
#    Skouterides, Mesarites, Doranites) sat alongside their Genoese-Caffan
#    counterparts (the Gattilusio, the Centurione, the Di Negro houses)
#    and the Sinope-Anatolian Greek merchant families. The council met
#    seasonally at Trebizond and Caffa under joint Trapezuntine-Genoese
#    presidency, fixed the kommerkion duties on the grain, salt-fish,
#    caviar, alum, silk and slave traffic of the Black Sea, and arbitrated
#    inter-city prize cases. It is the Trapezuntine antecedent of the
#    Phanariot mercantile commonwealth.

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003

	potential = {
		has_or_had_tag = PON
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		OR = {
			owns = location:trebizond
			owns = location:giresun

			owns = location:sinop

	}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {

	# Working council: the kommerkion-rates fix Black Sea customs, the

	# emporium.

		global_trade_center_power = 0.10
		trade_sea_efficiency = 0.10
		burghers_estate_satisfaction_recovery = 0.001

	# Decayed council: the Genoese houses outvote the Pontic archontic

	# raiding rather than concede commercial precedence to the burghers.

		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
