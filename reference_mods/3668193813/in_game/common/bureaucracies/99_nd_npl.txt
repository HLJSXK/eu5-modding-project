###############################################################
# National Destinies -- NPL (Nepal) bureaucracies
#
# Five institutions spanning the Malla city-states (pre-1768),
# Gorkha unification under Prithvi Narayan Shah, and the
# Bhimsen Thapa regency of the Mukhtiyari era (1806-1837).
#
#   1. Bhardari Sabha       -- council of bhardars (kajis, sardars)
#   2. Mukhtiyari           -- office of the chief minister
#   3. Pajani               -- annual personnel renewal review
#   4. Newar Guthi          -- Kathmandu Valley endowment trusts
#   5. Jagir Land Tenure    -- military service land assignments
#
# Slot accounting: 5 bureaucracies, +2 from unlock advance,
# +1 from capital = 3 effective slots; player picks 3 of 5.
###############################################################


# =====================================================================
# Bhardari Sabha -- the Council of Bhardars
#
# The bhardari was the Nepali council of state, made up of the great
# noble houses (Pande, Thapa, Basnyat, Chautariya) holding the kaji
# and sardar offices. After Prithvi Narayan's death in 1775, factional
# struggles in the Bhardari Sabha effectively ran the kingdom during
# the long minorities of the Shah kings, until Bhimsen Thapa broke
# its collegiality. A genuine aristocratic council that constrained
# the throne -- distinct from the rubber-stamp courts of imperial
# bureaucracies elsewhere.
# =====================================================================
npl_bhardari_sabha_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = NPL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	#neutral: noble houses sit at the council table
	neutral_modifier = {
		nobles_estate_allowed_leading_military = yes
	}

	#positive: collegial governance, factional balance, stable transitions
	positive_modifier = {
		scale = { value = scope:maintenance }
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
		stability_cost_efficiency = 0.1
		monthly_legitimacy = 0.05
	}

	#negative: the great houses turn the council into a feud arena;
	#royal minorities become regency civil wars
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.05
		nobles_estate_satisfaction_decay = 0.05
	}
}


# =====================================================================
# Mukhtiyari -- the Office of the Chief Minister
#
# The mukhtiyar was the all-powerful chief minister of Nepal,
# institutionalised by Bhimsen Thapa from 1806 to 1837 under the
# minor king Girvan Yuddha and the boy-king Rajendra. With control
# of the seal, the army, and the treasury, the mukhtiyar superseded
# the bhardari council and centralised authority into a single hand.
# A counterweight to the noble factions -- a deliberate executive
# concentration outside the older collegial institutions.
# =====================================================================
npl_mukhtiyari_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = NPL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	#neutral: a single hand on the seal
	neutral_modifier = {
		country_cabinet_efficiency = 0.10
	}

	#positive: the mukhtiyar coordinates army, treasury, and chancery
	positive_modifier = {
		scale = { value = scope:maintenance }
		cabinet_trait_impact_modifier = 0.20
		hire_for_cabinet_efficiency = 0.2
		army_maintenance_efficiency = 0.05
	}

	#negative: a single overworked minister becomes the bottleneck
	#and the kajis resent the upstart who has eclipsed the bhardari
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		country_cabinet_efficiency = -0.15
	}
}


# =====================================================================
# Pajani -- the Annual Personnel Review
#
# The pajani was the yearly review at which every civil and military
# officeholder had to present themselves at court for reappointment;
# tenure expired by default each year. A Shah-era device for keeping
# the kajis, sardars, and revenue collectors directly accountable to
# the throne -- the institutional opposite of hereditary office. The
# pajani gave the king a recurring lever over the entire administrative
# class but at the cost of constant patronage churn.
# =====================================================================
npl_pajani_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = NPL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	#neutral: the throne holds the appointment lever
	neutral_modifier = {
		global_crown_estate_power = 0.05
	}

	#positive: annual review keeps officials sharp, corruption checked
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_estate_power = -0.20
		tax_income_efficiency = 0.05
		monthly_legitimacy = 0.05
	}

	#negative: pajani becomes an auction; every position has a price,
	#and patronage churn ruins continuity in the provinces
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		estate_enrichment = 0.15
		global_monthly_control = -0.002
	}
}


# =====================================================================
# Newar Guthi -- the Kathmandu Valley Endowment Trusts
#
# The guthi system is the Newari institution of perpetual trusts that
# fund temple maintenance, festival processions, irrigation channels,
# and rest-houses. Held collectively by clans or castes, guthis own
# land outright and administer it across generations independent of
# royal patronage. The Malla kings ruled with the guthi system; the
# Shahs inherited it and were never able to dissolve it. A standing
# civic-religious infrastructure parallel to the state.
# =====================================================================
npl_newar_guthi_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = NPL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:kathmandu
	}

	estates_that_like = {
		clergy_estate
		burghers_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	#neutral: the guthis hold their lands in perpetuity
	neutral_modifier = {
		global_estate_satisfaction_from_legitimacy = 0.10
	}

	#positive: festivals run, temples rebuild, irrigation flows
	positive_modifier = {

	#negative: guthi land sits outside crown reach; revenue leaks,

		global_crown_estate_power = -0.05
	}
}


# =====================================================================
# Jagir Land Tenure -- the Military Service Assignments
#
# Jagir was the assignment of revenue from a tract of land to a soldier
# or officer in lieu of cash pay; held for the term of service, in
# principle revocable at pajani. Alongside birta (tax-free grants to
# brahmins, often hereditary), jagir provided the financial backbone
# of the Gorkhali army -- it let the kingdom field large forces without
# a treasury that could pay them. The cost was a permanent transfer
# of revenue from the peasantry to the military aristocracy.
# =====================================================================
npl_jagir_tenure_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.002
	}

	potential = {
		has_or_had_tag = NPL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
	}

	estates_that_like = {
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		peasants_estate
	}

	#neutral: nobles lead the levies on jagir-funded service
	neutral_modifier = {
		nobles_estate_allowed_leading_military = yes
	}

	#positive: the army musters cheaply, fed by assigned revenue
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction

}
