# National Destinies -- Knights of the Temple (Templar Sovereign State)
# Four institutions of the alt-history Templar realm, drawn from the medieval
# brotherhood (1119-1312) projected forward through the Order of Christ
# (1319-1789) and the Iberian-Atlantic captaincy system. Auto-modifier
# (nd_tkt_bureaucracy_authority) grants 2 slots; the Tomar Convent capital
# adds 1 = 3 effective. The player picks 3 of 4.
#
# Note on estates: in the Templar constitution the "clergy_estate" maps to
# the brother-knights and chaplains (the brotherhood is a religious order
# under monastic Rule), the "nobles_estate" to the lay-noble preceptors and
# the hereditary brother-knight families of Iberia, the "crown_estate" to
# the Master's magistral household at Tomar, and the "burghers_estate" to
# the Lisbon-Lagos mercantile pops, factor agents, and the Templar bank's
# clearing-house clientele. This mapping drives the alignments below.

# 1. Sacro Capitulo Generale -- the Sacred Convent General Chapter. Under the
#    Latin Rule of Bernard (Council of Troyes 1129) the Master is elected by
#    the Convent and rules by the consent of the General Chapter, which sits
#    annually at Tomar to confirm or reject magistral measures, judge serious
#    breaches of Rule, and admit or deny postulants. The Chapter draws its
#    members from the senior commanderies (the Provincial Masters of Aragon,
#    Castile, Portugal, Provence-in-exile) and the conventual officers (the
#    Seneschal, Marshal, Drapier, Almoner, Treasurer). Magistral acts taken
#    without Chapter consent are nullified by the Visitor-General.
tkt_sacred_convent_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = TKT
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		clergy_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The brother-knights' constitutional consent legitimates magistral
	# acts. The Order's name carries weight in chancelleries.
	neutral_modifier = {
		clergy_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Working Chapter: the Provincial Masters convene, the Visitor-General
	# audits the commanderies, the Latin Rule is read aloud at every session,
	# and decisions of the magistral household pass under proper conventual
	# seal. The brotherhood holds together across an Iberian-Provencal
	# preceptory network.
	positive_modifier = {
		scale = { value = scope:maintenance }
		stability_importance_modifier = 0.10
		clergy_estate_satisfaction_recovery = 0.001
		legislative_efficiency = 0.05
	}

	# Stalled Chapter: the Provincial Masters refuse to convene, the
	# Visitor-General is bypassed, and the brotherhood drifts into national
	# factions (the Aragonese against the Portuguese, the Castilian against
	# both). Magistral decrees are challenged in chapter-house and quietly
	# ignored in the field commanderies.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_cost_efficiency = -0.1
		clergy_estate_satisfaction_decay = 0.020
	}
}

# 2. Apparatus Thesaurarii -- the Treasurer-General Apparatus, the Order's
#    signature institution and the first international banking corporation
#    of medieval Christendom. Under the Treasurer-General at Tomar the
#    preceptories of every province held vaults open to deposit; a pilgrim
#    or factor depositing silver at one preceptory withdrew bullion at
#    another under a sealed bill of exchange. The Apparatus took fees on
#    transfer, lent on bond to crowned heads (Philip II of France banked
#    his ransom of the Plantagenets through the Paris preceptory; James I
#    of Aragon mortgaged his crown jewels through the Barcelona house),
#    and held mortuaries (the estates of deceased brothers). Its officers
#    were chaplain-clerks under monastic discipline; its lending was the
#    chamber of Catholic state finance until the trial of 1307-1314.
tkt_treasurer_general_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = TKT
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		clergy_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The Treasurer's chest at Tomar underwrites the Order's continental
	# credit. The chaplain-clerks of the chamber become the brotherhood's
	# real fiscal cabinet.
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# Working Apparatus: deposits flow into the Tomar vault from preceptories
	# in every province, bills of exchange clear through Saragossa, Lyon,
	# and Acre under the Treasurer's seal, and the Order's loan capacity
	# scales with the chamber's gold reserves. The bank's prestige is the
	# Convent's credit.
	positive_modifier = {
		scale = { value = scope:maintenance }
		total_loan_capacity_modifier = 0.20
		bank_interest = -0.01
		burghers_estate_satisfaction_recovery = 0.001
	}

	# Decayed Apparatus: the chaplain-clerks turn rentier, the Provincial
	# Masters siphon mortuaries before they reach the Treasurer's chest,
	# and the brother-knight commanderies resent the chamber's claim on
	# preceptory revenue. The Tomar bills of exchange lose their continental
	# standing as default rumors spread.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}

# 3. Capitulum Occultum -- the Secret Chapter. The trial of 1307-1314 fixed
#    in Catholic memory the rumour of a hidden inner chapter beneath the
#    open conventual one: closed initiations under sealed doors, the rite

#    in Provencal dialect. In the alt-history sovereign Order, the
#    Capitulum Occultum is institutionalised at Tomar as the Master's own
#    inquiry chamber: it audits the brothers' own loyalty, runs intelligencers
#    under cover of preceptory-business, and weaponises the Baphomet-rumour
#    heritage into mystique that travels in pamphlets and tavern-talk from
#    Lisbon to Lyon. Its officers answer only to the Master; its sessions

# 4. Capitanias do Mar -- the Maritime Captaincies. Built on the alt-history
#    Order of Christ inheritance from Henry of Sagres (the Atlantic mandate
#    funded from Tomar's chambers as it was funded historically from the
#    Order of Christ), the Capitanias system divides every Atlantic colony
#    into hereditary captaincy-grants under brother-knight donatarios.
#    Madeira (Funchal, Machico), the Azores, Sao Tome, the African factor-
#    coast: each is governed by a knight-captain who answers to the Lagos
#    Factor-General and through him to the Tomar Convent. The captain holds
#    judicial, fiscal, and military authority over the captaincy in exchange
#    for a tenth of revenues to the Treasurer-General. The pattern is
#    half-feudal, half-conventual; it is the Order's actual instrument of
