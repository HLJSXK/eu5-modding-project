###############################################################
# National Destinies -- KAZ Kazakh Khanate bureaucracies
#
# Vanilla EU5 1.2 ships no KAZ bureaucracies. These five model
# the durable institutions of the Kazakh Khanate from Kerei and
# Janibek's 1465 split out of the Uzbek confederation through
# the Three Zhuz era and the Russian protectorate of 1731:
#
#   1. Jeti Jargy           -- Tauke Khan's 1680 Seven Codes
#   2. Three Zhuz           -- Senior/Middle/Junior horde confederation
#   3. Ak-Suiek Tore        -- Chinggisid white-bone ruling caste
#   4. Aksakal Aul          -- village-elder governance of nomadic camps
#   5. Batyr War Bands      -- warrior-champion irregular organization
#
# IDs are prefixed kaz_ per the National Destinies bureaucracy spec.
# All are gated on `has_game_rule = nd_bureaucracies_enabled` so the
# player's game-rule choice is honoured.
#
# Auto-modifier grants 2 slots; capital adds 1 = 3 total. Player
# picks 3 of 5 -- forcing a clear institutional shape on the steppe.
###############################################################


# =====================================================================
# Jeti Jargy -- the Seven Codes of Tauke Khan
#
# In the closing decades of the seventeenth century Tauke Khan
# convened the three greatest biys of the Kazakh nation -- Tole Bi
# of the Senior Zhuz, Kazybek Bi of the Middle, Aiteke Bi of the
# Junior -- and codified the customary law of the steppe into the
# Jeti Jargy, the Seven Codes. The codes regulated kun blood-money,
# bride-price, livestock theft, and the punishment of crimes; they
# enshrined the biy court as the steppe's true judicial authority.
# Where the codes were maintained, common Kazakhs found protection
# from arbitrary clan violence; where they decayed, blood-feud and
# barymta cattle-raiding returned to corrode the confederation.
# =====================================================================
kaz_jeti_jargy_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_or_had_tag = KAZ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		peasants_estate
		tribes_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# always-on: the codes' mere existence stabilises clan disputes
	neutral_modifier = {
		legislative_efficiency = 0.10
	}

	# enforced: biy courts arbitrate, common law binds the steppe
	positive_modifier = {
		scale = { value = scope:maintenance }
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction
		tribes_estate_satisfaction_recovery = 0.05
		stability_investment = small_stability_investment
	}

	# neglected: blood-feud and barymta return, courts ignored
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_decay = 0.00025
		monthly_war_exhaustion = 0.005
	}
}


# =====================================================================
# Three Zhuz Confederation -- Senior, Middle, and Junior Hordes
#
# The genius of Kazakh governance after Kasym Khan lay not in the
# single khan but in the balance of three zhuz: the Ulu (Senior)
# pasturing in Zhetysu, the Orta (Middle) on the central steppe,
# the Kishi (Junior) along the Yaik. Each zhuz kept its own pastures,
# its own biys, its own line of khans descended from Janibek and
# Kerei. Unity arose from consensus, never command. The arrangement
# made the steppe nearly impossible to conquer -- no single defeat
# shattered the whole -- but each zhuz could and did treat with
# foreign powers independently, as Abulkhair did with Russia in 1731.
# =====================================================================
kaz_three_zhuz_confederation_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price

		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = KAZ
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		tribes_estate

# =====================================================================
# Ak-Suiek Tore Hierarchy -- the Chinggisid white-bone caste
#
# Kazakh society was rigidly bisected between Ak-Suiek (white bone)
# and Kara-Suiek (black bone). Only descendants of Chinggis Khan
# through Jochi -- the Tore -- could claim the title of khan or
# sultan; only Tore could speak in the Kuralay assembly with binding
# authority. Below them, Kara-Suiek clans were led by their own biys
# and batyrs but barred from sovereign office. The Tore caste gave
# the Kazakhs an unbroken chain of legitimate rule from Janibek to
# Kenesary, but their endogamous insularity slowly hollowed out the
# political role of the lesser clans -- and provided the Russians,
# from 1822 onwards, with a list of names whose privileges could
# be selectively abolished.
# =====================================================================

# =====================================================================
# Aksakal Aul Governance -- the white-bearded elders of the camps
#
# Below the khans and beyond the reach of any chancery, Kazakh
# nomadic life ran through the aul -- a winter or summer camp of
# perhaps a dozen yurts belonging to a single extended kin group.
# The aksakal, literally "white-beard", was the senior elder whose
# word settled grazing rights, marriage contracts, and the daily
# disputes that never reached a biy court. The institution has no
# building, no scroll, no salary; it operates wherever Kazakhs
# pitch their tents. When khans tried to extract more than the aul
# elders thought just, the aul simply migrated beyond reach. When
# the Russian Empire later tried to register Kazakh families to
# fixed addresses, the aksakal system was the deepest obstacle.
# =====================================================================

# =====================================================================
# Batyr War Bands -- the warrior-champions of the steppe
#
# Outside the formal Tore hierarchy stood the batyrs: warriors of
# any clan who had earned the title through deeds of arms. Bogenbai,
# Kabanbai, Nauryzbai -- the names ring through Kazakh tradition as
# the men who broke the Dzungar invasion at Anyrakay in 1730 after
# the catastrophic Aktaban Shubyryndy retreat of 1723. A batyr
# commanded a war-band drawn from his own kin and admirers; he
# answered to the khan only by choice, fought when there was honour
# or plunder to be had, and dispersed when the campaign ended. The
# institution was the steppe's deepest reservoir of military manpower,
# but it ran on barymta -- the cattle-raid -- and could not be turned
# off when the campaign was done.
# =====================================================================
