# National Destinies -- universal governance bureaucracies.
# Available to any formed ND nation that researches the matching age-3
# governance advance (nd_provincial_governance / nd_maritime_governance).
# Each grants ONE minor governor building slot (neutral, always-on) plus a
# maintenance-slider tradeoff: a modest reward at high maintenance, a brutal
# control-ceiling collapse at low maintenance. local_proximity_source from the
# governor buildings is very strong, so the downside is deliberately severe.

# Provincial Governance -- noble-staffed provincial estates administering the
# distant interior. Unlocks the Provincial Estate building. The nobility
# resent central oversight; if the institution is starved, the provinces
# slip beyond the reach of the crown entirely.
nd_provincial_governance_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_advance = nd_provincial_governance
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		global_monthly_control = 0.003
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_max_rural_control = -0.10
		global_separatism = 0.15
		monthly_rebel_growth = 0.0004
	}
}

# Maritime Governance -- port commanderies projecting royal authority across
# the water to overseas harbors. Unlocks the Port Commandery building. Left
# underfunded, the port governors turn to smuggling and piracy, and the
# crown's grip on its coasts collapses.
nd_maritime_governance_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price

	potential = {
		has_advance = nd_maritime_governance
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	neutral_modifier = {
		num_nd_port_commanderies = 1
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		trade_income = 0.05
		global_sailors_modifier = 0.10
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_max_control = -0.05
		global_pirate_spawn_chance = 0.50
		trade_income = -0.15
	}
}
