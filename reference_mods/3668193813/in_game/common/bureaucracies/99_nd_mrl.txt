# ===== MRL (Maharlika) Bureaucracies =====
# L3 endgame Filipino formable. Five toggleable institutions reflecting the
# warrior-noble maharlika ideal, datu federalism, syncretic religious matrix,
# baybayin literacy, and pan-archipelago naval doctrine.

mrl_datu_court_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = MRL
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
		crown_estate
	}
	estates_that_dislike = {
		burghers_estate
	}

	# Maharlika exemption codified: warriors hold tax-free fiefs in exchange for service
	neutral_modifier = {
		nobles_estate_max_tax = -0.05
		nobles_estate_satisfaction_recovery = 0.001
	}

	# Maharlika serving the realm: levy mobilization is fast, retainer warriors plentiful
	positive_modifier = {
		scale = { value = scope:maintenance }
		levy_recovery_modifier = 0.15
		army_auxiliary_power = 0.1
		monthly_legitimacy = 0.1
	}

	# Code decays into hereditary rent-seeking: maharlika hoard land, peasants flee
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_nobles_estate_power = 0.15
		peasants_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
	}
}
