###############################################################
# National Destinies -- YMP (Yamato Empire) bureaucracies
#
# Endgame L5 formable: Hideyoshi's 1592 Imjin War succeeded.
# Korea pacified, Ming heartland conquered, samurai become
# the imperial officer corps from Pyongyang to Nanjing. The
# bakufu projects continental administration; the chrysanthemum
# throne sits over an East Asian empire.
#
# Six bureaucracies model the institutional choices a
# continental Yamato regime must make. Slot pool = 3 of 6.
#
#   1. Tairiku Bakufu        -- Continental Edo administration
#   2. Sankin-Kotai (cont.)  -- Daimyo alternate attendance, scaled
#   3. Kokudaka Governance   -- Daimyo as continental governors
#   4. Sankyo Itchi          -- Buddhist-Confucian-Shinto concord
#   5. Metsuke Censorate     -- Inspector-general scholar-officials
#   6. Suigun Bugyo          -- Atakebune admiralty + wakou marque
###############################################################


# =====================================================================
# Tairiku Bakufu -- Continental Bakufu
#
# The Edo administration's roju (council of elders), wakadoshiyori,
# and bugyo system rebuilt with branch chambers at Hanseong and at
# the conquered Ming capitals. A standardized continental bureaucracy
# replaces patchwork military government. The crown gains direct
# administrative reach into Korea and the Ming heartland; daimyo find
# their autonomy circumscribed by a parallel apparatus that does not
# answer to clan custom.
# =====================================================================
ymp_tairiku_bakufu_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = YMP
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:kyoto
		OR = {
			owns = location:dadu
			owns = location:shangyuan
			owns = location:namgyeong
		}
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: branch chambers exist; crown sits over continental admin
	neutral_modifier = {
		burghers_estate_allowed_in_cabinet = yes
	}

	# positive: bakufu chambers function -- crown reach, urban control,
	# administrative throughput across thousands of miles
	positive_modifier = {
		scale = { value = scope:maintenance }
		num_nd_provincial_estates = 2
		global_max_urban_control = 0.05
		country_cabinet_efficiency = 0.10
	}

	# negative: chambers ossify into venal sinecures -- daimyo go around
	# them, controllers chase paperwork instead of compliance
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.015
		global_separatism = 0.05
	}
}
