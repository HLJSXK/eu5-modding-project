# National Destinies -- SVB (Suvarnabhumi) bureaucracies
# Mainland SEA Theravada Buddhist universal-monarch empire.
# 6 institutions, 3 effective slots (2 from auto_modifier + 1 capital).

##########################################################################
# 1. Royal Sangha Council
# State-sponsored Theravada monastic hierarchy. Bayinnaung's 1560s
# sasana-purification reforms; the Sangharaja / Thathanabaing model
# where the king appoints a supreme patriarch and a council of elder
# theras (the Sudhamma Council in Burma) to police monastic discipline,
# certify ordinations, and arbitrate doctrinal disputes across the
# trans-mainland sangha.
##########################################################################
svb_royal_sangha_council_bureaucracy = {
    implementation_price = price:implement_bureaucracy_price
    maintenance_price = price:maintain_bureaucracy_price
    removal_price = price:remove_bureaucracy_price

    potential = {
        has_or_had_tag = SVB
        has_game_rule = nd_bureaucracies_enabled
    }

    allow = {
        OR = {
            religion = religion:theravada
            religion = religion:mahayana
        }
    }

    estates_that_like = {
        clergy_estate
        crown_estate
    }
    estates_that_dislike = {
        nobles_estate
    }

    # The mere existence of an enthroned Sangharaja gives the realm
    # one shared ecclesiastical address.
    neutral_modifier = {
        cultural_influence = 0.10
    }

    # Active sasana-purification: ordinations are certified, heterodox
    # forest-monk lineages submit to the council, doctrinal cohesion
    # binds disparate cultures under a single Theravada banner.
    positive_modifier = {
        scale = { value = scope:maintenance }
        global_pop_conversion_speed_modifier = 0.20
        clergy_estate_target_satisfaction = small_permanent_target_satisfaction
        global_pop_assimilation_speed_modifier = 0.10
    }

    # Neglected: the council fragments into rival nikayas, forest-monk
    # schisms, and monastic land disputes that the council can no
    # longer adjudicate. Devotion suffers and the abbots side with
    # local nobles instead of the throne.
    negative_modifier = {
        scale = {
            value = 1
            subtract = scope:maintenance
        }
        monthly_devotion = -0.005
        clergy_estate_satisfaction_decay = 0.02
    }
}

##########################################################################
# 2. Sakdina Hierarchy
# The dignity-rank system codified by Borommatrailokkanat in 1454,
# assigning every subject a numerical rank (sakdina) corresponding to
# rai of land entitlement. It is at once a precedence code, a service-
# land system, and a legal-procedure ranking (testimony weight in court
# scaled with sakdina). Defining feature of Ayutthaya governance.
##########################################################################
svb_sakdina_hierarchy_bureaucracy = {
    implementation_price = price:implement_bureaucracy_price
    maintenance_price = price:maintain_bureaucracy_price
    removal_price = price:remove_bureaucracy_price

    potential = {
        has_or_had_tag = SVB
        has_game_rule = nd_bureaucracies_enabled
    }

    allow = {}

    estates_that_like = {
        nobles_estate
        crown_estate
    }
    estates_that_dislike = {
        peasants_estate
    }

    # Codified rank means everyone knows their place at court audience.
    neutral_modifier = {
        government_size = -0.05
    }

    # The register is current: noble service-grants flow correctly,
    # courts apply rank-weighted procedure, and the crown can mobilize
    # nobles confidently against rivals or rebels.
    positive_modifier = {
        scale = { value = scope:maintenance }
        nobles_estate_target_satisfaction = small_permanent_target_satisfaction
        country_cabinet_efficiency = 0.10
        global_levy_size_modifier = 0.10
    }

    # Decayed: rank-grants outpace land-supply, the register is years
    # out of date, and the gap between sakdina theory and field reality
    # produces peasant flight from corvée and noble factional strife.
    negative_modifier = {
        scale = {
            value = 1
            subtract = scope:maintenance
        }
        peasants_estate_satisfaction_decay = 0.02
        stability_decay = 0.0005
    }
}

##########################################################################
# 3. Mandala Tributary Ring
# The Indianized SEA "circle of kings" -- paramount center plus
# ritually-bound tributary peripheries paying bunga mas, kneeling at
# coronation, and providing levies in war. Bayinnaung's 1564 reduction
# of Maha Chakkraphat to vassal-king is the textbook case. Power
# radiates outward and dissipates with distance, not via fixed borders.
##########################################################################
svb_mandala_tributary_ring_bureaucracy = {
    implementation_price = price:implement_bureaucracy_price
    maintenance_price = price:maintain_bureaucracy_price
    removal_price = price:remove_bureaucracy_price

    potential = {
        has_or_had_tag = SVB
        has_game_rule = nd_bureaucracies_enabled
    }

    allow = {}

    estates_that_like = {
        crown_estate
        nobles_estate
    }
    estates_that_dislike = {
        burghers_estate
    }

    # The cakkavatti's prestige radius is itself the institution.
    neutral_modifier = {
        diplomatic_range_modifier = 0.15
    }

    # Tributaries pay bunga mas and provide levies. The bond is
    # personal (king-to-king oath) and cheap to maintain because it
    # leaves local rule intact -- ritual submission is the substance.
    positive_modifier = {
        scale = { value = scope:maintenance }
        subject_loyalty = 0.10
        subject_pays_tributary_cost_modifier = -0.20
        monthly_legitimacy = 0.15
    }

    # Slack maintenance: tributaries openly resent the paramount,
    # coronation oaths lapse without enforcement, and burghers of the
    # paramount center see the routes through reluctant vassals
    # tighten with brigandage and arbitrary tolls. The mandala
    # collapses inward into hostile diplomacy.
    negative_modifier = {
        scale = {
            value = 1
            subtract = scope:maintenance
        }
        antagonism_received_modifier = 0.25
        burghers_estate_satisfaction_decay = 0.02
    }
}

##########################################################################
# 6. Phrai Corvée Registry
# The Ayutthaya-style annual labor draft. Free commoners (phrai luang)
# were registered to nai (master-officials) and owed six months of royal
# labor per year -- public works, royal granary, military mobilization.
# The registry was the realm's actual mobilization base; nobles fought
# over phrai-counts more than over land. Codified alongside sakdina,
# but mechanically distinct: this is the population schedule, not the
# rank schedule.
##########################################################################

    implementation_price = price:implement_bureaucracy_price

    removal_price = price:remove_bureaucracy_price
    maintenance_price_modifier = {
        value = country_economical_base
        multiply = 0.003

    potential = {
        has_or_had_tag = SVB

    allow = {}

    estates_that_like = {
        crown_estate
