# National Destinies -- Greater Burgundy bureaucracies
# 4 historically-grounded institutions of the Burgundian-Lotharingian state under
# Charles the Bold and his alt-history sovereign successors. Auto-modifier
# (nd_gbu_bureaucracy_authority) grants 2 slots; the capital adds 1 = 3 effective.
# Player picks 3 of 4.
#
# Identity: Burgundy is the composite-state paradox -- a duchy that out-administered
# its kingdoms, with a representative diet that pre-figured the Dutch States General,
# a financial chamber as advanced as any in 15th c. Europe, a chivalric order
# binding nobles to dynasty, and (under Charles) the first standing army of
# Northern Europe. The four institutions divide along the constitutional axis:
# Etats Generaux and Grand Council are estate/legal counterweights to the crown;
# Toison d'Or and Chambre des Comptes are crown-strengthening fiscal-chivalric
# instruments. No two share an estate alignment.
#
# All IDs are gbu_-prefixed to prevent collisions with other tags' bureaucracies.

# 1. Etats Generaux de Bourgogne -- the standing representative diet of the
#    Burgundian arc, first convoked at Bruges by Philip the Good in January 1464
#    as a single assembly of the Three Estates of all the duke's lands "by the
#    sea and by the land". Where the French Estates General sat at the king's
#    summons and dispersed at his pleasure, the Burgundian Etats Generaux became
#    a permanent constitutional partner: the duke (and later king) could levy
#    no aide without their consent, and the Grand Privilege of 1477 codified
#    their right to convoke themselves. The institution was the direct ancestor
#    of the States General of the Dutch Republic.
gbu_etats_generaux_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = GBU
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The diet's mere standing convocation gives every duchy and county a seat
	# at the kingdom-wide table.
	neutral_modifier = {
		allow_estate_parliament = yes
	}

	# Working diet: the cloth halls of Ghent and Bruges, the Brabantine prelates,
	# the noble houses of Hainaut and Holland all send deputies. Aides are voted
	# more readily than they are extracted by force; the kingdom's legislation
	# carries the moral weight of its constituent provinces.
	positive_modifier = {
		scale = { value = scope:maintenance }
		legislative_efficiency = 0.15
		parliament_duration_modifier = 0.20
		parliament_base_support = 0.10
	}

	# Deadlocked diet: the urban deputies and the noble bench quarrel over the
	# imposition of taxes; the Grand Privilege is invoked against the crown's
	# every initiative; the king's own legitimacy frays as he is forced to
	# bargain with provincial particularism.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		monthly_legitimacy = -0.05
		crown_estate_satisfaction_decay = 0.020
	}
}

# 2. Ordre de la Toison d'Or -- the Order of the Golden Fleece, founded at
#    Bruges on 10 January 1430 by Philip the Good on the day of his marriage
#    to Isabella of Portugal. Twenty-four knights at the founding (raised to
#    thirty-one in 1432, fifty-one under Charles V), each great house of the
#    Burgundian inheritance bound to the dynasty by personal chivalric oath.
#    The Sovereign-King of Arms (the Toison d'Or king of arms) maintained the
#    armorial register; the Statutes regulated chapters, expulsions, and the
#    inheritance of stalls. The Order was simultaneously a chivalric corps,
#    a diplomatic instrument (electing foreign sovereigns as members), and a
#    political mechanism for tying the high nobility to the magisterial house.

	potential = {

		has_game_rule = nd_bureaucracies_enabled

	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		nobles_estate

	# revenue the burghers themselves voted.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		burghers_estate_satisfaction_decay = 0.020
		monthly_complacency = 0.005
	}
}

# 3. Chambre des Comptes de Lille -- the Burgundian financial accounting chamber,
#    established at Lille in 1386 by Philip the Bold (a Brussels chamber was
#    added in 1404 for the Brabantine domain, and from 1473 the audit work was
#    progressively centralized at Mechelen under Charles the Bold). Its
#    auditeurs and treasurers reviewed every demesne revenue, customs farm,
#    and feudal aide across the duke's lands; the Burgundian gros silver coin
#    set a kingdom-wide monetary standard. By the mid-fifteenth century it was
#    one of the most disciplined treasury bureaucracies in Latin Europe, and
#    the institutional model for the later Habsburg Council of Finance.

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = GBU
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		owns = location:lille
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The auditeurs reach into every seigneurial revenue stream; demesne
	# accounting is detached from local feudal politics.
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# Working Chamber: the Lille auditeurs reconcile every customs farm; the
	# Burgundian gros holds its weight in markets from Hamburg to Genoa; the
	# urban patriciate -- which staffs the chamber -- entrenches itself as
	# the kingdom's fiscal-administrative class.
	positive_modifier = {
		scale = { value = scope:maintenance }
		tax_income_efficiency = 0.08
		global_max_urban_control = 0.05
		estate_enrichment = 0.05
	}

	# Embezzled Chamber: the auditeurs sell off receivers' offices, the gros
	# is debased to plug deficits, the noble landholders are squeezed for
	# back-aides on disputed feudal accounts, and the kingdom's fiscal
	# legitimacy frays.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.020
		monthly_inflation = 0.005
	}
}

# 4. Grand Conseil de Malines -- the Grand Council of Mechelen, the supreme
#    judicial council and chancery of the Burgundian Netherlands, established
#    by Charles the Bold's edict of 8 December 1473 at Mechelen as the central
#    tribunal of last resort for the duke's seventeen provinces. It heard
#    appeals from the provincial councils of Holland, Brabant, Flanders, and
#    Hainaut; sealed magisterial decrees; and bypassed the seigneurial courts
#    of the high nobility. Suspended after 1477 under the Grand Privilege,
#    re-established at Mechelen under Maximilian in 1504 as the Conseil prive,
#    and the institutional ancestor of the Brussels-based central councils
#    of the Habsburg Netherlands.

	allow = {
