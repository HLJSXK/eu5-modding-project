# National Destinies -- Sovereign Order of Saint John (Knights Hospitaller)
# Four institutions of the Order's mature constitutional state, drawn from the
# Rhodian and Maltese periods (1310-1798). Auto-modifier (nd_hos_bureaucracy_authority)
# grants 2 slots; the Convent capital adds 1 = 3 effective. The player picks 3 of 4.
#
# Note on estates: in the Hospitaller constitution the "nobles_estate" is the
# professed knight-brothers (eight langues of European aristocracy), the
# "clergy_estate" is the Order qua religious order (the brother chaplains and
# the Conventual Church), the "crown_estate" is the Grand Master's magisterial
# household, and "burghers_estate" the Maltese mercantile and corso-investing
# class. This mapping drives the estate alignments below.

# 1. Sacro Consiglio Compito -- the Council of the Tongues, the Order's
#    federal supreme council. Each of the eight langues (Provence, Auvergne,
#    France, Italy, Aragon, Castile-Portugal, Germany, England) sat under its
#    pilier (conventual bailiff): the Marshal of Provence, the Grand Commander
#    of Auvergne, the Hospitaller of France, the Admiral of Italy, the Drapier
#    of Aragon, the Grand Chancellor of Castile, the Grand Bailiff of Germany,
#    and the Turcopolier of England. The Statutes of Margat (1206) and the
#    Statutes of Roger de Pins (1357) fixed the federal compromise: no langue's
#    pilier could be deposed by the Grand Master alone, and the Council's
#    consent was required for war, alliance, and the levy of responsions.
hos_council_of_tongues_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = HOS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# The federal council's mere existence raises the standing of the eight
	# pilier-knights as the Order's constitutional pillars.
	neutral_modifier = {
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Working Council: the langues bring contingents from every European priory
	# to the Convent, the Marshal coordinates the field-army, and the federal
	# legitimacy of magisterial decrees gives every European priory a voice
	# in the magistral household.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		prestige_decay = -0.0025
		nobles_estate_satisfaction_recovery = 0.001
	}

	# Decayed Council: the langues lapse into national factionalism. The French
	# and Spanish pilieri quarrel over magisterial elections; the German bailiff
	# is bypassed; the English langue (suppressed by Henry VIII) leaves a
	# permanent vacancy. Magisterial elections deadlock for months.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		character_cabinet_efficiency = -0.15
		nobles_estate_satisfaction_decay = 0.020
	}
}

# 2. Tribunale degli Armamenti -- the Order's Maritime Tribunal, established
#    at Valletta in 1605 to license, regulate, and adjudicate the corso, the
#    state-licensed Hospitaller privateering campaign against Muslim shipping.
#    The Tribunal issued patents under the magisterial seal (the patente del
#    corso), heard prize cases, and took the tenth (la decima) for the Common
#    Treasury and another tenth for the Grand Master. Its judges came from
#    the Italian and Spanish langues; investors were Maltese burghers, knights
#    in command, and silent partners across Catholic Mediterranean. By the
#    1670s the Tribunal handled hundreds of cases per decade. Abuses --
#    licensed corsairs preying on Greek and Venetian shipping -- forced
#    repeated reforms and ultimately the Tribunal's gradual suppression after
#    1740 under French and Bourbon diplomatic pressure.
hos_tribunale_armamenti_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = HOS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		burghers_estate
		crown_estate
	}
	estates_that_dislike = {
		clergy_estate
	}

	# The Tribunal's patente-system formalizes Hospitaller commerce-raiding as
	# an institution of state. Maltese capital flows into corso ventures.
	neutral_modifier = {
		burghers_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# Regulated corso: prize-courts sitting at Valletta, the Order's tenth
	# flowing into the Common Treasury, the silent partners across the
	# Catholic Mediterranean drawing dividends, and the Maltese burgher class
	# enriched on the back of the patente system.
	positive_modifier = {
		scale = { value = scope:maintenance }
		burghers_estate_max_tax = 0.10
		estate_enrichment = 0.05
		burghers_estate_satisfaction_recovery = 0.001
	}

	# Unregulated corso: the patenti are sold to anyone with capital and a
	# galley, prizes are taken from Catholic Greek and Venetian shipping,
	# the Order's standing collapses across the Christian Mediterranean,
	# and the Conventual Church protests against unjust seizures.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		prestige_decay = 0.0025
		monthly_devotion = -0.05
	}
}

# 3. Sacra Infermeria -- the Holy Infirmary, the Order's defining religious
#    institution. The first Hospitaller infirmary was founded at Jerusalem
#    around 1080 by Brother Gerard; the Maltese Sacra Infermeria opened at
#    Valletta in 1574 and was extended through the seventeenth century into
#    one of the largest hospitals in Europe, with a Great Ward 155 metres
#    long. The Statutes obligated every brother-knight to serve a turn at
#    the patients' bedside (the servitio dei poveri); patients were addressed
#    as "Our Lords the Sick" and dined off silver plate. The Infirmary
#    pioneered standardized casebooks, surgical anatomy through the Anatomy
#    Theatre, and the systematic compounding of medicaments through the
#    Apothecary. Patients of every faith were admitted without distinction --
#    a unique feature among confessional hospitals of the period.
hos_sacra_infermeria_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = HOS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		religion = religion:catholic
	}

	estates_that_like = {
		clergy_estate
		burghers_estate
	}
	estates_that_dislike = {}

	# The hospital is the Order's religious raison d'etre. The brother-knights'
	# servitio dei poveri legitimates the magistral household before God.
	neutral_modifier = {
		monthly_devotion = 0.05
	}

	# Working Infirmary: standardized medicaments compounded at the Apothecary,
	# anatomical training under the Sicilian and Padovan masters, the Maltese
	# medical reputation drawing patients from across Catholic Europe, and
	# the Statutes' admit-all-faiths rule giving the Order a unique standing
	# with non-Catholic populations along its Mediterranean trade routes.
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_disease_resistance = 0.05
		tolerance_heretic = 1
		tolerance_heathen = 1
	}

	# Decayed Infirmary: bedside service neglected by aristocratic novices,
	# medicaments adulterated by underpaid apothecaries, and the Conventual
	# Church increasingly resentful of the magisterial household's diversion
	# of revenue from charitable to fiscal-military ends.
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		clergy_estate_satisfaction_decay = 0.020
		monthly_devotion = -0.05
	}
}

# 4. Tribunale del Comun Tesoro -- the Common Treasury Chamber. The Order
#    supported its Convent in part from the responsions: each commandery in
#    every European priory remitted one-third of its annual revenue to the
#    Treasury. The Comun Tesoro audited these returns, collected mortuaries
#    (the estates of deceased knights), and managed the Order's banking
#    relations with the Genoese and the papal Curia. Its officers (the
#    Receivers and the Procurator General) were appointed by the Grand
#    Master and were the magisterial counterweight to the langue-based
#    Council. The Treasury's centralization under Grand Master Lascaris
#    (1636-1657) and the Codice del Ordine of Rohan (1782) progressively
#    bound the langues into magisterial fiscal authority.

		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = HOS
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# The Receivers reach into every European priory; magisterial revenue is
	# detached from langue politics.
	neutral_modifier = {
		country_cabinet_efficiency = 0.05
	}

	# Audited responsions: priory revenues flow regularly to the Convent,
	# the Procurator General invests Treasury surpluses through Genoese banks,

			subtract = scope:maintenance

	}
