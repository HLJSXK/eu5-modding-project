# National Destinies -- Aztec Empire (AZT) bureaucracies
# Seven Mexica institutions; the player picks 3

# Excan Tlatoloyan -- the Triple Alliance Council.
# Tenochtitlan, Tetzcoco, and Tlacopan agreed in 1428 to share
# imperial tribute on a 2:2:1 basis after overthrowing Azcapotzalco.
# The huey tlatoani of Tenochtitlan presided as senior partner,
# but campaigns and tributary administration were jointly directed.
azt_excan_tlatoloyan_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = AZT
		has_game_rule = nd_bureaucracies_enabled
	}

	# The alliance was anchored in the three lake cities -- without
	# them the institutional fiction collapses.
	allow = {
		owns = location:tenochtitlan
		owns = location:tetzcoco
		owns = location:tlacopan
	}

	estates_that_like = {
		nobles_estate
		crown_estate
	}
	estates_that_dislike = {}

	# neutral: tripartite tribute split lets the realm reach further
	neutral_modifier = {
		diplomatic_capacity = 1
	
		country_cabinet_efficiency = -0.05
}

	# positive: aligned alliance partners co-administer the empire
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_integration_speed_modifier = 0.1
		annexation_speed_modifier = 0.1
		monthly_legitimacy = 0.025
	}

	# negative: tribute-share grievances simmer into separatism
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.15
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
	}
}
