###############################################################
# National Destinies -- DEC (Deccan) bureaucracies
#
# Seven historically grounded institutions of the Deccan plateau,
# spanning the Bahmani Sultanate (1347-1527), the five Deccan
# Sultanates that succeeded it (Ahmadnagar, Bijapur, Berar, Bidar,
# Golkonda 1490-1687), the Vijayanagara Empire (1336-1646), and
# the Asaf Jahi Hyderabad State (founded 1724). Auto-modifier
# grants 3 slots; capital adds 1 = 4 effective. Player picks 4 of 7.
#
# 1. Tarafdari System -- Bahmani provincial governorships
# 2. Mahmud Gawan's Centralizing Reforms -- 1463-1481 reorganization
# 3. Deccani-Pardesi Court Factionalism
# 4. Vijayanagara Nayaka System -- amaram military prebends
# 5. Persianate Diwan -- Qutb Shahi / Adil Shahi chancery
# 6. Deshmukh Watandari -- hereditary Marathi village chiefs
# 7. Asaf Jahi Subadari -- Hyderabad governance
#
# All IDs are tag-prefixed (dec_) to avoid collisions.
###############################################################


# =====================================================================
# 1. Tarafdari System
#
# Bahmani founder Hasan Gangu (Ala-ud-din Bahman Shah, r. 1347-1358)
# divided his realm into four atraf (provinces): Daulatabad, Berar,
# Bidar and Gulbarga. Each was ruled by a tarafdar -- a great noble
# who collected revenue, raised troops, and held the citadels of his
# province. The system extended royal authority across the plateau,
# but by the 1480s the tarafdars had become a kingmaking aristocracy
# whose disputes carved Bahmani lands into the five Deccan Sultanates.
# =====================================================================
dec_tarafdari_system_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = DEC
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {}

	estates_that_like = {
		nobles_estate
	}
	estates_that_dislike = {
		crown_estate
	}

	# neutral: a permanent layer of provincial governance staffed by tarafdars
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# positive: tarafdars deliver garrisons, levies and military discipline
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_levy_size_modifier = 0.10
		global_max_rural_control = 0.03
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	# negative: tarafdars feud, build personal dynasties, and break the realm apart
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_separatism = 0.05
		monthly_legitimacy = -0.05
	}
}


# =====================================================================
# 2. Mahmud Gawan's Centralizing Reforms
#
# Khwaja Mahmud Gawan, an Iranian-born wazir who served Bahmani sultans
# Humayun, Nizam Shah and Muhammad Shah III, restructured the realm
# between 1463 and 1481. He doubled the four atraf into eight, removed
# the citadels of each province from the tarafdar's command and placed
# them under royal officers, conducted a revenue survey of arable land,
# and established the Madrasa Mahmud Gawan at Bidar. The reforms briefly
# arrested Bahmani decline; Gawan was executed on a forged charge in
# 1481, and the realm fragmented within a decade of his death.
# =====================================================================

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price

		value = country_economical_base
		multiply = 0.005
	}

	potential = {
		has_or_had_tag = DEC
		has_game_rule = nd_bureaucracies_enabled
	}

	allow = {
		stability >= 1
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: doubling the provinces produces a denser administrative grid
	neutral_modifier = {
		num_nd_provincial_estates = 1
	}

	# positive: surveyed revenue, royal-officered fortresses, ordered chancery
	positive_modifier = {

# =====================================================================
# 3. Deccani-Pardesi Court Factionalism
#
# Deccan courts were divided between the Deccani party -- locally born
# Sunni Muslims, Habshi (African) military slaves and converts -- and
# the Pardesi or Afaqi party of Iranian, Turkic and Iraqi newcomers.
# The two factions drew different stipends, served in separate cavalry
# regiments, and patronized rival theological schools (Deccani Sunni,
# Pardesi Twelver Shia after Qara Yusuf). Bahmani sultans alternated
# Deccani and Pardesi wazirs to keep both in check; under the Adil
# Shahis of Bijapur the factions repeatedly took the city in arms.
# Maintained as a structural compromise, the rivalry produced the
# brilliantly cosmopolitan Deccan court culture; neglected, it
# produced succession civil wars.
# =====================================================================

# =====================================================================
# 4. Vijayanagara Nayaka System
#
# The amara-nayankara system, formalized under Krishnadevaraya
# (r. 1509-1529) and the later Tuluva and Aravidu dynasties, granted
# military commanders revenue assignments (amaram) over villages and
# forts in exchange for maintaining specified contingents of cavalry,
# infantry and elephants. The nayakas of Madurai, Tanjavur, Gingee,
# Ikkeri and Mysore answered to the Rayas of Vijayanagara through
# annual tribute and the obligation to march; after Talikota in 1565
# they became the de facto sovereign powers of Tamil and Kannada
# country, and several outlived the empire by more than a century.

# =====================================================================
# 5. Persianate Diwan
#
# The Qutb Shahis of Golkonda (1518-1687) made Persian their court and
# administrative language, importing chancery practice from Safavid
# Iran along with Twelver Shia ulema; Adil Shahi Bijapur followed in
# parallel under Yusuf Adil Shah, a Persian-born founder said to be
# of Aq Qoyunlu descent. The diwans drafted firmans in nasta'liq,
# audited the karkhanas, and supervised waqf endowments and Hajj
# subsidies forwarded through Mocha. Persian remained the language of
# Deccan administration into the Asaf Jahi era, fading only under
# the British residency.

# =====================================================================
# 6. Deshmukh Watandari
#
# The deshmukh and his fiscal-clerical counterpart the deshpande were
# hereditary Marathi village chiefs whose watan -- patrimony of office
# -- carried rights to a fixed share of the village revenue, judicial
# authority over caste disputes, and the privilege of maintaining a
# small armed retinue. The watandars predated the Bahmanis, persisted
# under the Adil Shahis, and supplied Shivaji with both his earliest
# allies and his most stubborn opponents. Their rights were so deeply
# entrenched that Mughal, Maratha and Asaf Jahi states alike chose to
# co-opt the watandars rather than dispossess them.
# =====================================================================

	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price

	}

	allow = {
		OR = {
			owns = location:pune
			owns = location:daulatabad
			owns = location:ahmednagar

		nobles_estate
		peasants_estate

	estates_that_dislike = {

	#         remote Marathi villages but freezes the social order
