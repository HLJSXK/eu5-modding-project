###############################################################
# National Destinies -- AYU (Ayutthaya) bureaucracies
#
# Five Ayutthaya-era institutions distinct from the broader SIA
# (Thailand) palette. SIA already covers the Trailok Reform
# spine (Chatusadom, Sakdina, Phra Khlang Sinkha, Phrai Suay,
# Maha Sangharaja, Three Seal Law). AYU's bureaucracies focus
# on what makes Ayutthaya specifically distinct: its cosmopolitan
# foreign-officered court, dual port administration, palace
# enforcement apparatus, war-elephant corps reformed under
# Naresuan, and the Mahatlek pages system that bound noble
# scions to the throne.
#
# Auto-modifier (nd_ayu_bureaucracy_authority) grants 2 slots;
# capital adds 1 = 3 total. Player picks 3 of 5.
#
# IDs prefixed ayu_; loc keys match the bare bureaucracy ID.
###############################################################


# =====================================================================
# Krom Tha -- the Twin-Port Administration
#
# Ayutthaya's maritime customs were uniquely organized into TWO
# co-equal port departments: the Krom Tha Khwa (Right Port) handled
# Chinese and East Asian trade and was traditionally led by an officer
# of Chinese descent, while the Krom Tha Sai (Left Port) handled
# Indian, Persian, Arab, and European traffic and was run by Persian
# Shia administrators -- most famously the Bunnag family, descendants
# of the Persian merchant Sheikh Ahmad Qomi who arrived in 1602. The
# bifurcation gave the throne two parallel customs revenues, two
# diplomatic apparatuses, and an institutionalized place for foreign-
# born officers in the highest ranks of court. It is the structural
# expression of Ayutthaya's cosmopolitan trade culture and is
# distinct from the Phra Khlang Sinkha royal-trade monopoly (which
# the throne ran for its own profit). Krom Tha governed everyone
# else's trade in the kingdom.
# =====================================================================
ayu_krom_tha_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.003
	}

	potential = {
		has_or_had_tag = AYU
		has_game_rule = nd_bureaucracies_enabled
	}

	# The twin-port system requires a working oceanic port: the capital
	# Ayodhya, the Andaman-side transit at Mergui, or the secondary
	# Lopburi customs house used in the 17th century.
	allow = {
		OR = {
			owns = location:ayodhya
			owns = location:mergui
			owns = location:lopburi
		}
	}

	estates_that_like = {
		burghers_estate
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	# neutral: foreign-born officers hold senior administrative seats
	neutral_modifier = {
		burghers_estate_allowed_in_cabinet = yes
	}

	# positive: dual customs running -- two trade catchments,
	# foreign merchants treated as kingdom subjects, urban prosperity
	positive_modifier = {
		scale = { value = scope:maintenance }
		global_trade_center_power = 0.15
		diplomatic_capacity = 1
		cultural_influence = 0.10
	}

	# negative: customs become extortionate and the foreign quarters
	# turn parasitic, the Thai nobility resents the strangers in office
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		nobles_estate_satisfaction_decay = 0.04
		antagonism_received_modifier = 0.10
	}
}


# =====================================================================
# Tamruat Phon -- the Palace Police
#
# Within the Wang (palace) pillar of the Chatusadom, the Tamruat Phon
# served as the royal household's enforcement arm: a guard corps
# distinct from the field army that policed the palace precinct,
# escorted the king during processions, conducted secret arrests of
# court rivals, and investigated treason cases brought before the
# Mahatthai. Recruited from trusted Mon and Khmer families and answer-
# able only to the Maha Uparat or the king himself, the Tamruat Phon
# functioned as Ayutthaya's praetorian guard. Their willingness (or
# refusal) to act decided the outcome of every Ayutthayan succession
# crisis from Borommatrailokkanat through King Suriyamarin's deposition
# in 1767.

# =====================================================================
# Yodfa Elephant Corps -- the Naresuan Military Reform
#
# After his 1593 single combat with the Burmese crown prince
# Mingyi Swa at Nong Sarai, King Naresuan reorganised the Ayutthayan
# elephant corps into a hereditary professional cadre. The Yodfa
# established war-elephant breeding stations in the upper Chao Phraya,
# permanent mahout dynasties from Lan Na and Lao client states, and a
# graded officer hierarchy that placed the Maha Uparat at the head of
# the elephantry. The corps' pikeman escorts -- drawn from Mon refugee
# communities granted Ayutthayan land in exchange for service --
# fielded the kingdom's most reliable infantry through the
# 17th century, fending off the renewed Toungoo invasions and
# extending Ayutthayan suzerainty into Cambodia and the northern
# Malay sultanates.

# =====================================================================
# Krom Mahatlek -- the Royal Pages Court
#
# The Krom Mahatlek enrolled the sons of Ayutthayan nobles, vassal
# rulers, and high-ranking foreign officers as royal pages in the
# inner palace. Pages slept on the floor outside the king's chambers,
# served his meals, ran his messages, and learned governance by
# observing the king at council. On reaching majority they entered
# the four-pillar ministries as junior officers -- ensuring that the
# next generation of provincial governors and Krom officials had been
# personally vetted by the throne and bound to it by years of intimate
# service. The system also functioned as a hostage apparatus: a
# rebellious vassal's son was always a sword's-stroke from the king's
# guard. King Narai's pages included Constantine Phaulkon, the Persian
# Aqa Muhammad, and the Bunnag heirs.

# =====================================================================
# Kromakan Asa -- the Foreign Officer Corps
#
# More than any other Southeast Asian state, Ayutthaya integrated
# foreign-born officers into the upper military and civilian
# administration. The Kromakan Asa enrolled Persian Shia under
# Sheikh Ahmad Qomi (1602), the Japanese Yamada Nagamasa as governor
# of Nakhon Si Thammarat (1620s), the Greek Constantine Phaulkon as
# Phra Khlang and de facto vizier under King Narai (1685-88),
# Portuguese gunners at the Asa Cherokat artillery park, and French
# military engineers under General Desfarges in the 1687-88
# misadventure. Each foreigner-officer was granted Sakdina rank,
# Thai noble honorifics, and a Krom of his own client community.
# When the system worked it gave Ayutthaya unique access to firearms,
# fortification science, and Persian Gulf diplomacy. When it broke
# down -- as it did in the 1688 Revolution that killed Phaulkon --
# the Thai nobility's reaction was xenophobic and absolute.
