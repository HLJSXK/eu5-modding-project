# Fire formation events at game start for tags that exist as already-formed countries.

		if = {
			limit = {
				exists = c:BOH

			c:BOH = { trigger_event_non_silently = nd_boh.1 }
		}

		# Byzantine succession crisis disaster + dense early-game content

		# our timed modifier at game start would undercut that. Game-start
		# BYZ players earn the formation event by reconquering the 3-city
		# Komnenian core (see on_four_yearly_check_for_script below).

# Danubian Monarchy succession-question chain advancement.
# The Reich situation's on_monthly used to do this, but coupling chain
# advancement to situation activity meant that if the situation ended
# early (strength path: 80% war score vs Emperor), the chain stopped
# firing. The chain is a SEPARATE internal pressure from the external
# pariah situation -- it has to run regardless of situation state.
#
# Each chain step uses a TIMED variable set in nd_dnm.1: years = 2/5/8.
# When the timed var has decayed AND the matching _fired guard is unset,
# the event fires once and the guard is set. Sequential dependency on
# the previous step's _fired guard means events fire in narrative order
# even if game time jumps oddly.
on_country_yearly_pulse = {
	effect = {
		if = {
			limit = {
				tag = DNM
				has_variable = nd_dnm_formed
				NOT = { has_variable = nd_dnm_chain_30_pending }
				NOT = { has_variable = nd_dnm_chain_30_fired }
			}
			hidden_effect = { set_variable = nd_dnm_chain_30_fired }
			trigger_event_non_silently = nd_dnm.30
		}
		if = {
			limit = {
				tag = DNM
				has_variable = nd_dnm_chain_30_fired
				NOT = { has_variable = nd_dnm_chain_31_pending }
				NOT = { has_variable = nd_dnm_chain_31_fired }
			}
			hidden_effect = { set_variable = nd_dnm_chain_31_fired }
			trigger_event_non_silently = nd_dnm.31
		}
		if = {
			limit = {
				tag = DNM
				has_variable = nd_dnm_chain_31_fired
				NOT = { has_variable = nd_dnm_chain_32_pending }
				NOT = { has_variable = nd_dnm_chain_32_fired }
			}
			hidden_effect = { set_variable = nd_dnm_chain_32_fired }
			trigger_event_non_silently = nd_dnm.32
		}
	}
}

# Achievement-triggered formation events for starting+formable overlaps.
# The country-start player can't use the formable button (you can't form
# what you already are), so we check four-yearly whether they have reached
# the formable's imperial-reach allow conditions. When met, fire the
# formation event chain. Variable check prevents re-firing.
on_four_yearly_check_for_script = {
	effect = {
		# BYZ achievement: 3-city Komnenian core (Constantinople + Athens +
		# Iznik). Forces game-start BYZ to actually reconquer territory --
		# meaningful imperial recovery beyond the starting position --
		# before earning our formation event and content. Preserves the
		# vanilla 1.2 DLC's stabilization-focused early game.
		if = {
			limit = {
				exists = c:BYZ
				NOT = { c:BYZ = { has_variable = nd_byz_formed } }
				c:BYZ = {
					# Orthodox is the canonical restoration, but d008 lets BYZ
					# restore Hellenism (the old Roman religion). Accept both so a
					# Hellenic revival isn't locked out of the formation event;
					# nd_byz.1 offers a religion-appropriate choice for each.
					OR = {
						religion = religion:orthodox
						religion = religion:hellenism_religion
					}
					owns = location:constantinople
					owns = location:athens
					owns = location:iznik
				}
			}
			c:BYZ = { trigger_event_non_silently = nd_byz.1 }
		}
	}
}
