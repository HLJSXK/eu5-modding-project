	fire_only_once = yes
	
	dynamic_historical_event = {
		tag = ENG
		tag = GBR
		from = 1500.1.1
		to = 1550.1.1
		monthly_chance = 1
	}

	image = "gfx/interface/illustrations/trade_market/trade.dds"

	trigger = { 
		hidden_trigger = { 
			at_war = no
		}
		owns = location:bristol
	}

	immediate = { 
		goods:cheese = { save_scope_as = cheese }
		location:bristol = { save_scope_as = bristol }
		location:bristol.owner = { save_scope_as = target_nation }
	}

	option = { #Accept Cheese Innovation - Rural
		name = cheese_events.1.b

		trigger = { 
			location:bristol = { 
				is_city = no
			}
		}

		location:bristol = { 
			add_location_modifier = cheese_hub
			construct_building = {
				building_type = building_type:rural_cheesemaker
				instant = yes
				owner = PREV
				cost_multiplier = 0.5
				cost_multiplier_reason = "cheese_event"
			}
		}

		location:bristol.dominant_culture = { add_cultural_influence = cultural_influence_weak_bonus }

		ai_will_select  = { value = 100 }
		historical_option = yes
	}

	option = { #Accept Cheese Innovation - City
		name = cheese_events.1.b

		trigger = { 
			location:bristol = { 
				is_city = yes
			}
		}

		location:bristol = {
