clocks_events.1 = { # The Frauenkirche Clock

	trigger = {
		hidden_trigger = {
			at_war = no
		}

		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
		save_scope_as = target_nation
		location:nuremberg = { save_scope_as = nuremberg }
	}

	option = {
		name = clocks_events.1.a

		add_gold = {
			value = country_tax_base
			multiply = -1
		}

		location:nuremberg = {
			create_art = {
				quality = 55
				type = work_of_art_type:astronomical_clock
				key = frauenkirche_clock
			}
		}

		add_prestige = prestige_mild_bonus

		ai_will_select  = { value = 100 }
		historical_option = yes
	}

	option = {
		name = clocks_events.1.b

		add_prestige = prestige_weak_penalty

		ai_will_select  = { value = 20 }
	}
}

	trigger = {
		hidden_trigger = {
			at_war = no
		}
		character:giovanni_de_dondi ?= {
			is_alive = yes
			owner ?= ROOT
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
		save_scope_as = target_nation
		character:giovanni_de_dondi = { save_scope_as = giovanni_de_dondi }
		capital = { save_scope_as = target_location }
		random_rival = { save_scope_as = rival_nation }
	}

	option = {
		name = clocks_events.3.a

		add_gold = {
			value = country_tax_base
			multiply = -2
		}

		capital = {
			create_art = {
				artist = scope:giovanni_de_dondi
				quality = 90
				type = work_of_art_type:astronomical_clock
				key = astrarium
			}
		}

		add_research_progress = research_progress_weak_bonus
		add_prestige = prestige_severe_bonus

		ai_will_select  = { value = 100 }
		historical_option = yes
	}

	option = {
		name = clocks_events.3.b

		if = {
			limit = {
				exists = scope:rival_nation
			}
			scope:giovanni_de_dondi = { move_country = scope:rival_nation }
		}
		else = {
			hidden_effect = { kill_character_silently = scope:giovanni_de_dondi }
		}

		add_prestige = prestige_mild_penalty

		ai_will_select  = { value = 20 }
	}
}
