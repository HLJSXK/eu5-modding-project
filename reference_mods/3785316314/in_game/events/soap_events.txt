	trigger = { 
		hidden_trigger = { 
			at_war = no

				owns = location:kran
			}
		}
	}

	immediate = { 
		location:kran = { save_scope_as = kran }
		save_scope_as = target_nation
	}

	option = {
		name = soap_events.1.a

		location:kran = { 
			change_raw_material = goods:aromatics
			add_location_modifier = soap_rose_valley

				add_cultural_influence = cultural_influence_weak_bonus

			value = country_tax_base
			multiply = -1.05
		}

		historical_option = yes
	}

	option = { 
		name = soap_events.1.b
