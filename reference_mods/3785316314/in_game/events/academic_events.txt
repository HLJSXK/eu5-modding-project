	desc = academic_events.100.desc
	outcome = neutral

	image = "gfx/interface/illustrations/institutions/legalism.dds"

	immediate = {
		scope:location.dominant_religion = { save_scope_as = target_religion }

		every_country = {
			limit = {
				NOT = { this = scope:winner }
				NOT = { this = scope:loser }
				capital.sub_continent = scope:location.sub_continent
				dominant_religion = scope:target_religion
			}
			add_to_list = academies_diaspora_candidates
		}

		while = {
			count = 5
			limit = { list_size = { name = academies_diaspora_candidates value > 0 } }
			random_in_list = {
				list = academies_diaspora_candidates
				save_scope_as = diaspora_target
			}
			scope:diaspora_target = {
				trigger_event_non_silently = { id = academic_events.101 days = 14 }
				remove_from_list = academies_diaspora_candidates
			}
		}
	}

	option = {
		name = academic_events.100.a
		trigger = {
			scope:winner = {
				academy_count < modifier:academy_max_level
			}
		}

		add_gold = {
			value = country_tax_base
			multiply = -2
		}

		scope:location = {
			add_academic_momentum = { amount = academic_momentum_severe_penalty }
		}

		scope:winner = { refresh_national_academy_list = yes }
		scope:loser = { refresh_national_academy_list = yes }

		ai_will_select = { value = 100 }
	}

	option = {
		name = academic_events.100.b

		scope:location ?= {
            remove_location_modifier = clocks_academy_site
			remove_variable = academic_momentum
        }
        scope:winner = { refresh_national_academy_list = yes }
