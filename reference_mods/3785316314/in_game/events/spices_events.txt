# Nutmeg & Banda Islands
spices_events.1 = { # The Banda Islands
	type = country_event
	title = spices_events.1.title
	desc = spices_events.1.desc
	outcome = neutral
	historical_info = spices_events.1.historical_info

	fire_only_once = yes
	
	dynamic_historical_event = {
		tag = ENG
		tag = POR
		tag = GBR
		tag = HOL
		tag = NED
		from = 1437.1.1
		to = 1637.1.1
		monthly_chance = 100
	}

	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_nutmeg.dds"

	trigger = { 
		hidden_trigger = { 
			at_war = no
		}
		location:banda_island = { 
			OR = { 
				has_owner = no
				NOT = {	owner ?= { capital ?= { continent = continent:europe } } }
			}
			is_discovered_by = ROOT	
		}	
	}

	immediate = { 
		location:banda_island = { save_scope_as = banda }
		culture:venetian = { save_scope_as = venetian }
		culture_group:arabian_group = { save_scope_as = arabian }
		goods:nutmeg = { save_scope_as = nutmeg }
	}

	option = { # Establish a Trade Post
		name = spices_events.1.a

		trigger = { 
			location:banda_island = { 
				is_city = yes
				has_owner = yes
				NOT = { owner ?= { capital ?= { continent = continent:europe } } }
			}
		}

		location:banda_island = { 
			if = { 
				limit = { 
					is_city = no
				}
				construct_building = {
					building_type = building_type:overseas_trading_post
					instant = yes
					owner = PREV
				}
			}
			else = { 
				construct_building = {
					building_type = building_type:trade_office
					instant = yes
					owner = PREV
				}
			}
		}

		ai_will_select  = { value = 100 }
		historical_option = yes
	}

	option = { # Return with Larger Force
		name = spices_events.1.b

		trigger = { 
			location:banda_island = {
				has_owner = yes 
			}
		}

		add_casus_belli = { 
			type = casus_belli:cb_conquer_province 
			province = location:banda_island.province
			target = location:banda_island.owner
			years = 25
		}

		ai_will_select  = { value = 30 }
	}

	option = { 
		name = spices_events.1.c

		trigger = { 
			location:banda_island = { 
				has_owner = no
			}
		}

		capital.market = { 
			add_goods_supply = { 
				goods = goods:nutmeg
				amount = 25
			}
		}

		add_gold = { 
			value = monthly_income_trade_and_tax	
			divide = 10 
		}

		ai_will_select  = { value = 100 }
	}

	option = { 
		name = spices_events.1.d

		add_prestige = prestige_mild_penalty

		trigger = { 
			location:banda_island = { 
				has_owner = yes
			}
		}

		ai_will_select  = { value = 20 }
	}
}
