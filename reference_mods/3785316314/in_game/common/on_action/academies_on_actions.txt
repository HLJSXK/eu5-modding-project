# Academy Events
yearly_country_pulse = {
    on_actions = {
        check_academy_overcrowding
    }
}

check_academy_overcrowding = {
    trigger = {
        has_game_rule = tt_expanded_on
        academy_count > modifier:academy_max_level
        academy_count > 1
    }
    effect = {
        trigger_event_non_silently = academic_events.80
    }
}

### Global Uni Pulse
weather_monthly_pulse = { 
	on_actions = { 
		academic_global_pulse
	}
}

academic_global_pulse = { 
	trigger = { has_game_rule = tt_expanded_on }
	effect = {
		refresh_global_academy_list = yes
	}
}

### Calculating Academic Tradition & Monthly Influence
monthly_country_pulse = {
	on_actions = {
    	pulse_academic_tradition
		pulse_academic_influence
    }
}

pulse_academic_tradition = { 
	trigger = { has_game_rule = tt_expanded_on }
	effect = { 
		every_in_list = {
			if = {
				limit = {
					has_variable = academic_momentum
				}
				change_variable = {
					name = academic_momentum
					multiply = {
						value = 1
						subtract = owner.modifier:academic_momentum_decay
					}
				}
			}
			
			# Check if has chair.
			if = { 
				limit = {
					NOT = { has_building = building_type:rectorate_chair_clergy }
					NOT = { has_building = building_type:rectorate_chair_nobility }
					NOT = { has_building = building_type:rectorate_chair_burghers }
				}
				clocks_swap_academic_chair = yes
			}

			# Check event.
			if = { 
				limit = { 
					NOT = { has_variable = academy_cooldown }
				}
				save_scope_as = academy_location
				root = { trigger_academy_event = yes }
				set_variable = { name = academy_cooldown value = yes days = { 1460 2920 } }
			}

			variable = national_academies_list
			set_academic_tradition = yes
		}
	}
}

pulse_academic_influence = {
	trigger = { has_game_rule = tt_expanded_on }
    effect = {
		set_variable = {
			name = academic_influence_gain_universities
			value = {
				value = total_effective_building_levels:university
				multiply = 0.01
				max = 1
			}
		}

		set_local_variable = { name = agt value = 0 }
		every_in_list = {
			variable = national_academies_list
			change_local_variable = {
				name = agt
				add = {
					value = var:academic_tradition_value
					multiply = 0.0005
				}
			}
		}
		set_variable = { 
			name = academic_influence_gain_academies 
			value = { 
				value = local_var:agt
				max = 5
			}
		}

		set_variable = {
			name = academic_influence_gain_total
			value = {
				value = var:academic_influence_gain_universities
				add = var:academic_influence_gain_academies
				add = modifier:monthly_academic_influence
			}
		}

		add_academic_influence = { amount = var:academic_influence_gain_total }
	}
}

### Removing Sites of Learning after Conquest
on_took_location_in_peace_treaty = {
    on_actions = {
        clocks_remove_academy_site_on_conquest
    }
}

clocks_remove_academy_site_on_conquest = {
	trigger = { 
		scope:location = { has_location_modifier = clocks_academy_site }
	}
    effect = {
		scope:winner ?= {
			trigger_event_non_silently = { id = academic_events.100 days = 7 }
		}
    }
}
