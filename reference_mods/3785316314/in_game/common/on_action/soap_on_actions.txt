add_potash_rgos = { 
	effect = { 
		location:balho = { 
			change_raw_material = goods:potash
			if = { 
				limit = { has_building = building_type:farming_village }
				destroy_building = "building(building_type:farming_village|owner)"
			}
			else_if = { 
				limit = { has_building = building_type:forest_village }
