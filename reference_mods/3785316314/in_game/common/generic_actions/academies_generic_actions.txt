appoint_scholar = {
    type = owncountry
    cooldown = {
        type = appointed_scholar
        years = 1
    }
    allow = {
        scope:actor = {
            custom_tooltip = {
                text = academic_influence_cost_45_tt
                has_variable = academic_influence
                var:academic_influence >= 45
            }
        }
        hidden_trigger = { 
            has_game_rule = tt_expanded_on
        }
    }
    select_trigger = {
        looking_for_a = character
        source = actor
        target_flag = target_character
        name = "choose_character"
        column = { data = character_info }
        visible = {
            is_adult = yes
            is_courtier = yes
            age_in_years >= 40
            
            trigger_if = {
                limit = { root = { modifier:nobles_academic_authority = yes } }
                has_estate = estate_type:nobles_estate
            }
            trigger_else_if = {
                limit = { root = { modifier:burghers_academic_authority = yes } }
                has_estate = estate_type:burghers_estate
            }
            trigger_else_if = { 
                limit = { root = { modifier:crown_academic_authority = yes } }
                is_ruler = no
                is_heir = no
                is_regent = no
            }
            trigger_else = {
                has_estate = estate_type:clergy_estate
            }
        }
        enabled = {
            is_ruler = no
            is_heir = no
            is_regent = no
        }
    }
    effect = {
        scope:actor = {
            change_variable = { name = academic_influence subtract = 45 }
            custom_tooltip = { 
                text = appoint_scholar_tt
                scope:target_character ?= {
                    add_trait = trait:academic
                    add_adm = { 3 10 }
                    add_dip = { 3 10 }
                    add_mil = { 3 10 }
                }
            }
        }
    }
    ai_will_do = { 
        add = 5 
    }
}

promote_center_learning = {
    type = owncountry
    allow = {
        scope:actor = {
            at_war = no
            gold >= country_economical_base
            modifier:academy_max_level > academy_count
            is_subject = no
        }
        hidden_trigger = { 
            has_game_rule = tt_expanded_on
        }
    }
    select_trigger = {
        looking_for_a = location
        source = actor
        target_flag = target_location
        name = "choose_location"
        none_available_msg_key = "no_locations_available"
        column = { data = name }
        visible = {
            owner ?= scope:actor
            has_building = building_type:university
            NOT = {
                province = { 
                    any_location_in_province = {
                        has_location_modifier = clocks_academy_site
                    }
                }
            }
        }
    }
    effect = {
        scope:actor = {
            add_gold = {
                value = country_economical_base
                multiply = -1
            }
            custom_tooltip = {
                text = promote_center_learning_tt
                if = { 
                    limit = { 
                        modifier:nobles_academic_authority = yes
                    }
                    scope:target_location ?= {
                        if = { 
                            limit = { NOT = { has_building = building_type:rectorate_chair_nobility } }
                            construct_building = {
                                building_type = building_type:rectorate_chair_nobility
                            }
                        }
                    }
                }
                else_if = { 
                    limit = { 
                        modifier:burghers_academic_authority = yes
                    }
                    scope:target_location ?= {
                        if = { 
                            limit = { NOT = { has_building = building_type:rectorate_chair_burghers } }
                            construct_building = {
                                building_type = building_type:rectorate_chair_burghers
                            }
                        }
                    }
                }
                else = { 
                    scope:target_location ?= {
                        if = { 
                            limit = { NOT = { has_building = building_type:rectorate_chair_clergy } }
                            construct_building = {
                                building_type = building_type:rectorate_chair_clergy
                            }
                        }
                    }
                }
                scope:target_location ?= {  
                    add_location_modifier = { modifier = clocks_academy_site years = -1 } 
                    add_variable = { name = cademy_cooldown value = yes days = { 100 360 } }
                }
                scope:actor = { refresh_national_academy_list = yes }
            }
        }
    }
    ai_will_do = { add = 5 }
}

dissolve_academy = {
    type = owncountry
    allow = {
        hidden_trigger = { 
            has_game_rule = tt_expanded_on
        }
    }
    select_trigger = {
        looking_for_a = location
        source = actor
        target_flag = target_location
        name = "choose_location"
        none_available_msg_key = "no_locations_available"
        column = { data = name }
        visible = {
            owner ?= scope:actor
            has_location_modifier = clocks_academy_site
        }
    }
    effect = {
        scope:target_location ?= {
            remove_location_modifier = clocks_academy_site
            remove_variable = academic_momentum
        }
        scope:actor = { refresh_national_academy_list = yes }
    }
    ai_will_do = { add = -100 }
}
