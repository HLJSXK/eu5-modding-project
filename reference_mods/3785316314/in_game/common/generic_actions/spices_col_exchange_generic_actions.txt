move_ow_spice_to_new_location = {
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 60
	automation_tick = never
	automation_tick_frequency = 12

	maximum_targets_in_one_tick = 20
	disallowed_duplicates_of_targets_for_ai = { target_location }

	potential = {
	}

	allow = {
		scope:actor ?= {
			any_market_present_in_country = {
				has_old_world_goods_in_market = yes
			}
		}
	}
	
	price = price:move_good_to_new_location

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:columbian_exchange = this
			situation_is_active = yes
		}
	}

	#goods in scope:goods
	select_trigger = {
		looking_for_a = goods
		target_flag = target_good
		ai_interaction_source_list = {# Has a market that demands the good
			every_goods = {
				limit = {
					has_tag = old_world_spices
				}
				save_temporary_scope_as = temp_good
				scope:actor = { 
					random_market_present_in_country = {
						limit = {
							"goods_demand_in_market(scope:temp_good)" > "goods_supply_in_market(scope:temp_good)"
						}
						scope:temp_good = { add_to_list = source }
					}
				}
			}
		}
		name = "spread_select_goods"
		none_available_msg_key = "move_ow_spice_to_new_location_no_available_spice"
		column = { data = name }
		visible = {
			has_tag = old_world_spices
			scope:actor ?= {
                OR = { 
                    any_owned_location = { 
                        raw_material = root
                    }
                    any_subject = { 
                        any_owned_location = { 
                            raw_material = root 
                        }
                    }
                }
			}
		}
	}
	
	#Select Region
	select_trigger = {
		looking_for_a = region
		source = actor
		target_flag = target_region
		name = "choose_a_region"
		none_available_msg_key = "no_regions_available"
		column = { data = name }
		visible = {
			OR = {
				continent = continent:america
				continent = continent:oceania
				this = region:macaronesia_region
			}
			scope:actor = {
				owns_or_has_subject_in = prev
			}
		}
	}
	#Select the owned location where to spread the new resource
	select_trigger = {
		looking_for_a = location
		source = target_region
		source_flags = only_actual_locations
		target_flag = target_location
		ai_interaction_source_list = {
			scope:target_region = {
				every_area_in_region = {
					limit = {
						scope:actor = { owns_or_has_subject_in = prev }
					}
					every_province_definition_in_area = {						
						limit = {
							scope:actor = { owns_or_has_subject_in = prev }
							#Maximum of 2 of the same resource per Province
							"raw_material_amount(scope:target_good)" < 2
						}
						every_ownable_location_in_province_definition = {
							limit = {
								owner ?= {
									is_subject_or_below_of = scope:actor
								}
								is_candidate_for_OW_good = yes
								NOT = {raw_material ?= scope:target_good}								
							}
							add_to_list = source
						}
					}
				}
			}
		}
		name = "choose_location"
		none_available_msg_key = "move_ow_spice_to_new_location_no_available_spice"
		column = { data = name }
		column = { data = raw_material }
		visible = {
			is_ownable = yes
			has_owner = yes
			OR = {
				owner ?= scope:actor
				top_owner ?= scope:actor
			}
			OR = {
				continent = continent:america
				continent = continent:oceania
				region = region:macaronesia_region
			}

			#Turmeric
			trigger_if = {
				limit = {
					scope:target_good ?= goods:turmeric
				}
				NOT = { raw_material ?= goods:turmeric }
                OR = { 
                    vegetation = forest
                    vegetation = farmland
                    vegetation = sparse
                }
				OR = {
					topography = plateau
					topography = hills
                    topography = flatland
				}
				OR = {					
					climate = tropical
                    climate = subtropical
				}
			}

            #Cardamom
			trigger_if = {
				limit = {
					scope:target_good ?= goods:cardamom
				}
				NOT = { raw_material ?= goods:cardamom }
                OR = { 
                    vegetation = forest
                    vegetation = jungle
                }
				OR = {
					topography = mountains
					topography = hills
                    topography = plateau
				}
				OR = {					
					climate = tropical
                    climate = subtropical
				}
			}

            #Cinnamon
            trigger_if = {
				limit = {
					scope:target_good ?= goods:cinnamon
				}
				NOT = { raw_material ?= goods:cinnamon }
                OR = { 
                    vegetation = forest
                    vegetation = farmland
                    vegetation = sparse
                }
				OR = {
					topography = flatland
					topography = hills
                    topography = plateau
				}
				OR = {					
					climate = tropical
                    climate = subtropical
				}
			}

            #Ginger
            trigger_if = {
				limit = {
					scope:target_good ?= goods:ginger
				}
				NOT = { raw_material ?= goods:ginger }
                OR = { 
                    vegetation = farmland
                    vegetation = forest
                    vegetation = sparse
                }
				OR = {
					topography = flatland
					topography = hills
                    topography = plateau
				}
				OR = {					
					climate = tropical
                    climate = subtropical
				}
			}

            #Nutmeg
            trigger_if = {
				limit = {
					scope:target_good ?= goods:nutmeg
				}
				NOT = { raw_material ?= goods:nutmeg }
                OR = { 
                    vegetation = forest
                    vegetation = jungle
                }
				OR = {
					topography = flatland
					topography = hills
				}
				OR = {					
					climate = tropical
				}
			}

            #Star Anise
            trigger_if = {
				limit = {
					scope:target_good ?= goods:star_anise
				}
				NOT = { raw_material ?= goods:star_anise }
                OR = { 
                    vegetation = woods
                    vegetation = jungle
                }
				OR = {
					topography = plateau
					topography = hills
                    topography = mountains
				}
				OR = {					
					climate = subtropical
				}
			}
		}
	}

	effect = {
		scope:target_location ?= {			
			change_max_raw_material_workers = {
				value = rgo_workers
				subtract = 1
				min = 0
				multiply = -1
			}
			change_prosperity = prosperity_weak_penalty
			change_raw_material = scope:target_good
		}
	}
	
	ai_will_do = {
		#If the action is doable by the AI, 
		#meaning there is a combination of target good and target location where it can replace the RGO it should do it
		add = 200
	}
}
