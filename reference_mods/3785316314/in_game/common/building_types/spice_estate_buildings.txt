# Extra Plantations
cocoa_plantation = {
	audio_tier = 2
	is_foreign = no
	pop_type = slaves
	max_levels = plantation_cap
	startup_ramp_target = rural_startup_ramp_target
	
	employment_size = plantation_employment
	
	category = colonial_category
	
	location_potential = {
		raw_material = goods:cocoa
		OR = {
			is_overseas_for_owner = yes
			owner ?= { is_colonial_subject = yes }
		}
	}
	
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	price = expand_rgo_farming

	always_add_demands = yes

	build_time = plantation_build_time
	
	unique_production_methods = {
		cocoa_plantation_maintenance = {
			produced = cocoa
			output = 1
		
			slaves_goods = 0.1333
			leather = 0.0666
			tools = 0.0666
			lumber = 0.0666
			debug_max_profit = -1  
			category = building_maintenance
		}
	}
	
	modifier = {
		local_cocoa_output_modifier = 0.1
		local_slave_pop_satisfaction = -0.005
		building_enslavement_power = 0.01
	}

	construction_demand = plantation_construction
}
