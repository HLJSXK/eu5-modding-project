observatory_building = {
	is_foreign = no
	pop_type = clergy
	max_levels = 1
	audio_tier = 4
	
	category = cultural_category
	employment_size = religious_building_employment
	
	forbidden_for_estates = yes

	city = yes
	megalopolis = yes

	location_potential = {
		development >= 40
	}

	AI_ignore_available_worker_flag = yes
	expensive = yes
	build_time = large_cultural_building_time
	forbidden_for_estates = yes

	allow = {	
		has_building = building_type:university
		OR = { 
			development > 50

		}
	}
	
	modifier = {
		local_burghers_max_literacy = 5
		local_nobles_max_literacy = 5
		local_monthly_literacy = 0.1
		local_migration_attraction = 0.05
		local_pop_promotion_speed = 0.0025
		local_monthly_development_modifier = 0.1
		local_academic_tradition_modifier = 0.25
	}
	
	unique_production_methods = {
		observatory_maintenance = {
			paper = 0.5
			books = 0.25
			precision_instruments 0.25
			optics = 0.25
			mirrors = 0.25

			category = building_maintenance
		}
	}
	
	construction_demand = university_construction
}

rectorate_chair_clergy = { 
	audio_tier = 3
	is_special = yes
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	
	employment_size = religious_building_employment

	forbidden_for_estates = yes

	town = yes
	city = yes
	megalopolis = yes

	category = religious_category
	
	allow = {
		always = no 
	}

	country_potential = {
		custom_tooltip = {
			text = seat_of_rector_why
			always = no 
		}
	}
	
	build_time = religious_building_time
	
	possible_production_methods = {
		academy_seat_maintenance
	}
	
	construction_demand = cardinal_construction

	modifier = {
		local_max_control = 0.05
		local_clergy_max_literacy = 2.5
		local_monthly_literacy = 0.1
		local_clergy_desired_pop = 0.05
		local_migration_attraction = 0.05
		local_pop_promotion_speed = 0.001
		local_institution_growth_modifier = 0.10
	}

	is_indestructible = yes
	can_close = no

	remove_if = {
    	location ?= { 
        	OR = {
            	NOT = { 
                	has_building = building_type:university
                	has_location_modifier = clocks_academy_site
            	}
            	owner ?= { 
                	OR = {
                    	modifier:burghers_academic_authority = yes
                    	modifier:nobles_academic_authority = yes
						modifier:crown_academic_authority = yes
                	}
            	}
        	}
    	}
	}
}
