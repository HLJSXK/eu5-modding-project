perfume_guild_max_level = {
	add = guild_max_level

	if = {
		limit = {
			modifier:local_perfume_guild_building_levels > 0
		}
		add = {
			desc = "perfume_rights"
			value = modifier:local_perfume_guild_building_levels
		}
	}
}

soap_current_age_oil_multiplier = { 
	if = { 
		limit = { has_embraced_institution = institution:industrialization }
		value = 10
	}
	else_if = { 
		limit = { has_embraced_institution = institution:manufactories }
		value = 5
	}
	else_if = { 
		limit = { has_embraced_institution = institution:global_trade }
		value = 2.5
	}
	else = { 
		value = 1
	}
}
