clocks_guild_max_level = {
	add = guild_max_level

	if = {
		limit = {
			modifier:local_clocks_guild_building_levels > 0
		}
		add = {
			desc = "clocks_rights"
			value = modifier:local_clocks_guild_building_levels
		}
	}
}

academy_count = {
    value = 0
    every_owned_location = {
        limit = {
            has_location_modifier = clocks_academy_site
        }
        add = 1
    }
}

total_research_bonus = { 
	value = 0
	every_in_list = {
        variable = national_academies_list
		if = { 
			limit = { has_variable = local_research_speed_bonus }
			add = var:local_research_speed_bonus
		}
	}
}
