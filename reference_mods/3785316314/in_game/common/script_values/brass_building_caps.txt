silverware_guild_max_level = {
	add = guild_max_level

	if = {
		limit = {
			modifier:local_silverware_guild_building_levels > 0
		}
		add = {
			desc = "silverware_rights"
			value = modifier:local_silverware_guild_building_levels
		}
	}
}

brass_current_age_multiplier = { 
	if = { 
		limit = { has_embraced_institution = institution:industrialization }
		value = 5
	}
	else_if = { 
		limit = { has_embraced_institution = institution:manufactories }
		value = 2.5
	}
	else = { 
		value = 1
	}
}
