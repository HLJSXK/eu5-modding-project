refresh_global_academy_list = {
    # Create the list
    clear_global_variable_list = global_academies_list
    every_country = {
        every_in_list = {
            variable = national_academies_list
            add_to_global_variable_list = { name = global_academies_list target = this }
        }
    }

    # Sort the List
    sort_global_variable_list = {
        name = global_academies_list
        order = {
            value = var:academic_tradition_value
        }
    }

    # Track the highest tradition value for map mode scaling
    set_global_variable = { name = highest_academic_tradition value = 0 }
    ordered_in_global_list = {
        variable = global_academies_list
        position = 0
        set_global_variable = { name = highest_academic_tradition value = var:academic_tradition_value }
    }
}

refresh_national_academy_list = { 
    clear_variable_list = national_academies_list
    every_owned_location = { 
        limit = { 
            has_location_modifier = clocks_academy_site
        }
        owner = { add_to_variable_list = { name = national_academies_list target = prev } }
    }
}

set_academic_tradition = { 
    # Get Global Modifier
    owner ?= { 
        set_local_variable = { 
			name = gatm
			value = modifier:global_academic_tradition_modifier
		}
    }
    
    # Get Language Strength
    set_variable = { 
        name = language_bonus_multiplier 
        value = { 
            value = owner.liturgical_language.language_power
            multiply = 0.5
        }
    }
    

    # Get Total Modifier
    set_local_variable = {
        name = total_academic_tradition_modifier
        value = {
            # Base 
            value = 1
            # Flat Local Modifier Bonusses
            add = { 
                value = modifier:local_academic_tradition_modifier
            }
            # Language
            add = { 
                value = var:language_bonus_multiplier
            }
            # Flat Global Modifier Bonusses
            add = { 
                value ?= local_var:gatm
            }
        }
    }
    
    ## Make sure location has academic momentum
    if = { limit = { NOT = { has_variable = academic_momentum } } set_variable = { name = academic_momentum value = 0 } }

    set_variable = { 
        name = academic_tradition_value
        value = {
            value = average_location_literacy
            add = {
                value = var:academic_momentum
            }
            multiply = { 
                value = local_var:total_academic_tradition_modifier
            }
        }
    }

    ## Set Research Speed Bonus
    set_variable = { 
        name = local_research_speed_bonus
        value = { 
            value = var:academic_tradition_value
            multiply = 0.0005
            min = 0.01
            max = 0.20
        }
    }
}

add_academic_momentum = {
    set_local_variable = { name = mom_t value = $amount$ }

    if = {
        limit = { local_var:mom_t >= 0 }
        custom_tooltip = {
            text = academic_momentum_gain_tt
            subject = THIS
            change_variable = { name = academic_momentum add = local_var:mom_t }
        }
    }
    else = {
        custom_tooltip = {
            text = academic_momentum_loss_tt
            subject = THIS
            change_variable = { name = academic_momentum add = local_var:mom_t }
        }
    }
    set_academic_tradition = yes
}

add_academic_influence = {

    if = { 
        limit = { NOT = { has_variable = academic_influence } }
        set_variable = { name = academic_influence value = 0 }
    }

    set_local_variable = { name = infl_t value = $amount$ }
    if = {
        limit = { local_var:infl_t >= 0 }
        custom_tooltip = {
            text = academic_influence_gain_tt
            if = { 
                limit = { NOT = { has_variable = academic_influence } }
                set_variable = { name = academic_influence value = 0 }
            }
            change_variable = { name = academic_influence add = local_var:infl_t }
            if = {
                limit = { var:academic_influence > modifier:maximum_academic_influence }
                set_variable = { name = academic_influence value = modifier:maximum_academic_influence }
            }
        }
    }
    else = {
        custom_tooltip = {
            text = academic_influence_loss_tt
            if = { 
                limit = { NOT = { has_variable = academic_influence } }
                set_variable = { name = academic_influence value = 0 }
            }
            change_variable = { name = academic_influence add = local_var:infl_t }
            if = {
                limit = { var:academic_influence > modifier:maximum_academic_influence }
                set_variable = { name = academic_influence value = modifier:maximum_academic_influence }
            }
        }
    }
    check_academic_influence_alert = yes
}

grant_academy_effect = {
    if = {
        limit = { NOT = { has_building = building_type:university } }
        construct_building = {
            building_type = building_type:university
            cost_multiplier = 0
            cost_multiplier_reason = "academies_start"
            instant = yes
        }
    }

    add_location_modifier = { modifier = clocks_academy_site years = -1 }

    if = {
        limit = { NOT = { has_building = building_type:rectorate_chair_clergy } }
        construct_building = {
            building_type = building_type:rectorate_chair_clergy
            cost_multiplier = 0
            cost_multiplier_reason = "academies_start"
            instant = yes
        }
    }

    owner = { refresh_national_academy_list = yes }
}

clocks_swap_academic_chair = {
    if = { 
        limit = { 
            owner = { modifier:burghers_academic_authority = yes }
        }
        construct_building = {
            building_type = building_type:rectorate_chair_burghers
            cost_multiplier = 0
            cost_multiplier_reason = "academies_swap"
            instant = yes
        }
    }
    else_if = { 
        limit = { 
            OR = { 
                owner = { modifier:nobles_academic_authority = yes }
                owner = { modifier:crown_academic_authority = yes }
            }
        }
        construct_building = {
            building_type = building_type:rectorate_chair_nobility
            cost_multiplier = 0
            cost_multiplier_reason = "academies_swap"
            instant = yes
        }
    }
    else = { 
        construct_building = {
            building_type = building_type:rectorate_chair_clergy
            cost_multiplier = 0
            cost_multiplier_reason = "academies_swap"
            instant = yes
        }
    }
}

# CMF Alert
check_academic_influence_alert = {
    if = { 
        limit = { has_variable = academic_influence }
        if = {
            limit = {
                var:academic_influence >= {
                    value = modifier:maximum_academic_influence
                    multiply = 0.9
                }
                NOT = { has_variable = academic_influence_alert_shown }
            }
            cmf_show_alert = { alert = academic_influence_alert }
            set_variable = { name = academic_influence_alert_shown value = 1 }
        }
        if = {
            limit = {
                var:academic_influence < {
                    value = modifier:maximum_academic_influence
                    multiply = 0.9
                }
                has_variable = academic_influence_alert_shown
            }
            cmf_remove_alert = { alert = academic_influence_alert }
            remove_variable = academic_influence_alert_shown
        }
    }
}
