# ============================================================
# Static modifiers for the Farr Proof of Fortune disaster.
#
# Three layers:
#   1. EVENT-DRIVEN BATTLE/WAR MODIFIERS — conquest glory,
#      defeat shame, capital lost. Applied by the on_action
#      hooks (04_farr_proof_hooks.txt). Time-limited.
#
#   2. APPARATUS, IN TWO HALVES — each apparatus event grants
#      both a REFUND and a LEGACY modifier.
#        - Refund (farr_X_refund): exactly cancels the penalty
#          the disaster's standing modifier applies for the
#          unclaimed recognition. Cleared by farr_proof_clear
#          when the disaster ends — the penalty goes, and the
#          refund goes with it.
#        - Legacy (farr_sikka_granted / farr_khutba_in_your_name
#          / farr_hunt_and_nawruz / farr_padishah): the
#          institutional trace of having held the recognition.
#          Persists past Vindicated, Endured, Broken-lapse.
#          Stripped only by Broken-formal (.201) — the doctrine
#          is gone with the dynasty.
#
#   3. RESOLUTION MODIFIERS — applied by the four resolution
#      events. Farr Manifest (Vindicated), Farr Endured
#      (Endured), the prestige scar (Broken-lapse). The
#      Broken-formal usurpation strips the reform and applies
#      no resolution modifier — the realm is now an amīr
#      sultanate.
# ============================================================

# ============================================================
# LAYER 1 — Battle / war event modifiers
# ============================================================

# Conquest glory — fires from on_winning_war when farr disaster
# active. Mode add_and_extend means each new conquest extends
# the duration to 12 months from the latest victory. Sustained
# conquest = sustained positive legitimacy growth.
farr_conquest_glory = {
	game_data = {
		category = country
		decaying = no
	}

	monthly_legitimacy = 0.4
	monthly_prestige = 0.05
}

# Defeat shame — fires from on_losing_war. Heaven visibly withdrew.
# Six months of crushing legitimacy bleed.
farr_defeat_shame = {
	game_data = {
		category = country
		decaying = no
	}

	monthly_legitimacy = -1.0
	monthly_prestige = -0.10
	amirs_estate_target_satisfaction = -0.05
}

# Capital lost — applied from on_location_changed_owner when the
# capital changes hands. Duration -1 = permanent until removed
# by the on_action when the capital returns to root.
farr_capital_lost = {
	game_data = {
		category = country
		decaying = no
	}

	monthly_legitimacy = -2.0
	monthly_prestige = -0.20
	stability_decay = 0.10
}

# Cleared by farr_proof_clear on disaster end.
# ============================================================
