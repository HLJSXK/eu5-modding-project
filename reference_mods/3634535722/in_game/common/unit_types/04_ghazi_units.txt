# ------------------------------------------------------------
# THE GHAZIS — clergy light cavalry, the volunteer pole
# ------------------------------------------------------------
a_ghazi_volunteers = {
	category = army_light_cavalry
	copy_from = a_age_1_traditions_light_cavalry

	buildable = no
	levy = yes

	# numerous and eager, lightly equipped:
	morale_damage_done = 0.20          # zeal in the charge
	morale_damage_taken = -0.20        # martyrdom doctrine: they do not break easily
	strength_damage_taken = 0.20       # ...but they die when hit; no armor
	strength_damage_done = -0.10       # little staying power in a grind
	initiative = 1                     # they come fast and willing
	supply_weight = -0.15              # they live light, off the land and the faith

	# the frontier and the open field — raiders, not line-breakers:
	combat = {
		sparse = 0.15
		flatland = 0.10
		desert = 0.10                  # the ghaza rode the dry marches
		forest = -0.15
		wetlands = -0.15
	}

	gfx_tags = { light_tag }
}
