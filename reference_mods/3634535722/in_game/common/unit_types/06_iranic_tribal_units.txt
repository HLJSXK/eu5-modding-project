# =====================================================================
# IRANIC TRIBAL UNITS — the Zagros levy
#
# The steppe equation (tribesman = horseman) does not hold in the
# mountains. Zagros transhumance runs on foot, through country mass
# cavalry cannot enter; the Iranic tribes' military meaning was hardy
# irregular infantry — the geographers' javelin-and-bow men of the
# passes, the ambush war that kept the Hazaraspids alive for two
# centuries. When this manpower finally enters royal armies at scale
# it enters as MUSKET infantry (the Safavid tofangchi drew on Lur,
# Bakhtiyari and Kurdish levies; the Zand army was Lak/Lur foot with
# firearms) — never through a cavalry phase.
#
# The mounted exception (Shabankara clan elites, the Ayyubid Kurdish
# askar) was a warrior ELITE in service, not a mass — modeled as the
# small levy_iranic_tribal_retinue, not a unit of its own: they ride
# as amir cavalry.
# =====================================================================

# Age 1-2: the mountain skirmisher. Javelin, bow, and the high ground.
a_zagros_skirmishers = {
	category = army_light_infantry
	copy_from = a_age_1_traditions_light_infantry

	buildable = no
	levy = yes

	# At home in country that kills cavalry: punishes pursuit, melts
	# uphill. Fragile in the open field.
	morale_damage_done = 0.10
	strength_damage_taken = 0.10
	morale_damage_taken = -0.15
	initiative = 2
	combat = {
		# raised in the mountains.

		hills = 0.3
		mountains = 0.3
		# (seasonal_irrigated is not a combat terrain key; the valley
		# half of the transhumant year is the plateau basins)
		plateau = 0.15

	}
}
