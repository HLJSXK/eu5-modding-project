# Persianate-world AI evaluation lists — one list per action family,
# each gate a CHEAP SUPERSET of the family's own potential/allow
# conditions. A list's potential EXCLUDES: countries failing it never
# evaluate the actions at all, so a too-narrow gate silently disables
# AI use. A too-broad gate merely wastes a check — hence superset.
#
# (tariqa network actions live in tariqa_actions_list; the two
# shattered-rum situation actions in shattered_rum_list.)

# --- Custody of the Ulus: every action checks custody_fired ---------
custody_actions_list = {
	potential = {
		has_variable = custody_fired
	}
	actions = {
		custody_press_claim
		custody_demand_fealty
		custody_proclaim_true
		custody_secure_household
		custody_sikka_everywhere
		custody_summon_kuriltai
		custody_drop_khan
	}
}
