# -----------------------------------------
# GHULAM SYSTEM — Matter of Tork, Approach 1
#
# The corps needs an ADMINISTRATOR (crown via ghulam_system,
# OR the diwan, OR the amirs — one of the two privileges below) and
# an IMPLEMENTER (the fostering households that raise the boys —
# peasants privilege; a tribes variant is a planned follow-on).
#
# Administration is a real choice, not a tax:
#  - Diwan: the CORRECT ghulam system — salaried, registered,
#    answerable to no estate. The amirs' monopoly on violence broken
#    by accountants; they are EXTRA angry.
#  - Amirs: politically cheap — but the Saljuq failure mode: slave
#    soldiers administered by amirs become amir households within a
#    generation. The corps leaks into the estate it was built to
#    replace (amir power gain on this privilege; the First Muster
#    event also rolls worse under amir administration).
#
# Availability is sticky: matter_tork_ghulam_program is set the
# first time Approach 1 is chosen and never removed — a granted
# privilege survives an approach switch (earned-gains rule), and
# remains grantable (the program exists as an institution once
# founded).
#
# TIMELINE (design decision 36): these privileges are the PALACE
# form of the institution — the Samanid/Ghaznavid precedent, live
# from the Matter (~1390s) with no reform required. The Age 3
# ghulam_system is the CONSTITUTIONAL form (Abbas's
# systematization); its has_reform branch below is the late root
# for realms that never had the Matter. Nothing here needs the
# reform before 1444.
#
# Design: docs/design/matter_of_tork_design.md §4A, decisions 20-22.
# -----------------------------------------

diwan_ghulam_administration = {
	estate = diwan_estate
	content_priority = 610

	potential = {
		government_type = government_type:persianate_sultanate
		OR = {
			has_variable = matter_tork_ghulam_program
			has_reform = government_reform:ghulam_system
		}
		NOT = { has_estate_privilege = estate_privilege:amirs_ghulam_administration }
	}

	country_modifier = {
		can_build_ghulam_barracks = yes
		# The diwan al-'ard: muster rolls, salary registers, the
		# army as a column in the chancery's books.
		diwan_estate_target_satisfaction = 0.05
		global_diwan_estate_power = 0.20
		country_cabinet_efficiency = -0.05
		# The amirs' monopoly on violence, broken by accountants.
		amirs_estate_target_satisfaction = -0.10
		monthly_amirs_estate_rebel_growth = 0.005
		monthly_towards_centralization = 0.05
		monthly_towards_tazik = 0.05
	}
}

amirs_ghulam_administration = {
	estate = amirs_estate
	content_priority = 610

	potential = {
		government_type = government_type:persianate_sultanate
		OR = {
			has_variable = matter_tork_ghulam_program
			has_reform = government_reform:ghulam_system
		}
		NOT = { has_estate_privilege = estate_privilege:diwan_ghulam_administration }
		# Foreclosed by Expulsion/Eradication: the refounding purge
		# transfers amir-held commands to the chancery, permanently.
		NOT = { has_variable = matter_turk_dissolved }
	}

	country_modifier = {
		can_build_ghulam_barracks = yes
		# Politically painless — the amirs supervise the new corps
		# as they supervised everything military.
		amirs_estate_target_satisfaction = 0.05
		# The Saljuq failure mode, priced: the corps leaks into amir
		# households. You are building your replacement army inside
		# the institution you are trying to replace.
		global_amirs_estate_power = 0.15
		monthly_towards_decentralization = 0.05
	}
}

ghulam_fostering_households = {
	estate = peasants_estate
	content_priority = 610

	potential = {
		government_type = government_type:persianate_sultanate
		OR = {
			has_variable = matter_tork_ghulam_program
			has_reform = government_reform:ghulam_system
		}
	}

	country_modifier = {
		# The implementer: village households paid to raise and
		# school the purchased boys before palace service — the
		# fostering layer every ghulam system ran on.
		peasants_estate_target_satisfaction = 0.05
		global_peasants_estate_power = 0.10
		peasants_estate_max_tax = -0.025
	}
}
