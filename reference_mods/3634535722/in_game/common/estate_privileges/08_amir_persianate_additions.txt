# === CAPSTONE — administrative AND fiscal immunity ===
# Pair: soyurghal ⟷ diwan_musadara_confiscation
soyurghal_grants = {
	estate = amirs_estate
	content_priority = 600

	potential = {
		government_type = government_type:persianate_sultanate
	}

	allow = {
		NOT = { has_estate_privilege = estate_privilege:diwan_musadara_confiscation }
	}

	# The amir capstone — hereditary, fully tax-immune, and
	# administratively immune (state agents barred from entering).
	# Late-Timurid, Aq Qoyunlu, early Safavid. Unlike the tuyul, the
	# grant is not service-conditioned and is in principle perpetual.
	# Because the immunity is administrative as well as fiscal, the
	# state's control cannot grow on the land at all — the chancery
	# loses the right to send its registers, not just its tax-farmers.
	country_modifier = {
		amirs_estate_target_satisfaction = ez_priv_sat_huge
		global_amirs_estate_power = ez_priv_power_huge
		global_max_rural_control = ez_soyurghal_rural_malus
		global_max_urban_control = ez_amir_urban_malus_big
		province_integration_speed = -0.20
		tax_income_efficiency = -0.10
		amirs_estate_max_tax = -0.20
		diwan_estate_target_satisfaction = -0.10
		global_diwan_estate_power = -0.20
		monthly_towards_turk = 0.10
		monthly_towards_decentralization = 0.10
	}

}

yarligh_tarkhan = {
	estate = amirs_estate

	potential = {
		government_type = government_type:persianate_sultanate
	}

	# Tarkhan yarlighs — Mongol chancery patents granting personal and
	# fiscal immunity. The Ilkhans handed them out lavishly; the
	# Timurids and Aq Qoyunlu kept the practice. The state writes
	# itself out of the revenue stream household by household.
	country_modifier = {

		amirs_estate_max_tax = -0.1

		# chiefs alike — a light urban touch alongside the heavier

		global_max_urban_control = ez_amir_urban_malus
		diwan_estate_target_satisfaction = -0.05
		monthly_towards_turk = 0.05
	}
}

# === CAPSTONE — the tarkhan principle generalized ===
# The yarligh as a household-by-household exemption is the ordinary
# practice; the universal tarkhan is the maximalist version — the
# whole amir caste raised to tarkhan rank as a class. Aq Qoyunlu and
# early Safavid drift in this direction by default; the qizilbāsh
# khans simply assumed their persons were beyond the assessor's reach.
# The state writes its own cavalry aristocracy out of the tax base
# entirely, and the chancery is left to extract revenue from peasants,
# burghers, and itself.
universal_tarkhan = {
	estate = amirs_estate
	content_priority = 700

	potential = {
		government_type = government_type:persianate_sultanate
	}

	country_modifier = {

		diwan_estate_target_satisfaction = -0.15
		global_diwan_estate_power = -0.25
		monthly_towards_turk = 0.15
		monthly_towards_decentralization = 0.20
	}

}

# === The Umarāʾ at the Dīwān ===
# Mirror of diwan_tazik_command_of_arms. The Türco-Mongol military
# aristocracy is admitted to chancery seats. Niẓām al-Mulk warns
# against exactly this in the Siyāsatnāma — the pen and the sword
# in the same hand ruins the registers — and the diwan responds
# accordingly.
amirs_open_muster_rolls = {
	estate = amirs_estate
	content_priority = 600

	potential = {
		government_type = government_type:persianate_sultanate
	}
	allow = { }
	can_revoke = { }

	# The corps opens to any creed: non-Muslim military men enter the
	# amirs estate as service cavalry rather than sitting outside it as
	# hereditary lords (is_amir_pop reads this privilege). Historical
	# arc: the Deccan sultanates enrolling Maratha silahdars, Shahji
	# Bhonsle's jagirs, the Rajput mansabdars — manpower necessity made
	# constitutional practice, over the ulama's standing objection.
	country_modifier = {
		amirs_estate_target_satisfaction = 0.03
		# The recruitment pool widens dramatically.
		global_amirs_estate_power = 0.20
		global_manpower_modifier = 0.10
		# The ulama do not forget which order the realm is supposed to
		# uphold.
		clergy_estate_target_satisfaction = -0.05
	}
}

amirs_umara_at_the_diwan = {
	estate = amirs_estate
	content_priority = 600

	potential = {
		government_type = government_type:persianate_sultanate
		# Matter of Tork: foreclosed once the Türk military class is
		# dissolved — the umara this licenses no longer exist.
		NOT = { has_variable = matter_turk_dissolved }
	}

	# Türk amīrs holding administrative posts is the bête noire of
	# Persianate political theory. Niẓām al-Mulk's Siyāsatnāma has a
	# whole register of complaints about military men sitting at the
	# dīwān and ruining the records. The Aq Qoyunlu and Qara Qoyunlu
	# practiced it openly; the late-Timurid Herat dīwān tilted the
	# same way under pressure from the umarāʾ. To formally license it
	# is to repudiate the wuzarāʾ tradition, and the chancery feels it.
	country_modifier = {
		amirs_estate_target_satisfaction = ez_priv_sat_huge
		global_amirs_estate_power = ez_priv_power_large
		amirs_estate_allowed_in_cabinet = yes
		country_cabinet_efficiency = -0.10
		monthly_towards_decentralization = 0.05
		monthly_towards_turk = 0.10
		diwan_estate_target_satisfaction = -0.10
		global_diwan_estate_power = -0.15
	}

}

# === RIVALRY — judicial autonomy ===

	estate = amirs_estate

	potential = {
		government_type = government_type:persianate_sultanate
	}

	allow = {
		NOT = { has_estate_privilege = estate_privilege:diwan_qanun_codification }
	}

	# Tribal country is carved out of the maẓālim/qāḍī system. The
	# amir judges by ʿurf and ʿādat (Türk: töre) in his own jurisdiction.
	# Urban courts are untouched — the bazaar still sees a qāḍī — but
	# the chancery's legal uniformity ends at the city wall.
	country_modifier = {
		amirs_estate_target_satisfaction = ez_priv_sat_large
		global_amirs_estate_power = ez_priv_power_large
		global_max_rural_control = ez_amir_rural_malus
		# The chancery's writ ends at the city wall — in tribal country
		# the amir's own ʿurf adjudication shields his dependents, and
		# himself, from chancery assessment.
		amirs_estate_max_tax = -0.05
		global_diwan_estate_power = -0.15
		diwan_estate_target_satisfaction = -0.10
		stability_cost_efficiency = -0.10
		province_integration_speed = -0.10
		monthly_towards_decentralization = 0.05
		monthly_towards_turk = 0.05
	}
}
