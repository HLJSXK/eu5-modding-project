# BURGHERS — Persianate additions and overrides
#
# The Persianate urban estate isn't the Latin burgher of a chartered
# commune. It is the bazaar: asnaf guilds shading into futuwwa lodges,
# ayyar street-bands, Karimi/Tabrizi caravan houses, and a state-
# appointed Malik al-Tujjar sitting on top.
#
# This file:
#   - Adds four Persianate-specific privileges.
#   - Overrides four base privileges that the diwan/chancery should
#     own (mint, treasury, coinage) by re-declaring them with a
#     potential gate that blocks government_type:persianate_sultanate.
#     Re-declaration overrides the base file via mod load order.
# -----------------------------------------

# === OVERRIDES — block on persianate_sultanate ===
# Mint, treasury, coinage are diwan/chancery prerogatives in the
# Persianate state — Rashid al-Din's coin reform is the case in point.
# Re-declared with original modifier sets intact; only potential is added.

REPLACE:treasury_rights = {
	estate = burghers_estate

	potential = {
		NOT = { government_type = government_type:persianate_sultanate }
	}

	country_modifier = {
		bank_interest = -0.01
		tax_income_efficiency = small_tax_income_efficiency_bonus
		burghers_estate_target_satisfaction = medium_privilege_target_satisfaction
		global_burghers_estate_power = 0.5
		monthly_towards_innovative = societal_value_monthly_move
	}
}
