# -----------------------------------------------------------------------
# THE WAQF CHARTERS — the deceptively good privileges
#
# Each unlocks the GRAND waqf for its estate (the building's
# country_potential checks has_estate_privilege). The bait is built to
# be seized gladly: a large loyalty equilibrium, only a SMALL and non-
# commensurate power concession (looks like a steal), and — because
# endowment is meritorious piety — a clergy-satisfaction and legitimacy
# boon (the just, generous ruler). No fiscal sweetener: the entire
# tax/credit bleed lives in the Grand Waqf the estate then builds out,
# and the privilege is sticky (revoke_privilege_cost_modifier) once the
# endowments are entrenched. The dead hand only tightens.
#
# See in_game/common/building_types/09_waqf_buildings.txt.
# -----------------------------------------------------------------------

# Aʿyān Waqf — "Inviolable Endowments". The chancery class's muṣādara
# shelter, dressed as the protection of pious charity.
ayyan_waqf_charter = {
	estate = diwan_estate
	content_priority = 600

	potential = {
		government_type = government_type:persianate_sultanate
		country_has_estate = estate_type:diwan_estate
	}

	country_modifier = {
		# The bait: a loyal, contented diwan, and a pious, lawful ruler.
		diwan_estate_target_satisfaction = 0.10
		clergy_estate_target_satisfaction = 0.025
		monthly_legitimacy = 0.025
		# A small power concession — far less than the loyalty would seem
		# to cost. The real price is deferred into the Grand Waqf.
		global_diwan_estate_power = 0.03
		# Entrenched endowments do not unwind politely.
		revoke_privilege_cost_modifier = 0.5
	}
}
