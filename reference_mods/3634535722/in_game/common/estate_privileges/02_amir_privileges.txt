amirs_iqta_grant = {
	estate = amirs_estate

	# The basic iqtāʿ — service-conditioned, theoretically revocable
	# revenue grant on land. Buyid origin, Seljuq systematization. The
	# mechanical bite is carried by the grant content itself; the
	# privilege is the legal frame under which it sits.
	country_modifier = {
		amirs_estate_target_satisfaction = ez_priv_sat_major
		global_amirs_estate_power = ez_priv_power_major
		### this is actually a CORE privilege of the whole system.
		amirs_estate_levy_size = 0.50
		amirs_estate_max_tax = -0.1
		global_nobles_rural_desired_pop_scaled = 0.015 # adding some rural population due to small scale iqtas being embedded into this privilege.


	}
}

amirs_hereditary_tuyul = {
	estate = amirs_estate

	# Rival: diwan_mavajib_cash_pay. Cash army pay and land-grant army
	# pay are mutually exclusive systems of cavalry funding.
	allow = {
		NOT = { has_estate_privilege = estate_privilege:diwan_mavajib_cash_pay }
	}

	country_modifier = {
		amirs_estate_target_satisfaction = 0.05
		global_amirs_estate_power = 0.50
		tax_income_efficiency = -0.025
		tribes_estate_levy_size = 0.1
		amirs_estate_levy_size = 0.15
		army_heavy_cavalry_power = 0.15
		army_light_cavalry_power = 0.15
		# Tuyul takes the iqtaʿ off the service contract and locks it
		# into the holder's lineage. The chancery cannot revoke; the
		# next assessment cycle never comes. A much more total carve-out
		# than the basic iqtaʿ.
		amirs_estate_max_tax = -0.15
		# Tuyul is the rural revenue grant — its bite is overwhelmingly
		# in the village. Urban touch acknowledges the amir's garrison
		# in the regional capital of his grant.
		global_max_rural_control = ez_amir_rural_malus
		global_max_urban_control = ez_amir_urban_malus_big
		monthly_towards_turk = 0.2
	}
	
}

amirs_tribal_autonomy_rights = {
	estate = amirs_estate

	# Matter of Tork: see amirs_tribal_hosts — the old charter is
	# foreclosed once the Türk constituency is dissolved.
	potential = {
		NOT = { has_variable = matter_turk_dissolved }
	}

	country_modifier = {
		amirs_estate_target_satisfaction = 0.05
		global_amirs_estate_power = 0.15

		# The Lock — tribesmen stay tribesmen (no promotion to settled
		# pop types), and stay culturally Türk (no assimilation into
		# Persian/Arab culture pools). Religious conversion is NOT
		# blocked: Türkmen converted to Sunni Hanafi en masse under
		# the Seljuqs, and later to Shi'i Islam under the Safavids.
		
		global_tribesmen_assimilation_blocked = yes
		global_tribesmen_conversion_blocked = yes
		# Tribal levy weight — the umara-tribal compact pulls a
		# disproportionate share of the realm's manpower from these
		# two estates relative to the country's actual size.
		block_tribal_promotion = yes
		tribes_estate_levy_size = 0.1

		# Drawbacks — purely steppe phenomenon. Tribesmen are not in
		# cities, and the chancery's hold on the cities is unchanged.
		global_max_rural_control = ez_amir_rural_malus
		tolerance_heretic = 1
		monthly_towards_turk = 0.05
	}
}
