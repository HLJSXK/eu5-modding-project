# -----------------------------------------
# CLERGY (ulama) — Persianate additions and overrides
#
# The Persianate Sunni clergy is the ulama: qadis, mudarrisin,
# muhtasibs, anchored in waqf endowments and the great madrasas.
# Sayyid lineages and Sufi tariqa networks sit alongside, sometimes
# overlapping. This file fills the gap the base game leaves between
# Catholic dioceses and Confucian scholars.
#
# Adds five Persianate-specific privileges. Overrides clergy_lecture_oaths
# (which requires base-game "university" buildings — not the madrasa).
# -----------------------------------------

# === NEW PERSIANATE CLERGY PRIVILEGES ===

naqib_al_ashraf = {
	estate = clergy_estate

	potential = {
		religion ?= { group ?= religion_group:muslim }
		OR = {
			government_type = government_type:persianate_sultanate
			capital = { 
				OR = {
					region = region:persia_region
					region = region:khorasan_region
					area = area:iraq_arabi_area # Baghdad/Iraq
					area = area:jazira_area # Mosul and surroundings
					area = area:cappadocia_area

				}
			}
		}
	}

	country_modifier = {
		clergy_estate_target_satisfaction = medium_privilege_target_satisfaction
		global_clergy_estate_power = 0.5
		clergy_estate_max_tax = -0.10
		tolerance_own = 1
		monthly_legitimacy = 0.05
	}
}

# === OVERRIDE — block on persianate_sultanate ===
# clergy_lecture_oaths requires base-game "university" buildings.
# In a Persianate state the analogous institution is the madrasa,
# which sits structurally inside madrasa_endowment above.

REPLACE:clergy_lecture_oaths = {
	estate = clergy_estate

	potential = {
		NOT = { government_type = government_type:persianate_sultanate }
		total_effective_building_levels:university > 0
	}

	allow = {
		total_effective_building_levels:university > 0
	}

	country_modifier = {
		global_pop_promotion_speed_modifier = -0.1
		country_cabinet_efficiency = 0.1
		stability_cost_efficiency = 0.1
		clergy_estate_target_satisfaction = medium_privilege_target_satisfaction
		university_construction_blocked = yes
		global_clergy_estate_power = 0.1
	}
}
