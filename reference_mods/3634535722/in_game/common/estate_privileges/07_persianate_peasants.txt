# -----------------------------------------
# PEASANTS — Persianate additions and override
#
# Persianate rural life ran on smallholder cultivation under iqta'
# tenure, organized in boneh collectives, watered by qanats. There was
# no European-style serfdom; instead the friction was between amir
# tax-farmers and village heads. Boneh System (01_persian_peasant_privileges.txt)
# already covers the village-collective layer; this file adds the
# water-infrastructure and tenant-protection layers, and overrides
# the base communal_lands which would otherwise double-stack.
# -----------------------------------------

# === NEW PERSIANATE PEASANT PRIVILEGES ===

qanat_maintenance_compact = {
	estate = peasants_estate

	potential = {
		OR = {
			government_type = government_type:persianate_sultanate
			capital = { region = region:persia_region }
			capital = { region = region:khorasan_region }
		}
	}

	country_modifier = {
		peasants_estate_target_satisfaction = 0.1 # this is a very major privilege, as qanat maintenance is absolutely essential to rural life in the relevant regions. It's also a major source of peasant power, as the peasants are the ones who actually do the work of maintaining the qanats and can withhold that labor if they are dissatisfied enough.
		global_peasants_estate_power = 0.50
		peasants_estate_max_tax = -0.075
		global_monthly_food_modifier = 0.05
		global_max_rgo_size_modifier_in_rural = 0.05
		monthly_towards_traditional_economy = 0.05
	}
}

iqta_cultivator_protections = {
	estate = peasants_estate

	potential = {
		OR = {
			government_type = government_type:persianate_sultanate
			capital = { 
				OR = {
					region = region:persia_region
					region = region:khorasan_region
					area = area:iraq_arabi_area # Baghdad/Iraq
					area = area:jazira_area # Mosul and surroundings
					area = area:cappadocia_area

				}
			}
		}
	}

	country_modifier = {
		peasants_estate_target_satisfaction = large_privilege_target_satisfaction
		global_peasants_estate_power = 0.75
		global_diwan_estate_power = 0.15
		global_amirs_estate_power = -0.15
		amirs_estate_target_satisfaction = -0.05
		monthly_towards_tazik = 0.05
		monthly_towards_centralization = 0.05
	}
}

# Central Asian institution with no European analogue.

#   monthly_towards_decentralization
#
# Needs a real counterweight before going in. Options to explore:

#     in exchange for satisfaction loss),
#   - or unlock by an institution/centralization tier so the player
#     has actually paid for the autonomy hit elsewhere.
# Leaving as TODO; not declared.

# === OVERRIDE — block on persianate_sultanate ===
# communal_lands overlaps boneh_system 1:1 conceptually. Re-declared
# with the original modifier set intact; only potential is added,
# so non-Persianate states still see it normally.

REPLACE:communal_lands = {
	estate = peasants_estate

	potential = {
		NOT = { government_type = government_type:persianate_sultanate }
	}

	country_modifier = {
		monthly_towards_decentralization = societal_value_large_monthly_move
		global_amirs_estate_power = -0.05
		global_peasant_enfranchisment = 0.05
		peasants_estate_target_satisfaction = medium_privilege_target_satisfaction
		global_peasants_estate_power = 0.33
	}
}
