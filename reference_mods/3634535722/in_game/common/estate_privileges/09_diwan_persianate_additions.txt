# === RIVALRY — funding the cavalry ===
# Pair: diwan_mavajib_cash_pay ⟷ amirs_hereditary_tuyul
diwan_mavajib_cash_pay = {
	estate = diwan_estate
	content_priority = 550

	potential = {
		government_type = government_type:persianate_sultanate
	}

	allow = {
		NOT = { has_estate_privilege = estate_privilege:amirs_hereditary_tuyul }
	}

	# Mavājib — cash salary for the cavalry, paid out of the dīwān-i
	# istīfāʾ via hawāla drafts on provincial revenue. The Sasanian-
	# Abbasid model that later under-girded the ghulām corps and the
	# Ottoman kapikulu. Cash pay keeps the countryside in the state's
	# hands instead of ceding it to land-grant army-holders — the
	# single strongest rural-control instrument the chancery owns.
	# Treasury feels it: army maintenance climbs, and the amirs lose
	# the economic base they had under tuyul.
	country_modifier = {
		diwan_estate_target_satisfaction = ez_priv_sat_huge
		global_diwan_estate_power = ez_priv_power_huge
		global_max_rural_control = ez_mavajib_rural_bonus
		global_manpower_modifier = 0.10
		land_morale = 0.05
		army_maintenance_efficiency = -0.15
		tax_income_efficiency = -0.05
		# Cash pay strips the amirs of their tuyul-and-iqtaʿ base of
		# fiscal opacity. They draw a salary like everyone else now,
		# and a salary is recorded line-by-line in the dīwān-i istīfāʾ.
		amirs_estate_max_tax = 0.10
		global_amirs_estate_power = -0.20
		amirs_estate_target_satisfaction = -0.10
		monthly_towards_centralization = 0.10
		monthly_towards_tazik = 0.10
	}

}

# === The modernist anti-musādara ===
# Mirror of diwan_musadara_confiscation. The chancery formally
# renounces its prerogative of muṣādara — confiscation of dead,
# disgraced, or fallen officials' estates — and codifies the
# inviolability of mulk as positive law. The institution Niẓām al-
# Mulk listed as the great suppressor of long-run prosperity is
# finally removed; the institution every late-Safavid and Qajar
# reformist wanted to be rid of; the institution the Constitutional
# Revolution of 1906 finally codified out of existence in Iran.
#
# Anti-thetical to the political DNA of a Persianate sultanate in
# 1337 — the privilege is age-gated to the Reformation onwards, at
# the earliest. A 14th-century court cannot conceive of it. Comes
# with notable prosperity and development bonuses (the chancery's
# complaint of muṣādara was always that nobody improved seizable
# land) and a steady drain on the throne's legitimacy (the shāh
# who cannot seize is hardly a shāh).
mulk_protection_charter = {
	estate = diwan_estate
	content_priority = 400

	potential = {
		government_type = government_type:persianate_sultanate
		OR = {
			current_age = age_4_reformation
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
	}

	allow = {
		NOT = { has_estate_privilege = estate_privilege:diwan_musadara_confiscation }
		NOT = { modifier:access_to_musadara = yes }
	}

	country_modifier = {
		global_monthly_prosperity = 0.01 # yes. this is ... massive.
		global_monthly_development = 0.0075 # yes. this is also ... massive.
		# Property security flips the cap downward across the board —
		# the chancery can no longer reach into any estate's estate
		# on the death of the holder. Diwan benefits most because the
		# chancery class is the densest in heritable urban property.
		burghers_estate_max_tax = -0.10
		amirs_estate_max_tax = -0.20
		diwan_estate_max_tax = -0.30
		# The throne pays the political price — a shāh whose ministers
		# die with their fortunes intact is a shāh whose prerogative
		# has been formally curtailed. The drain is steady, not violent.
		monthly_legitimacy = -0.30
		diwan_estate_target_satisfaction = -0.025 ## diwan is mixed on it. On the one hand, their wealth is safe. on the other hand, musadara HELPED them build wealth by squeezing others.
		burghers_estate_target_satisfaction = 0.1 ## this is ... what burghers have dreamed of. for UNTOLD generations.
		amirs_estate_target_satisfaction = 0.1

		global_diwan_estate_power = -0.10
	}

}

# === The Tazik Command of Arms ===
# A direct affront to the steppe inheritance — diwan officers
# admitted to field command. The amirs feel it as Ṭoghā Temür
# felt the Kartid Muʿizz al-Dīn: "what business has a tāzīk
# leading horsemen?" No mutual exclusion — this is a unilateral
# break with the baseline, not a fork.
diwan_tazik_command_of_arms = {
	estate = diwan_estate
	content_priority = 600

	potential = {
		government_type = government_type:persianate_sultanate
	}

	# Persianate viziers had led armies before — Niẓām al-Mulk on
	# campaign, Rashīd al-Dīn on the Ilkhanid frontier — but the
	# Türco-Mongol military aristocracy never accepted it as their
	# due. Ṭoghā Temür, the Chinggisid claimant in Gorgan, is the
	# stock voice of the prejudice: a tāzīk has no business at the
	# head of a host. To formally license it is to spit on that
	# inheritance, and the amirs respond accordingly.
	country_modifier = {
		diwan_estate_target_satisfaction = ez_priv_sat_huge
		global_diwan_estate_power = ez_priv_power_large
		diwan_estate_allowed_leading_military = yes
		country_cabinet_efficiency = 0.10
		monthly_towards_centralization = 0.05
		monthly_towards_tazik = 0.10
		amirs_estate_target_satisfaction = -0.10
		global_amirs_estate_power = -0.15
	}
}
