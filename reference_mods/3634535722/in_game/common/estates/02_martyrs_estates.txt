# -----------------------------------------------------------------------------
# Estates of the Commonwealth of Martyrs
#
# Two novel estates that replace the standard Diwan/Amirs duality for
# Sarbedārī-lineage polities. Existing burghers_estate (urban bāzāris) and
# peasants_estate carry forward unchanged from base game and aren't redefined
# here.
#
# Structural antagonism: Sufyānī notables hate Shaykhī power, Shaykhī
# dervishes hate Sufyānī power. Their oscillation IS the polity's politics.
# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------
# SUFYĀNĪ NOTABLES — urban-aristocratic conservative faction
# Named after Sufyānī faction of Sarbedārs (more conservative, malik-led).
# Provides bureaucratic continuity, regional administration, fiscal extraction.
# In Mushaʿshaʿ-lineage polities this slot is filled by the sayyid lay elite.
# -----------------------------------------------------------------------------

sufyani_notables_estate = {
	color = rgb { 110 90 60 }

	power_per_pop = 25
	tax_per_pop = 100

	rival = 0
	alliance = 0.01

	revolt_court_language = court_language
	characters_have_dynasty = always
	can_generate_mercenary_leaders = yes
	bank = no

	satisfaction = {
		country_cabinet_efficiency = 0.12
		tax_income_efficiency = 0.08
		global_max_control = 0.10
		stability_cost_efficiency = 0.05
	}

	high_power = {
		country_cabinet_efficiency = 0.25
		tax_income_efficiency = 0.15
		global_max_control = 0.15
		change_policy_cost_modifier = -0.10

		# The conservative wing dampens millenarian fervor — when they
		# dominate, the polity drifts toward conventional sultanate norms.
		monthly_devotion = -0.05
		monthly_towards_centralization = 0.10
	}

	low_power = {
		# Sufyānī collapse → full Shaykhī mahdist rule. Admin breaks down,
		# but devotional intensity spikes.
		country_cabinet_efficiency = -0.30
		tax_income_efficiency = -0.20
		stability_cost_efficiency = -0.15
		monthly_devotion = 0.08
	}

	power = {
		# Power-scaled effect: notables contribute civic-administrative weight.
		parliament_base_support = 0.10
		trade_income = 0.05
	}

	opinion = {
		add = {
			desc = "ESTATE_OPINION_BASE"
			value = "opinion(scope:target)"
			multiply = 0.30
		}

		# Hates Shaykhī power — the historical Sufyānī/Shaykhī factional war.
		add = {
			desc = "SUFYANI_HATES_SHAYKHI_POWER"
			value = "scope:target.estate_power(estate_type:shaykhi_dervishes_estate)"
			subtract = 0.20
			multiply = 12
		}

		if = {
			limit = { stability > 0 }
			add = { desc = "ESTATE_OPINION_STABILITY" value = 10 }
		}

		if = {
			limit = { average_control_in_home_region > 50 }
			add = { desc = "ESTATE_OPINION_HIGH_CONTROL" value = 5 }
		}

		# Loves Tazik alignment (sedentary-urban-bureaucratic).
		add = {
			desc = "turk_vs_tazik"
			value = {
				value = societal_value:turk_vs_tazik
				divide = 5
			}
		}
	}
}


# -----------------------------------------------------------------------------
# SHAYKHĪ DERVISHES — armed religious order, charismatic-militant
# Named after the Shaykhī faction (descended from Shaykh Khalīfa → Shaykh
# Ḥasan Jūrī's dervish network). Historical Sarbedār militancy at its peak
# fielded ~5,000 armed dervishes. This is the polity's primary war-machine
# and its mahdist legitimacy reservoir simultaneously.
# In Mushaʿshaʿ-lineage polities this slot is filled by the founder's
# charismatic following (Muḥammad ibn Falāḥ's mahdī-army).
# -----------------------------------------------------------------------------

shaykhi_dervishes_estate = {
	color = rgb { 35 35 100 }

	power_per_pop = 30
	tax_per_pop = 3           # Dervishes only pay Khoms

	rival = 0
	alliance = 0.015

	revolt_court_language = common_language
	characters_have_dynasty = never    # Individual charisma, not lineage
	can_generate_mercenary_leaders = yes
	bank = no

	satisfaction = {
		land_morale_modifier = 0.20
		levy_combat_efficiency_modifier = 0.20
		global_manpower_modifier = 0.15
		monthly_devotion = 0.04
		a_dervish_corps_build_cost_modifier = -0.5
	}

	high_power = {
		land_morale_modifier = 0.30
		global_manpower_modifier = 0.25
		amount_looted_modifier = 0.40
		a_dervish_corps_maintenance_cost_modifier = -0.5 ### they pay way less.
		a_dervish_corps_reinforce_cost_modifier = -0.5 ### they pay way less.
		# Militant religious fervor
		monthly_devotion = 0.10
		tolerance_own = 2

		# But anti-bureaucratic, anti-fiscal
		country_cabinet_efficiency = -0.20
		tax_income_efficiency = -0.20

		# Mahdist enthusiasm doesn't sit well with stability
		stability_cost_efficiency = -0.1

		# Pushes hard toward decentralization (each dervish lodge is its own
		# authority node)
		monthly_towards_decentralization = 0.15
	}

	low_power = {
		# Shaykhī collapse = military thin, legitimacy crisis
		land_morale_modifier = -0.20
		global_manpower_modifier = -0.15
		monthly_devotion = -0.08
	}

	power = {
		# Power-scaled: dervishes bolster wartime mobilization
		global_manpower_modifier = 0.05
	}

	opinion = {
		add = {
			desc = "ESTATE_OPINION_BASE"
			value = "opinion(scope:target)"
			multiply = 0.30
		}

		# Hates Sufyānī power — the conservative wing throttles the
		# mahdist mission.
		add = {
			desc = "SHAYKHI_HATES_SUFYANI_POWER"
			value = "scope:target.estate_power(estate_type:sufyani_notables_estate)"
			subtract = 0.20
			multiply = 15
		}

		if = {
			limit = { at_war = yes }
			add = { desc = "ESTATE_OPINION_AT_WAR" value = 25 }
		}

		if = {
			limit = { religion = religion:shia }
			add = { desc = "SHAYKHI_OPINION_TWELVER_RULER" value = 15 }
		}

		# Loves Tazik alignment in the urban-populist sense (anti-tribal,
		# anti-amir) — but less strongly than Sufyānī because Shaykhī
		# politics is its own thing, not bureaucratic.
		add = {
			desc = "turk_vs_tazik"
			value = {
				value = societal_value:turk_vs_tazik
				divide = 8
			}
		}
	}
}
