	power_per_pop = 35
	tax_per_pop = 200
	## each of their privileges is GOING to give a reduction in max tax. this number is going to go way up soon.
	rival = 0
	alliance = 0.012

	revolt_court_language = court_language
	characters_have_dynasty = always
	can_generate_mercenary_leaders = no
	bank = yes

	satisfaction = {
		country_cabinet_efficiency = 0.15
		tax_income_efficiency = 0.10
		change_policy_cost_modifier = -0.10
		global_max_control = 0.075
		global_monthly_prosperity = 0.0025
	}

	high_power = {
		country_cabinet_efficiency = 0.25
		
		research_speed_modifier = 0.15
		global_max_urban_control = 0.15
		global_max_rural_control = 0.075

		bank_interest = 0.05
		revoke_privilege_cost_modifier = 0.75
		change_policy_cost_modifier = 0.10

		# they shelter their own tax
		diwan_estate_max_tax = -0.4
		## but they gleefully squeeze others
		burghers_estate_max_tax = 0.1
		tax_income_efficiency = 0.15
		clergy_estate_max_tax = 0.15
		peasants_estate_max_tax = 0.05
	}

	low_power = {
		country_cabinet_efficiency = -0.20
		stability_cost_efficiency = -0.2
		global_max_urban_control = -0.05
		global_max_rural_control = -0.10
		### they start to get squeezed, themselves
		tax_income_efficiency = -0.10
		diwan_estate_max_tax = 0.25
		### but fail to squeeze others as effectively
		burghers_estate_max_tax = -0.1
		clergy_estate_max_tax = -0.1
		peasants_estate_max_tax = -0.1
		## banks can spawn easier, and thus interest is lower.
		bank_interest = -0.05

	}



	opinion = {
		add = {
			desc = "ESTATE_OPINION_BASE"
			value = "opinion(scope:target)"
			multiply = 0.35
		}

		add = {
			desc = "ESTATE_OPINION_TARGET_D_ESTATE_POWER"
			value = "scope:target.estate_power(estate_type:diwan_estate)"
			subtract = 0.15
			multiply = 6
		}

		if = {
			limit = { court_language = scope:target.court_language }
			add = { desc = "ESTATE_OPINION_SAME_COURT_LANGUAGE" value = 5 }
		}

		if = {
			limit = { expected_army_size < scope:target.expected_army_size }
			add = { desc = "ESTATE_OPINION_HIGHER_MIL_POWER" value = -5 }
		}
		
		if = {
			limit = { stability > 0 }
			add = { desc = "ESTATE_OPINION_STABILITY" value = 10 }
		}
		
		if = {
			limit = { average_control_in_home_region > 50 }
			add = { desc = "ESTATE_OPINION_HIGH_CONTROL" value = 5 }
		}
		
		if = {
			limit = { modifier:tax_income_efficiency > 1.0 }
			add = { desc = "ESTATE_OPINION_EFFICIENT_TAXES" value = 5 }
		}
	}
}
