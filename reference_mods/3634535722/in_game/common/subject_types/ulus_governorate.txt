# ============================================================
# Ulus Governorate — the deferring administrator's vassalage.
#
# What an ilkhanate_governorate becomes when it declares fealty for
# a steward (custody_declare_fealty). NOT a conquered vassal: the
# governorate keeps strong internal autonomy (it chose this, and it
# administers its own corner as it always did) — the overlord gets
# its levies, its loyalty in war, and its weight in the custody
# contest, but cannot easily annex or micromanage it. The Injuid
# under the Ilkhan, the Kart under whoever held Khurasan.
#
# Design: docs/design/custody_of_the_ulus_design.md
# ============================================================

ulus_governorate = {
	subject_pays = subject_pays_vassal
	color = subject_vassal
	level = 2
	annullment_favours_required = 20

	# Created only through the declare-fealty action.
	creation_visible = { always = no }
	visible_through_diplomacy = { always = no }
	visible_through_treaty = { always = no }
	subject_creation_enabled = { always = no }
	release_country_enabled = { always = no }

	# Sworn for the war the steward is fighting — they march together.
	join_offensive_wars_always = {
		always = no
	}
	join_defensive_wars_always = { always = no }
	join_offensive_wars_can_call = { always = yes }
	join_defensive_wars_can_call = { always = yes }
	will_join_independence_wars = yes
	# The custodian is NOT auto-dragged into the governorate's wars
	# (engine default protects subjects from external attackers — which
	# let losing governorates kneel mid-war and hand the regent the
	# bill; it bled CHB and JAL white). Defense is a REQUEST instead:
	# custody.35, fired from on_war_declared (23_custody_defense_hooks).
	overlord_protects_external = no

	# The governorate chose this — it does not strain at the leash, but
	# it keeps its own house. High autonomy, hard to annex.
	has_overlords_ruler = no
	strength_vs_overlord = -0.25
	annexation_speed = 0.5
	annexation_min_years_before = 15
	annexation_min_opinion = 175
	overlord_can_cancel = yes

	diplomatic_capacity_cost_scale = 0.75
	has_limited_diplomacy = yes

	# The daimyo pattern: a governorate may make war on anyone except
	# its own custodian — fellow governorates above all (the Ilkhanid
	# successor melee), and outsiders too. The one bond it keeps is
	# the khan's name in the khutbah; everything else it settles by
	# the sword. (root = subject type; scope:overlord / scope:subject
	# the relation; scope:attacker / scope:defender the war.)
	allow_declaring_wars = { always = yes }
	can_attack = {
		NOT = { scope:defender = scope:overlord }
	}

	can_overlord_build_roads = no
	can_overlord_build_buildings = no
	can_overlord_build_rgos = no

	can_change_rank = yes
	can_change_heir_selection = yes

	great_power_score_transfer = 0.25
	diplo_chance_accept_subject = {
		base = -92
		current_strength = 0.2
		border_distance = -0.3
		negative_opinion = -5
		positive_opinion = 0.25
		rank_difference = -5
		giving_defensive_support = 2
		receiving_defensive_support = 5
		royal_ties = 1
		trust_in_actor = 0.2

		num_of_limited_subjects = -5
		culture_war = 3
		diplomatic_reputation = 1
		different_religion = -20
		same_common_language = 5
		same_court_language = 10
		different_culture = -10
		different_government_type = -25
		actor_at_war = -20
		competing_power = -200
		tax_base = -0.25
		
		reject_subjugation_reasons = -1
		accept_subjugation_reasons = 1
	}

	diplo_chance_accept_overlord = {
		base = -52
		border_distance = -0.3
		negative_opinion = -5
		positive_opinion = 0.25
		same_common_language = 5
		same_court_language = 10
		different_culture = -10
		different_government_type = -25
		actor_at_war = -20
		competing_power = -200
		trust_in_actor = 0.2
	}

	ai_wants_to_be_overlord = {
        scope:subject = {
			add = {
                desc = "COALITION_GRADE_ANTAGONISM_TOWARDS_THEM_SUBJECT"
                value = {
                    every_country_with_coalition_grade_antagonism_against_us = {                    
                        add = {
                            value = -10
                        }
                    }
                }
            }
        }
	}
}
