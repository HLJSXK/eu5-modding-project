# -----------------------------------------------------------------------------
# Shaykhī Dervishes — maẓālim cases (parliament agendas)
#
# Cases brought AGAINST the dervish wing before the dyarchic council. The
# asymmetry with 09_sufyani is deliberate: the brethren hoard no fortunes —
# their capital is devotion. Ruling against the regime's own base costs the
# commonwealth add_devotion every time (the revolution disciplining itself
# dampens the fervor it runs on) and buys back civic order instead of gold:
# stability, control, and the satisfaction of the propertied benches.
# -----------------------------------------------------------------------------

pa_shaykhi_case_bazaar_extortion = {
	type = country
	estate = shaykhi_dervishes_estate
	importance = 1.85
	potential = {
		country_has_estate = estate_type:shaykhi_dervishes_estate
	}
	on_accept = {
		add_estate_satisfaction = { type = estate_type:shaykhi_dervishes_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = { add = { value = "estate_power(estate_type:burghers_estate)" multiply = 0.2 } add = 0.10 } }
		add_estate_satisfaction = { type = estate_type:sufyani_notables_estate value = { add = { value = "estate_power(estate_type:sufyani_notables_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = { add = { value = "estate_power(estate_type:clergy_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = { add = { value = "estate_power(estate_type:peasants_estate)" multiply = 0.2 } } }

		# Return the "alms": what the lodges took at knife-point goes back
		# to the stalls, two parts restitution, one part fines.
		# Inlined seizure (kept inline for tooltip preview).
		"estate(estate_type:shaykhi_dervishes_estate)" = { save_scope_as = seize_target }
		set_local_variable = {
			name = seized_pool
			value = {
				value = scope:seize_target.estate_gold
				multiply = 0.25
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:shaykhi_dervishes_estate
			value = {
				value = local_var:seized_pool
				multiply = -1
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = local_var:seized_pool multiply = 0.6667 }
		}
		add_gold = { value = local_var:seized_pool multiply = 0.3333 }

		# Disciplining the brethren dampens the fire.
		add_devotion = -5
		add_country_modifier = { modifier = shaykhi_verdict_bazaar_peace years = 8 }
	}
	can_bribe = { can_bribe_default_trigger = yes }
	on_bribe = { on_bribe_default_effect = yes }
	ai_will_do = { base = 100 }
	chance = 10
}
