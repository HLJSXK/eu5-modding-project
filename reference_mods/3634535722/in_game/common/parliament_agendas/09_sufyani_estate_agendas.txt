# -----------------------------------------------------------------------------
# Sufyānī Notables — maẓālim cases (parliament agendas)
#
# Cases brought AGAINST the notable wing before the dyarchic council, in the
# style of 07_diwan / 08_amirs: accepting the case rules against the estate
# (flat satisfaction loss) and pleases every other bench (estate_power × 0.2).
# The notables hold property, so verdicts seize estate gold and redistribute
# it to the wronged parties — the amirs-file economics.
# -----------------------------------------------------------------------------

pa_sufyani_case_grain_hoarding = {
	type = country
	estate = sufyani_notables_estate
	importance = 2.5
	potential = {
		country_has_estate = estate_type:sufyani_notables_estate
	}
	on_accept = {
		add_estate_satisfaction = { type = estate_type:sufyani_notables_estate value = -0.20 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = { add = { value = "estate_power(estate_type:peasants_estate)" multiply = 0.2 } add = 0.10 } }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = { add = { value = "estate_power(estate_type:burghers_estate)" multiply = 0.2 } add = 0.05 } }
		add_estate_satisfaction = { type = estate_type:shaykhi_dervishes_estate value = { add = { value = "estate_power(estate_type:shaykhi_dervishes_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = { add = { value = "estate_power(estate_type:clergy_estate)" multiply = 0.2 } } }

		# Break open the granaries: a large share of the notables' hoard,
		# split three ways — the hungry, the bakers, and the treasury.
		# Inlined seizure (kept inline for tooltip preview).
		"estate(estate_type:sufyani_notables_estate)" = { save_scope_as = seize_target }
		set_local_variable = {
			name = seized_pool
			value = {
				value = scope:seize_target.estate_gold
				multiply = 0.40
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:sufyani_notables_estate
			value = {
				value = local_var:seized_pool
				multiply = -1
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:peasants_estate
			value = { value = local_var:seized_pool multiply = 0.3333 }
		}
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = local_var:seized_pool multiply = 0.3333 }
		}
		add_gold = { value = local_var:seized_pool multiply = 0.3334 }

		add_country_modifier = { modifier = sufyani_verdict_granaries_opened years = 5 }
	}
	can_bribe = { can_bribe_default_trigger = yes }
	on_bribe = { on_bribe_default_effect = yes }
	ai_will_do = { base = 100 }
	chance = 10
}
