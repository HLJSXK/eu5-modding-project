pa_amirs_case_land_seizure = {
	type = country
	estate = amirs_estate
	importance = 2.5
	potential = {
		country_has_estate = estate_type:amirs_estate
	}
	on_accept = {
		add_estate_satisfaction = { type = estate_type:amirs_estate value = -0.20 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = { add = { value = "estate_power(estate_type:peasants_estate)" multiply = 0.2 } add = 0.05 } }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = { add = { value = "estate_power(estate_type:diwan_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = { add = { value = "estate_power(estate_type:clergy_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = { add = { value = "estate_power(estate_type:burghers_estate)" multiply = 0.2 } } }
		add_army_tradition = -7.5
		add_manpower = { value = monthly_manpower multiply = -12 }
		random_owned_location = {
			limit = { is_capital = no }
			add_location_modifier = { modifier = amir_purge_raw_output years = 10 }
		}
	}
	can_bribe = { can_bribe_default_trigger = yes }
	on_bribe = { on_bribe_default_effect = yes }
	ai_will_do = { base = 100 }
	chance = 10
}

pa_amirs_case_caravan_theft = {
	type = country
	estate = amirs_estate
	importance = 1.85
	on_accept = {
		add_estate_satisfaction = { type = estate_type:amirs_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = { add = { value = "estate_power(estate_type:burghers_estate)" multiply = 0.2 } add = 0.10 } }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = { add = { value = "estate_power(estate_type:diwan_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = { add = { value = "estate_power(estate_type:clergy_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = { add = { value = "estate_power(estate_type:peasants_estate)" multiply = 0.2 } } }

		# Fine the Amirs: a large share of their accumulated coffers, split
		# two parts to the wronged merchants, one part to the crown.
		# Inlined seizure (kept inline for tooltip preview).
		"estate(estate_type:amirs_estate)" = { save_scope_as = seize_target }
		# LOCAL variable: scopes to this effect execution, not the
		# country entity — no residue, no cross-effect collisions.
		set_local_variable = {
			name = seized_pool
			value = {
				value = scope:seize_target.estate_gold
				multiply = 0.40
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:amirs_estate
			value = {
				value = local_var:seized_pool
				multiply = -1
			}
		}
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = local_var:seized_pool multiply = 0.6667 }
		}
		add_gold = { value = local_var:seized_pool multiply = 0.3333 }

		add_country_modifier = { modifier = amir_purge_trade_buff years = 10 }
	}
	can_bribe = { can_bribe_default_trigger = yes }
	on_bribe = { on_bribe_default_effect = yes }
	ai_will_do = { base = 100 }
	chance = 10
}
