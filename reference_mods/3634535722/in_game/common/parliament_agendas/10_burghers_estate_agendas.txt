	}
	on_accept = {
		add_estate_satisfaction = { type = estate_type:burghers_estate value = -0.20 }
		add_estate_satisfaction = { type = estate_type:peasants_estate value = { add = { value = "estate_power(estate_type:peasants_estate)" multiply = 0.2 } add = 0.10 } }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = { add = { value = "estate_power(estate_type:clergy_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = { add = { value = "estate_power(estate_type:diwan_estate)" multiply = 0.2 } } }
		add_estate_satisfaction = { type = estate_type:amirs_estate value = { add = { value = "estate_power(estate_type:amirs_estate)" multiply = 0.2 } } }

		# Fine the Burghers (12 months)
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = root.monthly_income_trade_and_tax multiply = -12 }
		}
		# Compensate Peasants (9 months)
		add_gold_to_estate = {
			estate_type = estate_type:peasants_estate
			value = { value = root.monthly_income_trade_and_tax multiply = 9 }
		}
		# Crown Share (3 months)
		add_gold = { value = root.monthly_income_trade_and_tax multiply = 3 }

		add_stability = 1
