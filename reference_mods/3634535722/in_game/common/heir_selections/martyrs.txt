# =====================================================================
# MARTYRS DYARCHIC ELECTION — succession of the Commonwealth of Martyrs
#
# Base-game diarchic_election is Genoa's: it draws candidates from
# nobles/burghers and scores Spinola/Doria dynasty membership — none of
# which the commonwealth has. This is the real thing: candidates come
# from the two wings (Sufyānī notables, Shaykhī dervishes), and the
# election is the dyarchic seesaw itself — war elevates the dervish
# shaykh, peace favors the malik, and each wing votes its own virtue
# (the lodges weigh the sword, the notables weigh the ledger).
# Historical calibration: twelve rulers in ~45 years, alternating
# between the factions, most transitions violent.
# =====================================================================

martyrs_dyarchic_election = {
	all_in_country = yes
	ignore_ruler = yes
	allow_foreign_ruler = no
	use_election = yes
	term_duration = 0	# until ruler death — or the knife
	max_possible_candidates = 5
	allowed_estates = { sufyani_notables_estate shaykhi_dervishes_estate }
	heir_is_allowed = {
	}
	potential = {
		government_type = government_type:commonwealth_of_martyrs
	}

	calc = {
		if = {
			limit = {
				exists = root
			}
			# Each wing votes its own virtue.
			if = {
				limit = { root ?= { has_estate = estate_type:shaykhi_dervishes_estate } }
				add = {
					desc = "HEIR_SELECTION_SHAYKHI_SWORD"
					value = root.mil
				}
			}
			if = {
				limit = { root ?= { has_estate = estate_type:sufyani_notables_estate } }
				add = {
					desc = "HEIR_SELECTION_SUFYANI_LEDGER"
					value = root.adm
				}
			}
			# The seesaw: war elevates the dervish candidate, peace the malik.
			if = {
				limit = {
					root ?= { has_estate = estate_type:shaykhi_dervishes_estate }
					owner ?= { at_war = yes }
				}
				add = {
					desc = "HEIR_SELECTION_SHAYKHI_WAR"
					value = 150
				}
			}
			if = {
				limit = {
					root ?= { has_estate = estate_type:sufyani_notables_estate }
					owner ?= { at_war = no }
				}
				add = {
					desc = "HEIR_SELECTION_SUFYANI_PEACE"
					value = 100
				}
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
			}
			add = {
				desc = "HEIR_SELECTION_ADM_DESC"
				value = root.adm
			}
			add = {
				desc = "HEIR_SELECTION_DIP_DESC"
				value = root.dip
			}
			add = {
				desc = "HEIR_SELECTION_MIL_DESC"
				value = root.mil
			}
			if = {
				limit = { modifier:blocked_from_being_ruler = yes }
				add = {
					desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
					value = -1000
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = root }
			}
			add = {
				desc = "HEIR_SELECTION_SHAYKHI_WAR"
				value = 150
			}
			add = {
				desc = "HEIR_SELECTION_SUFYANI_PEACE"
				value = 100
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_ADM_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_DIP_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_MIL_DESC"
				value = 1
			}
		}
	}

	modifier = {
		care_about_producing_heirs = yes
	}
}
