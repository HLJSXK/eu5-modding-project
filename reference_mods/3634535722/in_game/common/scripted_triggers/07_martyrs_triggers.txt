# =====================================================================
# Commonwealth of Martyrs — shared triggers
#
# is_turco_mongol_polity: the standing enemy of the vow. RULER-CULTURE
# definition (design decision pending review): a polity is Turco-Mongol
# if its ruler belongs to the Turkic, Mongolian, or Tatar culture
# groups. This correctly EXCLUDES the Kartids (Tazik rulers) despite
# their being historical enemies — the vow is against the ordu's
# dispensation, not against every rival.
# =====================================================================

is_turco_mongol_polity = { #country
	ruler ?= {
		culture = {
			OR = {
				has_culture_group = culture_group:turkic_group
				has_culture_group = culture_group:mongolian_group
				has_culture_group = culture_group:tatar_group
			}
		}
	}
}
