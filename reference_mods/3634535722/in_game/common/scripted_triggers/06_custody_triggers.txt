# Triggers for the Custody of the Ulus disaster (Chinggisid Regency).
# Design: docs/design/custody_of_the_ulus_design.md

# A rival custody: any other Eranzamin member that either runs its
# own regency fiction or is ruled by the Borjigin line outright
# (Togha Temur's GRG is the seeded standing case).
custody_is_rival = {
	NOT = { this = root }
	OR = {
		has_reform ?= government_reform:chinggisid_regency
		ruler ?= { dynasty ?= dynasty:borjigin_dynasty }
	}
}

custody_has_sole_custody = {
	has_variable = custody_rivals
	var:custody_rivals = 0
}

custody_can_summon_kuriltai = {
	has_variable = custody_sole_months
	var:custody_sole_months >= 24
}

custody_can_endure = {
	has_variable = custody_months
	var:custody_months >= 300
	has_variable = charade_counter
	var:charade_counter >= 2
}

custody_resolved_trigger = {
	OR = {
		has_variable = custody_quiet_crown
		has_variable = custody_restored_yarliq
		has_variable = custody_endured
		has_variable = custody_expired
	}
}
