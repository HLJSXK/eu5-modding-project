# ============================================================
# Scripted triggers for the Farr Proof of Fortune disaster.
#
# Sustained-zone resolutions: the player must hold high or low
# legitimacy for 60 cumulative months (5 years' worth, not
# necessarily consecutive) for the houses to read it as a
# verdict. A flash-in-the-pan victory or a single-month dip
# does not resolve the wager.
# ============================================================

# Vindicated — 60 cumulative months at legitimacy ≥ 75.
farr_proof_can_vindicate = {
	has_variable = farr_high_months
	var:farr_high_months >= 60
}

# Broken-lapse — 60 cumulative months at legitimacy ≤ 25.
# The doctrine dies of inanition; no usurper, no battle.
farr_proof_can_break_lapse = {
	has_variable = farr_low_months
	var:farr_low_months >= 60
}

# Endured — 25 years (300 months) have elapsed and no other
# resolution has fired. Realm intact, claim survived, quiet
# vindication.
farr_proof_can_endure = {
	has_variable = farr_disaster_months
	var:farr_disaster_months >= 300
}
