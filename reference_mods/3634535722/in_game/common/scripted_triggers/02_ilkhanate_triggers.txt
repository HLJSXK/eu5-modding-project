is_eranzamin_region_province = {
	OR = {
		region = region:persia_region
		region = region:khorasan_region
		region = region:caucasus_region
		area = area:cappadocia_area
		area = area:iraq_arabi_area # Baghdad/Iraq
		area = area:jazira_area # Mosul and surroundings
	}
}

# Iranshahr proper — the heartland of a United Iran, used by the
# Erānzamīn victory dividend and the two victory cabinet actions.
# Deliberately NARROWER than is_eranzamin_region_province: it drops
# Cappadocia (Anatolia is contested ground, not Iranshahr) and keeps
# only the plateau, Khurasan, the Caucasus, and the two Iraqi/Jaziran
# arcs. Resolves from both location and province scope.
is_iranshahr_province = {
	OR = {
		region = region:persia_region
		region = region:khorasan_region
		region = region:caucasus_region
		area = area:iraq_arabi_area
		area = area:jazira_area
	}
}

is_eranzamin_contender = {
	is_subject = no
	# Tariqa orders (building-countries) are not contenders — they deny
	# the throne question entirely; no flavor.1, no warlord modifier.
	NOT = { country_type = building }
	# Straight off the capital LOCATION (region/area resolve there too):
	# the province link returned unset scopes for some countries in the
	# any_country sweeps (error.log 2026-08-25).
	capital ?= { is_eranzamin_region_province = yes }

	# Must be a relevant state
	# OR = {
	# 	country_rank_level > 1
	# 	num_locations_owned_or_owned_by_subjects_or_below > 10
	# }
}
