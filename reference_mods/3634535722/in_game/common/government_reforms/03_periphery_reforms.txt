# =====================================================================
# PERSIANATE PERIPHERY — starter reforms (persianate_periphery_plan.md)
#
# REPLACE:mamluk_government — modified version of the base reform
# (country_specific.txt:3571). Keeps the base mamluk mechanics
# (promote/barracks) and adds: the chancery flag and the military
# surgery — the fellahin FUND the army, they never fight. NOTE: until
# the MAM government-type switch lands, the estate-targeted lines are
# inert under the base monarchy; the chancery flag however activates
# diwan pop growth immediately (Arabic urban literate pops qualify).
#
# indian_sultanate_reform — the Delhi/Tughluq chassis, NEW (base has
# no Delhi reform). The bureaucracy reform's structural skeleton with
# Indian tuning: harder extraction (the fisc delta that kept the
# subcontinent sultanate country), garrison-city control profile,
# partial rather than total peasant demilitarization (the paik
# tradition). Locked — granted by setup and by successor-spawn
# effects, like persianate_bureaucracy_reform.
# =====================================================================

REPLACE:mamluk_government = {
	unique = yes
	major = yes
	content_priority = 1100
	potential = {
		has_or_had_tag = MAM
		NOT = { has_reform = government_reform:persianate_bureaucracy_reform }
	}
	locked = {
		is_locked_mechanic = {
			mechanic = government_reform
			type = mamluk_government
		}
	}
	country_modifier = {
		# --- base mamluk mechanics, kept ---
		can_promote_mamluks = yes
		can_build_mamluk_barracks = yes
		monthly_army_tradition = 0.1
		court_spending_efficiency = -0.1

		# --- the chancery: rawk surveys, diwan al-jaysh, the inshāʾ ---
		has_nizam_ul_mulk_chancery = yes
		tax_income_efficiency = 0.05

		# Nobles estate is EMPTY under the chancery (the nobles pop ladder
		# reassigns every noble pop to amirs before the fallback) — full
		# enfranchisement keeps peasant/dhimmi wealth from draining into
		# the ghost estate. The askari are paid from the fisc, not from a
		# manorial wealth share.
		global_peasant_enfranchisment = 1.0

		# --- military surgery: no peasant hosts, ever ---
		peasants_estate_levy_size = -1.0
		burghers_estate_levy_size = -0.5
		# The askari: small, elite, expensive. (Amirs line arms when the
		# government-type switch lands; nobles line covers the interim.)
		amirs_estate_levy_size = 0.75
		nobles_estate_levy_size = 0.5
		land_morale_modifier = 0.10

		# --- the chancery state, Egyptian dress (structural block from
		# the bureaucracy reform, tuned; the diwan capped junior) ---
		country_cabinet_efficiency = 0.05
		diwan_estate_target_satisfaction = 0.05
		global_diwan_estate_power = 0.15
		diwan_estate_allowed_in_cabinet = yes
		# Cairo's difference from Iran: the amirs SIT in government — the
		# dawadar system. No Siyasatnama complaint register can undo it.
		amirs_estate_allowed_in_cabinet = no
		nobles_estate_allowed_in_cabinet = no
		burghers_estate_allowed_in_cabinet = no
		clergy_estate_blocked_from_cabinet = yes
		burghers_estate_blocked_from_cabinet = yes
		nobles_estate_blocked_from_cabinet = yes
		amirs_estate_allowed_leading_military = yes
		diwan_estate_allowed_leading_military = no
		nobles_estate_allowed_leading_military = no
		burghers_estate_allowed_leading_military = no
		diwan_estate_levy_size = -0.9
		monthly_diwan_estate_rebel_growth = -0.01
		global_clergy_desired_pop_scaled = 0.01
		# The Nile: the most governable countryside on earth — higher
		# RURAL ceiling than the Iranian norm, same urban.
		global_max_urban_control = 0.15
		global_max_rural_control = 0.10
		global_estate_max_tax = 0.1
		tribes_estate_levy_size = 0.1
		government_size = 1
		cultures_capacity = 2
	}
	years = 2
}

# REPLACE: base ships indian_sultanate_reform (country_specific.txt:2799,
# held at start by DLH, MAB, RJR, SMA, SND, HNV — so NO government line:
# the five minors stay on their base governments until their own batch).
# Base content (bandagan barracks) kept; base's aristocracy drift dropped —
# taluqdar_nobility carries the aristocratic pole, this reform carries the
# Tughluq centralizing one. Deliberately ADOPTABLE (no locked block, like
# base): any Muslim power in India can take up the Delhi chassis — that is
# the self-sustaining-Persianate-India mechanism.
REPLACE:indian_sultanate_reform = {
	age = age_1_traditions
	potential = {
		religion.group = religion_group:muslim
		capital ?= { sub_continent = sub_continent:south_asia }
		NOT = { has_reform = government_reform:persianate_bureaucracy_reform }
	}
	country_modifier = {
		has_nizam_ul_mulk_chancery = yes
		# NO enfranchisement override here, deliberately: with the
		# tripartite ladder the Indian nobles estate is REAL (Hindu
		# rajas/taluqdars — non-Muslim noble pops stay nobles), and the
		# peasant/dhimmi wealth share flowing to them IS the taluqdar
		# extraction, working as intended.
		# The bandagan institution — military slavery, base-modeled via
		# the mamluk barracks chassis.
		can_build_mamluk_barracks = yes

		# The Tughluq fisc — the most articulated iqṭāʿ-diwan system of
		# its day: empire-wide revenue registers, harder extraction than
		# the Iranian norm. THIS is the delta that kept the subcontinent
		# sultanate country.
		global_nobles_estate_power = -0.6 # nobles are stronger here, because it's india.
		country_cabinet_efficiency = 0.1
		tax_income_efficiency = 0.1
		global_estate_max_tax = 0.15
		diwan_estate_target_satisfaction = 0.05
		global_diwan_estate_power = 0.25
		monthly_diwan_estate_rebel_growth = -0.01
		monthly_towards_centralization = 0.15
		global_clergy_desired_pop_scaled = 0.01
		# Populate the elite strata the base pop data leaves empty: the
		# rural lever grows the HINDU taluqdar/raja layer (non-Muslim
		# nobles stay in the nobles estate), the city lever feeds the
		# Muslim urban notables who flow to diwan and amirs.
		global_nobles_rural_desired_pop_scaled = 0.001
		global_nobles_city_desired_pop_scaled = 0.001

		# Cabinet structure: the vizier state.
		diwan_estate_allowed_in_cabinet = yes
		amirs_estate_allowed_in_cabinet = no
		nobles_estate_allowed_in_cabinet = no
		burghers_estate_allowed_in_cabinet = no
		amirs_estate_allowed_leading_military = yes
		diwan_estate_allowed_leading_military = no
		clergy_estate_blocked_from_cabinet = yes
		burghers_estate_blocked_from_cabinet = yes
		nobles_estate_blocked_from_cabinet = yes
		# Military structure: Turco-Persian cavalry elite + the slave
		# corps; paiks keep a partial peasant contribution (NOT the
		# Iranian -0.9 — the paik tradition is real infantry).
		amirs_estate_levy_size = 0.4
		peasants_estate_levy_size = -0.5
		diwan_estate_levy_size = -0.9
		tribes_estate_levy_size = 0.1

		# Garrison-city state: Delhi's writ runs through fortified urban
		# centers; its rural reach was always thinner (the Doab risings).
		global_max_urban_control = 0.15

		government_size = 1
		cultures_capacity = 2
	}
	years = 2
}
