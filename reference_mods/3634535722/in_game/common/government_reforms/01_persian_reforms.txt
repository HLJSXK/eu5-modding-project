#   Age 3 — the Safavid fusion and its disciplining

#   - All reforms top-gated by `government = persianate_sultanate`

#     reform is intrinsically Muslim / Shiʿi.

#   - One-shot adoption costs (stability hits, etc.) go in
#     `on_activate`. Ongoing modifiers stay in `country_modifier`.
#   - Magnitudes via `ez_*` script values in flavor_values.txt.

# Divine royal fortune — the mirror-for-princes path. You cease
# to be a Turco-Mongol warlord and become shāh bearing the farr-i
# īzadī, the Islamic-Persian synthesis the ulema themselves wrote
# (Niẓām al-Mulk's Siyāsatnāma, Ṭūsī's Akhlāq-i Nāṣirī, Ghazālī's
# Naṣīḥat al-Mulūk, later Davānī). Injuid–Muzaffarid–Kart practice.
#
# The friction is NOT theological — by 1337 farr, zill Allāh, and
# Anūshīrvān-as-just-king are standard Islamic kingship doctrine,
# praised in the very mirrors the clergy read and teach. The
# friction is institutional: by what right do you crown yourself
# when the only sanctioning authority is the ʿAbbāsid in Cairo,
# and you have just bypassed him? And political: a settled Persian
# shāh has no place for the steppe aristocracy that fights his wars.
farr_i_izadi = {
	age = age_1_traditions
	government = persianate_sultanate
	locked = {
		NOT = { has_variable = farr_was_broken_lapse }
	}
	potential = {
		is_member_of_international_organization = international_organization:eranzamin
		NOT = { has_reform = government_reform:chinggisid_regency }
		NOT = { has_reform = government_reform:ghazi_sultanate }
		NOT = { has_reform = government_reform:amir_sultanate }
		NOT = { has_reform = government_reform:dar_al_jihad }
		NOT = { has_reform = government_reform:ilkhanate_governorate }
		NOT = { has_reform = government_reform:tariqa_order }
		# Re-declaration lock: a dynasty that has tried farr and failed
		# cannot re-declare. The attempted_farr_and_failed modifier is
		# applied by the Broken-lapse and Broken-formal resolutions of
		# the farr_proof_of_fortune disaster. Removing the modifier in
		# console / by external event clears the lock.
		NOT = { has_country_modifier = attempted_farr_and_failed }
	}

	country_modifier = {
		# Legitimacy bleed has moved to the farr_proof_of_fortune disaster
		# (-0.5/mo there, vs the -0.25 here). The reform is the STANDING
		# claim; the disaster is the wager under test. Keeping the bleed
		# on the reform double-taxes the same political fact.
		stability_decay = -0.0025 # the realm bleeds while the kingship doctrine is being established — the project is its own stress.
		monthly_towards_centralization = 0.1
		monthly_towards_tazik = 0.05
		global_max_urban_control = ez_diwan_rural_bonus

		# The natural constituency of mirror-for-princes kingship
		# rallies: the chancery sees its own theory of the state
		# vindicated (this IS the doctrine of their own books); the
		# burghers see their cities affirmed as the seat of justice.
		diwan_estate_target_satisfaction = ez_priv_sat_large
		burghers_estate_target_satisfaction = ez_priv_sat_major
		government_size = 1
		# The price: the steppe-coded layers reject a crown that owes
		# nothing to them — the amirs have no place in a settled shāh's
		# court, the tribes see a state that does not want them.
		amirs_estate_target_satisfaction = -0.05
		tribes_estate_target_satisfaction = -0.10
		monthly_amirs_estate_rebel_growth = 0.005
		monthly_tribes_estate_rebel_growth = 0.005

		# The jurists' grumble is quiet and institutional, not
		# theological: a small, persistent unease about a sultan who
		# declares himself God's shadow without caliphal investiture.
		clergy_estate_target_satisfaction = -0.05

		# The wider world's hostility is political-status, not
		# religious: the Caliph in Cairo resents the bypass, rival
		# amirs see a non-Borjigin claiming sovereign kingship as a
		# status threat. Targeted CB hostility from chinggisid_regency
		# / Türk-leaning persianate sultanates belongs in a separate
		# antagonism_against trigger, not on this country_modifier.
		antagonism_received_modifier = 0.15
	}
	
	years = 0 # this is a CLAIM.
}

# Dār al-Jihād — the verified ghaza claim. What the Ghāzī Sultanate
# was reaching for, now institutionalized: the wager has been won,
# the corps is a standing fact, and the realm is reorganized around
# the doctrine. Cemal Kafadar's frame for the early Ottoman polity
# (Between Two Worlds), but also the Ghaznavid-Ghūrid template —
# the polity that exists FOR the ghaza and is held together by it.
# The akıncı become a vocational estate, the ribats become tax-
# exempt waqfs with mustering halls, the qāḍī reads the khutba
# without flinching. Requires that you have already declared ghazi
# and that the wager closed in your favor — Vindicated (Sword of
# Islam) or Endured (Guardian of the Thughur).
dar_al_jihad = {
	age = age_2_renaissance
	government = persianate_sultanate

	potential = {
		NOT = { has_reform = government_reform:timurid_synthesis }
		NOT = { has_reform = government_reform:farr_i_izadi }
		NOT = { has_reform = government_reform:turkmen_confederation }
		NOT = { has_reform = government_reform:iranian_kingship }
		NOT = { has_reform = government_reform:tariqa_order }
		OR = {
			has_variable = ghazi_sword_of_islam
			has_variable = ghazi_guardian_of_thughur
		}
	}
	on_activate = {

		add_stability = 15 ## refunds most of the cost for the change.
		# The King title: a vindicated ghazi doctrine is a crown —
		# both Rum roads and the Iranian one pass through kingship
		# before anyone reaches for empire.
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
	}
	country_modifier = {
		# The wager is won. The doctrine is no longer under audit;
		# it is the constitutional fact of the realm. Standing
		# legitimacy, in the iranian_kingship register.
		monthly_legitimacy = 0.25
		government_reform_slots = 1
		# The corps is institutionalized. Manpower deepens past the
		# base reform's pulse; the field army moves as if the muster
		# were already standing — because, in this realm, it is.
		global_manpower_modifier = 0.20
		global_levy_size_modifier = 0.25
		land_morale = 0.10
		army_tradition_from_battle = 0.20

		# Sword of the Faith named in chanceries from Cairo to Delhi.
		# Antagonism falls; the diplomatic register changes shape.
		antagonism_received_modifier = -0.20
		diplomatic_reputation = 1

		# The clergy patronage deepened. The qāḍīs read the khutba
		# without flinching; orthodoxy institutionalized as a state
		# apparatus, not a constituency to be courted.
		clergy_estate_target_satisfaction = ez_priv_sat_huge
		global_clergy_estate_power = 0.40
		tolerance_own = 2

		# The amīrs and tribes are the corps' constituency; the
		# frontier marches have been integrated as a vocational
		# warrior estate rather than a problem to be solved.
		amirs_estate_target_satisfaction = ez_priv_sat_major
		tribes_estate_target_satisfaction = ez_priv_sat_minor

		# Drift: the doctrine has institutionalized, so the chancery
		# and the corps now coexist rather than the chancery being
		# subordinated to the frontier. Light centralization; the
		# spiritualist / jurisprudential pulls persist as the
		# doctrinal apparatus deepens.
		monthly_towards_centralization = 0.1
		monthly_towards_spiritualist = 0.1
		monthly_towards_jurisprudence = 0.15

		cultural_tradition_modifier = 0.05
	}

	years = 2
}
