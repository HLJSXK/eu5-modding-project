# =====================================================================
# TARIQA ORDER — the Sufi order as extraterritorial building-country
#
# The khalīfa system as mechanics: the order lives at its shrine-seat
# (capital it need not own — the Peruzzi-in-Florence pattern) and
# plants tariqa_khanaqah lodges in other realms' towns via the normal
# foreign-building interface (kontor chassis). Gated on
# country_type = building, mirroring banking_country.
#
# Phase-2 (designed, Part D.2): base-choice reforms layered on top —
#   tariqa_tribal_base   (Türkmen devotees → the Qizilbash road)
#   tariqa_clerical_base (madrasa-adjacent respectability)
#   tariqa_popular_base  (village networks → rural millenarianism)
# and the radicalization arc (the Junayd moment).
# =====================================================================

tariqa_order = {
	potential = {
		country_type = building
		religion.group = religion_group:muslim
	}
	allow = {
	}
	locked = {
		is_locked_mechanic = {
			mechanic = government_reform
			type = tariqa_order
		}
	}
	country_modifier = {
		allows_tariqa_buildings = yes
		# The order's currency is reverence and reach, not tax.
		improve_relation_impact = 0.10
		monthly_towards_mysticism = 0.2 ## THIS IS mysticism
		diplomatic_reputation = 1
		global_levy_size_modifier = 0.5
		
	}
	years = 2
}

# =====================================================================
# THE THREE BASES — who the order recruits determines what it becomes.
# Mutually exclusive; each unlocks a base-specific building whose
# EMPLOYMENT is the census of devotees (the khuruj event-troop math
# reads it later). Endgames per docs/design D.2.1:
#   tribal  → khuruj conquest (Safaviyya)
#   popular → defensive rising, inherit the patron (Marʿashi)
#   urban   → court capture, wealth, infiltration (Hurufi)
# =====================================================================

# The Türkmen road: khalifas among the pastoral confederations —
# devotion circles in the camps. Not yet armed: the swords come only
# with radicalization (qizilbash_lodge, Two Pirs completion).
tariqa_tribal_base = {
	age = age_1_traditions
	potential = {
		country_type = building
		has_reform = government_reform:tariqa_order
		NOT = { has_reform = government_reform:tariqa_popular_base }
		NOT = { has_reform = government_reform:tariqa_urban_base }
	}
	country_modifier = {
		allows_tariqa_tribal_devotions = yes
		land_morale_modifier = 0.10
		monthly_towards_mysticism = 0.1
		tribes_estate_levy_size = 0.5
	}
	years = 5
}

# The village road: zawiyas among the peasantry — the network that
# inherits its patron's realm when he strikes it too late.
tariqa_popular_base = {
	age = age_1_traditions
	potential = {
		country_type = building
		has_reform = government_reform:tariqa_order
		NOT = { has_reform = government_reform:tariqa_tribal_base }
		NOT = { has_reform = government_reform:tariqa_urban_base }
	}
	country_modifier = {
		allows_tariqa_rural_zawiyas = yes
		global_manpower_modifier = 0.10
		monthly_towards_mysticism = 0.1
	}
	years = 5
}

# The court road: madrasa endowments, litterateurs, and money — no
# armies at all; the endgame is converting the throne itself.
tariqa_urban_base = {
	age = age_1_traditions
	potential = {
		country_type = building
		has_reform = government_reform:tariqa_order
		NOT = { has_reform = government_reform:tariqa_tribal_base }
		NOT = { has_reform = government_reform:tariqa_popular_base }
	}
	country_modifier = {
		allows_tariqa_urban_waqfs = yes
		diplomatic_reputation = 1
		global_max_literacy = 5
	}
	years = 5
}
