# -----------------------------------------------------------------------------
# Reform tree for the Commonwealth of Martyrs
#
# Starter: Bāshtīn Proclamation — auto-applies at game start to any
# commonwealth_of_martyrs polity. Encodes the founding moment (the 1337
# Bāshtīn revolt led by ʿAbd al-Razzāq Bāshtīnī) and the polity's enduring
# character: anti-aristocratic, militant, mahdist-awaiting, dyarchic.
#
# Evolution path: a single jurisprudential arc from traditionist communal
# mahdism toward formal clerical sovereignty. Each step is a real moment in
# Twelver legal-political development; the full sequence is a counterfactual
# (Sarbedārs collapsed by 1381, never reached Usūlī consolidation), but every
# stage is historically grounded as a moment of Shīʿī jurisprudence.
#
#   Bāshtīn Proclamation (proto-Akhbārī, communal traditionist)
#     → Usūlī Consolidation     (rationalist jurisprudence; jurist ijtihād)
#       → Marjaʿiyya             (single jurist authority binding emulators)
#         → Wilāyat-i Faqīh      (jurist direct sovereignty; Khomeini-doctrine
#                                  endpoint, anachronistic but consistent
#                                  alt-history with the others)
# -----------------------------------------------------------------------------

bashtin_proclamation_reform = {
	age = age_1_traditions
	government = commonwealth_of_martyrs

	potential = { }

	country_modifier = {
		# Anti-aristocratic foundation: nobles, amirs, diwan all suppressed.
		global_nobles_estate_power = -0.50
		global_amirs_estate_power = -0.75
		global_diwan_estate_power = -0.30
		global_nobles_rural_desired_pop_scaled = -0.002
		global_nobles_city_desired_pop_scaled = 0.002
		# Dyarchic estate participation enabled
		sufyani_notables_estate_allowed_in_cabinet = yes
		shaykhi_dervishes_estate_allowed_in_cabinet = yes
		sufyani_notables_estate_can_participate_in_parliament = yes
		shaykhi_dervishes_estate_can_participate_in_parliament = yes
		# The polity's military profile: heavy infantry + urban militia +
		# dervish corps. NO peasant infantry levies; peasants are taxbase
		# (and occasional mobilization base in crisis), not standing levy.
		peasants_estate_levy_size = -0.50
		burghers_estate_levy_size = 0.25
		global_max_control = 0.1
		tribes_estate_levy_size = -0.80   # Sarbedārs reject tribal-cavalry warfare
		# The dervish corps IS the army — ~5,000 armed brethren at peak from
		# a three-district state. The notables field the urban militia.
		shaykhi_dervishes_estate_levy_size = 0.75
		sufyani_notables_estate_levy_size = 0.25
		a_dervish_corps_build_cost_modifier = -0.5 ### they pay way less.
		a_dervish_corps_maintenance_cost_modifier = -0.5 ### they pay way less.
		a_dervish_corps_reinforce_cost_modifier = -0.5 ### they pay way less.
		global_pop_promotion_speed_modifier = 1.5 ### there is no nobility.
		global_peasant_enfranchisment = 1
		# Devotional state
		tolerance_own = 2
		monthly_devotion = 0.05
		global_manpower_modifier = 0.25
		land_morale_modifier = 0.10

		# Commonwealth structure: decentralized, low control ceiling,
		# permanent internal Sufyānī/Shaykhī tension
		global_max_control = -0.15
		stability_cost_efficiency = -0.1

		# Coin in the Hidden Imām's name — legitimacy of conventional
		# sultanic kind is rejected
		monthly_legitimacy = -0.10

		# Sliders push: militant, communal, spiritualist
		monthly_towards_tazik = 0.05
	}

	days = 0
}

# The Niyāba ʿĀmma Charter — the ledger becomes lawful.
# The commonwealth's entry into the chancery world (design doc §7):
# with Nishapur's chancery privilege established, the council charters
# a diwan in the absent Imām's name — the general vicariate given an
# administrative body. Grants has_nizam_ul_mulk_chancery, THE
# apparatus gate of the Persianate world (02_chancery_modifier_types),
# and cancels the Bāshtīn diwan suppression. The Usūlī line now hangs
# off this: jurisprudential consolidation presupposes a chancery to
# consolidate INTO.
niyaba_amma_charter_reform = {
	age = age_2_renaissance
	government = commonwealth_of_martyrs

	potential = {
		religion = religion:shia
		has_reform = government_reform:bashtin_proclamation_reform
		has_estate_privilege = estate_privilege:sufyani_chancery_of_the_vow
	}

	country_modifier = {
		has_nizam_ul_mulk_chancery = yes

		# The ledger serves the trust: the Bāshtīn diwan suppression
		# (-0.30) is cancelled.
		global_diwan_estate_power = 0.30
		country_cabinet_efficiency = 0.10
		global_max_control = 0.05
		tax_income_efficiency = 0.05

		monthly_towards_centralization = 0.10
		monthly_towards_tazik = 0.05
	}

	years = 5
}
