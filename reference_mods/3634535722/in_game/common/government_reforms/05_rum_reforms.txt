# =====================================================================
# SHATTERED RUM — reform family (persianate_periphery_plan.md §3)
#
# The Anatolian answers to the same legitimacy question the death of
# the ulus of Hülegü posed in Iran. The beyliks are switched to the
# persianate_sultanate government type at situation start (near-inert
# without a chancery); the marches are crisis-exempt via
# anatolian_beylik until they claim more than a march.
#
#   seljuk_claimant  — the declaration: custody of Konya. Nobody has
#                      the blood (the line died in 1308); the claim is
#                      possession + apparatus + recognition. Opens the
#                      Seljuk Inheritance disaster (06_seljuk_
#                      inheritance.txt): the audit is STATE-BUILDING.
#   sultanate_of_rum — the vindicated claim. Kingdom rank, the dowry
#                      of the dead sultanate. Self-declared: you hold
#                      the dar al-mulk and the diwan; Cairo's opinion
#                      is noted and filed.
#
# The ghazi road reuses ghazi_sultanate / Burden of the Sword /
# dar_al_jihad UNCHANGED (audited: no Eranzamin, region, or culture
# gates anywhere in that machine).
# =====================================================================

seljuk_claimant = {
	age = age_1_traditions
	government = persianate_sultanate

	potential = {
		religion.group = religion_group:muslim
		# Konya is deliberately NOT required to declare: taking the
		# dar al-mulk is the first step of the Inheritance PROGRAM
		# (the disaster's Old Capital watcher), not a prerequisite of
		# the word. The Karamanids start one step ahead by holding it
		# — that is their entire head start.
		capital ?= { region = region:anatolia_region }
		NOT = { has_reform = government_reform:ghazi_sultanate }
		NOT = { has_reform = government_reform:dar_al_jihad }
		NOT = { has_reform = government_reform:chinggisid_regency }
		NOT = { has_reform = government_reform:farr_i_izadi }
		NOT = { has_reform = government_reform:amir_sultanate }
		NOT = { has_reform = government_reform:ilkhanate_governorate }
		NOT = { has_reform = government_reform:tariqa_order }
		NOT = { has_reform = government_reform:mamluk_government }
		NOT = { has_reform = government_reform:indian_sultanate_reform }
		NOT = { has_reform = government_reform:timurid_empire }
		NOT = { has_reform = government_reform:sultanate_of_rum }
		# A house that declared and broke cannot re-declare — the
		# Karamanid fate, set by the disaster's Broken resolution.
		NOT = { has_variable = rum_claim_broken }
	}

	# Unremovable while the audit runs: the claim can be VINDICATED
	# or BROKEN, never quietly shelved (the AI was dropping it the
	# moment the disaster bit). The resolution variables are set
	# before rum.20/rum.21 fire, so script removal in those events
	# passes the lock.
	locked = {
		NOR = {
			has_variable = rum_seljuk_broken
			has_variable = rum_seljuk_vindicated
		}
	}

	country_modifier = {
		# The claim is a boast: every beg who refused it is now your
		# rival, and the world watches whether you can be a state.
		antagonism_received_modifier = 0.20
		monthly_prestige = 0.1
		# The claim's natural constituency — the qadis and scribes who
		# remember what a sultanate WAS pay attention; the begs bristle.
		clergy_estate_target_satisfaction = 0.03
		nobles_estate_target_satisfaction = -0.05
		monthly_towards_centralization = 0.05
	}

	years = 0 # a claim, spoken in the khutba at Konya.
}

# The vindicated claim — granted solely by the Seljuk Inheritance
# disaster's Vindicated resolution (rum.20), which removes
# seljuk_claimant in the same stroke. TWO-REFORM STACK, the Iranian
# pattern: the chancery lives on persianate_bureaucracy_reform
# (erected mid-disaster by Restore the Diwan and KEPT — chancery
# access continues through it), and this reform is the CROWN on top,
# the iranian_kingship analogue. A broken claimant likewise keeps the
# bureaucracy reform he built: the apparatus outlives the claim,
# which is why the Karamanids stayed dangerous for 150 years.
sultanate_of_rum = {
	age = age_1_traditions
	government = persianate_sultanate

	potential = {
		has_variable = rum_seljuk_vindicated
		NOT = { has_reform = government_reform:seljuk_claimant }
		NOT = { has_reform = government_reform:chinggisid_regency }
		NOT = { has_reform = government_reform:tariqa_order }
	}

	on_activate = {
		# The dowry of the dead sultanate: the crown of Kayqubad.
		add_stability = 15
		if = {
			limit = { country_rank_level < 3 }
			set_country_rank = country_rank:rank_kingdom
		}
	}

	country_modifier = {
		# iranian_kingship's register, Rum dress: the verified claim.
		monthly_legitimacy = ez_reform_legitimacy_major
		tax_income_efficiency = 0.10
		country_cabinet_efficiency = 0.15
		global_diwan_estate_power = 0.15
		diwan_estate_target_satisfaction = ez_priv_sat_major
		# The begs swore at vindication — cavalry for a sultan they
		# bent the knee to.
		amirs_estate_target_satisfaction = ez_priv_sat_minor
		monthly_towards_centralization = 0.10
		global_max_urban_control = ez_diwan_urban_bonus
		cultural_tradition_modifier = 0.05
		diplomatic_reputation = 1
	}

	years = 10
}

# REPLACE: base anatolian_beylik (country_specific.txt:3478) — kept
# verbatim plus the late-founder switch: a beylik founded or freed
# AFTER situation start (uc_bey breaking loose, a new march) takes
# this reform to enter the race, and the government form comes with
# it. Also the crisis exemption anchor: this reform sits in the
# unsettled_legitimacy_crisis NOT-list — a march has no khutba
# problem until it claims more than a march.
REPLACE:anatolian_beylik = {

	potential = {
		OR = {
			culture = culture:turkish_culture
			culture = culture:turkoman_culture
			is_subject_type = uc_bey
		}
	}
	allow = {
		OR = {
			culture = culture:turkish_culture
			culture = culture:turkoman_culture
			is_subject_type = uc_bey
		}
	}
	locked = {
		OR = {
			current_year < 1340
			is_situation_active = situation:rise_of_the_ottomans
		}
	}
	country_modifier = {
		war_on_same_religion_no_cb_cost_modifier = -0.25
		wrong_culture_levy_size = 0.25
		government_size = 1
		government_reform_slots = 1
		allow_tributary_subject = yes
		global_nobles_rural_desired_pop_scaled = 0.0075
		global_nobles_city_desired_pop_scaled = 0.005
		# Beg squabbles are background noise to Christendom and Cairo
		# alike — playtest showed a beylik conquering HALF of western
		# Anatolia drawing a coalition of the entire Balkans plus
		# Venice, Genoa and the Mamluks. Nobody crusaded over which
		# march held Denizli.
		antagonism_received_modifier = -0.30
	}

	on_activate = {
		if = {
			limit = {
				is_situation_active = situation:rise_of_the_ottomans
				NOT = {
					is_target_in_global_variable_list = {
						name = eligible_beylik_list
						target = root
					}
				}
			}
			add_to_global_variable_list = { name = eligible_beylik_list target = root }
		}
		# Late founders join the shattered world on its terms.
		if = {
			limit = {
				is_situation_active = situation:rise_of_the_ottomans
				NOT = { government_type = government_type:persianate_sultanate }
			}
			change_government_type = government_type:persianate_sultanate
			change_heir_selection = heir_selection:favorite_son_elective_succession
		}
		# A march is a duchy, not a county (rank parity with the
		# game-start correction in 20_rum_hooks.txt).
		if = {
			limit = { country_rank_level < 2 }
			set_country_rank = country_rank:rank_duchy
		}
	}
	years = 2
}

# REPLACE: base legacy_of_osman (country_specific.txt:1257) — one-line
# potential fix for a base quirk we flagged and playtest confirmed:
# the reform is granted by rise_of_the_ottomans.2 to WHOEVER is the
# strongest beylik at situation start (the event sets
# tur_rise_of_ottoman_flag on them), but base's potential also
# demanded has_or_had_tag = TUR — so a non-Ottoman leader's grant
# silently failed and they never got the -0.33 antagonism package
# base balanced the whole race around. Tag requirement dropped;
# content verbatim.
REPLACE:legacy_of_osman = {
	unique = yes
	potential = {
		has_variable = tur_rise_of_ottoman_flag
	}

	locked = {
		is_locked_mechanic = {
			mechanic = government_reform
			type = legacy_of_osman
		}
	}

	country_modifier = {

		global_integration_speed_modifier = 0.25
		government_size = 1
		antagonism_received_modifier = -0.33
		auto_slave_raid_different_religion = yes
		government_reform_slots = 1
		global_monthly_control = 0.05
	}
	years = 2
}
