# =====================================================================
# THE KHURŪJ — the order in arms, and the crown that ends it
#
# Script-only reforms (potential gates on variables no player action
# sets): the tariqa loses its order reforms the moment the government
# flips to persianate_sultanate for the rising, so the army phase and
# the settled phase each need a reform of their own.
#
#   khuruj_rising_reform  — the RISING (tariqa.2.b): the devotee army
#     in the field. Ferocious and unsustainable; deliberately NOT
#     exempted from the unsettled-legitimacy crisis — the khurūj
#     bleeds legitimacy until it takes a great city and crowns its
#     pir (tariqa.6). The bleed IS the clock.
#   crowned_murshid_reform — the CROWNED MURSHID (tariqa.6): the
#     pir on a throne, the Qizilbāsh formula a century early (the
#     mature Age-3 doctrine is the persian tree's murshid_i_kamil —
#     distinct reform, distinct key). Held only while the age reads
#     mystically: the potential requires societal mysticism beyond
#     50, so the reform is REMOVED if the jurists claw the age back —
#     arguably what happened to the Safavids.
# =====================================================================

khuruj_rising_reform = {
	age = age_1_traditions
	government = persianate_sultanate

	potential = {
		has_variable = tariqa_khuruj_underway
	}

	country_modifier = {
		land_morale_modifier = 0.20
		discipline = 0.10
		# The lodge network feeds the column: bread waits in every town
		# the brethren ever preached in.
		global_supply_limit_modifier = 0.50
	}

	days = 0
}

crowned_murshid_reform = {
	age = age_1_traditions
	government = persianate_sultanate

	potential = {
		has_variable = tariqa_crowned_murshid
		# Maintained only while mysticism rules the age: below the
		# threshold the potential fails and the reform is removed.
		societal_value:mysticism_vs_jurisprudence < -50
	}

	country_modifier = {
		# The crown answers the khuṭba question (also exempted from the
		# unsettled-legitimacy crisis in 03_eranzamin_legitimacy_crisis).
		monthly_legitimacy = 0.1
		land_morale_modifier = 0.10
		discipline = 0.05
	}

	days = 0
}
