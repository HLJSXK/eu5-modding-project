# =====================================================================
# Commonwealth of Martyrs — cabinet actions
#
# charter_of_the_liberated: the integration fast-lane. A province the
# dawa has already won (Shia majority) is not a conquest but a
# homecoming — the council charters it and it integrates at a
# multiple of the normal rate. Modeled on integrate_into_diwan
# (01_diwan_actions.txt), stronger and gated on religion instead of
# chancery buildings.
# =====================================================================

charter_of_the_liberated = {
	ability = adm
	icon = taluqdar_tax_collection

	potential = {
		government_type = government_type:commonwealth_of_martyrs
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target

		name = "charter_of_the_liberated_select_province"
		none_available_msg_key = "charter_of_the_liberated_no_provinces"

		column = { data = name }
		column = { data = population }
		column = { data = province_tax_base }

		visible = {
			# The dawa has done its work: the province recites with us.
			religion_percentage = {
				religion = religion:shia
				value >= 0.5
			}
			# There must be owned, not-yet-cored land left to charter.
			any_location_in_province = {
				owner = scope:actor
				integration_level = conquered
			}
			NOT = {
				cabinet_already_performing_same_task_on_target = {
					cabinet_action = charter_of_the_liberated
					interaction_target = target
				}
			}
		}
	}

	province_modifier = {
		province_integration_speed = 8
	}
	progress = {
		scope:target = {
			value = province_average_integration
		}
	}
	is_finished = {
		scope:target = {
			any_location_in_province = {
				OR = {
					integration_level = integrated
					integration_level = core
				}
				count = all
			}
		}
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		value = 100
		modifier = {
			factor = 1.5
			scope:target = { province_tax_base > 20 }
		}
	}
}
