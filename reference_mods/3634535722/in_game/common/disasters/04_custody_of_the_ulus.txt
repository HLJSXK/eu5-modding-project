# ============================================================
# The Custody of the Ulus — the regency's fiction under audit
#
# The third legitimacy disaster. Farr audits the person's fortune;
# ghazi audits the realm's deeds; the custody audits THE FICTION
# ITSELF. The regent's legitimacy is cheap, high, and borrowed —
# what fails is the mandate, and it fails two ways: RIVALRY
# (someone else holds a khan too: the Jalayirid-Chupanid wars were
# fought over whose khan was the khan) and EXPIRY (khans die; each
# replacement is less credible — the existing charade_counter /
# charade_continues_mod machinery, reused not duplicated).
#
# Resolutions:
#   - Sole Custody fork (player-timed via the Kuriltai action,
#     24 months at zero rivals): the Quiet Crown (drop the fiction
#     on your own terms — the Shaykh Uvays move; the Matter of
#     Tork and Persia open up behind it) or the Restored Yarliq
#     (the khanate made real — the Mongol endpoint).
#   - Endured (300 months, >= 2 successions survived): the weaker,
#     dated fork — the Gurkani Regime (worse but stable) or a
#     late Quiet Crown at a steeper price.
#   - Expired: the fiction died (forced -100 exit at rock bottom)
#     -> amir_sultanate, the honest version of what you were.
#
# Design: docs/design/custody_of_the_ulus_design.md
# ============================================================

custody_of_the_ulus = {

	image = "gfx/interface/illustrations/disaster/custody_of_the_ulus.dds"

	monthly_spawn_chance = 1

	can_start = {
		has_reform = government_reform:chinggisid_regency
		custom_tooltip = {
			text = custody_not_yet_completed_tt
			NOT = { has_variable = custody_completed }
		}
		custom_tooltip = {
			text = custody_not_yet_fired_tt
			NOT = { has_variable = custody_fired }
		}
	}

	can_end = {
		OR = {
			custom_tooltip = {
				text = custody_end_quiet_crown_tt
				has_variable = custody_quiet_crown
			}
			custom_tooltip = {
				text = custody_end_yarliq_tt
				has_variable = custody_restored_yarliq
			}
			custom_tooltip = {
				text = custody_end_endured_tt
				has_variable = custody_endured
			}
			custom_tooltip = {
				text = custody_end_expired_tt
				AND = {
					NOT = { has_reform = government_reform:chinggisid_regency }
					has_variable = custody_expired
				}
				
				
			}
		}
	}

	# The fiction-maintenance drag: light — the reform's own pump
	# still runs; this is the cost of KEEPING the mask on. The
	# contested-custody auto-modifier (rivals >= 1) is the heavier
	# slice and eases automatically as custodies fall.
	modifier = {
		monthly_legitimacy = -0.10
		monthly_diwan_estate_rebel_growth = 0.005
	}

	on_start = {
		custody_init = yes
		set_variable = custody_fired
		trigger_event_non_silently = custody.1
	}

	on_monthly = {
		custody_monthly_tick = yes
		custody_resolution_poll = yes
	}

	on_end = {
		custody_clear = yes
		set_variable = custody_completed
	}
}
