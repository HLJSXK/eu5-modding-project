# -----------------------------------------------------------------------------
# Sufyānī Notables — parliament issues
#
# The urban-aristocratic faction's bench in the dyarchic council. Issues
# express the Sufyānī program: civic order, fiscal extraction, qāḍī courts,
# reconciliation with surviving Mongol/Türk neighbors, dervish containment.
#
# All issues gate on government_type = commonwealth_of_martyrs so they cannot
# leak into other council parliaments. parliament_type = council suffices —
# the Bāshtīn Proclamation pins this type onto every Commonwealth polity.
# -----------------------------------------------------------------------------

sufyani_codify_qadi_courts = {
	estate = sufyani_notables_estate

	modifier_when_in_debate = {
		global_max_control = 0.05
	}

	allow = {
		government_type = government_type:commonwealth_of_martyrs
		parliament_type = parliament_type:sarbadar_council
		NOT = { has_country_modifier = sufyani_codified_courts_mod }
	}

	chance = {
		add = 1
		add = {
			if = {
				limit = { stability < 0 }
				value = 2
			}
		}
	}

	on_debate_passed = {
		add_country_modifier = { modifier = sufyani_codified_courts_mod years = 10 }
		add_estate_satisfaction = { type = estate_type:sufyani_notables_estate value = 0.10 }
		change_societal_value = { type = turk_vs_tazik value = 3 }
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}

sufyani_curb_dervish_excesses = {
	estate = sufyani_notables_estate

	modifier_when_in_debate = {
		stability_cost_efficiency = 0.05
	}

	allow = {
		government_type = government_type:commonwealth_of_martyrs
		parliament_type = parliament_type:sarbadar_council
		"estate_power(estate_type:shaykhi_dervishes_estate)" > 0.55
	}

	chance = {
		add = 1
		add = {
			if = {
				limit = { "estate_power(estate_type:shaykhi_dervishes_estate)" > 0.7 }
				value = 3
			}
		}
	}

	on_debate_passed = {
		add_estate_satisfaction = { type = estate_type:shaykhi_dervishes_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:sufyani_notables_estate value = 0.15 }
		add_country_modifier = { modifier = sufyani_curbed_dervishes_mod years = 8 }
		add_stability = 1
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}
