# -----------------------------------------------------------------------------
# Shaykhī Dervishes — parliament issues
#
# The mahdist-militant faction's bench in the dyarchic council. Issues express
# the Shaykhī program: jihād declarations, anti-aristocratic purges, dervish
# mobilization, mahdist proclamation, and the excommunication of Sufyānī
# moderates as enemies of the Imām's cause.
#
# All issues gate on government_type = commonwealth_of_martyrs.
# -----------------------------------------------------------------------------

shaykhi_proclaim_jihad = {
	estate = shaykhi_dervishes_estate

	modifier_when_in_debate = {
		war_no_cb_cost_modifier = -0.10
	}

	allow = {
		government_type = government_type:commonwealth_of_martyrs
		parliament_type = parliament_type:sarbadar_council
		at_war = no
		NOT = { has_country_modifier = shaykhi_jihad_proclaimed_mod }
	}

	chance = {
		add = 1
		add = {
			if = {
				limit = { "estate_power(estate_type:shaykhi_dervishes_estate)" > 0.6 }
				value = 2
			}
		}
	}

	on_debate_passed = {
		add_country_modifier = { modifier = shaykhi_jihad_proclaimed_mod years = 5 }
		add_estate_satisfaction = { type = estate_type:shaykhi_dervishes_estate value = 0.15 }
		add_devotion = 10
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}

		change_societal_value = { type = turk_vs_tazik value = 5 }
	}

	on_debate_failed = {
		parliament_debate_failed_effect = yes
	}
}
