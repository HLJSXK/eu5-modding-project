# Auto-modifier: at-war bonus during Farr Proof of Fortune.
#
# While the farr_proof disaster is active and the realm is at
# war, the bleed is partially offset. Net while at war = -0.2/mo
# (-0.5 base + 0.3 here). Net while at peace = -0.5/mo.
#
# This makes war *slow* the bleed without paying for itself —
# only conquest (the farr_conquest_glory static modifier, +0.4/mo
# for 12mo) pushes the math positive. The player who maintains
# perpetual stalemated war still bleeds; the player who wins
# decisively breaks even and then some.
# ============================================================

	potential_trigger = {
		has_variable = farr_proof_fired
		at_war = yes
	}
	amirs_estate_target_satisfaction = 0.075 
	monthly_legitimacy = 0.3
}
