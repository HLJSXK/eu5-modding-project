# Unsettled Legitimacy — the cost of deferring the choice. 
# A persianate_sultanate in iran-zamīn that has not committed to one of
# the three Age 1 legitimacy paths (Chinggisid Regency, Ghazi Sultanate,
# Farr-i-Izadi) bleeds legitimacy and estate satisfaction equilibrium.
#
# Historically: the post-Ilkhanid amir who could not answer the khuṭba
# question — in whose name is the Friday sermon read? — was the amir
# his rivals devoured first. The qāḍīs, the mustawfīs, and the umarāʾ
# all need an answer; "we are still deciding" is the worst one.
#
# Lifts as soon as any one of the three reforms is taken. The reforms
# are mutually exclusive among themselves; this modifier vanishes the
# moment any of them is enacted.

unsettled_legitimacy_crisis = {
	potential_trigger = {
		government_type = government_type:persianate_sultanate
		NOT = { has_reform = government_reform:chinggisid_regency }
		NOT = { has_reform = government_reform:gurkani_regime }
		NOT = { has_reform = government_reform:ghazi_sultanate }
		NOT = { has_reform = government_reform:farr_i_izadi }
		NOT = { has_reform = government_reform:dar_al_jihad }
		NOT = { has_reform = government_reform:iranian_kingship }
		NOT = { has_reform = government_reform:ilkhanate_governorate }
		# The Ghaznavid template IS an answer: the sword holds the realm
		# and Cairo's diploma names it lawful. (Also the constitution of
		# released Great Raid marches — they must not be born bleeding.)
		NOT = { 
			AND = {
				has_reform = government_reform:amir_sultanate
				has_variable = tributary_march_released_by_legitimate_ruler
			}
		}
		NOT = { has_reform = government_reform:tariqa_order }
		# The crowned murshid IS an answer to the khuṭba question — the
		# coronation (tariqa.6) settles the claim. The khurūj-in-arms
		# deliberately is NOT excluded: the rising bleeds legitimacy
		# until it takes a great city and crowns its pir. The bleed is
		# the clock.
		NOT = { has_reform = government_reform:crowned_murshid_reform }
		# Periphery starter reforms answer the khuṭba question their own
		# way — Cairo's caliph (Mamluks), the Delhi dispensation (India).
		NOT = { has_reform = government_reform:mamluk_government }
		NOT = { has_reform = government_reform:indian_sultanate_reform }
		# The Ṣāḥib-Qirān's answer is built into his reform: puppet khan,
		# gurkan title, and the conjunction itself. Nothing gnaws at him.
		NOT = { has_reform = government_reform:timurid_empire }
		# Shattered Rum: a MARCH has no khutba problem — a beg's
		# legitimacy is his sword and his father's tents. The question
		# becomes acute only when he claims more than a march, and both
		# claims (the Seljuk custody and its vindicated crown) carry
		# their own answer.
		NOT = { has_reform = government_reform:anatolian_beylik }
		NOT = { has_reform = government_reform:seljuk_claimant }
		NOT = { has_reform = government_reform:sultanate_of_rum }
		NOT = {
			ruler ?= { dynasty = dynasty:borjigin_dynasty }				
		}
	}

	monthly_legitimacy = -0.6
	amirs_estate_target_satisfaction = -0.20
	diwan_estate_target_satisfaction = -0.20
	clergy_estate_target_satisfaction = -0.15
	burghers_estate_target_satisfaction = -0.15
	
}


ghazi_lord_at_war = {
	# The at-war offset for the Burden of the Sword disaster.
	# Bleed during the disaster is -0.5/mo; this offset is +0.75
	# so the realm nets +0.25 at qualifying war, -0.5 in peace —
	# the doctrine's signature: decay-at-rest.
	#
	# Qualifying target: religion != root.religion (covers tier-3
	# dar al-harb AND tier-2 heterodox Muslim states with their
	# own religion entry), OR explicit mongol_law_policy holders
	# (the tent-law violation case the doctrine names by name).
	potential_trigger = {
		government_type = government_type:persianate_sultanate

		OR ={
			has_reform = government_reform:ghazi_sultanate
			has_reform = government_reform:dar_al_jihad
		}
		at_war = yes
		any_country_at_war_with = {

			religion != root.religion

			
		}
	}

	# The +0.75 here exceeds the disaster's -0.5 bleed by exactly
	# the +0.25 net the design specifies. Outside the disaster
	# this stacks on the reform's own +0.1, which is fine.
	monthly_legitimacy = 0.75
	monthly_prestige = 0.10
	global_levy_size_modifier = 0.25
	global_manpower_modifier = 0.25
}
