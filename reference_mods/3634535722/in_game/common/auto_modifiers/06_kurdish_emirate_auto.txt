# =====================================================================
# KURDISH EMIRATE — the defensive levy
#
# The tribal host turns out in full ONLY in defense of the mountains. When the
# emirate is the aggressor, the aghas stay home and the levy collapses — so the
# bonus is gated on the country NOT being an attacker. This is the mechanical
# face of the whole design: devastating at home, useless abroad.
#
# (Auto-modifiers re-evaluate on state change — cf. amirs_war_auto_modifier,
# which keys off at_war the same way.)
# =====================================================================

kurdish_emirate_defensive_levy = {
	potential_trigger = {
		government_type = government_type:iranic_tribal_polity
		at_war = yes
		NOT = { is_an_attacker = yes }
	}

	# MUCH greater levy when defending — tune to taste.
	global_levy_size_modifier = 0.50
}
