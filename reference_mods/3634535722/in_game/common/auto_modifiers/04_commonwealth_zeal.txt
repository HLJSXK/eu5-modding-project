# =====================================================================
# Revolutionary Zeal — the defensive surge
#
# While the founding generation still stands (revolutionary_zeal
# running) and the commonwealth fights a DEFENSIVE war, the zeal
# doubles: a second full helping of the bonuses, and the maluses
# rise by half. The surge is flat while the base decays, so a late
# defensive war is proportionally MORE than double the withered
# base — deliberate: existential defense rekindles the founding
# fire, and a cornered commonwealth fights like 1337 again.
#
# Chassis: ghazi_lord_at_war (02.. auto_modifiers) — war-conditional
# offset via potential_trigger. is_in_defensive_war is a base
# scripted trigger (country_triggers.txt).
# =====================================================================

zeal_defensive_surge = {
	potential_trigger = {
		government_type = government_type:commonwealth_of_martyrs
		has_country_modifier = revolutionary_zeal
		is_in_defensive_war = yes
	}

	global_manpower_modifier = 0.50
	army_maintenance_efficiency = 0.25
	regiment_reinforcement_speed = 0.25
	land_morale_modifier = 0.20
	land_morale_recovery = 0.10
	tax_income_efficiency = 0.10

	global_monthly_prosperity = -0.0005
	global_monthly_development = -0.0005
}
