# Base values for the Commonwealth of Martyrs estates — mirrors
# diwan_base_values / amirs_base_values in 01_diwan_auto_modifiers.txt.
# Without a base target satisfaction the estates start hyper-rebellious
# and can be neither levied nor taxed.
#
# Gated on estate presence (not government type) so any later polity that
# acquires the estates — Mushaʿshaʿ-type rebellion-fork spawns — gets the
# floor automatically.
#
# NOTE: amirs_base_values is deliberately NOT extended to the commonwealth.
# The expropriated amirs living under Sarbadar rule SHOULD be furious —
# the anti-aristocratic foundation is the polity's whole premise.

sufyani_base_values = {
    potential_trigger = {
        government_type = government_type:commonwealth_of_martyrs
    }

    sufyani_notables_estate_target_satisfaction = 0.40
    global_sufyani_notables_estate_power = 0.05
}

shaykhi_base_values = {
    potential_trigger = {
        government_type = government_type:commonwealth_of_martyrs
    }

    shaykhi_dervishes_estate_target_satisfaction = 0.40
    global_shaykhi_dervishes_estate_power = 0.05
}
