# ============================================================
# Seljuk Inheritance — at-war offset (mirrors farr_proof_at_war_bonus)
#
# The ambient bleed is -1.0/mo; at war the claimant nets -0.7
# before conquest glory. A claim to the dead sultanate is proven
# in the field as much as in the registers — perpetual stalemate
# still bleeds; the claimant who wins decisively breaks even and
# climbs. The begs (nobles estate) respect a claimant at war.
# ============================================================

rum_claim_at_war = {
	potential_trigger = {
		has_variable = rum_inheritance_fired
		at_war = yes
	}
	nobles_estate_target_satisfaction = 0.075
	monthly_legitimacy = 0.3
}

# ============================================================
# The march without a mandate — a beylik that answers the
# shattering with NOTHING bleeds. The deferral (rum.2 option C)
# is a real choice, but it is not free: the Ilkhan who confirmed
# your fathers is gone, and until you claim SOMETHING — ghaza or
# Konya — every month erodes the house's word a little.
# ============================================================

rum_march_without_mandate = {
	potential_trigger = {
		has_reform = government_reform:anatolian_beylik
		NOT = { has_reform = government_reform:ghazi_sultanate }
		NOT = { has_reform = government_reform:seljuk_claimant }
		NOT = { has_reform = government_reform:dar_al_jihad }
		NOT = { has_reform = government_reform:sultanate_of_rum }
	}
	monthly_legitimacy = -0.5
}
