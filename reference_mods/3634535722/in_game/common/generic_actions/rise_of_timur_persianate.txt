# =====================================================================
# RISE OF TIMUR — the reform lands on the Persianate chassis
#
# REPLACE: base rot_reform_into_monarchy (generic_actions/
# rise_of_timur.txt:467) converts TIM to a generic monarchy. But the
# Timurid state IS the Persianate sultanate at maximum extension —
# Persian chancery, iqṭāʿ/soyurghal grants, tümen amīrs, a puppet
# khan for legitimacy. Land it on persianate_sultanate; the REPLACEd
# timurid_empire reform (04_timurid_reform.txt) now carries the full
# chancery block itself, so no separate bureaucracy-reform grant.
# Everything else — price, allow, select_trigger, AI — base verbatim.
# =====================================================================

REPLACE:rot_reform_into_monarchy = {
	type = situation
	potential = {
		scope:actor = {
			tag = TIM
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
	price = price:rot_reform_into_monarchy_price
	allow = {
		scope:actor = {
			is_rebel_country = no
			at_war = no
			NOT = { country_type = location }
			custom_tooltip = {
				text = rot_has_selected_core_region_tt
				has_variable = rot_has_selected_core_region
			}
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_timur = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			change_country_type = location
			change_government_type = government_type:persianate_sultanate
			unlock_government_reform_effect = { type = timurid_empire }
			add_reform = government_reform:timurid_empire
			# Timur designated Pīr Muḥammad and the amīrs ignored him —
			# succession by designation-then-contest, the same chassis
			# Delhi runs.
			change_heir_selection = heir_selection:favorite_son_elective_succession
			# The language of the Ẓafarnāma and of every farmān out of
			# Samarqand.
			set_court_language = culture:farsi_culture.dialect
			set_country_rank = country_rank:rank_empire
		}
	}

	ai_will_do = {
		add = 250	#Ai should always do it asap
	}
}
