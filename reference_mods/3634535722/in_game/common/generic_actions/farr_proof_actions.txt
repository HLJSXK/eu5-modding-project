# ============================================================
# Farr Proof of Fortune — disaster actions (apparatus claims)
#
# Four buttons surfaced on the disaster panel. Each represents the
# player's deliberate act of consolidating a doctrinal apparatus
# at a legitimacy threshold:
#
#   claim_sikka              — legitimacy ≥ 50 — fires farr_proof.50
#   claim_khutba             — legitimacy ≥ 65 — fires farr_proof.65
#   proclaim_hunt_and_nawruz — legitimacy ≥ 80 — fires farr_proof.80
#   assume_padishah          — legitimacy ≥ 90 — fires farr_proof.90
#
# Threshold flags (farr_50_fired etc.) are SET HERE in the action's
# effect block, not by the old apparatus_dispatch scripted effect
# (which has been retired — see 03_farr_proof_effects.txt). The
# flag's role is to hide the button after claim; one-shot is enforced
# by the allow block reading NOT = has_variable.
#
# Buttons stay enabled forever once threshold met — the player owns
# the timing. Sitting on legitimacy = 92 and waiting for a key
# conquest before assuming pādishāh is a real strategic choice.
# ============================================================

claim_sikka = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never
	automation_tick_frequency = 12

	allow = {
		scope:actor = {
			legitimacy >= 50
			NOT = { has_variable = farr_50_fired }
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:farr_proof_of_fortune
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = farr_50_fired
			trigger_event_non_silently = farr_proof.50
			add_legitimacy = -10
		}
		 ## you lose some of the banked legitimacy by claiming the sikka, but you are now on the path to recovery.
		### should COST legitimacy.
		international_organization:eranzamin = {
			every_international_organization_member = {
				limit = {
					has_reform = government_reform:chinggisid_regency
				}
				add_opinion = {
					modifier = opinion_betrayed_genghis
					target = scope:actor
				}
				add_casus_belli = {
					type = casus_belli:cb_eranzamin_succession
					target = scope:actor
				}
			}
		}
	}

	ai_will_do = {
		add = 100
	}
}
