# ============================================================
# Ghazi: Burden of the Sword — disaster actions (apparatus claims)
#
# Four buttons surfaced on the disaster panel. Each represents the
# player's deliberate act of consolidating a doctrinal apparatus:
#
#   claim_jihad_within         — legitimacy ≥ 50, no yasa law
#                                — fires ghazi_burden.50
#   claim_ribat_network        — legitimacy ≥ 60, gold ≥ 100
#                                — fires ghazi_burden.65
#   claim_caliphal_investiture — legitimacy ≥ 60
#                                — fires ghazi_burden.80
#   claim_sultan_of_ghazis     — legitimacy ≥ 75 with investiture,
#                                else ≥ 90  — fires ghazi_burden.90
#
# Apparatus claims COST legitimacy. The cost is the price of
# spending banked credibility for the institutional fact; the
# refund modifier then cancels the corresponding penalty in the
# disaster's standing modifier. Without a legitimacy cost the
# claims auto-cascade — see lessons learned in farr_proof_actions.
#
# Buttons stay enabled forever once threshold met — the player
# owns the timing. Sitting on legitimacy 92 and waiting for a key
# tier-3 conquest before assuming Sultan of the Ghazis is a real
# strategic choice.
# ============================================================

claim_jihad_within = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never
	automation_tick_frequency = 12

	allow = {
		scope:actor = {
			ghazi_can_claim_jihad_within = yes
			NOT = { has_variable = ghazi_50_fired }
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:ghazi_burden_of_sword
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = ghazi_50_fired
			trigger_event_non_silently = ghazi_burden.50
			add_legitimacy = -10
		}
	}

	ai_will_do = {
		add = 100
	}
}

# ============================================================
# INTERNAL CAMPAIGN — tier-1 purification of the realm itself.
#
# Gated on:
#   • disaster active
#   • ghazi_hisba_apparatus legacy (apparatus 4.1 claimed)
#   • hisba_law on the strict / muhtasib_regime side
#   • at least one hisba_office building owned
#   • ghazi_internal_campaigns counter < 3 (diminishing-returns cap)
#
# Cost: 50 gold (the campaign has expenses — printed proclamations,
# whippings paid for, wine smashed, lawyers' fees) and a small
# legitimacy bite: -3 leg up front. The credit lives in the event
# branches (vice / heterodoxy / sectarian) and tapers with each
# subsequent campaign.
# ============================================================
launch_internal_campaign = {
	type = disaster
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never
	automation_tick_frequency = 12

	allow = {
		scope:actor = {
			ghazi_can_launch_internal_campaign = yes
		}
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:ghazi_burden_of_sword
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			# Three months' income: proclamations, whippings, lawyers.
			add_gold = {
				value = monthly_income_total
				multiply = -3
			}
			add_legitimacy = -3
			trigger_event_non_silently = ghazi_burden.500
		}
	}

	ai_will_do = {
		add = 100
	}
}

ghazi_declare_holy_war = {
	show_message = no
	type = disaster

	potential = {
		scope:actor = {
			has_reform = government_reform:ghazi_sultanate
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 3

	cooldown = {
		type = ghazi_declare_holy_war
		months = 6
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = { data = name }
		visible = {
			disaster_type = disaster_type:ghazi_burden_of_sword
			disaster_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_neighbor_country = {
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "ghazi_holy_war_target"
		none_available_msg_key = "ghazi_no_holy_war_target_countries"
		column = { data = name }
		column = { data = expected_army_size }
		visible = {
			is_real_country = yes
			OR = {
				NOT = {religion = religion:sunni } ## a heathen or heretic.
				AND = {
					religion = religion:sunni ## a co-religionist, but of a different sect.
					scope:actor = {
						has_variable = ghazi_fatwa_acquired ## at least the first apparatus claimed, so the claim has some doctrinal weight.
					}
				}
			}
			scope:actor = {
				can_declare_war_on = root
				NOT = { has_truce_with = root }
				NOT = { is_allied_with = { target = root } }
			}
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = { exists = scope:target }
				add_casus_belli = {
					target = scope:target
					type = casus_belli:cb_ghozva
				}
			}
		}
	}

	ai_will_do = {
		add = 100
		# modifier = {
		# 	factor = 0
		# 	at_war = yes
		# }
	}
}
