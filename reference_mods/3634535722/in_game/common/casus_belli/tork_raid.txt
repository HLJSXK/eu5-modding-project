# The Great Raid's sword — granted by designating a theater (Matter of
# Tork, approach 4). Visible against any owner of land in the
# designated region. No allowed_locations fence: a raid war ranges
# where the raid ranges; the CB's own visibility is the fence.

cb_great_raid = {

	create_visible = {
		has_variable = matter_tork_raid_region
		var:matter_tork_raid_region ?= {
			any_location_in_region = {
				exists = owner
				owner = scope:target
			}
		}
	}

	declare_enabled = {
		has_variable = matter_tork_raid_region
		var:matter_tork_raid_region ?= {
			any_location_in_region = {
				exists = owner
				owner = scope:target
			}
		}
	}

	speed = 20

	war_goal_type = great_raid_wargoal
}
