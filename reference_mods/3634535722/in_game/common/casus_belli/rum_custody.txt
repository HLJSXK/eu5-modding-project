# The custodian's CB — granted by rum_summon_to_obedience (Shattered
# Rum situation action, unlocked by proclaiming custody of the Old
# Capital). Anatolia-limited wargoal, cb_eranzamin_succession pattern:
# the claim of Konya legitimates war on the recalcitrant marches, and
# ONLY there — the custodian of Rum has no writ over Syria.

cb_rum_custody = {

	create_visible = {
		has_variable = rum_old_capital
		scope:target = {
			OR = {
				has_reform = government_reform:anatolian_beylik
				capital ?= { region = region:anatolia_region }
			}
		}
	}

	declare_enabled = {
		has_variable = rum_old_capital
		scope:target = {
			OR = {
				has_reform = government_reform:anatolian_beylik
				capital ?= { region = region:anatolia_region }
			}
		}
	}

	speed = 20

	war_goal_type = rum_custody_wargoal
}
