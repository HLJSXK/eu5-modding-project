# =====================================================================
# SHIA RELIGIOUS ASPECTS — doctrine as contested positions
#
# Three-layer architecture (see docs/design/shia_rebuild_plan.md Part A):
#   aspects  = doctrinal commitments of the state (this file)
#   schools  = the establishment/network running religious institutions
#   policies = madhhab/fiqh (the sharia_law policy layer, base game)
#
# Axis 1 — IMAMATE (mutually exclusive; the constitutive question).
#   A state may hold NONE: that is post-Mongol confessional ambiguity
#   (Pfeiffer), and it is the 1337 default for most courts.
# Axis 2 — OCCULTATION POLITICS (Twelver-gated).
# Axis 3 — AUTHORITY OF THE JURIST (Twelver-gated).
# Axis 4 — POSTURE TOWARD THE MAJORITY.
#
# Icons reuse base aspect keys via icon= (no custom dds needed).
# =====================================================================

# The manifestation: the ruler IS the Mahdi (or His visible vessel) —
# fabricated sayyid nasab and all. Ibn Falah 1436, Nurbakhsh 1423,
# Ismaʿil's ghuluww poetry. Enormous zeal, catastrophic respectability,
# and a built-in crisis: the Mahdi dies. Rival mahdis hate each other
# more than they hate the quietists.
messianism = {
	icon = baptism_by_fire
	religion = shia

	enabled = {
		has_religious_aspect = religious_aspect:imamate_ithna_ashari
		NOT = { has_religious_aspect = religious_aspect:intizar_quietism }
		NOT = { has_religious_aspect = religious_aspect:niyaba_amma }
		# An order (building-country) cannot claim the manifestation
		# without a circulated sayyid nasab — spread_sayyid_genealogy
		# (Two Pirs ladder) licenses the claim. States claim freely:
		# their genealogists were already on the payroll.
		OR = {
			NOT = { country_type = building }
			has_variable = sayyid_genealogy_established
		}
	}

	modifier = {
		land_morale_modifier = 0.10
		global_pop_conversion_speed_modifier = 0.20
		stability_cost_efficiency = -0.10
		monthly_towards_spiritualist = societal_value_monthly_move
	}

	opinions = {
		messianism = -20
		intizar_quietism = -15
		niyaba_amma = -15
		taqiyya = -10
	}
}

# --------------------------------------------------------------------
# AXIS 2b — THE ZAYDI ROAD (no occultation, so no occultation politics:
# the activist counterpart slot for those who follow the risen Imam)
# --------------------------------------------------------------------

# Qiyam bi'l-Sayf — the rising by the sword. Zaydi doctrine makes the
# imamate itself conditional on it: the Imam is whichever qualified
# sayyid RISES against unjust rule; a claimant who sits at home
# forfeits the title. Doctrinally incompatible with taqiyya — the
# Zaydis' rejection of dissimulation is precisely what made them the
# permanently-insurgent branch.
qiyam_bil_sayf = {
	icon = war_sermons
	religion = shia

	enabled = {
		has_religious_aspect = religious_aspect:imamate_zaydi
		NOT = { has_religious_aspect = religious_aspect:taqiyya }
		# Same nasab gate as messianism: an order claims khuruj only
		# with the genealogy in circulation — the imamate is sayyid
		# property, and the Two Pirs ladder is where orders forge it.
		OR = {
			NOT = { country_type = building }
			has_variable = sayyid_genealogy_established
		}
	}

	modifier = {
		global_manpower_modifier = 0.10
		land_morale_modifier = 0.05
		monthly_towards_mysticism = societal_value_monthly_move
	}

	opinions = {
		qiyam_bil_sayf = 10
		intizar_quietism = -15
		taqiyya = -10
	}
}
