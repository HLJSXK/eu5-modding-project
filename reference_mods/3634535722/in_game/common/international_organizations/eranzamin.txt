# Erānzamīn — the operative IO for the post-Ilkhanid succession contest.
#
# Replaces the base ilkhanate IO as the real driver. Design:
#   -Flat federation: no leader, no engine-driven claimant promotion.
#   -Membership confers the "contender" passive (lower AE impact,
#     cheaper integration, modest army morale, increased unrest from
#     the contested legitimacy).
#   -Members can leave voluntarily (abandon claim).
#   -The Erānzamīn succession *situation* owns lifecycle and victory;
#     this IO auto-disbands when the situation flips its resolved flag.
#
# Starting members are seeded in main_menu/setup/start/ via a mod-side
# 17_eranzamin_organization.txt (created separately). The roster mirrors
# the base ilkhanate setup plus Sarbedars, minus a few tags that don't
# belong in the Persianate-succession frame.

eranzamin = {

	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no
	max_circles_at_formation = 10
	# Flat federation: no leader country, no claimant promotion.
	has_leader_country = no

	only_leader_country_joins_defensive_wars = no
	gives_military_access_to_all_when_at_war = no

	can_declare_war = {
		always = yes
	}

	# The "I am pursuing the throne of Erānzamīn" passive, applied to
	# every member. Eases conquest within the contested region but the
	# realm is volatile.
	modifier = {
		global_war_score_efficiency = 0.33
		antagonism_received_modifier = -0.50
		diplomatic_annexation_cost = -0.25
		diplomatic_annexation_efficiency = 0.25
		pop_join_rebel_threshold = 0.1
		land_morale_modifier = 0.10
	}

	# Player-creation is gated off — the IO is seeded at game start.
	create_visible_trigger = {
		always = no
	}

	# Members in the Erānzamīn region can extend invitations to their
	# fellow regional powers. Gated on capital, not on owning a province,
	# because the latter let Mamluk-style outsiders qualify off a single
	# foothold in Anatolia or the Crescent.
	invite_visible_trigger = {
		scope:target ?= {
			capital = { is_eranzamin_region_province = yes }
		}
		
	}

	# Organic joins: tier >= duchy, capital inside the region, sovereign.
	# Capital-based — not any_owned_location — so Mamluks (Cairo) and
	# similar adjacent powers can't slip in by holding a border province.
	can_join_trigger = {
		country_rank_level > 1

		capital = { is_eranzamin_region_province = yes }
		
		is_subject = no
	}

	# Members cannot voluntarily abandon the claim.
	can_leave_trigger = {
		always = no
	}
	
	auto_leave_trigger = {
		always = no
	}

	# Disband when the succession situation resolves (path-victory or
	# timeout). The situation sets eranzamin_succession_resolved in
	# its on_ended.
	auto_disband_trigger = {
		has_global_variable = eranzamin_succession_resolved
	}

	on_joined = {
	}

	on_left = {
	}

	# No engine cadence — the situation handles monthly logic.
	monthly_effect = {
	}

	variables = {
	}

	special_statuses_implemented = {
	}

	ai_desire_to_join = {
		add = {
			desc = "PERSIANATE_LEGITIMACY"
			value = 50
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}
