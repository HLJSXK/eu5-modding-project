# Inert override of the base game's ilkhanate IO.
#
# The base IO has 9 hardcoded references in the game (custom localization,
# formable_countries, generic_actions, peace_treaties, casus_belli,
# special_statuses, AI lists, flavor_chb event, and the IO file itself).
# We can't delete it without breaking those references, but we don't want
# its mechanics either:
#   -block_from_change_to_empire_rank blocks members from ranking up
#   -can_leave_trigger=always=no traps members forever
#   -auto_disband_trigger keyed to the claimant special status
#   -monthly_effect auto-promotes a sole claimant to leader
# All of those fight the Erānzamīn design (flat federation, path-driven
# victory, no engine-owned claimant promotion).
#
# This file replaces the base ilkhanate with an inert shell: file present,
# key resolves, special_status type still declared, but no mechanics.
# The real work happens in international_organization:eranzamin.

ilkhanate = {

	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no

	has_leader_country = yes
	leader_title_key = "ILKHANATE_LEADER"
	leader_type = character
	leader_change_trigger_type = none
	leader_change_method = none

	can_lead_trigger = {
		always = no
	}

	# No defensive-war auto-call. Inert.
	only_leader_country_joins_defensive_wars = yes

	can_declare_war = {
		always = no
	}

	# No member-wide modifier. Removing block_from_change_to_empire_rank
	# is the most important effect of this override.
	modifier = {
	}

	leader_modifier = {
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_join_trigger = {
		always = no
	}

	# Members can leave freely.
	can_leave_trigger = {
		always = yes
	}

	auto_leave_trigger = {
		always = no
	}

	# Do NOT auto-disband. Erānzamīn-side logic governs the lifecycle.
	auto_disband_trigger = {
		always = no
	}

	on_joined = {
	}

	on_left = {
	}

	# No monthly claimant promotion. The Erānzamīn situation tracks
	# contenders; this IO is a dead-letter shell.
	monthly_effect = {
	}

	variables = {
	}

	# Keep the special status declared so the 9 base references resolve.
	# Never granted, never read — just present for the schema.
	special_statuses_implemented = {
		ilkhan_claimant
	}

	ai_desire_to_join = {
	}

	ai_desire_to_allow_new_member = {
	}
}
