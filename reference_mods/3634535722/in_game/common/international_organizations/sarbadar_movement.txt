# =====================================================================
# THE SARBADĀR MOVEMENT — the vow as an organization
#
# The movement is prior to the state: the sworn network exists whether
# one city holds it or ten do. Modeled structurally on the mod's
# eranzamin IO (flat federation, no leader, seeded at start).
#
# Design (sarbadar_commonwealth_plan.md §5): the commonwealth starts
# DECENTRALIZED — SRB is the movement's only founding member, but the
# vessel is built for growth: every new commonwealth born of a rising
# (tariqa khurūj landings, the rebel fork) joins at formation. The
# decentralized endings (Covenant of Chapters, Orders of the Sword,
# Islamic Commonwealth, United Emirates) inherit and transform this
# IO; the centralized endings dissolve it into the unitary state via
# the sarbadar_movement_dissolved global variable.
#
# A member that stops being a commonwealth (e.g. martyrs.4.a
# normalization into a sultanate) falls out automatically — the
# movement is the government form's fellowship, not a diplomatic bloc.
# =====================================================================

sarbadar_movement = {

	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no
	max_circles_at_formation = 10
	# Flat: the chapters acknowledge no senior by right. Seniority
	# contests are ending-content, not day-1 structure.
	has_leader_country = no

	only_leader_country_joins_defensive_wars = no
	# The brethren open their roads to each other.
	gives_military_access_to_all_when_at_war = yes

	can_declare_war = {
		always = yes
	}

	# Membership passive: the network carries the cause. Light on
	# purpose — the movement's real payoffs arrive with the Reckoning.
	modifier = {
		monthly_devotion = 0.02
		land_morale_modifier = 0.05
	}

	# Seeded at game start; never player-created.
	create_visible_trigger = {
		always = no
	}

	# Members may extend the oath to fellow commonwealths.
	invite_visible_trigger = {
		scope:target ?= {
			government_type = government_type:commonwealth_of_martyrs
		}
	}

	# Only commonwealths of the vow belong.
	can_join_trigger = {
		government_type = government_type:commonwealth_of_martyrs
		is_subject = no
	}

	# The vow is not resigned. Normalization (auto_leave) or the
	# movement's own dissolution are the only exits.
	can_leave_trigger = {
		always = no
	}

	# A member that stops being a commonwealth has already left in
	# every way that matters.
	auto_leave_trigger = {
		NOT = {
			government_type = government_type:commonwealth_of_martyrs
		}
	}

	# The centralized endings dissolve the movement into the unitary
	# state (future Reckoning content sets the flag).
	auto_disband_trigger = {
		has_global_variable = sarbadar_movement_dissolved
	}

	on_joined = {
	}

	on_left = {
	}

	monthly_effect = {
	}

	variables = {
	}

	special_statuses_implemented = {
	}

	ai_desire_to_join = {
		add = {
			desc = "THE_VOW_BINDS_US"
			value = 100
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 50
		}
	}
}
