# =====================================================================
# THE CORONATION HOOK — the khurūj ends where the great cities are
#
# A rising order (khuruj_rising_reform) that takes any of the six
# metropoles of Erānzamīn — Tabriz, Isfahan, Baghdad, Nishapur,
# Shiraz, Herat — is offered its coronation (tariqa.6): the army
# becomes a kingdom, the pir becomes a crowned murshid.
#
# Engine scopes per 14_martyrs_conquest: root = location,
# scope:winner = new owner. The conquered city is passed to the event
# through a variable (scopes do not survive the event boundary).
# =====================================================================

on_location_changed_owner = {
	on_actions = {
		khuruj_coronation_hook
	}
}

khuruj_coronation_hook = {
	trigger = {
		scope:winner ?= {
			has_reform = government_reform:khuruj_rising_reform
			NOT = { has_variable = tariqa_crowned_murshid }
		}
		OR = {
			this = location:tabriz
			this = location:isfahan
			this = location:baghdad
			this = location:nishapur
			this = location:shiraz
			this = location:herat
		}
	}
	effect = {
		scope:winner ?= {
			set_variable = {
				name = khuruj_coronation_city
				value = root
			}
			trigger_event_non_silently = tariqa.6
		}
	}
}
