# The amirs' iqṭāʿ demand (iqta.1) — yearly check behind a cheap gate;
# the event's own trigger handles the count and the cooldown.
# Extends base's yearly_country_pulse (merges).

yearly_country_pulse = {
	on_actions = {
		yearly_iqta_demand_pulse
	}
}

yearly_iqta_demand_pulse = {
	trigger = {
		government_type = government_type:persianate_sultanate
		country_has_estate = estate_type:amirs_estate
	}
	events = {
		iqta.1
	}
}
