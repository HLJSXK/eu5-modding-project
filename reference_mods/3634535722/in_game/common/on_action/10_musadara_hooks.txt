# =====================================================================
# MUṢĀDARA HOOKS — the desperation prompt on the loan on_actions
#
# Engine on_actions (re-declared here; Paradox merges, so this EXTENDS
# rather than overrides). Scopes per the engine:
#   root = country, scope:target = the loan, scope:owner = the lender.
#
# When a realm that holds the right of muṣādara signs or renews a loan
# while drowning — more than ten loans and bleeding income, or a loan
# capacity that no longer covers a year of revenue — the crown is offered
# the desperation confiscation (musadara.1). A 1-year cooldown variable
# (set in the event's immediate) keeps a flurry of loans from spamming it.
# =====================================================================

on_loan_taken = {
	effect = {
		if = {
			limit = {
				modifier:access_to_musadara = yes
				NOT = { has_variable = musadara_desperation_recent }
				OR = {
					AND = {
						num_loans > 10
						monthly_income_total < 0
					}
					country_loan_capacity < {
						value = monthly_income_total
						multiply = 12
					}
				}
			}
			trigger_event_non_silently = { id = musadara.1 }
		}
	}
}

on_loan_renewed = {
	effect = {
		if = {
			limit = {
				modifier:access_to_musadara = yes
				NOT = { has_variable = musadara_desperation_recent }
				OR = {
					AND = {
						num_loans > 10
						monthly_income_total < 0
					}
					country_loan_capacity < {
						value = monthly_income_total
						multiply = 12
					}
				}
			}
			trigger_event_non_silently = { id = musadara.1 }
		}
	}
}
