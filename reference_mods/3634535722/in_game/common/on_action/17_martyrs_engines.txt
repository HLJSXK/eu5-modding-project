# =====================================================================
# Commonwealth of Martyrs — devotion engines
#
# 1. MARTYRS' ROLLS (privilege-gated): the Karbala paradigm as a
#    loop — the commonwealth gains devotion from its own great
#    battles against the Turco-Mongol enemy, and MORE from the lost
#    ones than the won ones. The gallows keeps its promise; losses
#    are never wasted. Battle scopes per 06_ghazi_hooks:
#    scope:actor = own unit, scope:target = enemy unit.
#
# 2. CONQUEST DEVOTION: every location taken is the vow vindicated —
#    the metronome of the coming Reckoning.
#
# 3. MARTYRS FLAVOR PULSE: commonwealth branch of the government
#    flavor pulse (wired in 03_persianate_flavor_pulse's first_valid
#    list — the sultanate gate fails for the commonwealth and falls
#    through to this one). Hosts the ʿĀshūrāʾ commemoration.
# =====================================================================

on_great_battle_won = {
	on_actions = {
		martyrs_rolls_battle_won
	}
}

on_great_battle_lost = {
	on_actions = {
		martyrs_rolls_battle_lost
	}
}

on_location_changed_owner = {
	on_actions = {
		martyrs_conquest_devotion
	}
}

martyrs_rolls_battle_won = {
	trigger = {
		scope:actor.owner ?= {
			government_type = government_type:commonwealth_of_martyrs
			has_estate_privilege = estate_privilege:shaykhi_martyrs_rolls
		}
		scope:target.owner ?= {
			# unit scope -> owner: is_rebel_country is a country trigger.
			is_rebel_country = no
			is_turco_mongol_polity = yes
		}
	}
	effect = {
		scope:actor.owner = {
			add_devotion = 2
		}
	}
}

martyrs_rolls_battle_lost = {
	trigger = {
		scope:actor.owner ?= {
			government_type = government_type:commonwealth_of_martyrs
			has_estate_privilege = estate_privilege:shaykhi_martyrs_rolls
		}
		scope:target.owner ?= {
			# unit scope -> owner: is_rebel_country is a country trigger.
			is_rebel_country = no
			is_turco_mongol_polity = yes
		}
	}
	effect = {
		scope:actor.owner = {
			# The fallen are entered in the rolls. A lost field fills
			# more pages than a won one.
			add_devotion = 4
		}
	}
}

martyrs_conquest_devotion = {
	trigger = {
		scope:winner ?= {
			government_type = government_type:commonwealth_of_martyrs
		}
	}
	effect = {
		scope:winner ?= {
			add_devotion = 1
		}
	}
}

martyrs_flavor_pulse = {
	trigger = {
		government_type = government_type:commonwealth_of_martyrs
	}

	random_events = {
		# 100 = 0 keeps the pulse mostly silent on any given tick.
		100 = 0

		10 = martyrs.10   # ʿĀshūrāʾ (1-year cooldown in-event)
	}
}
