# Seeds initial characters into the mod's custom estates at game start so they
# can exert power and provide bonuses before natural promotion + generation
# catches up. Without this, estates start empty for the first few years.
#
# -Persianate Sultanate: 2 diwan + 2 amir characters
# -Commonwealth of Martyrs: 2 sufyani notables + 2 shaykhi dervishes

on_game_start = {
	on_actions = {
		on_seed_mod_estates
	}
}

on_seed_mod_estates = {
	effect = {
		# Persianate Sultanate estate seeding
		every_country = {
			limit = {
				government_type = government_type:persianate_sultanate
			}
		
			# Diwan — Persian scribal class, hardcoded farsi regardless of country culture
			create_character = {
				estate = estate_type:diwan_estate
				age = { min = 30 max = 50 }
				culture = culture:farsi_culture
				religion = root.religion
				adm = { 50 90 }
				dip = { 40 70 }
				mil = { 10 40 }
			}
			create_character = {
				estate = estate_type:diwan_estate
				age = { min = 20 max = 30 }
				culture = culture:farsi_culture
				religion = root.religion
				adm = { 40 80 }
				dip = { 40 70 }
				mil = { 10 40 }
			}
			# Amirs — military commanders; culture follows the country (Turco-Mongol
			# for CHB/JAL/ERE, Lur for HZP/LCK, Kurdish for SHB, etc.)
			create_character = {
				estate = estate_type:amirs_estate
				age = { min = 30 max = 50 }
				culture = root.culture
				religion = root.religion
				adm = { 10 40 }
				dip = { 40 70 }
				mil = { 50 90 }
			}
			create_character = {
