# Persianate flavor pulse — hosts the non-DHE generic and eranzamin flavor
# events. Piggybacks on religion_flavor_pulse so it runs on the engine's
# religion-pulse cadence (which the muslim sub-pulse already uses).
#
# Per-event triggers do the heavy filtering (capital region, culture group,
# tag, estate availability, etc.); this outer trigger is just a cheap
# religion gate so the engine can skip the whole pulse for 80% of countries.

government_flavor_pulse = {
	first_valid_on_action = {
		persianate_flavor_pulse
		# Falls through when the sultanate gate fails — the commonwealth
		# branch (defined in 17_martyrs_engines.txt).
		martyrs_flavor_pulse
	}
}

persianate_flavor_pulse = {
	trigger = {
		government_type = government_type:persianate_sultanate
	}

	random_events = {
		# 100 = 0 keeps the pulse mostly silent on any given tick; raise
		# event weights or lower this empty option to make Persianate
		# flavor fire more often.
		100 = 0

		10 = persianate_generic.4   # Year of Plague (weight_multiplier boosts in 1346-1500 window)
		10 = persianate_generic.5   # Caravanserai Project
		10 = persianate_generic.6   # Pilgrim Caravan

		# Erānzamīn flavor — non-DHE subset (DHE-gated ones — Hafez,

		10 = eranzamin_flavor.4     # Yasa or Sharia

	}
