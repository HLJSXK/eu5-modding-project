# =====================================================================
# IRANIC TRIBAL POLITY — the great-city conversion hook
#
# When a mîrnishîn seizes one of the great seats of Iranian/Iraqi empire, the
# chancery comes within its grasp and it is offered the road to becoming a
# Persianate Sultanate. Wired onto the engine's owner-change hook; re-declaring
# on_location_changed_owner EXTENDS it (Paradox merges on_actions).
#
# Engine-provided scopes on this on_action:
#   root        = the location whose owner changed
#   scope:winner = the new owner
#   scope:loser  = the previous owner
# =====================================================================

on_location_changed_owner = {
	on_actions = {
		iranic_tribal_great_city_conquest
	}
}

iranic_tribal_great_city_conquest = {
	trigger = {
		scope:winner ?= { government_type = government_type:iranic_tribal_polity }
		# root = the conquered location. One of the great seats of empire.
		OR = {
			this = location:tabriz
			this = location:isfahan
			this = location:shiraz
			this = location:nishapur
			this = location:baghdad
			this = location:herat
		}
	}
	effect = {
		scope:winner ?= {
			trigger_event_non_silently = { id = kurdish_emirate.1 }
		}
	}
}
