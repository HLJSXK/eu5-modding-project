on_game_start = {
	on_actions = {
    	on_diwan_start
    }
}

on_diwan_start = {
	effect = {
		# Set the global flag that gates eranzamin_succession's can_start.
		# The situation's on_start handles contender flagging, warlord
		# modifier grant, and flavor.1 dispatch — no per-country setup
		# events needed here.
		set_global_variable = start_eranzamin_succession
		activate_situation = situation:eranzamin_succession
		activate_situation = situation:sarbadar_phenomenon
		trigger_event_non_silently = sarbadar_phenomenon.1
		# Fire the one-shot cleanup pass (kill non-_pw character duplicates,
		# revoke base communal_lands where Boneh System should rule).
		# Needs a country root for the event scope; CHB is always present at
		# game start and the character kills inside are absolute-scoped.
		add_damaged_qanat = yes
		random_country = {
			limit = { exists = this }
			trigger_event_silently = { id = persianate_cleanup.1 }
		}
		international_organization:eranzamin = {
			every_international_organization_member = {
				every_owned_location = {
					limit = {
						OR = {
						vegetation = farmland
						vegetation = qanat_farmland
						vegetation = seasonal_irrigated
						vegetation = crumbling_qanat
						raw_material ?= { is_food = yes } 
						}

					}
					construct_building = {
						cost_multiplier = 0
						cost_multiplier_reason = "game_concept_event"
						instant = yes
						building_type = building_type:farming_village
					}
					
					
				}
				every_owned_location = {
					limit = {
						OR = {
						vegetation = farmland
						vegetation = qanat_farmland
						vegetation = seasonal_irrigated
						raw_material ?= { is_food = yes } 
						}

					}
					construct_building = {
						cost_multiplier = 0
						cost_multiplier_reason = "game_concept_event"
						instant = yes
						building_type = building_type:farming_village
					}
					change_max_raw_material_workers = 1
				}
				every_owned_location = {
					limit = {
						OR = {
						vegetation = farmland
						vegetation = seasonal_irrigated
						raw_material ?= { is_food = yes } 
						}

					}
					construct_building = {
						cost_multiplier = 0
						cost_multiplier_reason = "game_concept_event"
						
						instant = yes
						building_type = building_type:farming_village
					}
					change_max_raw_material_workers = 1
				}
				every_owned_location = {
					limit = {
						OR = {
						vegetation = farmland
						raw_material ?= { is_food = yes } 
						}

					}
					construct_building = {
						cost_multiplier = 0
						cost_multiplier_reason = "game_concept_event"
						
						instant = yes
						building_type = building_type:farming_village
					}
					change_max_raw_material_workers = 1
				}
				every_owned_location = {
					limit = {
						NOT = {has_building = building_type:damaged_qanat}
						vegetation = qanat_farmland

					}
					construct_building = {
						cost_multiplier = 0
						cost_multiplier_reason = "game_concept_event"
						
						instant = yes
						building_type = building_type:qanat_network
					}
				}
				every_owned_location = {
					limit = {
						OR = {
						vegetation = seasonal_irrigated
						vegetation = farmland
							}

					}
					construct_building = {
						cost_multiplier = 0
						cost_multiplier_reason = "game_concept_event"
						
						instant = yes
						building_type = building_type:irrigation_systems
					}
				}

			}
		}
	}
}
