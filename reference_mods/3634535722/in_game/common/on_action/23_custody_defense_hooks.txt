# =====================================================================
# THE GOVERNORATE CALLS — defense of sworn governorates as a request
#
# ulus_governorate no longer auto-drags its custodian into wars
# (overlord_protects_external = no). Instead, when a sworn governorate
# is declared on by anyone but its own custodian, the custodian gets
# custody.35 and decides.
#
# Engine scopes (base _hardcoded header): scope:actor = declarer,
# scope:recipient = declared-on, scope:war = the war.
# =====================================================================

on_war_declared = {
	on_actions = {
		custody_ulus_defense_request
	}
}

custody_ulus_defense_request = {
	trigger = {
		scope:recipient ?= {
			is_subject_type = ulus_governorate
			overlord ?= {
				NOT = { this = scope:actor }
				# Already in this fight (attacked alongside the
				# governorate) — nothing to ask.
				NOT = { is_at_war_with = scope:actor }
			}
		}
	}
	effect = {
		scope:recipient = { save_scope_as = sworn_governorate }
		scope:actor = { save_scope_as = ulus_attacker }
		scope:war = { save_scope_as = ulus_war }
		scope:recipient.overlord = {
			trigger_event_non_silently = custody.35
		}
	}
}
