# Custody of the Ulus — kill credit.
#
# Paradox merges on_actions across files, so re-declaring the engine
# on_military_annex here EXTENDS it (does not override). Fired before
# the annexation: root = the annexer, scope:target = the annexed.
#
# If a steward running the custody disaster annexes a country that was
# a rival custody (its own regency fiction, or a Borjigin sovereign),
# that is a kill — it unlocks the fiction-apparatus gates ahead of the
# passive rivals-remaining thresholds. Design:
# docs/design/custody_of_the_ulus_design.md

on_military_annex = {
	effect = {
		if = {
			limit = {
				has_variable = custody_fired
				NOT = { has_variable = custody_completed }
				scope:target = { custody_is_rival = yes }
			}
			change_variable = { name = custody_kills add = 1 }
		}
	}
}
