# The dīwān's ultimatum (diwan_vizierate.1) — yearly check, cheap
# outer gate so the ~290 non-persianate countries never evaluate the
# event trigger. Extends base's yearly_country_pulse (merges).

yearly_country_pulse = {
	on_actions = {
		yearly_diwan_vizierate_pulse
	}
}

yearly_diwan_vizierate_pulse = {
	trigger = {
		government_type = government_type:persianate_sultanate
		country_has_estate = estate_type:diwan_estate
	}
	events = {
		diwan_vizierate.1
	}
}
