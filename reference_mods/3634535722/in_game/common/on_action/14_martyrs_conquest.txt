# =====================================================================
# COMMONWEALTH OF MARTYRS — the two conquests that matter
#
# Mashhad: the shrine of the eighth Imam, ʿAlī al-Riḍā — the only Imam
#   buried in Iranian soil. For a polity minting coin in the Hidden
#   Imam's name, taking the Threshold is a constitutional event.
#   (Historically: ʿAlī-yi Muʾayyad took Mashhad and made the shrine
#   a pillar of Sarbadār legitimacy.)
# Nishapur: the old metropolis of Khurasan — courts, ledgers, the
#   notables' prize. (Historically: Wajīh al-Dīn Masʿūd took it in
#   the polity's first expansion.)
#
# Wired onto the engine's owner-change hook; re-declaring
# on_location_changed_owner EXTENDS it (Paradox merges on_actions).
# Engine-provided scopes: root = location, scope:winner = new owner,
# scope:loser = previous owner. Once-only via winner variables.
# =====================================================================

on_location_changed_owner = {
	on_actions = {
		martyrs_shrine_city_conquest
	}
}

martyrs_shrine_city_conquest = {
	trigger = {
		scope:winner ?= { government_type = government_type:commonwealth_of_martyrs }
		OR = {
			AND = {
				this = location:mashhad
				scope:winner ?= { NOT = { has_variable = martyrs_mashhad_fired } }
			}
			AND = {
				this = location:nishapur
				scope:winner ?= { NOT = { has_variable = martyrs_nishapur_fired } }
			}
		}
	}
	effect = {
		if = {
			limit = { root = location:mashhad }
			scope:winner ?= {
				set_variable = martyrs_mashhad_fired
				trigger_event_non_silently = martyrs.2
			}
		}
		if = {
			limit = { root = location:nishapur }
			scope:winner ?= {
				set_variable = martyrs_nishapur_fired
				trigger_event_non_silently = martyrs.3
			}
		}
	}
}
