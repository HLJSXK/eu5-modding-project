# =====================================================================
# IRANIC TRIBAL POLITY — the mîrnishîn government form
#
# A hereditary mountain principality on a tribal-confederate base: the Kurdish
# emirate and its kin. Persianate in CULTURE (chancery Persian, court forms),
# but NOT in STRUCTURE — there is no dīwān, no surveyed fisc, no iqṭāʿ apparatus
# here. The mîr rules through tribal aghas and valley lords who keep their own
# autonomy. Durable on the defensive, inert on the offensive (see the
# kurdish_emirate reform + the kurdish_emirate_defensive_levy auto_modifier).
#
# Converts UP to persianate_sultanate via kurdish_emirate.1 if the mîr seizes a
# great seat of empire and takes up the chancery.
# =====================================================================

iranic_tribal_polity = {
	use_regnal_number = yes
	generate_consorts = yes

	# Hereditary princely rule, contested within the ruling house — the chronic
	# fraternal succession strife the suzerain empires always exploited.
	heir_selection = tribal_oldest_male
	heir_selection = tanistry_elective
	heir_selection = ritual_selection
	heir_selection = matrilineal_non_exclusive

	map_color = gov_tribe

	government_power = tribal_cohesion

	# No dīwān here — the ruling class is the tribal-military aristocracy.
	default_character_estate = nobles_estate

	modifier = {
		care_about_producing_heirs = yes
		# Thin chancery, light footprint — survives as a nominal vassal of
		# whichever empire is ascendant.
		diplomatic_upkeep_modifier = -0.5
	}
}
