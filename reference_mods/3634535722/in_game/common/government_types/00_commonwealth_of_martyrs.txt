# -----------------------------------------------------------------------------
# Commonwealth of Martyrs
#
# Government type for the Sarbedārān of Sabzevar (1337-1381) and any later
# millenarian Shīʿī polity in their lineage (Mushaʿshaʿ-type, Nūrbakhshī,
# Marʿashī, etc.). The name preserves the sar-be-dār ("head on the gallows")
# martyrdom-vow that the dynastic name itself constitutes, and travels cleanly
# from the Karbalāʾ paradigm through Hurūfī/Nūrbakhshī persecution to the
# Qizilbāsh ghāzī-martyr ethos.
#
# Structural model: dyarchic commonwealth oscillating between
#   -Sufyānī notables (urban-aristocratic, malik-led, conservative)
#   -Shaykhī dervishes (armed religious order, shaykh-led, mahdist-militant)
# with the Hidden Imām as absent sovereign — coinage minted in His name as
# the constitutional claim of general vicariate (niyāba ʿāmma).
# -----------------------------------------------------------------------------

commonwealth_of_martyrs = {
	# Sarbedār rulers did not use regnal numbering — the Imām is the only sovereign.
	use_regnal_number = no

	# Dyarchic election is the historical Sufyānī/Shaykhī structure. Other
	# options cover periods of factional dominance and military regency.
	# (Custom selection — base diarchic_election is Genoa's and draws from
	# nobles/burghers, which the commonwealth suppresses/lacks.)
	heir_selection = martyrs_dyarchic_election
	heir_selection = oligarchic_elective
	heir_selection = theocratic_elective
	heir_selection = judicial_election
	heir_selection = military_dictatorship
	heir_selection = general_heir_selection
	heir_selection = matrilineal_non_exclusive

	map_color = gov_theocracy

	# Devotion is the right resource: Sarbedārī legitimacy IS religious
	# devotion to the Hidden Imām + the Karbalāʾ paradigm + the martyr-ethos.
	government_power = devotion

	default_character_estate = sufyani_notables_estate

	modifier = {
		# The court speaks the language of jurists and dervishes.
		court_language_is_liturgical_language_importance_modifier = 8

		# Awaiting the Mahdī: the polity sees itself as ante-room, not endpoint.
		# Doesn't auto-convert subjects — Sarbedārs were militant against
		# specific aristocratic enemies, not a conversion regime.
	}
}
