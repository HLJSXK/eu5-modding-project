# ============================================================
# ERANZAMIN — Religious buildings, 1337 set
#
# Design logic: religious infrastructure is not neutral. The
# madrasa was invented (Nizam al-Mulk, 1060s) as the orthodox
# state's weapon against heterodoxy — pen-aligned. The  /
# astana is charisma infrastructure — the soil of the messianic
# pole. The building set therefore mirrors the diwan/iqta split
# on the religious plane:
#
#   madrasa + hisba_office   → din-panah / orthodox path
#   sufi_lodge → astana      → charismatic / messianic path
#   hawza + imam shrines     → the Shia pipeline (Hilla 1337 →
#                              Jabal Amil imports → Usuli mujtahids)
#
# Cross-system hooks (for the reform/action wiring):
#   - astana at Ardabil = the Safaviyya. Holding it should
#     discount / enable murshid_i_kamil.
#   - astana count/density could gate or cheapen
#     mushasha_messianism.
#   - madrasa + hisba presence are natural vindication targets
#     and prerequisites for the din-panah path.
#   - hawza count gates mujtahid_establishment (age 4) and
#     supplies the ulama for imami_tashayyu.
#
# Modifier names marked [verify] where the engine key is unknown.
# ============================================================


# --- Tier 2 sufi: the astana / great shrine-lodge complex ---
# Upgrade of the ordinary khanqah. Represents the mother lodges
# and saint-shrine complexes: Ardabil (Safaviyya), Chisht
# (Chishtiyya), Turbat-i Jam (Ahmad-i Jam), Bastam (Bayazid),
# Yasi (Ahmad Yasavi), Konya (Mevlevi). Waqf-endowed pilgrimage
# economies with a resident dervish population.
astana_complex = {
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	employment_size = 0.3 

	category = religious_category
	graphical_tags = { large_religious }

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	expensive = yes
	build_time = large_religious_building_time  

	country_potential = {
		religion.group = religion_group:muslim
	}

	allow = {
		owner ?= { country_has_sufism = yes }
		# Built on top of an existing lodge — the order matures in place.
		has_building = building_type:sufi_lodge
		# One per province: a mother lodge is a regional institution.
		custom_tooltip = {
			text = astana_one_per_province_tt
			location.province = {
				NOT = {
					any_location_in_province = {
						has_building = building_type:astana_complex
					}
				}
			}
		}
	}

	possible_production_methods = {
		astana_maintenance
	}

	modifier = {
		# The user-requested cap expansion: the shrine economy carries
		# a resident dervish/sayyid/servitor population beyond what the
		# location would otherwise sustain.
		local_clergy_desired_pop = 0.5             
		local_clergy_max_literacy = 10
		local_pop_conversion_speed_modifier = 0.20  # sufi conversion: the historical engine of Islamization
		# Pilgrimage economy — traffic, alms, the shrine bazaar.
		local_monthly_prosperity = 0.001            
		local_clergy_estate_power = 0.5            
	}

	construction_demand = town_building_construction
}


# --- The madrasa: legalist orthodoxy as infrastructure ---
# The Nizamiyya's descendant form, ubiquitous by 1337 (the Baghdad
# Nizamiyya itself still taught when Ibn Battuta passed through).
# Orthodox fiqh education: literacy, conversion *to the state
# religion*, and the qadi class. The counter-building to the
# astana on the heterodoxy axis.
REPLACE:madrasa = {
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	employment_size = 0.1

	category = religious_category
	graphical_tags = { medium_religious }

	town = yes
	city = yes
	megalopolis = yes
	# No rural_settlement — the madrasa is an urban institution.

	build_time = religious_building_time

	country_potential = {
		religion.group = religion_group:muslim
	}

	possible_production_methods = {
		madrasa_maintenance
	}

	modifier = {
		local_clergy_max_literacy = 15
		local_nobles_max_literacy = 5               # the qadi/katib overlap — madrasa graduates staff courts and chanceries
		local_pop_conversion_speed_modifier = 0.10

	}

	construction_demand = town_building_construction
}


# --- The hisba office: the muhtasib's jurisdiction ---
# Market-law enforcement: weights, prices, public morals. The
# building Mubariz al-Din made famous by closing the taverns of
# Shiraz (and that Hafiz spent a divan complaining about).
# Religious-legal control infrastructure with burgher friction —
# the enforcement arm of the din-panah path.
hisba_office = {
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	employment_size = 0.05

	category = religious_category
	graphical_tags = { small_religious }

	town = yes
	city = yes
	megalopolis = yes

	build_time = religious_building_time

	country_potential = {
		religion.group = religion_group:muslim
	}



	possible_production_methods = {
		hisba_maintenance
	}

	modifier = {
		local_max_control = 0.05                         
		local_pop_conversion_speed_modifier = 0.05
		local_burghers_estate_unrest = 0.05          
		local_clergy_estate_power = 0.15            
	}

	construction_demand = town_building_construction
}


# --- The hawza: Shia scholarly center ---
# In 1337 this means Hilla above all — twelve years after Allama
# al-Hilli, the systematizer of ijtihad and converter of Öljaitü.
# The intellectual pipeline that later supplies imami_tashayyu its
# ulama and mujtahid_establishment its mujtahids.
hawza_ilmiyya = {
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	employment_size = 0.2

	category = religious_category
	graphical_tags = { medium_religious }

	town = yes
	city = yes
	megalopolis = yes

	build_time = religious_building_time

	country_potential = {
		religion = religion:shia
	}

	allow = {
		# A hawza needs a Shia community to teach. 10% is the floor at
		# which the mujtahids have enough students and litigants to seat
		# a scholarly circle — below that, the lodge has no reason to be.
		custom_tooltip = {
			text = hawza_requires_shia_population_tt
			location = { "religion_percentage(religion:shia)" > 0.1 }
		}
	}

	possible_production_methods = {
		hawza_maintenance
	}

	modifier = {
		local_clergy_max_literacy = 20               # the most scholarly building in the set
		local_clergy_desired_pop = 0.3              
		local_pop_conversion_speed_modifier = 0.10
		local_clergy_estate_power = 0.05            
	}

	construction_demand = town_building_construction
}


# --- The imam shrine: the atabat and the great mashhads ---
# Unique, site-locked: Najaf, Karbala, Kazimayn, Samarra, Mashhad,
# Qom. Not buildable — these exist or they don't. Pilgrimage
# economies of a different magnitude than any astana, drawing
# traffic across confessional lines (Sunni rulers patronized the
# atabat throughout).
imam_shrine = {
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	employment_size = 0.5

	category = religious_category
	graphical_tags = { large_religious }

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	build_time = large_religious_building_time

	# Not buildable. The atabat and great mashhads exist or they don't —
	# Twelver imam-shrines are not a thing a state seats by fiat. Seeded
	# at the six site locations via main_menu/setup/start/. Modifiers and
	# upkeep apply normally once the building exists at a location.
	location_potential = {
		always = no
	}

	country_potential = {
		religion.group = religion_group:muslim
	}

	possible_production_methods = {
		imam_shrine_maintenance
	}

	modifier = {
		local_clergy_desired_pop = 1.0
		local_clergy_max_literacy = 15
		local_monthly_prosperity = 0.003
		local_pop_conversion_speed_modifier = 0.15
		local_clergy_estate_power = 1.0
	}
}
