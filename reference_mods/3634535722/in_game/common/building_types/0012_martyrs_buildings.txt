# ============================================================
# Commonwealth of Martyrs — the khānaqāh as war institution
#
# dervish_khanaqah : the Ḥasan Jūrī lodge — recruiting hall,
#   armory, and indoctrination house of the armed dervish order.
#   The historical Shaykhiyya organized craftsmen and villagers
#   through exactly this network (Sabzavār and its districts;
#   the Marʿashī and Samarqand replications ran on the same
#   institution). This is where the SALARIED army lives:
#   can_recruit_regiment_in_this_location is the whole point —
#   the commonwealth pays soldiers from confiscation revenue,
#   and the lodge is the mustering hall where pay meets zeal.
#
#   Modeled on mutatawwia (clergy-staffed military lodge) and
#   ghulam_barracks (the recruit-enabling salaried-corps
#   building).
# ============================================================

dervish_khanaqah = {
	is_foreign = no
	max_levels = 5
	# The lodge houses the armed brethren — soldiers pops, which are what
	# feed the shaykhi_dervishes_estate (the sober ulama stay clergy).
	pop_type = soldiers
	employment_size = 0.05

	category = military_category
	graphical_tags = { medium_religious }

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	build_time = religious_building_time

	country_potential = {
		government_type = government_type:commonwealth_of_martyrs
	}


	raw_modifier = {
		local_soldiers_desired_pop_scaled = 0.025 ### draws sworn brethren (soldiers pops = the dervish corps proper) to settle around the lodge.
	}
	modifier = {
		# The lodge feeds men into the corps and keeps a standing watch.
		local_manpower = 0.02
		local_garrison_size = 0.125
		# The lodge IS the Shaykhī order's local presence.
		local_shaykhi_dervishes_estate_power = 0.75
		# The salaried army musters here — this is what unlocks regular
		# recruitment for a polity with no barracks tradition.
		can_recruit_regiment_in_this_location = yes
	}

	unique_production_methods = {
		dervish_khanaqah_maintenance = {
			cloth = 0.1
			weaponry = 0.1
			category = building_maintenance
		}
	}
	construction_demand = town_building_construction
}
