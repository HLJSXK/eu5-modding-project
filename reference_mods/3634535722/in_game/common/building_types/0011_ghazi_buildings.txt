# ============================================================
# Ghazi — military infrastructure of the volunteer and the city
#
# Two buildings:
#   - mutatawwia    : the recruitment lodge — ribat-mosque-lodge
#                     hybrid that draws volunteer holy warriors
#                     (al-mutaṭawwiʿa, "those who present
#                     themselves"). Gated to dar_al_jihad: this is
#                     the institutional form behind the reform's
#                     recruitment ledger. Clergy-staffed; the
#                     muezzin, preacher, and ribat-shaykh are who
#                     keep volunteers fed and pious.
#   - zurkhanah     : the "house of strength" — Persian gymnasium
#                     and futuwwa brotherhood. Javānmardī ethic, the
#                     ʿayyārān tradition of urban militia, the same
#                     wrestling-and-zikr space that fed Sarbedar
#                     and later Safavid cadres. Urban only — the
#                     zurkhanah is a quarter-of-the-bazaar
#                     institution, not a village one.
#
# Both feed unit / levy work downstream.
# ============================================================


# --- The mutaṭawwiʿa lodge -----------------------------------
# Gated to dar_al_jihad. The reform names a recruitment ledger;
# this building is the place that ledger is kept. Clergy here are
# ribat-shaykhs, preachers, and the muezzin — they recruit, feed,
# and pray over volunteers who muster from the surrounding country.
# The garrison bonus represents the lodge's standing guard of
# muraābiṭūn: men who live at the ribat in rotation.
mutatawwia = {
	is_foreign = no
	max_levels = 1
	pop_type = clergy
	employment_size = 0.05

	category = military_category
	graphical_tags = { medium_religious }

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	build_time = religious_building_time

	country_potential = {
		has_reform = government_reform:dar_al_jihad
	}

	possible_production_methods = {
		mutatawwia_maintenance
	}

	modifier = {
		# The recruitment ledger: the lodge attracts a resident clerical
		# cadre — preachers, ribat-shaykhs, muezzins — that the location
		# would not otherwise sustain.
		local_clergy_desired_pop = 0.25
		local_manpower = 0.01
		# Standing rotation of muraābiṭūn at the lodge.
		local_garrison_size = 0.125
		## the clergy have an armed wing here, ofc it majorly increases their power.
		local_clergy_estate_power = 0.75
	}

	construction_demand = town_building_construction
}


# --- The zurkhanah -------------------------------------------
# The "house of strength". Wrestling pit, training floor, futuwwa
# brotherhood, javānmardī oath. Urban — a quarter institution, the
# kind of place ʿayyārān gathered in fourteenth-century Khorasan
# and Sarbedari mobilizers in Sabzavar. Burgher-staffed: the
# pahlavāns are sons of the bazaar, not peasants. Reduces unrest
# because the futuwwa code mediates quarter disputes; raises
# prosperity because the bazaar and the zurkhanah share an oath.
zurkhanah = {
	is_foreign = no
	max_levels = 1
	pop_type = burghers
	employment_size = 0.15

	category = military_category
	graphical_tags = { small_military }

	town = yes
	city = yes
	megalopolis = yes
	# No rural_settlement — the zurkhanah is a quarter institution.

	build_time = guild_build_time

	country_potential = {
		religion.group = religion_group:muslim
		OR = {
			AND = {
				modifier:has_nizam_ul_mulk_chancery = yes
				has_estate_privilege = estate_privilege:ayyar_militia_recognition
			}
			government_type = government_type:commonwealth_of_martyrs
		}

	}

	possible_production_methods = {
		zurkhanah_maintenance
	}

	modifier = {
		local_burghers_desired_pop_scaled = 0.01
		local_unrest = -1
		local_monthly_prosperity = 0.001
	}

	construction_demand = guild_construction
}
