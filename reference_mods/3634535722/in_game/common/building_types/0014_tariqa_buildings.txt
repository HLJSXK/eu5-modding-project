# =====================================================================
# TARIQA KHANAQAH — the order's lodge in another realm's city
#
# The extraterritorial building of a tariqa_order country (kontor
# chassis: is_foreign + need_good_relation + pop_size_created). The
# lodge houses a resident clergy pop linked to the ORDER — which is
# the scripting surface for everything downstream: the radicalization
# sweep is every_owned_foreign_building → linked_pop →
# change_pop_religion, verbatim the base Reformation trade-spread.
# =====================================================================

tariqa_khanaqah = {
	is_special = yes
	is_foreign = yes
	need_good_relation = yes
	pop_type = clergy

	category = religious_category
	# One of the great quiet functions of the orders: Islamization.
	# 1/10 the Sarbadar preachers' rate — the base game's own number
	# for sufi missionary work (sufi_mosque, unique_buildings.txt).
	conversion_religion = sunni

	max_levels = 3
	employment_size = 0.1
	pop_size_created = 0.05

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	increase_per_level_cost = 0.5

	build_time = religious_building_time

	country_potential = {
		modifier:allows_tariqa_buildings = yes
		# Once the recitation has changed, new lodges are radical ones.
		NOT = { is_radical_tariqa = yes }
	}


	modifier = {
		# The lodge answers to the pir before the governor — quietly.
		local_max_control = -0.1

		# What the HOST gets: the langar feeds the poor, the lodge
		# mediates quarter disputes, pilgrims and travelers spend.
		local_unrest = -0.1
		local_monthly_prosperity = 0.00125
		building_missionary_effort_scaled = 0.001

		# What the ORDER gets: the futuh — donations flowing monthly
		# to the shrine-seat, per lodge level. Reach funds reach.
		gold_to_building_owner = 1
	}

	unique_production_methods = {
		tariqa_khanaqah_maintenance = {
			cloth = 0.05
			category = building_maintenance
		}
	}
	construction_demand = khanqah_building_demand
}

	}

		tariqa_rural_zawiya_maintenance = {
			cloth = 0.05
			category = building_maintenance
		}
	}

}
