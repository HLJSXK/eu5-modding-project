# Matter of Tork, Approach 1 — the ghulam barracks.
#
# Modeled on the base mamluk_barracks, which is the inherited
# 'classical' institution (Mamluk Egypt et al. start with it).
# This is the NEW, purpose-built version — stronger, because it is
# designed from scratch around the corps rather than inherited.
#
# Gated by can_build_ghulam_barracks (granted by the diwan/amirs
# ghulam administration privileges and by ghulam_system).
# Corps tiers are gated on built barracks: tier II needs >= 1,
# tier III needs >= 3 (the physical apparatus, not an abstraction).

ghulam_barracks = {
	is_special = yes
	is_foreign = no
	pop_type = slaves

	max_levels = 3
	employment_size = slave_soldier_barracks

	category = military_category

	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes

	build_time = soldier_building

	expensive = yes

	country_potential = {
		modifier:can_build_ghulam_barracks = yes
	}

	always_add_demands = yes

	modifier = {
		local_manpower = 0.015
		can_recruit_regiment_in_this_location = yes
	}

	unique_production_methods = {
		ghulam_barracks_maintenance = {
			leather = 0.15
			cloth = 0.1
			weaponry = 0.15
			slaves_goods = 0.1
			category = building_maintenance
		}
	}
	construction_demand = soldier_building_construction
}
