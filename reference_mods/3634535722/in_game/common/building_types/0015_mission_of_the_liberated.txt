# =====================================================================
# MISSION OF THE LIBERATED — the commonwealth's owned-territory dawa
#
# The overt counterpart of sarbadar_preachers (which is the COVERT
# cell in other people's cities and dissolves under commonwealth
# rule). The commonwealth builds missions in its own conquered land:
# engine-side conversion at the sarbadar rate, plus a monthly
# scripted pass in 02_sarbadar_phenomenon that tips the LOWER ORDERS
# first — laborers, peasants, tribesmen. The revolution recruits from
# the bottom; the notables convert last and resentfully.
#
# No pop-condition allow gate (playtest: allow-with-any_pop blocked
# construction unpredictably); the building is simply useless where
# there is no one left to convert.
# =====================================================================

mission_of_the_liberated = {
	max_levels = 1
	pop_type = clergy
	employment_size = 0.05

	category = religious_category

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	conversion_religion = shia

	build_time = religious_building_time

	country_potential = {
		government_type = government_type:commonwealth_of_martyrs
	}

	# The mission is the commonwealth's organ: it does not survive
	# the commonwealth's rule ending here.
	remove_if = {
		NOT = {
			owner ?= {
				government_type = government_type:commonwealth_of_martyrs
			}
		}
	}

	modifier = {
		building_missionary_effort_scaled = 0.01
		# The mission feeds, arbitrates, and preaches: the quarter calms.
		local_unrest = -0.1
	}

	construction_demand = town_building_construction
}
