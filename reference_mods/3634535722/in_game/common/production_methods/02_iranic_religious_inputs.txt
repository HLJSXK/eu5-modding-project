# Maintenance production methods for the Iranic religious building set.
# Scaling notes:
#   imam_shrine ≈ central_diwan tier (the great mashhads run as miniature
#     state economies in their own right — Mashhad-i Rezā in Khurasan, the
#     atabat at Najaf/Karbala/Kazimayn/Samarra, the Qom shrine of Fatima
#     Maʿṣūma). Heavy on textiles for the drapery and khilʿa flow.
#   hawza ≈ local_diwan tier on the bookish axis (heavier on paper/books
#     than any other building in the set — the mujtahids run on copying
#     and commentary).
#   astana = mid-tier shrine-lodge economy; lighter than imam_shrine but
#     with a textile/banner load that madrasas lack.
#   madrasa = mid-tier scholarly; paper/books-heavy, light on the rest.
#   hisba = small office, paper-light, books-light, nothing else.
#
# No goods_gold anywhere — the Iranian economy does not have bullion access
# at this scale, and the waqf endowments that funded these institutions ran
# on grain, real estate, and bazaar rents rather than coin.

astana_maintenance = {
	paper      = 0.3   # lodge registers, copies of the order's rasaʾil and silsila
	books      = 0.2   # hagiographies, Maṯnawī, the Awrād of the order
	fine_cloth = 0.3   # khirqa robes for the murids, banners, shrine drapery
	furniture  = 0.1   # the lodge's audience hall, the shaykh's seat

	category = building_maintenance
}

madrasa_maintenance = {
	paper      = 0.4   # the lecture-and-copying economy — students copying the manual under the master
	books      = 0.3   # the madhhab's standard texts, the fiqh manuals
	fine_cloth = 0.1   # the mudarris's robe of office

	category = building_maintenance
}

hisba_maintenance = {
	paper      = 0.2   # the muḥtasib's registers, the market complaints
	books      = 0.1   # fiqh manuals for market law and public morals

	category = building_maintenance
}

hawza_maintenance = {
	paper      = 0.5   # the heaviest paper load in the set — ijtihad treatises, mujtahids' marginalia, students' transcripts
	books      = 0.4   # Hilli's al-Mukhtalaf, the kutub al-arbaʿa, the rijāl works
	fine_cloth = 0.1   # the marjaʿ's robe

	category = building_maintenance
}

imam_shrine_maintenance = {
	paper      = 0.4   # shrine registers, pilgrimage records, the mutawalli's accounts
	books      = 0.3   # ziyāra-manuals, the hadith of the imam, the histories of the shrine
	fine_cloth = 0.4   # the heaviest textile load in the set — drapery, the ḍariḥ's coverings, robes of the khuddām
	furniture  = 0.1   # the shrine's appointments
	marble     = 0.1   # ongoing maintenance of the dome, the iwans, the courtyards

	category = building_maintenance
}

mutatawwia_maintenance = {
	paper      = 0.2   # the recruitment ledger, the muster roll, the ribat-shaykh's correspondence
	books      = 0.1   # ḥadīth of jihad, the futuwwa manuals, the preacher's homiletic stock
	fine_cloth = 0.2   # banners, the volunteers' standard-issue cloak
	weapons    = 0.4   # the lodge's racks — the muraābiṭūn arm themselves from the ribat's stores

	category = building_maintenance
}

zurkhanah_maintenance = {
	fine_cloth = 0.1   # the pahlavān's tunbān, the murshid's robe
	weapons    = 0.2   # mils, kabādeh, the training-floor arms
	furniture  = 0.1   # the gowd, the sardam, the audience benches

	category = building_maintenance
}
