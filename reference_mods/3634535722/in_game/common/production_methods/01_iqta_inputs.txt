# Iqṭāʿ Grant — production method (upkeep).
#
# The grant funds two intertwined economies. First, the amir's household:
# the khilʿa-robed family in the citadel, the chancery correspondence with
# the diwan, the small luxuries of status. Second, the cavalry retinue:
# the herd, the lamellar, the sabres, the lance heads, the banners and
# felts of camp. Both come out of the parcel's diverted revenue.
#
# Scale: heavier than jurchen_barracks_maintenance (which is purely
# barracks), lighter than ghilman_barracks_maintenance (which is paying
# a standing professional corps). Iqṭāʿ sits in between: part-time
# tribal cavalry plus the amir's seat.

iqta_grant_maintenance = {
	# Cavalry retinue
	horses    = 0.1  # the herd that mounts the askar
	weaponry  = 0.2  # sabres, lance heads, bows, mail
	leather   = 0.2  # saddles, harness, lamellar
	cloth     = 0.2  # uniforms, banners, tent felts
	# Amir household
	fine_cloth = 0.1 # khilʿa robes, family dress
	paper      = 0.05  # chancery correspondence with the diwan
	
	category = building_maintenance
}
