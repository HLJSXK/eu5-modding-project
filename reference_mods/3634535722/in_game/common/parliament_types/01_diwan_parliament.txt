diwan_council = {
	type = country
	potential = {
		modifier:allow_council_parliament = yes
		government_type = government_type:persianate_sultanate
	}
	allow = { 
		NOT = { has_country_modifier = supreme_authority_modifier }
	}
	modifier = {
		has_a_parliamentary_system = yes
		
		# Participation. this is a court of law, not a parliament in the western sense. all may bring cases in front of the shadow of god.
		clergy_estate_can_participate_in_parliament = yes
		diwan_estate_can_participate_in_parliament = yes
		amirs_estate_can_participate_in_parliament= yes
		peasants_estate_can_participate_in_parliament = yes
		tribes_estate_can_participate_in_parliament = yes
		# Nobles participate by default. that is not correct. they DO NOT exist.
		nobles_estate_can_participate_in_parliament = no
		nobles_estate_blocked_from_parliament = yes
		# Agenda Impact is higher against elites.
		amirs_estate_agenda_impact = 0.25
		clergy_estate_agenda_impact = 0.25
		diwan_estate_agenda_impact = 0.1
	}
}
