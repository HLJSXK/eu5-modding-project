# ============================================================
# Farr Proof of Fortune — events
#
# Structure:
#   .1            announce (disaster fired)
#   .50/.65/.80/.90  apparatus (sikka, khutba, hunt+nawrūz, pādishāh)
#                     — claimed by the player via the disaster panel's
#                     action buttons (common/generic_actions/farr_proof_actions.txt)
#                     when legitimacy crosses each threshold
#   .35/.20/.10   erosion (amīrs absent, khutba dropped, bī-farr)
#                     — fire once on first downward threshold-touch
#   .100/.101/.102   omens (comet, earthquake, plague)
#                     — random pulses while disaster active; .102
#                     weight-multiplied during 1346–50 plague window
#   .200          Vindicated — Farr Manifest
#   .201          Broken-formal — Türk amīr usurpation
#   .202          Endured — quiet vindication at year 25
#   .203          Broken-lapse — doctrine lapses, prestige scar
#   .300–.306     INDICATOR events for the on-action hooks. Each
#                     fires every time its hook applies, until the
#                     player silences it. Option (b) sets
#                     farr_silence_<type> (per-type mute); option
#                     (c) sets farr_silence_all (master mute).
# ============================================================

# ---201 BROKEN-FORMAL — Türk amīr usurpation -----------------------
# The legitimacy floor breached. A Türk amīr seizes the throne;
# the realm survives, the doctrine does not. The reform is stripped
# and the government tilts to chinggisid_regency.
farr_proof.201 = {
	type = country_event
	category = disaster_event
	title = farr_proof.201.title
	desc = farr_proof.201.desc
	outcome = negative

	immediate = {
		# The old ruler is killed in the coup; the engine generates an heir,
		# but the political fact of the realm has shifted hard toward the
		# amīrs. A dedicated Türk-character template can replace this in a
		# later pass; for now the modifier + estate shifts carry the meaning.
		
		create_character = {
			dynasty = dynasty:borjigin_dynasty
			estate = estate_type:amirs_estate
			religion = root.religion
			culture = culture:mongolian_culture
			age = {24 40}
			adm = {0 20} # not an administrator.
			dip = {33 66} ## not complex, but has certain charisma
			mil = {75 100}
			save_scope_as = the_military_usurper_scope
		}

		
		kill_character = ROOT.ruler
		set_new_ruler = scope:the_military_usurper_scope
		# Strip the farr reform; the doctrine dies with the dynasty.
		remove_reform = government_reform:farr_i_izadi
		# The Ghaznavid template fills the vacuum — a Türkic amīr
		# sultanate atop a fully Persianized administrative state.
		# The realm survives; the legitimating doctrine has changed.
		add_reform = government_reform:amir_sultanate
		# Broken-formal alone strips the apparatus LEGACIES. The
		# institutional traces (the dār al-ḍarb, the khutba-precedent,
		# the Nawrūz court, the pādishāh title) die with the dynasty;
		# the new amīr regime has its own legitimating script and will
		# not maintain the doctrinal apparatus of the dead house.
		if = {
			limit = { has_country_modifier = farr_sikka_granted }
			remove_country_modifier = farr_sikka_granted
		}
		if = {
			limit = { has_country_modifier = farr_khutba_in_your_name }
			remove_country_modifier = farr_khutba_in_your_name
		}
		if = {
			limit = { has_country_modifier = farr_hunt_and_nawruz }
			remove_country_modifier = farr_hunt_and_nawruz
		}
		if = {
			limit = { has_country_modifier = farr_padishah }
			remove_country_modifier = farr_padishah
		}
		# Permanent scar — the cousins remember, future declarations locked.
		add_country_modifier = {
			modifier = attempted_farr_and_failed
			years = 25
			mode = add_and_extend
		}
		# Mark for future events / IO contender logic.
		set_variable = farr_was_broken_formal
		# Settle the political fact: Türk amīrs win this round.
		add_estate_satisfaction = { type = estate_type:amirs_estate value = 0.25 }
		add_estate_satisfaction = { type = estate_type:tribes_estate value = 0.20 }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = -0.20 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = -0.10 }
		# Modest legitimacy restoration — the new regime has its own claim.
		add_legitimacy = 25
	}

	option = {
		name = farr_proof.201.a
		# The new ruler accepts the realm as it stands.
	}
}

# ===============================================================
# INDICATOR EVENTS — fire whenever their hook applies, until the
# player silences them.
#
# Each event has three options:
#   (a) Acknowledge — applies the effect, keeps notifying.
#   (b) Silence this notification type — applies the effect and
#       sets farr_silence_<type>, muting future notifications of
#       only this kind.
#   (c) Silence all of these notifications — applies the effect
#       and sets farr_silence_all, muting every remaining
#       notification for the run.
#
# The hook checks BOTH farr_silence_<type> and farr_silence_all;
# if either is set, the hook applies the effect directly and does
# not trigger the event (trigger_event_silently does not work on
# multi-option events).
# ===============================================================

# ---300 BATTLE WON --------------------------------------------------
farr_proof.300 = {
