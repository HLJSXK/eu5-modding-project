# ============================================================================
# 1. The Poet's Patron — Hafez of Shiraz
# ----------------------------------------------------------------------------
# Fires for whoever owns Shiraz, from 1340 onward. Hafez historically wrote
# panegyrics first for Abu Ishaq Inju, then Shah Shuja Muzaffarid, then
# Sultan Ahmad Jalayir — fortune followed the patron, not the man. He
# famously refused to leave Shiraz when called to Baghdad and Bengal.
#
#
# ============================================================================
eranzamin_flavor.1 = {
	type = country_event
	title = eranzamin_flavor.1.title
	desc = eranzamin_flavor.1.desc
	outcome = positive
	fire_only_once = yes

	# Realistic Shiraz holders during Hafez's lifetime (d. ~1390): the two
	# Shirazi dynasties (Injuids → Muzaffarids), plus the Jalayirids who
	# Hafez later panegyrized, and the regional neighbors that periodically
	# threatened or absorbed Fars.
	dynamic_historical_event = {
		tag = INJ
		tag = MZF
		tag = JAL
		tag = CHB
		tag = KRT
		tag = MKR
		from = 1340.1.1
		to = 1395.1.1
		monthly_chance = 4
	}

	illustration_tags = { 10 = interior 10 = happy }

	trigger = {
		any_owned_location = { this = location:shiraz }
	}

	immediate = {
		# Spawn Hafez at the Shiraz-owning court. Note: create_character
		# syntax assumed against base-game pattern; verify.
		create_character = {
			first_name = name_hafiz
			culture = farsi_culture
			religion = sunni
			birth_date = 1315.1.1
			artist = writer
			artist_skill = 0.98 ## this is Hafez, the undisputed master of the Persian ghazal form. He should be a top-tier poet, and if he is not, WHO IS?
			save_scope_as = hafez_scope
		}
	}

	# OPTION A: Lavish patronage — commission the Diwan
	option = {
		name = eranzamin_flavor.1.a
		change_societal_value = { type = turk_vs_tazik value = 0.025 }
		change_gold_effect = { scale = ez_endowment_cost }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = ez_minor_estate_sat }
		add_prestige = ez_major_prestige
		add_country_modifier = { modifier = hafez_patronage_active years = ez_long_years }
		capital = {
			create_art = {
				artist = scope:hafez_scope
				type = work_of_art_type:poem
				quality = 100 ## this is the diwan of hafiz. each and every household that speaks persian has a copy, reads from it on every holiday, and quotes it in times of joy and sorrow alike. it is the single most salient to every day life  work of persian literature, and it should be represented as such in the game.
				key = diwan_of_hafiz
			}
		}

		# scope:hafez_scope ?= {
		# 	start_work_of_art = { type = poem quality = 95 }
		# }
	}

	# OPTION B: Modest stipend
	option = {
		name = eranzamin_flavor.1.b
		change_societal_value = { type = turk_vs_tazik value = ez_minor_axis_tazik }
		change_gold_effect = { scale = ez_court_cost }
		scope:hafez_scope ?= {
			start_work_of_art_of_type_quality = { type = poem quality = 50 }
		}
	}

	# OPTION C: Dismiss as un-Islamic
	option = {
		name = eranzamin_flavor.1.c
		add_estate_satisfaction = { type = estate_type:amirs_estate value = ez_minor_estate_sat }
		add_stability = ez_minor_stab_cost
		add_country_modifier = { modifier = new_madrasa_built years = ez_short_years }
	}
}

	trigger = {
		OR = {
			any_neighbor_country = { culture = culture:lur_culture }
			any_neighbor_country = { culture = culture:kurdish_culture }

				OR = {
					culture_percentage = {
						culture = culture:lur_culture

						culture = culture:kurdish_culture
						value > 0.4
					}

						culture_percentage = {
							culture = culture:lur_culture
							value > 0.2
						}
						culture_percentage = {
							culture = culture:kurdish_culture

						}
					}
				}
			 }
		}
	}

	option = {
		name = eranzamin_flavor.10.a

		add_country_modifier = { modifier = lur_kurd_tribute years = ez_long_years }

		name = eranzamin_flavor.10.b
