namespace = persianate_generic

# Generic flavor events for any Persianate polity (Iran / Khurasan / Caucasus
# capital). NOT restricted to Erānzamīn contenders. Builds out the lived
# texture of an Iranic state regardless of which faction is in charge.

# Shared trigger:
#  -capital in persia / khorasan / caucasus region (matches turk_vs_tazik
#    axis allow)
#  -religion = muslim (Islamic frame; non-Muslim Persianate polities, if
#    any, don't fit these flavor frames)

# ============================================================================
# G1. The Tazkira Compiled
# ----------------------------------------------------------------------------
# A scholar presents a regional biographical dictionary of poets, sufis, or
# ulama. The genre that produced Dawlatshah's Tazkirat al-Shu'ara,
# Sam Mirza's Tuhfa-yi Sami, etc.
# ============================================================================
persianate_generic.1 = {
	type = country_event
	title = persianate_generic.1.title
	desc = persianate_generic.1.desc
	outcome = neutral
	fire_only_once = yes

	illustration_tags = { 10 = interior 10 = happy }

	trigger = {
		religion ?= { group = religion_group:muslim }
		capital = {
			OR = {
				region ?= region:persia_region
				region ?= region:khorasan_region
				region ?= region:caucasus_region
				area ?= area:iraq_arabi_area
				area ?= area:jazira_area
			}
		}
	}

	# OPTION A: Adopt as court reference
	option = {
		name = persianate_generic.1.a
		change_gold_effect = { scale = ez_court_cost }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = ez_minor_estate_sat }
		change_societal_value = { type = turk_vs_tazik value = ez_minor_axis_tazik }
	}

	# OPTION B: Reward and shelve
	option = {
		name = persianate_generic.1.b
		change_gold_effect = { scale = ez_bribe_cost }
	}

	# OPTION C: Reject
	option = {
		name = persianate_generic.1.c
		add_estate_satisfaction = { type = estate_type:diwan_estate value = ez_minor_estate_loss }
		change_societal_value = { type = turk_vs_tazik value = ez_minor_axis_tork }
	}
}

# ============================================================================
# G4. The Year of Plague
# ----------------------------------------------------------------------------
# Recurrent post-1346 plague waves. The 14th century saw 30+% urban mortality
# in some Persianate cities (Tabriz, Shiraz, Herat).
# ============================================================================
persianate_generic.4 = {
	type = country_event
	title = persianate_generic.4.title
	desc = persianate_generic.4.desc
	outcome = negative
	fire_only_once = no  # plague recurs

	illustration_tags = { 10 = exterior 10 = angry }

	trigger = {
		religion ?= { group = religion_group:muslim }
		NOT = { has_variable = generic_pers_plague_recently }  # don't fire twice in a row
		capital = {
			OR = {
				region ?= region:persia_region
				region ?= region:khorasan_region
				region ?= region:caucasus_region
				area ?= area:iraq_arabi_area
				area ?= area:jazira_area
			}
		}
	}

	# Weight up during the historical Black Death window so the plague
	# clusters in the 14th-15th century, then thins out afterward.
	weight_multiplier = {
		base = 1
		modifier = {
			factor = 5
			trigger = {
				current_year >= 1346
				current_year <= 1500
			}
		}
	}
	immediate = {
		set_variable = { name = generic_pers_plague_recently value = 1 months = 60 }  # don't fire again for 3 years
	}
	# OPTION A: Quarantine the bazaar
	option = {
		name = persianate_generic.4.a
		change_gold_effect = { scale = ez_endowment_cost }
		add_estate_satisfaction = { type = estate_type:diwan_estate value = ez_minor_estate_sat }
		add_country_modifier = { modifier = quarantine_imposed years = ez_short_years }
	}

	# OPTION B: Pious procession
	option = {
		name = persianate_generic.4.b
		change_gold_effect = { scale = ez_court_cost }
		add_country_modifier = { modifier = pious_procession years = ez_short_years }
	}

	# OPTION C: Endure it
	option = {
		name = persianate_generic.4.c
		add_stability = ez_minor_stab_cost
		add_estate_satisfaction = { type = estate_type:diwan_estate value = ez_minor_estate_loss }
	}
}
