namespace = diwan_flavor

# ==============================================================================
# TEMPLATE 1: THE VIZIER'S PARABLE (Siyasatnameh Style)
# Use this for events where the Vizier quotes a story about Anushirvan or Mahmud 
# to advise the ruler on "Justice" vs "Pragmatism".
# ==============================================================================
diwan_flavor.template_1 = {
	type = country_event
	title = diwan_flavor.template_1.title
	desc = diwan_flavor.template_1.desc
	
	trigger = {
		modifier:has_nizam_ul_mulk_chancery = yes
		# Optional: societal_value:turk_vs_tazik < 0 
	}
	
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:diwan_estate background = estate_type:diwan_estate }
	}

	# Option A: The Path of the Pen (Tazik/Centralization)
	# "Justice is the foundation of the state."
	option = {
		name = diwan_flavor.template_1.a
		
		add_estate_satisfaction = { type = estate_type:diwan_estate value = 0.10 }
		
		# Push towards Bureaucracy
		change_societal_value = { type = turk_vs_tazik value = 5 }
		
		add_gold = -50
	}

	# Option B: The Path of the Sword (Turk/Pragmatism)
	# "A King cannot rule without an Army."
	option = {
		name = diwan_flavor.template_1.b
		
		add_estate_satisfaction = { type = estate_type:amirs_estate value = 0.10 }
		
		# Push towards Military Aristocracy
		change_societal_value = { type = turk_vs_tazik value = -5 }
		
		add_manpower = 500
	}
}

# ==============================================================================
# TEMPLATE 2: THE CORRUPTION SCANDAL (Mushrif/Audit)
# Use this when the Spymaster/Inspector reports graft.
# ==============================================================================
diwan_flavor.template_2 = {
	type = country_event
	title = diwan_flavor.template_2.title
	desc = diwan_flavor.template_2.desc
	
	trigger = {
		modifier:has_nizam_ul_mulk_chancery = yes
		monthly_corruption > 0
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:diwan_estate background = estate_type:nobles_estate }
	}

	# Option A: Strict Justice (Punish the official)
	# "Confiscate his wealth (Musadara)!"
	option = {
		name = diwan_flavor.template_2.a
		
		add_estate_satisfaction = { type = estate_type:diwan_estate value = -0.20 }
		add_gold = 100
		change_corruption = -1
		
		change_societal_value = { type = turk_vs_tazik value = 5 }
	}

	# Option B: Lenience (Nepotism)
	# "He is useful to us."
	option = {
		name = diwan_flavor.template_2.b
		
		add_estate_satisfaction = { type = estate_type:diwan_estate value = 0.10 }
		change_corruption = 0.5
		
		change_societal_value = { type = turk_vs_tazik value = -5 }
	}
}

# ==============================================================================
# TEMPLATE 3: THE TRIBAL GRIEVANCE (Yasa vs Sharia)
# Friction between Amirs (Customary Law) and Clergy/State (Religious Law).
# ==============================================================================
diwan_flavor.template_3 = {
	type = country_event
	title = diwan_flavor.template_3.title
	desc = diwan_flavor.template_3.desc
	
	trigger = {
		country_has_estate = estate_type:amirs_estate
		"estate_power(estate_type:amirs_estate)" > 0.4
	}
	
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:amirs_estate background = estate_type:clergy_estate }
	}
	
	# Option A: Respect the Yasa (Tribal Custom)
	option = {
		name = diwan_flavor.template_3.a
		
		add_estate_satisfaction = { type = estate_type:amirs_estate value = 0.15 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = -0.15 }
		
		change_societal_value = { type = turk_vs_tazik value = -10 }
	}
	
	# Option B: Enforce Sharia/Kanun (State Law)
	option = {
		name = diwan_flavor.template_3.b
		
		add_estate_satisfaction = { type = estate_type:amirs_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.15 }
		
		change_societal_value = { type = turk_vs_tazik value = 10 }
	}
}
