# ============================================================
# Ghazi: Burden of the Sword — events
#
# Structure:
#   .1            announce (disaster fired)
#   .50/.65/.80/.90  apparatus (jihad_within, ribat_network,
#                     investiture, sultan_of_ghazis)
#                     — claimed by the player via the disaster
#                     panel's action buttons; legitimacy-neutral
#                     on claim (refund + legacy modifier pair only)
#   .35/.20/.10   erosion (preacher cool, khutba cooler, bī-ghaza)
#                     — fire once on first downward threshold-touch
#   .40           designation expectation failed (timer elapsed
#                     without progress; volunteers grumble)
#   .100–.104     omens (preacher turns, volunteers drift, fatwa
#                     question, infidel-victory surge, muhtasib
#                     turns)
#   .150          fatwa chain — a scholar at court issues the
#                     Mardin-style ruling against yasa-holders
#   .160–.162     frontier designation assemblies (caucasus / rum
#                     / hindustan)
#   .200          Sword of Islam (Vindicated)
#   .201          Dervish Rising (Broken-formal — strips legacies,
#                     strips reform, generative clause feeds the
#                     murshid_i_kamil lineage)
#   .202          Guardian of the Thughur (Endured)
#   .203          Doctrine Lapses (Broken-lapse)
#   .300–.307     INDICATOR events for the on-action hooks. Each
#                     fires every time its hook applies, until the
#                     player silences it. Option (b) sets
#                     ghazi_silence_<type> (per-type mute); option
#                     (c) sets ghazi_silence_all (master mute).
# ============================================================

# ===============================================================
# APPARATUS EVENTS — legitimacy-neutral on claim. Refund +
# legacy pair only. Crossing a threshold does NOT push the player
# toward the next; the apparatus settles the corresponding
# silence and is its own end.
# ===============================================================

# ---50 JIHAD WITHIN — the censor walks the bazaar ----------------
ghazi_burden.50 = {
	type = country_event
	category = disaster_event
	title = ghazi_burden.50.title
	desc = ghazi_burden.50.desc
	outcome = positive

	option = {
		name = ghazi_burden.50.a
		# Refund: cancels the clergy slice while disaster runs.
		add_country_modifier = {
			modifier = ghazi_jihad_within_refund
			months = -1
		}
		# Legacy: hisba apparatus institutionalized; persists past
		# resolution unless Dervish Rising strips it.
		add_country_modifier = {
			modifier = ghazi_hisba_apparatus
			months = -1
		}
		# A modest clergy bump — the jurists concede the house is
		# at least being cleaned.
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.05 }
	}
}

# FRONTIER DESIGNATION ASSEMBLIES
# Unlocked once the ribat network is claimed. Fired periodically
# (event-side trigger gates on apparatus 4.2). The player picks
# one of three theaters; the modifier engages; the expectation
# timer begins.
# ===============================================================

# ---160 THE VOLUNTEERS ARE GATHERED — Caucasus -------------------
ghazi_burden.160 = {
	type = country_event
	category = disaster_event
	title = ghazi_burden.160.title
	desc = ghazi_burden.160.desc
	outcome = neutral

	option = {
		name = ghazi_burden.160.a
		# Switch designation if one already held; one-time cost.
		if = {
			limit = { has_country_modifier = ghazi_frontier_rum }
			remove_country_modifier = ghazi_frontier_rum
			add_legitimacy = -3
		}
		if = {
			limit = { has_country_modifier = ghazi_frontier_hindustan }
			remove_country_modifier = ghazi_frontier_hindustan
			add_legitimacy = -3
		}
		add_country_modifier = {
			modifier = ghazi_frontier_caucasus
			months = -1
		}
		set_variable = { name = ghazi_designation_progress_months value = 0 }
	}
	option = {
		name = ghazi_burden.160.b
		if = {
			limit = { has_country_modifier = ghazi_frontier_caucasus }
			remove_country_modifier = ghazi_frontier_caucasus
			add_legitimacy = -3
		}
		if = {
			limit = { has_country_modifier = ghazi_frontier_hindustan }
			remove_country_modifier = ghazi_frontier_hindustan
			add_legitimacy = -3
		}
		add_country_modifier = {
			modifier = ghazi_frontier_rum

# INTERNAL CAMPAIGNS (tier-1)
# Three flavors of hisba-driven purification of the realm. Each
# branch calls ghazi_apply_internal_campaign_credit, which handles
# the diminishing-returns curve and increments the counter.
# Historical model: Mubariz al-Din at Shiraz, 1353 — closing the
# taverns, persecuting the philosophers, dispersing the heterodox
# circles. Each flavor pays a different political cost.
# ===============================================================

	type = country_event
	category = disaster_event
	title = ghazi_burden.500.title
	desc = ghazi_burden.500.desc
	outcome = neutral

	# Vice: the wine-shops, the gambling dens, the unveiled women
	# of the bazaar. The Shiraz-1353 image. Burghers pay; clergy
	# applaud; the chronicles record a city in mourning.
	option = {
		name = ghazi_burden.500.a
		ghazi_apply_internal_campaign_credit = yes
		add_country_modifier = {
			modifier = ghazi_vice_campaign_scar
			months = 60
			mode = add_and_extend
		}
		add_estate_satisfaction = { type = estate_type:burghers_estate value = -0.10 }
	}

	# Heterodoxy: ghulat shaykhs, Hurufi cells, the doctrinally
	# suspect ribats. Clergy applaud loudest here; but the
	# constituency that COULD have been the realm's own dervish
	# rising grows thinner by the same act. (Narrative effect; the
	# Dervish Rising resolution does not condition on this counter.)
	option = {
		name = ghazi_burden.500.b
		ghazi_apply_internal_campaign_credit = yes
		add_country_modifier = {
			modifier = ghazi_heterodoxy_campaign_scar
			months = 60
			mode = add_and_extend
		}
		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.05 }
	}

	# Sectarian: the other branch. In a Sunni realm this is shia
	# circles dispersed; in a Shia realm, the reverse. Modeled
	# symmetrically as conversion-speed boost + a tolerance penalty.
	option = {
		name = ghazi_burden.500.c
		ghazi_apply_internal_campaign_credit = yes
		add_country_modifier = {
			modifier = ghazi_sectarian_campaign_scar
			months = 60
			mode = add_and_extend
		}
		random_owned_location = { add_location_modifier = { modifier = ez_unrest_3 years = 3 } }
	}
}

# The constituency defects to a more credible vessel. A
# charismatic shaykh gathers the ribat network; messianic rising.
# Strips reform AND all four legacies. The generative clause:
# the rising's energy is the raw material the Safavid corridor
# captures.

	type = country_event
	category = disaster_event
	title = ghazi_burden.201.title
	desc = ghazi_burden.201.desc
	outcome = negative

	immediate = {
		# Strip the reform — the doctrine dies with the dynasty.
		remove_reform = government_reform:ghazi_sultanate
		# The Mushaʿshaʿ-style heterodox apocalyptic tilt: the
		# constituency went to a charismatic shaykh; the throne
		# now runs as a millenarian fusion. This reform sits in
		# the Age 2 heterodox axis and is the natural landing.
		# Acts as the corridor's first step; murshid_i_kamil in
		# Age 3 reads ghazi_birthed_messianic_rising as confirming
		# precedent.
		add_reform = government_reform:mushasha_messianism

		# dynasty's ghazi project is erased; the ribat network
		# now belongs to the shaykh.
		if = {
			limit = { has_country_modifier = ghazi_hisba_apparatus }
			remove_country_modifier = ghazi_hisba_apparatus

			limit = { has_country_modifier = ghazi_the_lodges_endowed }
			remove_country_modifier = ghazi_the_lodges_endowed

			limit = { has_country_modifier = ghazi_caliphal_diploma }
			remove_country_modifier = ghazi_caliphal_diploma
		}
		if = {
			limit = { has_country_modifier = ghazi_sultan_of_ghazis_in_protocol }
			remove_country_modifier = ghazi_sultan_of_ghazis_in_protocol
		}

		add_country_modifier = {
			modifier = ghazi_dervish_rising_scar
			years = 25
			mode = add_and_extend
		}
		set_variable = ghazi_was_dervish_rising

		# messianic rising. The murshid_i_kamil corridor reads this
		# variable as a precondition for the Safavid-track start.

		# disinherited (they followed the shaykh, not the throne).

			mil = {75 100}

		}

		set_new_ruler = scope:the_military_usurper_scope
		change_religion = religion:shia
	}

	option = {
		name = ghazi_burden.201.a
		# The realm survives. The doctrine does not.
	}
}
