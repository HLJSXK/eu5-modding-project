namespace = amirs_estate_events

# Amirs demand military leadership
amirs_estate_events.1 = {
	type = country_event
	title = amirs_estate_events.1.title
	desc = amirs_estate_events.1.desc

	trigger = {
		country_has_estate = estate_type:amirs_estate
		"estate_power(estate_type:amirs_estate)" >= 0.25
		NOT = { has_estate_privilege = estate_privilege:amir_al_umara } # Can use vanilla privilege or clone
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:amirs_estate background = estate_type:amirs_estate }
	}

	option = {
		name = amirs_estate_events.1.a
		add_estate_satisfaction = { type = estate_type:amirs_estate value = estate_satisfaction_severe_bonus }
		grant_estate_privilege = estate_privilege:amir_al_umara
		
	}
	option = {
		name = amirs_estate_events.1.b
		add_estate_satisfaction = { type = estate_type:amirs_estate value = estate_satisfaction_extreme_penalty }
	}
}

# Amirs take control of a location
amirs_estate_events.3 = {
	type = country_event
	title = amirs_estate_events.3.title
	desc = amirs_estate_events.3.desc
	trigger = {
		country_has_estate = estate_type:amirs_estate
		"estate_power(estate_type:amirs_estate)" >= 0.25
		any_owned_location = {
			modifier:local_amirs_estate_power > 0.5 # Using amirs power
			is_capital = no
			local_control > 0.1
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		random_owned_location = {
			limit = {
				modifier:local_amirs_estate_power > 0.5
				is_capital = no
				local_control > 0.1
			}
			save_scope_as = amir_encroachment
		}
	}

	option = {
		name = amirs_estate_events.3.a
		scope:amir_encroachment = {
			change_control = control_mild_penalty
		}
	}
	option = {
		name = amirs_estate_events.3.b
		add_estate_satisfaction = { type = estate_type:amirs_estate value = estate_satisfaction_weak_penalty }
	}
}
