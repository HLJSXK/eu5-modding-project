namespace = diwan_estate_events

# Diwan demand administrative control
diwan_estate_events.1 = {
	type = country_event
	title = diwan_estate_events.1.title
	desc = diwan_estate_events.1.desc

	trigger = {
		country_has_estate = estate_type:diwan_estate
		"estate_power(estate_type:diwan_estate)" >= 0.25
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:diwan_estate background = estate_type:diwan_estate }
	}

	option = {
		name = diwan_estate_events.1.a
		add_estate_satisfaction = { type = estate_type:diwan_estate value = estate_satisfaction_severe_bonus }
		# Grant an administrative privilege
		grant_estate_privilege = estate_privilege:diwan_merit_examinations
	}
	option = {
		name = diwan_estate_events.1.b
		add_estate_satisfaction = { type = estate_type:diwan_estate value = estate_satisfaction_extreme_penalty }
	}
}

# Diwan tightening control (Audit)
diwan_estate_events.3 = {
	type = country_event
	title = diwan_estate_events.3.title
	desc = diwan_estate_events.3.desc
	trigger = {
		country_has_estate = estate_type:diwan_estate
		"estate_power(estate_type:diwan_estate)" >= 0.25
		any_owned_location = {
			modifier:local_diwan_estate_power > 0.5
			is_capital = no
			local_control < 0.9
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		random_owned_location = {
			limit = {
				modifier:local_diwan_estate_power > 0.5
				is_capital = no
				local_control < 0.9
			}
			save_scope_as = diwan_audit_target
		}
	}

	option = {
		name = diwan_estate_events.3.a
		scope:diwan_audit_target = {
			change_control = control_mild_bonus
		}
	}
	option = {
		name = diwan_estate_events.3.b
		add_estate_satisfaction = { type = estate_type:diwan_estate value = estate_satisfaction_weak_penalty }
	}
}
