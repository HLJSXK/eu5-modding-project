namespace = flavor_tur

# =====================================================================
# BASE OTTOMAN FLAVOR — persianate gate widening
#
# Under Shattered Rum, TUR runs government_type persianate_sultanate
# for the whole campaign, which killed the only two monarchy-gated
# events in base's flavor_TUR.txt (audited: 2 gates in 18,903 lines):
#   flavor_tur.88 — The Imperial Harem (1437+)
#   flavor_tur.95 — The One True Emperor (1537+)
# REPLACE'd verbatim with the gate widened to monarchy-OR-persianate.
# NOTE: REPLACE on the events database is unverified in engine —
# first dev-branch load after this change must check the error log
# for these two keys.
# =====================================================================

REPLACE:flavor_tur.88 = { #The Imperial Harem
	type = country_event
	title = flavor_tur.88.title
	desc = flavor_tur.88.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = TUR
		from = 1437.1.1
		to = 1637.1.1
		monthly_chance = 20
	}

	historical_info = flavor_tur.88.historical_info


	trigger = {
		has_advance = janissaries_advance

		exists = capital

		has_policy = harem_policy

		has_ruler = yes

		country_rank_level >= 3

		OR = {
			government_type = government_type:monarchy
			government_type = government_type:persianate_sultanate
		}

	}

	immediate =  {
		capital = {
			save_scope_as = capital_location
		}

		root.ruler = { save_scope_as = target_ruler }
	}

	option = {
		name = flavor_tur.88.a
		historical_option = yes

		set_variable = { name = ottoman_imperial_harem value = 1 }
		add_reform = government_reform:ottoman_imperial_harem

		add_estate_satisfaction = { type = estate_type:nobles_estate value = -0.20 }

		add_policy = policy:public_harem_policy

	}
	option = {
		name = flavor_tur.88.b

			change_societal_value = { type = centralization_vs_decentralization value = societal_value_minor_move_to_right }

		add_estate_satisfaction = { type = estate_type:nobles_estate value = 0.10 }

		add_stability = 5

		add_policy = policy:dispersed_harem_policy
	}

}

scripted_trigger is_suitable_rival_for_crimea = {
	capital = { is_discovered_by = root }
	NOT = {
		is_allied_with = { target = c:TUR }
		is_at_war_with = root
	}
	opinion = { target = root value < 25 }
	OR = {
  		country_rank = country_rank:rank_empire
 		country_rank = country_rank:rank_kingdom
  	}
 	capital = {
  		distance_to_squared = {
    		location = location:kaffa
    		value < 150000
  		}
	}
}


REPLACE:flavor_tur.95 = { #The One True Emperor
	type = country_event
	title = flavor_tur.95.title
	desc = flavor_tur.95.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = TUR
		from = 1537.1.1
		to = 1737.1.1
		monthly_chance = 15
	}

	historical_info = flavor_tur.95.historical_info

	trigger = {

		has_ruler = yes

		government_power < 70

		country_rank = country_rank:rank_empire

		OR = {
			government_type = government_type:monarchy
			government_type = government_type:persianate_sultanate
		}

		"religion_group_percentage_in_country(religion_group:christian)" > 0

		custom_tooltip = {
			text = variable_dilsiz_tt
			has_variable = dilsiz
		}

		owns = location:constantinople

	}

	immediate =  {

		root.ruler = { save_scope_as = target_ruler }
	}

	option = {
		name = flavor_tur.95.a
		historical_option = yes

		change_gold_effect = { scale = -3 }

		custom_tooltip = {
			text = christian_support.tt
			every_owned_location = {
				every_pop = {
					limit = {
						owner = root
						religion.group = religion_group:christian
					}
					add_pop_satisfaction = pop_satisfaction_mild_bonus
				}
			}
		}

		random_list = {

			40 = {
				add_government_power = government_power_mild_bonus
			}
			60 = {
				add_government_power = government_power_severe_bonus
			}
		}
	}

	option = {
		name = flavor_tur.95.b

		change_gold_effect = { scale = -1 }

		add_government_power = government_power_mild_bonus
	}

	option = {
		name = flavor_tur.95.c

		add_estate_satisfaction = { type = estate_type:clergy_estate value = 0.15 }
	}
}
