namespace = musadara

# =====================================================================
# THE DESPERATION MUṢĀDARA — option 2 of the muṣādara system
#
# Fired from on_loan_taken / on_loan_renewed (see on_action/
# 10_musadara_hooks.txt) when the realm is drowning: more than ten loans
# and bleeding income, or a loan capacity that no longer covers a year of
# revenue. The crown, out of credit, reaches for the elites' coffers — and,
# in the Persianate idiom, dresses the seizure as justice and gains
# legitimacy for it. The "worse than the yoke" thesis, made mechanical:
# confiscation here is not merely tolerated, it is virtuous.
#
# Gating lives in the on_action (privilege + distress + cooldown); this
# event is triggered-only and sets its own 1-year cooldown so a flurry of
# loans does not spam it.
# =====================================================================

musadara.1 = {
	type = country_event
	title = musadara.1.title
	desc = musadara.1.desc

	illustration_tags = {
		10 = interior
		10 = angry
	}



	immediate = {
		# One desperation prompt per year, however many loans are signed.
		set_variable = {
			name = musadara_desperation_recent
			value = 1
			years = 1
		}
	}

	# 1. Declare Maṣlaḥa — repudiate the debt as a thing done for the
	#    common good. The lenders (the elites) eat the loss; the crown
	#    spins it as virtue.
	option = {
		name = musadara.1.a

		every_loan = { cancel_loan = this }

		add_estate_satisfaction = { type = estate_type:diwan_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:amirs_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = -0.15 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = -0.15 }

		add_country_modifier = { modifier = recent_musadara years = 5 }

		add_legitimacy = 10
		add_stability = -10
	}

	# 2. The Circle of Justice needs greasing — repudiate the debt AND
	#    seize a fifth of every wealthy estate's coffers. The deeper grab:
	#    more legitimacy spun, more ruin sown.
	option = {
		name = musadara.1.b

		every_loan = { cancel_loan = this }
		"estate(estate_type:diwan_estate)" = { save_scope_as = target_diwan_estate }
		"estate(estate_type:amirs_estate)" = { save_scope_as = target_amirs_estate }
		"estate(estate_type:burghers_estate)" = { save_scope_as = target_burghers_estate }
		"estate(estate_type:clergy_estate)" = { save_scope_as = target_clergy_estate }
		# Seize 20% of each elite estate's gold into the treasury. Each pair
		# reads the estate's gold before the seizure changes it; estates with
		# nothing simply yield nothing.
		add_gold = { value = scope:target_diwan_estate.estate_gold multiply = 0.2 }
		add_gold_to_estate = {
			estate_type = estate_type:diwan_estate
			value = { value = scope:target_diwan_estate.estate_gold multiply = -0.2 }
		}
		add_gold = { value = scope:target_amirs_estate.estate_gold multiply = 0.2 }
		add_gold_to_estate = {
			estate_type = estate_type:amirs_estate
			value = { value = scope:target_amirs_estate.estate_gold multiply = -0.2 }
		}
		add_gold = { value = scope:target_burghers_estate.estate_gold multiply = 0.2 }
		add_gold_to_estate = {
			estate_type = estate_type:burghers_estate
			value = { value = scope:target_burghers_estate.estate_gold multiply = -0.2 }
		}
		add_gold = { value = scope:target_clergy_estate.estate_gold multiply = 0.2 }
		add_gold_to_estate = {
			estate_type = estate_type:clergy_estate
			value = { value = scope:target_clergy_estate.estate_gold multiply = -0.2 }
		}

		add_estate_satisfaction = { type = estate_type:diwan_estate value = -0.30 }
		add_estate_satisfaction = { type = estate_type:amirs_estate value = -0.30 }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = -0.30 }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = -0.30 }

		add_country_modifier = { modifier = recent_musadara_severe years = 5 }

		add_legitimacy = 15
		add_stability = -20
	}

	# 3. Refuse. Confiscation is the rot that devours the realm.
	option = {
		name = musadara.1.c
	}
}
