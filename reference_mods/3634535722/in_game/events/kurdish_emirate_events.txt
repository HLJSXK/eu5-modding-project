namespace = kurdish_emirate

# =====================================================================
# THE CHANCERY OFFERED — a mîr takes a great seat of empire
#
# Fired from on_location_changed_owner (see 12_iranic_conquest.txt) when an
# Iranic Tribal Polity conquers Tabriz, Isfahan, Shiraz, Nishapur, Baghdad, or
# Herat. The mîr may take up the dīwān and become a Persianate Sultanate — or
# remain the prince of his mountains.
# =====================================================================

kurdish_emirate.1 = {
	type = country_event
	title = kurdish_emirate.1.t
	desc = kurdish_emirate.1.d
	outcome = positive

	illustration_tags = {
		10 = interior
		10 = regular
	}

	# Defensive re-check — only a mîrnishîn can take up the chancery.
	trigger = {
		government_type = government_type:iranic_tribal_polity
	}

	# Take up the chancery — become a Persianate Sultanate, unlock the dīwān.
	option = {
		name = kurdish_emirate.1.a
		change_government_type = government_type:persianate_sultanate
		remove_reform = government_reform:kurdish_emirate
		add_reform = government_reform:persianate_bureaucracy_reform
		add_legitimacy = -10
		add_stability = -10
		ai_chance = {
			factor = 95
		}
	}

	# Remain the mîr of the mountains. ALmost never happened historically, but the option is there.
	option = {
		name = kurdish_emirate.1.b
		add_legitimacy = 5
		ai_chance = {
			factor = 5
		}
	}
}
