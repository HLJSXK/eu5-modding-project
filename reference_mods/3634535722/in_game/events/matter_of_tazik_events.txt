# ---25 THRESHOLD: A petition from the bāzār ---------------------------

	category = disaster_event
	title = matter_of_tazik.25.title
	desc = matter_of_tazik.25.desc
	outcome = neutral

	# Hear in open court — concedes
	option = {
		name = matter_of_tazik.25.a

		name = matter_of_tazik.25.c
		change_variable = { name = matter_progress add = 5 }

# ---100: DISASTER FIRES (urban civil war) ----------------------------

matter_of_tazik.100 = {
	type = country_event
	category = disaster_event
	title = matter_of_tazik.100.title
	desc = matter_of_tazik.100.desc
	outcome = neutral

	immediate = {
		spawn_urban_breakaway = yes
		# Religious-unrest cascade on non-rolling urban provinces
		every_owned_province = {
			limit = { is_urban_diwan_capable_location = yes }
			every_location_in_province = {
				add_location_modifier = { modifier = ez_unrest_5 years = 5 }
				change_control = control_severe_penalty
			}
		}
		add_estate_satisfaction = { type = estate_type:amirs_estate value = 20 }
	}

	option = {
		name = matter_of_tazik.100.a
	}
}

# ---200: SAFAVID SYNTHESIS OVERRIDE (§20.4) --------------------------
# Fires when 75/75/0.80 thresholds met while disaster is active. Auto-
# resolves into the Qizilbash Synthesis exit — peacefully annexes the
# breakaway, sets both matter_resolved flags. Triggered by the
# threshold_dispatch effect setting matter_synthesis_override flag.

matter_of_tazik.200 = {
	type = country_event
	category = disaster_event
	title = matter_of_tazik.200.title
	desc = matter_of_tazik.200.desc
	outcome = positive
	trigger = {
		has_variable = matter_synthesis_override
		safavid_synthesis_thresholds_met = yes
	}

	immediate = {
		matter_apply_qizilbash_synthesis = yes
		# Peacefully annex the breakaway if it spawned
		if = {
			limit = { exists = var:matter_breakaway_ref }
			var:matter_breakaway_ref ?= {
				destroy_rebel = this
			}
		}
	}

	option = {
		name = matter_of_tazik.200.a
	}
}
