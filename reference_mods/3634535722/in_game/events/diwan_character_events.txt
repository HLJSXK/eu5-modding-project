namespace = diwan_character_events

# Spawn Diwan Character (Bureaucrat)
diwan_character_events.1 = {
	type = country_event
	hidden = yes
	
	trigger = {
		country_has_estate = estate_type:diwan_estate
		any_character = {
			count < 4
			has_estate = estate_type:diwan_estate
			is_female = no
		}
	}
	
	immediate = {
		create_character = {
			estate = estate_type:diwan_estate
			age = { min = 18 max = 45 }
			culture = root.culture # Or specific Persian logic? Default is root culture.
			religion = root.religion
			
			# Stats bias: Admin/Diplo
			# (Assuming game handles stats by job or randomly, but we can nudge traits)
		}
	}
}
