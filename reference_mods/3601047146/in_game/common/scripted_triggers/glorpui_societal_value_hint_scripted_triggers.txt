# Societal value hint takeability gates. Evaluated in country scope; PREV inside each object-literal block is that country.

glorpui_svh_reform_core = {
	government_reform:$KEY$ = {
		is_implementable_in = PREV
		NOT = { is_fully_implemented_in = PREV }
	}
}

# is_fully_implemented_in on a policy is true whenever its law is in force, so the current-policy
# exclusion uses has_policy instead.
glorpui_svh_policy_core = {
	policy:$KEY$ = {
		is_implementable_in = PREV
	}
	NOT = { has_policy = $KEY$ }
	law:$LAW$ = {
		OR = {
			is_fully_implemented_in = PREV
			is_implementable_in = PREV
		}
	}
}

glorpui_svh_privilege_takeable = {
	estate_privilege:$KEY$ = {
		is_implementable_in = PREV
		NOT = { is_fully_implemented_in = PREV }
	}
}
