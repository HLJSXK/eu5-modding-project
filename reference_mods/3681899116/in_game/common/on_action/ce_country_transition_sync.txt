on_new_country_formed = {
	on_actions = {
		ce_country_events_on_new_country_formed
	}
}

ce_country_events_on_new_country_formed = {
	effect = {
		if = {
			limit = { is_ai = no }
			ce_update_country_events_temporal_state = yes
		}
	}
}

on_game_start = {
	on_actions = {
		ce_country_events_on_game_start
	}
}

ce_country_events_on_game_start = {
	effect = {
		every_country = {
			limit = { is_ai = no }
			ce_update_country_events_temporal_state = yes
		}
	}
}
