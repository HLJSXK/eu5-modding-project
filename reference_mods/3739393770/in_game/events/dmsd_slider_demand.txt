namespace = dmsd_slider_demand

# ═══════════════════════════════════════════════════════════════════════════════
# SLIDER DEMAND EVENTS
# ═══════════════════════════════════════════════════════════════════════════════
# Phase 1 of the monthly demand cycle. Fires per-country via
# monthly_country_pulse. Each slider computes its gold, applies
# efficiency decay, then distributes across proximity sources into
# market location maps. Phase 2 runs later the same month in
# weather_monthly_pulse (which fires after all countries are handled),
# reading the freshly populated maps and applying market demands.

# --- Per-slider efficiency decay ---
# Diminishing returns on heavy spending. Each doubling of raw gold past
# the baseline costs 25% efficiency. Per-slider, not country-wide, so a
# large nation spending modestly is unpenalized.
@dmsd_eff_baseline = 250
# 0.415 = log(0.75)/log(0.5), the exponent that yields exactly ×0.75 per doubling.
@dmsd_eff_exponent = 0.415


# ═══════════════════════════════════════════════════════════════════════════════
# PHASE 1: PER-COUNTRY ACCUMULATORS
# ═══════════════════════════════════════════════════════════════════════════════

# --- Event .1: Court spending ---
# Gold = inferred_slider × 0.1 × eco_base / efficiency_divisor
# Court splits 50/50 between regional and religion demand baskets.
dmsd_slider_demand.1 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	immediate = {

		if = {
			limit = { NOT = { has_variable_list = dmsd_prox_sources } }
			dmsd_rebuild_prox_cache = yes
		}

		religion = { group = { save_temporary_scope_as = dmsd_rg } }
		dmsd_infer_court_slider = yes

		# Resolve demand variant IOs for GUI tooltip display.
		# Two-step lookup: subcontinent → demand type → IO.
		# Results stored as country variables read by GetVariable in GUI.
		capital = { sub_continent = { save_temporary_scope_as = dmsd_disp_sc } }

		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_disp_sc } }
			set_local_variable = { name = dmsd_disp_d value = "global_variable_map(dmsd_court_reg1|scope:dmsd_disp_sc)" }
		} else = {
			set_local_variable = { name = dmsd_disp_d value = demand:dmsd_court_demand1 }
		}
		set_variable = { name = dmsd_disp_court_reg_io value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_d)" }

		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_rg } }
			set_local_variable = { name = dmsd_disp_d value = "global_variable_map(dmsd_court_reg1|scope:dmsd_rg)" }
			set_variable = { name = dmsd_disp_court_rel_io value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_d)" }
		}

		# Diplo, stab, cult IO refs: same pattern, different registries
		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_diplo_reg1 target = scope:dmsd_disp_sc } }
			set_local_variable = { name = dmsd_disp_d value = "global_variable_map(dmsd_diplo_reg1|scope:dmsd_disp_sc)" }
		} else = {
			set_local_variable = { name = dmsd_disp_d value = demand:dmsd_diplomatic_demand1 }
		}
		set_variable = { name = dmsd_disp_diplo_io value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_d)" }

		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_stab_reg1 target = scope:dmsd_disp_sc } }
			set_local_variable = { name = dmsd_disp_d value = "global_variable_map(dmsd_stab_reg1|scope:dmsd_disp_sc)" }
		} else = {
			set_local_variable = { name = dmsd_disp_d value = demand:dmsd_stability_demand1 }
		}
		set_variable = { name = dmsd_disp_stab_io value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_d)" }

		# Court gold calculation.
		# court_maintenance trigger gives raw slider position (0-1) but doesn't
		# account for reduced spending when gov power is capped. We use the
		# inferred effective slider from dmsd_infer_court_slider instead.
		# Engine clamps the efficiency divisor to min 0.5 (max 2× penalty).
		set_local_variable = {
			name = dmsd_gold
			value = {
				value = 0
				if = {
					limit = { has_variable = dmsd_court_slider }
					add = {
						value = var:dmsd_court_slider
						multiply = 0.1
						multiply = "country_economical_base"
						divide = { value = 1 add = "modifier:court_spending_efficiency" min = 0.5 }
					}
				}
			}
		}
		# Efficiency decay applied to pre-decay gold; see constants above
		change_local_variable = {
			name = dmsd_gold
			multiply = {
				value = @dmsd_eff_baseline
				divide = { value = local_var:dmsd_gold min = @dmsd_eff_baseline }
				pow = @dmsd_eff_exponent
			}
		}
		dmsd_accumulate_court = yes
	}
}

# --- Event .2: Diplomatic spending ---
# Gold = diplomatic_maintenance × base_rate × eco_base / efficiency_divisor
# diplomatic_maintenance trigger works correctly here (no capping issue).
dmsd_slider_demand.2 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	immediate = {

		set_local_variable = {
			name = dmsd_gold
			value = {
				value = "diplomatic_maintenance"
				multiply = "modifier:diplomatic_spending_cost"
				multiply = "country_economical_base"
				divide = { value = 1 add = "modifier:diplomatic_upkeep_efficiency" min = 0.5 }
			}
		}
		change_local_variable = {
			name = dmsd_gold
			multiply = {
				value = @dmsd_eff_baseline
				divide = { value = local_var:dmsd_gold min = @dmsd_eff_baseline }
				pow = @dmsd_eff_exponent
			}
		}
		dmsd_accumulate_diplo = yes
	}
}

# --- Event .5: Cultural spending ---
# Unlike the other sliders, cultural cost comes from artist salaries, not a
# slider × eco_base formula. The base salary per artist scales with BOTH
# tax_base AND number of artists (quadratic in N), creating natural
# diminishing returns on hiring more artists.
#
# yearly_salary trigger exists but doesn't include the per-artist N scaling,
# so we use the known static formula instead:
#   total = ARTIST_SALARY_BASE_FACTOR × tax_base × N² / 12 × modifiers
# Country-wide artist_salary_modifier applies uniformly; per-character
# modifiers (e.g. humble trait = -25%) are averaged across all artists.
dmsd_slider_demand.5 = {
	type = country_event
	title = hidden_event.t
	hidden = yes
	immediate = {

		set_local_variable = { name = dmsd_num_artists value = 0 }
		set_local_variable = { name = dmsd_char_mod_sum value = 0 }
		every_artist = {
			change_local_variable = { name = dmsd_num_artists add = 1 }
			change_local_variable = { name = dmsd_char_mod_sum add = "modifier:artist_salary_modifier_on_character" }
		}
		set_local_variable = {
			name = dmsd_gold
			value = {
				value = 0.025
				multiply = "country_tax_base"
				multiply = local_var:dmsd_num_artists
				multiply = local_var:dmsd_num_artists
				divide = 12
				multiply = {
					value = 1
					add = "modifier:artist_salary_modifier"
					add = {
						value = local_var:dmsd_char_mod_sum
						divide = { value = local_var:dmsd_num_artists min = 1 }
					}
				}
			}
		}

		# Stored for tooltip; can't recompute in a script value without every_artist
		set_variable = { name = dmsd_cult_raw_gold value = local_var:dmsd_gold }

		change_local_variable = {
			name = dmsd_gold
			multiply = {
				value = @dmsd_eff_baseline
				divide = { value = local_var:dmsd_gold min = @dmsd_eff_baseline }
				pow = @dmsd_eff_exponent
			}
		}

		capital = { sub_continent = { save_temporary_scope_as = dmsd_disp_sc } }
		if = {
			limit = { is_key_in_global_variable_map = { name = dmsd_cult_reg1 target = scope:dmsd_disp_sc } }
			set_local_variable = { name = dmsd_disp_d value = "global_variable_map(dmsd_cult_reg1|scope:dmsd_disp_sc)" }
		} else = {
			set_local_variable = { name = dmsd_disp_d value = demand:dmsd_cultural_demand1 }
		}
		set_variable = { name = dmsd_disp_cult_io value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_d)" }

		dmsd_accumulate_culture = yes
	}
}
