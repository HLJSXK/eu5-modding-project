# PLAYER DISPLAY MAP UPDATE
# ═══════════════════════════════════════════════════════════════════════════════
# Called monthly for the player only (is_ai = no) during Phase 1.
# Populates global goods display maps with the specific variant goods
# that match the player's capital subcontinent and religion.
# The GUI iterates these maps via GetGlobalMapKeys to show goods icons.
#
# The active variant's IO holds the exact goods in its dmsd_weights map.
# We look up the IO from dmsd_demand_io and copy its contents.

dmsd_update_player_display = {
	# Determine the player's active court regional variant
	capital = { sub_continent = { save_temporary_scope_as = dmsd_disp_sc } }
	religion = { group = { save_temporary_scope_as = dmsd_disp_rg } }

	# Court regional goods: look up variant from registry, then IO, then copy weights
	clear_global_variable_map = dmsd_tt_court_goods
	if = {
		limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_disp_sc } }
		set_local_variable = { name = dmsd_disp_demand value = "global_variable_map(dmsd_court_reg1|scope:dmsd_disp_sc)" }
	} else = {
		set_local_variable = { name = dmsd_disp_demand value = demand:dmsd_court_demand1 }
	}
	set_global_variable = { name = dmsd_io_ref value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_demand)" }
	global_var:dmsd_io_ref = {
		every_key_in_variable_map = {
			variable = dmsd_weights
			save_temporary_scope_as = dmsd_disp_good
			add_to_global_variable_map = {
				name = dmsd_tt_court_goods
				key = scope:dmsd_disp_good
				value = "variable_map(dmsd_weights|scope:dmsd_disp_good)"
			}
		}
	}

	# Court religion goods: same pattern
	clear_global_variable_map = dmsd_tt_court_rel_goods
	if = {
		limit = { is_key_in_global_variable_map = { name = dmsd_court_reg1 target = scope:dmsd_disp_rg } }
		set_local_variable = { name = dmsd_disp_demand value = "global_variable_map(dmsd_court_reg1|scope:dmsd_disp_rg)" }
		set_global_variable = { name = dmsd_io_ref value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_demand)" }
		global_var:dmsd_io_ref = {
			every_key_in_variable_map = {
				variable = dmsd_weights
				save_temporary_scope_as = dmsd_disp_good
				add_to_global_variable_map = {
					name = dmsd_tt_court_rel_goods
					key = scope:dmsd_disp_good
					value = "variable_map(dmsd_weights|scope:dmsd_disp_good)"
				}
			}
		}
	}

	# Diplomatic goods
	clear_global_variable_map = dmsd_tt_diplo_goods
	if = {
		limit = { is_key_in_global_variable_map = { name = dmsd_diplo_reg1 target = scope:dmsd_disp_sc } }
		set_local_variable = { name = dmsd_disp_demand value = "global_variable_map(dmsd_diplo_reg1|scope:dmsd_disp_sc)" }
	} else = {
		set_local_variable = { name = dmsd_disp_demand value = demand:dmsd_diplomatic_demand1 }
	}
	set_global_variable = { name = dmsd_io_ref value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_demand)" }
	global_var:dmsd_io_ref = {
		every_key_in_variable_map = {
			variable = dmsd_weights
			save_temporary_scope_as = dmsd_disp_good
			add_to_global_variable_map = {
				name = dmsd_tt_diplo_goods
				key = scope:dmsd_disp_good
				value = "variable_map(dmsd_weights|scope:dmsd_disp_good)"
			}
		}
	}

	# Cultural goods
	clear_global_variable_map = dmsd_tt_cult_goods
	if = {
		limit = { is_key_in_global_variable_map = { name = dmsd_cult_reg1 target = scope:dmsd_disp_sc } }
		set_local_variable = { name = dmsd_disp_demand value = "global_variable_map(dmsd_cult_reg1|scope:dmsd_disp_sc)" }
	} else = {
		set_local_variable = { name = dmsd_disp_demand value = demand:dmsd_cultural_demand1 }
	}
	set_global_variable = { name = dmsd_io_ref value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_demand)" }
	global_var:dmsd_io_ref = {
		every_key_in_variable_map = {
			variable = dmsd_weights
			save_temporary_scope_as = dmsd_disp_good
			add_to_global_variable_map = {
				name = dmsd_tt_cult_goods
				key = scope:dmsd_disp_good
				value = "variable_map(dmsd_weights|scope:dmsd_disp_good)"
			}
		}
	}

	# Stability goods
	clear_global_variable_map = dmsd_tt_stab_goods
	if = {
		limit = { is_key_in_global_variable_map = { name = dmsd_stab_reg1 target = scope:dmsd_disp_sc } }
		set_local_variable = { name = dmsd_disp_demand value = "global_variable_map(dmsd_stab_reg1|scope:dmsd_disp_sc)" }
	} else = {
		set_local_variable = { name = dmsd_disp_demand value = demand:dmsd_stability_demand1 }
	}
	set_global_variable = { name = dmsd_io_ref value = "global_variable_map(dmsd_demand_io|local_var:dmsd_disp_demand)" }
	global_var:dmsd_io_ref = {
		every_key_in_variable_map = {
			variable = dmsd_weights
			save_temporary_scope_as = dmsd_disp_good
			add_to_global_variable_map = {
				name = dmsd_tt_stab_goods
				key = scope:dmsd_disp_good
				value = "variable_map(dmsd_weights|scope:dmsd_disp_good)"
			}
		}
	}
}
