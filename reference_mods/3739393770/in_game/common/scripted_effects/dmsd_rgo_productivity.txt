# ═══════════════════════════════════════════════════════════════════════════════
# RGO PRODUCTIVITY: PRICE-LINKED OUTPUT PENALTY
# ═══════════════════════════════════════════════════════════════════════════════
# Percentage-based penalty using per-good global variable maps.
# Reduces RGO output proportionally when market price falls below base.
# Accounts for existing output modifiers so the penalty is a true percentage
# of actual output, not a flat additive.

@dmsd_rgo_sensitivity = 1.0
@dmsd_rgo_price_step = 0.02
@dmsd_rgo_threshold = 0.007

# ═══════════════════════════════════════════════════════════════════════════════
# PER-LOCATION PENALTY LOGIC
# ═══════════════════════════════════════════════════════════════════════════════
# Location scope, $good$ known.
# Pre-filter: skip everything if price hasn't moved by more than price_step
# since last evaluation. Modifier reads only run when price moved enough.

dmsd_rgo_apply_pct = {
	if = {
		limit = { relative_raw_material_price < 0 }

		set_local_variable = { name = dmsd_rgo_price value = relative_raw_material_price }
		round_local_variable = { name = dmsd_rgo_price nearest = @dmsd_rgo_price_step }

		# Price pre-filter: skip if rounded price hasn't changed
		set_local_variable = { name = dmsd_rgo_skip value = 0 }
		if = {
			limit = {
				has_variable = dmsd_rgo_last_price
				local_var:dmsd_rgo_price = var:dmsd_rgo_last_price
			}
			set_local_variable = { name = dmsd_rgo_skip value = 1 }
		}

		if = { limit = { local_var:dmsd_rgo_skip = 0 }
			set_variable = { name = dmsd_rgo_last_price value = local_var:dmsd_rgo_price }

			set_local_variable = {
				name = dmsd_rgo_ext
				value = {
					value = modifier:local_raw_material_output
					add = owner.modifier:global_raw_material_output
					add = modifier:local_$good$_output_modifier
					add = owner.modifier:global_$good$_output_modifier
				}
			}
			if = { limit = { has_variable = dmsd_rgo_applied }
				change_local_variable = { name = dmsd_rgo_ext subtract = var:dmsd_rgo_applied }
			}

			set_local_variable = {
				name = dmsd_rgo_target
				value = {
					value = relative_raw_material_price
					multiply = @dmsd_rgo_sensitivity
					multiply = { value = 1 add = local_var:dmsd_rgo_ext }
				}
			}

			# Compare unrounded target against stored rounded applied
			set_local_variable = { name = dmsd_rgo_delta value = { value = local_var:dmsd_rgo_target } }
			if = { limit = { has_variable = dmsd_rgo_applied }
				change_local_variable = { name = dmsd_rgo_delta subtract = var:dmsd_rgo_applied }
			}

			if = { limit = { OR = {
				local_var:dmsd_rgo_delta > @dmsd_rgo_threshold
				local_var:dmsd_rgo_delta < { value = @dmsd_rgo_threshold multiply = -1 }
			} }

				round_local_variable = { name = dmsd_rgo_target nearest = 0.01 }
				set_variable = { name = dmsd_rgo_applied value = local_var:dmsd_rgo_target }
				add_location_modifier = {
					modifier = dmsd_rgo_price_penalty
					years = -1
					mode = replace
					size = { value = local_var:dmsd_rgo_target }
				}
			}
		}
	} else = {
		if = { limit = { has_variable = dmsd_rgo_applied }
			if = { limit = { has_location_modifier = dmsd_rgo_price_penalty } remove_location_modifier = dmsd_rgo_price_penalty }
			remove_variable = dmsd_rgo_applied
			remove_variable = dmsd_rgo_last_price

		}
	}
}

dmsd_rgo_init_vm_good = {
	if = { limit = { has_global_variable_map = dmsd_rgo_vm_$good$ } clear_global_variable_map = dmsd_rgo_vm_$good$ }
	every_location_in_the_world = {
		limit = { has_owner = yes raw_material = goods:$good$ }
		add_to_global_variable_map = { name = dmsd_rgo_vm_$good$ key = this value = 1 }
	}
}
