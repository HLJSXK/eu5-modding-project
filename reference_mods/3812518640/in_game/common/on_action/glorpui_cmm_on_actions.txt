#noinspection ALL
# Hook this mod into CMF shared registration on_action.
cmf_on_mod_registration = {
	on_actions = {
		glorpui_on_register_cmf_mod
	}
}

glorpui_on_register_cmf_mod = {
	effect = {
		glorpui_register_cmf_mod = yes
	}
}

# Unified callback hook for setting changes, alert clicks, action bar clicks.
cmf_on_callback = {
	on_actions = {
		glorpui_on_cmf_callback
	}
}

glorpui_on_cmf_callback = {
	effect = {
		glorpui_handle_cmf_callback = yes
	}
}

cmf_on_banner_registration = {
	on_actions = {
		glorpui_register_lobby_banner
	}
}

glorpui_register_lobby_banner = {
	effect = {
		cmf_register_lobby_banner = { mod_id = glorpui }
	}
}
