peace_humiliate = {
	
	blocks_full_annexation = yes

	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 5
		}
		add = {
			desc = "game_concept_prestige"
			value = scope:loser.prestige
			multiply = 0.5
		}
	}

	effect = {
		scope:winner = {
			add_prestige = {
				value = scope:loser.prestige
				multiply =  0.2
				min = 5
			}
			add_political_influence = {
				value = 100
				multiply = {
					value = "scope:loser.country_economical_base"
					divide = {
						value = 0.01
						add = "scope:winner.country_economical_base"
						add = "scope:loser.country_economical_base"
					}
					subtract = 0.45
					max = 20
					min = 0
				}
			}
			if = {
				limit = {
					scope:loser = { is_rival_of = scope:winner }
				}
				set_variable = {
					name = recently_humiliated_rival_variable
					years = 10
				}
			}
		}
		scope:loser = { 
			add_prestige = prestige_extreme_penalty 
			add_political_influence = {
				value = 100
				multiply = {
					value = "scope:loser.country_economical_base"
					divide = {
						value = 0.01
						add = "scope:winner.country_economical_base"
						add = "scope:loser.country_economical_base"
					}
					subtract = 0.45
					multiply = -1
					max = 0
					min = -20
				}
			}
		}
	}

	ai_desire = {
		if = {
			limit = {
				scope:winner = {
					is_rival_of = scope:loser
				}
			}
			add = {
				desc = "game_concept_rival"
				value = 5
			}
		}
		if = {
			limit = {
				scope:winner.prestige < scope:loser.prestige
			}
			add = {
				desc = "DIPLOREASON_LOWER_PRESTIGE"
				value = scope:loser.prestige
				subtract = scope:winner.prestige
				divide = 5
			}
		}
		scope:winner = {
			add = {
				value = "ai_target_subvert(scope:loser)"
				multiply = 0.1
			}
		}
	}
}
