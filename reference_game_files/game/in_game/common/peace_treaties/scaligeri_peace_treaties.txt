scaligeri_victory = {
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 70
		}
	}

	potential = {
		scope:war ?= {
			casus_belli ?= { this = casus_belli:cb_scaligeri_war }
			is_a_defender = scope:winner
		}
	}

	effect = {
		scope:winner = {
			custom_tooltip = {
				text = ver_add_integration_in_every_location
				every_owned_location = {
					limit = {
						is_in_scripted_geography = scripted_geography:scaligeri_conquests_geography
						NOT = { integration_progress = 100 }
					}
					change_integration_progress = 100
				}
			}
			custom_tooltip = {
				text = ver_remove_antagonism
				every_known_country = {
					remove_antagonism = {
						target = scope:winner
						modifier = antagonism_peace_treaty_demands
					}
				}
			}
		}
	}

	ai_desire = {
		add = {
			desc = "BASE"
			value = 10000
		}
	}
}

ver_milanese_demands = {
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			if = {
				limit = {
					scope:war.var:MLO_demands = flag:brescia_demand
				}
				value = 25
			}
			else_if = {
				limit = {
					scope:war.var:MLO_demands = flag:parma_demand
				}
				value = 20
			}
			else = {
				value = 33 # Base
			}
		}
	}

	potential = {
		country_exists = c:MLO
		scope:war ?= {
			casus_belli ?= { this = casus_belli:cb_scaligeri_war }
			allows_scaligeri_peace_treaty_for_country = { country = MLO }
			is_an_attacker = scope:winner
			is_an_attacker = c:MLO
		}
		exists = scope:loser.province:brescia_province
	}

	allow = {
		scaligeri_peace_treaty_demand_allow = { country = MLO }
	}

	effect = {
		if = {
			limit = {
				scope:war.var:MLO_demands = flag:brescia_demand
			}
			scope:loser = {
				province:brescia_province = {
					change_province_owner = c:MLO
				}
			}
		}
		else_if = {
			limit = {
				scope:war.var:MLO_demands = flag:parma_demand
			}
			province_definition:parma_province = {
				every_location_in_province_definition = {
					limit = {
						owner ?= {
							scope:war = {
								is_a_defender = PREV
							}
						}
					}
					change_location_owner = c:MLO
				}
			}
		}
		else = {
			error_log = "Tried to execute ver_milanese_demands peace deal without a valid flag"
		}
	}

	ai_desire = {
		add = {
			desc = "BASE"
			value = 10000
		}
	}
}
