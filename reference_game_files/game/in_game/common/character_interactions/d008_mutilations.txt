castrate_character = {
	message = yes
	on_own_nation = yes

	potential = {
		scope:actor = {
			has_or_had_tag = BYZ
			has_dlc = "d008_fate_of_the_phoenix"
		}
	}

	allow = {
		scope:actor = {
			has_ruler = yes
		}
	}

	ai_tick = never

	price = price:castrate_character_price

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
			is_alive = yes
			is_female = no
			NOR = { 
				has_trait = castrated
				has_trait = eunuch
			}
		}
		enabled = {
			is_adult = yes
			is_ruler = no
			is_regent = no
			NOT = { is_spouse_of = scope:actor.ruler }
		}
	}

	effect = {
		scope:actor = {
			add_prestige = 5
			if = {
				limit = { scope:recipient.estate_type = estate_type:crown_estate }
				add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_extreme_penalty }
			}
			else = {
				add_estate_satisfaction = { type = scope:recipient.estate_type value = estate_satisfaction_extreme_penalty }
			}
		}
		scope:recipient = {
			add_trait = trait:castrated
		}
	}

	ai_will_do = {}
}
