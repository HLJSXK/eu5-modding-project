# Culture and religion estate agendas
pa_accept_estate_culture = {
	type = country
	estate = nobles_estate
	estate = clergy_estate
	estate = burghers_estate
	estate = peasants_estate
	estate = dhimmi_estate
	estate = tribes_estate
	estate = cossacks_estate

	importance = 2

	potential = {
		NOT = {
			"estate(scope:target)" = {
				dominant_culture ?= {
					this = root.dominant_culture
				}
			}
		}
		NOT = {
			"estate(scope:target)" = {
				dominant_culture ?= {
					is_accepted_in = root
				}
			}
		}
		NOT = {
			"estate(scope:target)" = {
				dominant_culture ?= {
					this = root.culture
				}
			}
		}
	}

	on_accept = {
		"estate(scope:target)" = {
			dominant_culture = {
				save_scope_as = estate_dominant_culture
			}
		}
		add_accepted_culture = scope:estate_dominant_culture
		add_estate_satisfaction = { type = scope:target value = 0.1 }
	}

	can_bribe = {
		can_bribe_default_trigger = yes
	}
	on_bribe = {
		on_bribe_default_effect = yes
	}

	chance = 10
}

pa_change_court_language_to_estate_language = {
	type = country
	estate = nobles_estate
	estate = clergy_estate
	estate = burghers_estate
	estate = peasants_estate
	estate = dhimmi_estate
	estate = tribes_estate
	estate = cossacks_estate

	importance = 1.5

	potential = {
		# Don't propose if an overlord recently forced the court language, the subject status forbids change,
		# the player's own change_court_language is on cooldown, or parliament already changed it recently.
		modifier:prevented_from_changing_court_language_by_overlord = no
		NOT = { has_country_modifier = parliament_recently_changed_court_language }
		NOT = { has_cooldown = force_change_court_language }
		NOT = { has_cooldown = change_court_language }
		OR = {
			# Non-clergy estates: original culture-based behavior.
			AND = {
				NOT = { scope:target = { this = estate_type:clergy_estate } }
				NOT = {
					"estate(scope:target)" = {
						dominant_culture ?= {
							this = root.dominant_culture
						}
					}
				}
				NOT = {
					"estate(scope:target)" = {
						dominant_culture ?= {
							dialect = {
								this = root.court_dialect
							}
						}
					}
				}
				# Nobles only push for a change if their language differs from the country's primary culture's language.
				trigger_if = {
					limit = { scope:target = { this = estate_type:nobles_estate } }
					"estate(scope:target)" = {
						dominant_culture ?= {
							language = {
								NOT = { this = root.culture.language }
							}
						}
					}
				}
			}
			# Clergy: religion-first priority chain.
			AND = {
				scope:target = { this = estate_type:clergy_estate }
				OR = {
					# Case A: clergy religion = country religion, and court isn't already the liturgical dialect.
					AND = {
						"estate(scope:target)" = {
							dominant_religion ?= { this = root.religion }
						}
						NOT = { court_dialect = { this = root.liturgical_dialect } }
					}
					# Case B: clergy religion differs from country religion, has a fixed liturgical, court isn't it.
					"estate(scope:target)" = {
						dominant_religion ?= {
							NOT = { this = root.religion }
							has_fixed_liturgical_language = yes
							NOT = {
								language = { this = root.court_dialect.language }
							}
						}
					}
					# Case C (fallback): no liturgical preference applies — same triggers as the non-clergy path.
					AND = {
						NOT = {
							"estate(scope:target)" = {
								dominant_religion ?= { this = root.religion }
							}
						}
						NOT = {
							"estate(scope:target)" = {
								dominant_religion ?= { has_fixed_liturgical_language = yes }
							}
						}
						NOT = {
							"estate(scope:target)" = {
								dominant_culture ?= {
									this = root.dominant_culture
								}
							}
						}
						NOT = {
							"estate(scope:target)" = {
								dominant_culture ?= {
									dialect = {
										this = root.court_dialect
									}
								}
							}
						}
					}
				}
			}
		}
	}

	on_accept = {
		if = {
			limit = { scope:target = { this = estate_type:clergy_estate } }
			if = {
				limit = {
					"estate(scope:target)" = {
						dominant_religion ?= { this = root.religion }
					}
				}
				# Case A — country's current liturgical dialect.
				liturgical_dialect = {
					save_scope_as = court_language_target_dialect
				}
			}
			else_if = {
				limit = {
					"estate(scope:target)" = {
						dominant_religion ?= { has_fixed_liturgical_language = yes }
					}
				}
				# Case B — clergy religion's fixed liturgical language.
				"estate(scope:target)" = {
					dominant_religion = {
						language = {
							save_scope_as = court_language_target_dialect
						}
					}
				}
			}
			else = {
				# Case C — clergy's dominant culture dialect.
				"estate(scope:target)" = {
					dominant_culture = {
						dialect = {
							save_scope_as = court_language_target_dialect
						}
					}
				}
			}
		}
		else = {
			"estate(scope:target)" = {
				dominant_culture = {
					dialect = {
						save_scope_as = court_language_target_dialect
					}
				}
			}
		}
		set_court_language = scope:court_language_target_dialect
		# Use the action's own cooldown type so the engine auto-gates the Change Court Language button.
		add_cooldown = {
			type = change_court_language
			years = 20
		}
		add_country_modifier = {
			modifier = parliament_recently_changed_court_language
			years = 20
			mode = add_and_extend
		}
		add_estate_satisfaction = { type = scope:target value = 0.1 }
	}

	allow = {
		# Re-check at vote time in case any lockout was applied between proposal and acceptance.
		modifier:prevented_from_changing_court_language_by_overlord = no
		NOT = { has_country_modifier = parliament_recently_changed_court_language }
		NOT = { has_cooldown = force_change_court_language }
		NOT = { has_cooldown = change_court_language }
	}

	can_bribe = {
		can_bribe_default_trigger = yes
	}
	on_bribe = {
		on_bribe_default_effect = yes
	}

	chance = 100000
}
