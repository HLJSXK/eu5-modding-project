cb_crusade = {
	years = 25
	allow_white_peace = no
	create_visible = {
		always = no
	}
	create_enabled = {
		always = no
	}
	
	#scopes are actor/recipient for countries, while root = province checked.
	province = {
		any_location_in_province = {
			always = yes
		}
	}

	ai_cede_location_desire = {
		add = {
			if = {
				limit = {
					scope:location.area = area:levant_area
				}
				add = 50
			}
			if = {
				limit = {
					scope:location = {
						is_location_holy_site_for = root.religion
					}
				}
				add = 200
			}
			if = {
				limit = {
					scope:location = {
						dominant_religion.group = religion_group:christian
					}
				}
				add = 20
			}
			if = {
				limit = {
					scope:location = {
						OR = {
							sub_continent = sub_continent:western_europe
							region = region:balkan_region
							region = region:carpathia_region
							region = region:scandinavian_region
							region = region:baltic_region
							region = region:russian_region
							region = region:ruthenia_region
						}
					}
				}
				add = 10
			}			
		}
	}
	custom_tags = { religious_casus_belli }

	war_goal_type = crusade_conquest

	allow_ports_for_reach_ai = yes

	additional_war_enthusiasm = 0.1

	additional_war_enthusiasm_attacker = {
		value = 0.5
		subtract = {
			value = scope:war.war_length
			divide = 24
		}
		min = 0
	}
}
