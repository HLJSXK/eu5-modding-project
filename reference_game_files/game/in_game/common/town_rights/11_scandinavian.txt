scandinavian_fishing_privileges = {
	kept_at_conquest = no
	location_modifier = {
		local_sailors_modifier = 0.1
		local_maritime_presence = 0.1
		local_max_rgo_size_modifier = 0.5
	}

	country_modifier = {
		expand_rgo_gathering_cost_modifier = 0.05
	}

	allow = {
		scope:target = { is_port = yes }
	}

	color = goods_salt
}

scandinavian_bergslag_privileges = {
	kept_at_conquest = no
	location_modifier = {
		local_max_rgo_size_modifier = 0.5
		local_tools_output_modifier = 0.1
		local_peasant_enfranchisment = 0.05

		local_rgo_build_time = 0.5
	}

	country_modifier = {
		expand_rgo_mining_cost_modifier = 0.05
	}

	allow = {
		scope:target = {
			NOT = { has_town_rights = town_rights_type:royal_tooling_rights }
			OR = {
				raw_material = goods:iron
				raw_material = goods:copper
				raw_material = goods:silver
				raw_material = goods:lead
			}
		}
	}

	color = goods_copper
}
