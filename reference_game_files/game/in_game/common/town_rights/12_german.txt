#Magdeburg Rights North Germany
magdeburg_rights_town_rights = {
	kept_at_conquest = no

	location_modifier = {
		local_migration_attraction = 0.2
		local_merchant_capacity_modifier = 0.25

		local_pop_assimilation_speed_modifier = -0.5
	}
	country_modifier = {
		legislative_efficiency = -0.05
	}

	potential = {
		capital ?= {
			region = region:north_german_region
		}
		has_estate_privilege = estate_privilege:magdeburg_rights
	}

	allow = {
		scope:target = {
			NOR = {
				has_town_rights = town_rights_type:nuremberg_rights_town_rights
				has_town_rights = town_rights_type:kulm_town_rights
			}
		}
	}

	color = rgb { 189 115 99 } # tag MAG
}
