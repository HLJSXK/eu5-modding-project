pacify_rebel_pops = {

	custom_tooltip = pacify_rebel_pops_surrender_tt
	custom_tooltip = {
		text = every_supporter_pop_gains_20_satisfaction_tt
		every_pops_supporting_rebel = { add_pop_satisfaction = pop_satisfaction_extreme_bonus }
	}
}


grant_benefits_to_estate = {

	if = {
		limit = {
			exists = root.concession_privilege
		}
		root.owner = {
			grant_estate_privilege = root.concession_privilege
		}
	}

	if = {
		limit = {
			exists = root.concession_policy
		}
		root.owner = {
			add_policy = root.concession_policy
		}
	}

	owner = {
		add_estate_satisfaction = {
			type = root.estate_type
			value = estate_satisfaction_extreme_bonus
		}
	}
}
