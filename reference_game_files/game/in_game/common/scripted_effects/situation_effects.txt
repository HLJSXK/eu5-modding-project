enable_situation_effect = {
	custom_description = {
		text = enable_situation_effect
		value = situation:$type$
		set_global_variable = {
			name = enabled_situation_$type$
			value = yes
		}
	}
}
disable_situation_effect = {
	custom_description = {
		text = disable_situation_effect
		value = situation:$type$
		remove_global_variable = enabled_situation_$type$
	}
}

flag_situation_event = {
	$situation$ = {
		set_variable = situation_event_$event$_fired
	}
}

join_situation_faction = {
	$international_organization$ = {
		add_country_to_international_organization = $target$
	}
	hidden_effect = {
		$target$ = {
			set_variable = {
				name = $join_variable$
				value = $international_organization$
				years = $join_years$
			}
			if = {
				limit = {
					has_variable = $leave_variable$
					var:$leave_variable$ = $international_organization$
				}
				remove_variable = $leave_variable$
			}
		}
	}
}
leave_situation_faction = {
	if = {
		limit = { $target$ = { is_member_of_international_organization = $international_organization$ } }
		$international_organization$ = {
			if = {
				limit = {
					international_organization_has_leader = yes
					exists = leader_country
					leader_country = $target$
				}
				set_local_variable = io_revote
			}
			remove_country_from_international_organization = $target$
			if = {
				limit = { has_local_variable = io_revote }
				update_leader_country = yes
				remove_local_variable = io_revote
			}
		}
	}
	hidden_effect = {
		$target$ = {
			set_variable = {
				name = $leave_variable$
				value = $international_organization$
				years = $leave_years$
			}
			if = {
				limit = {
					has_variable = $join_variable$
					var:$join_variable$ = $international_organization$
				}
				remove_variable = $join_variable$
			}
		}
	}
}

rtr_change_allegiance_points = {
	custom_description = {
		text = rtr_change_allegiance_points
		value = $value$
		if = {
			limit = { NOT = { has_variable = rtr_yuan_allegiance } }
			set_variable = {
				name = rtr_yuan_allegiance
				value = 0
			}
		}
		change_variable = {
			name = rtr_yuan_allegiance
			add = $value$
		}
		if = {
			limit = { has_variable = rtr_yuan_allegiance }
			if = {
				limit = { var:rtr_yuan_allegiance > 100 }
				set_variable = {
					name = rtr_yuan_allegiance
					value = 100
				}
			}
			if = {
				limit = { var:rtr_yuan_allegiance < -100 }
				set_variable = {
					name = rtr_yuan_allegiance
					value = -100
				}
			}
		}
	}
}
rtr_recreate_middle_kingdom = {
	custom_tooltip = {
		text = rtr_recreate_middle_kingdom.tt
		trigger_event_non_silently = flavor_chi.115
		if = {
			limit = { NOT = { exists = international_organization:middle_kingdom } }
			create_international_organization = {
				type = international_organization_type:middle_kingdom
			}
			set_up_proper_china_for_middle_kingdom = yes
		}
		international_organization:middle_kingdom ?= {
			leader_country ?= {
				every_subject = {
					limit = { is_subject_type = tusi }
					overlord = { cancel_subject = prev }
				}
			}
			add_country_to_international_organization = prev
			set_leader_country = prev
		}
		change_celestial_authority = { value = celestial_authority_very_strong_bonus }
		if = {
			limit = { NOT = { has_bureaucracy_of_type = bureaucracy_type:six_boards_bureaucracy } }
			if = {
				limit = { NOT = { has_advance = chi_bureaucracy_advance } }
				research_advance = advance_type:chi_bureaucracy_advance
			}
			add_bureaucracy = bureaucracy_type:six_boards_bureaucracy
			every_current_bureaucracy = {
				limit = { bureaucracy_type = bureaucracy_type:six_boards_bureaucracy }
				set_entrenchment = 60
				set_maintenance = 0.50
			}
		}
		save_scope_as = country_to_pass_to_scripted_effect
		rtr_core_all_of_china_effect = { country = scope:country_to_pass_to_scripted_effect }
		every_present_culture_in_country = {
			limit = {
				has_culture_group = culture_group:chinese_group
				NOR = {
					is_primary_or_accepted_in = prev
					is_tolerated_in = prev
				}
			}
			prev = { add_tolerated_culture = prev }
		}
	}
}

rise_of_the_ottomans_recalculate_top_3 = {
	if = {
		limit = {
			any_in_global_list = {
				variable = eligible_beylik_list
				OR = {
					NOT = { country_exists = this }
					is_subject = yes
					NOR = {
						culture = culture:turkish_culture
						culture = culture:turkoman_culture
					}
				}
			}
		}
		every_in_global_list = {
			variable = eligible_beylik_list

			limit = {
				OR = {
					NOT = { country_exists = this }
					is_subject = yes
					NOR = {
						culture = culture:turkish_culture
						culture = culture:turkoman_culture
					}
				}
			}

			save_temporary_scope_as = target_country
			remove_list_global_variable = { 
				name = eligible_beylik_list 
				target = scope:target_country
			}
		}
	}
	ordered_in_global_list = {  
		variable = eligible_beylik_list
		limit = { is_subject = no }
		order_by = {
			
			add = {
				value = military_strength
			}
			add = {
				value = country_tax_base
			}
		} 
		max = 1
		check_range_bounds = no 
		situation:rise_of_the_ottomans = {
			set_variable = {
				name = strongest_beylik_variable
				value = prev
			}
			set_variable = {
				name = strongest_strongest_beylik_total_score_variable
				value = {
					add = var:strongest_beylik_variable.military_strength
					add = var:strongest_beylik_variable.country_tax_base 
				}
			}
		}
	}
	
	if = {
		limit = {
			global_variable_list_size = {
				name = eligible_beylik_list
				value > 1
			}
		}
		ordered_in_global_list = {	#2nd strongest  
			variable = eligible_beylik_list
			limit = { 
				this != situation:rise_of_the_ottomans.var:strongest_beylik_variable 
				is_subject = no
			}
			order_by = {
				
				add = {
					value = military_strength
				}
				add = {
					value = country_tax_base
				}
			} 
			max = 1
			check_range_bounds = no
			situation:rise_of_the_ottomans = {
				set_variable = {
					name = second_strongest_strongest_beylik_variable
					value = prev
				}
				set_variable = {
					name = second_strongest_strongest_beylik_total_score_variable
					value = {
						add = var:second_strongest_strongest_beylik_variable.military_strength
						add = var:second_strongest_strongest_beylik_variable.country_tax_base 
					}
				}
			}
		}
	}
	else = {
		situation:rise_of_the_ottomans = {
			remove_variable = second_strongest_strongest_beylik_variable
			remove_variable = second_strongest_strongest_beylik_total_score_variable
		}
	}

	if = {
		limit = {
			global_variable_list_size = {
				name = eligible_beylik_list
				value > 2
			}
		}
		ordered_in_global_list = { #3rd Strongest 
			variable = eligible_beylik_list
			limit = { 
				this != situation:rise_of_the_ottomans.var:strongest_beylik_variable 
				this != situation:rise_of_the_ottomans.var:second_strongest_strongest_beylik_variable 
				is_subject = no
			}
			order_by = {
				
				add = {
					value = military_strength
				}
				add = {
					value = country_tax_base
				}
			} 
			max = 1
			check_range_bounds = no 
			situation:rise_of_the_ottomans = {
				set_variable = {
					name = third_strongest_strongest_beylik_variable
					value = prev
				}
				set_variable = {
					name = third_strongest_strongest_beylik_total_score_variable
					value = {
						add = var:third_strongest_strongest_beylik_variable.military_strength
						add = var:third_strongest_strongest_beylik_variable.country_tax_base
					}
				}
			}
		}
	}
	else = {
		situation:rise_of_the_ottomans = {
			remove_variable = third_strongest_strongest_beylik_variable
			remove_variable = third_strongest_strongest_beylik_total_score_variable
		}
	}
}

council_of_trent_force_end_effect = {
	if = {
		limit = {
			is_situation_active = situation:council_of_trent
			NOT = {
				exists = international_organization:catholic_church
			}
		}
		end_situation = { situation = situation:council_of_trent outcome = "council_of_trent_church_dissolved" }
	}
	else_if = {
		limit = {
			is_situation_active = situation:council_of_trent
			situation:council_of_trent = {
				custom_tooltip = {
					text = council_of_trent_end_trigger_passed_debates_tt
					has_variable = council_of_trent_passed_debates
					has_variable = council_of_trent_passed_debates_max
					var:council_of_trent_passed_debates >= var:council_of_trent_passed_debates_max
				}
			}
		}
		end_situation = { situation = situation:council_of_trent outcome = "council_of_trent_reforms_adopted" }
	}
	else_if = {
		limit = {
			is_situation_active = situation:council_of_trent
			situation:council_of_trent = {
				custom_tooltip = {
					text = council_of_trent_end_trigger_failed_debates_tt
					has_variable = council_of_trent_failed_debates
					has_variable = council_of_trent_failed_debates_max
					var:council_of_trent_failed_debates >= var:council_of_trent_failed_debates_max
				}
			}
		}
		end_situation = { situation = situation:council_of_trent outcome = "council_of_trent_reforms_rejected" }
	}
	else_if = {
		limit = {
			is_situation_active = situation:council_of_trent
			situation:council_of_trent = {
				current_age_or_later = { age = age_5_absolutism }
			}
		}
		end_situation = { situation = situation:council_of_trent outcome = "council_of_trent_council_adjourned" }
	}
}

council_of_trent_proposal_effect = {
	if = {
		limit = {
			exists = scope:recipient
			exists = scope:target
			exists = scope:vote
			scope:recipient = international_organization:catholic_church
			is_situation_active = situation:council_of_trent
		}
		situation:council_of_trent = {
			set_variable = {
				name = council_of_trent_current_debate
				value = scope:vote
			}
			add_to_global_variable_list = { name = council_of_trent_voting_for_debate target = scope:proposer }
			remove_list_global_variable = { name = council_of_trent_unvoters target = scope:proposer }
		}
		international_organization:catholic_church = {
			every_international_organization_member = {
				limit = { NOT = { this = scope:proposer } }
				add_to_global_variable_list = { name = council_of_trent_unvoters target = this }
			}
		}
	}
}

council_of_trent_pass_effect = {
	if = {
		limit = {
			exists = scope:recipient
			scope:recipient = international_organization:catholic_church
			is_situation_active = situation:council_of_trent
		}
		if = {
			limit = {
				exists = scope:vote
				situation:council_of_trent = {
					has_variable = council_of_trent_current_debate
					var:council_of_trent_current_debate = scope:vote
				}
			}
			situation:council_of_trent = {
				change_variable = { name = council_of_trent_passed_debates add = 1 }
				remove_variable = council_of_trent_current_debate
			}
			international_organization:catholic_church = {
				if = {
					limit = { NOT = { has_variable = council_of_trent_reforms_done_counter } }
					set_variable = { name = council_of_trent_reforms_done_counter value = 1 }
				}
				else = {
					change_variable = { name = council_of_trent_reforms_done_counter add = 1 }
				}
			}
		}
		# Winner policy != proposed policy: engine routes through pass_effect, but the proposer's debate failed.
		else_if = {
			limit = {
				situation:council_of_trent = {
					has_variable = council_of_trent_current_debate
				}
			}
			situation:council_of_trent = {
				change_variable = { name = council_of_trent_failed_debates add = 1 }
				remove_variable = council_of_trent_current_debate
			}
			international_organization:catholic_church = {
				if = {
					limit = { NOT = { has_variable = council_of_trent_reforms_done_counter } }
					set_variable = { name = council_of_trent_reforms_done_counter value = -1 }
				}
				else = {
					change_variable = { name = council_of_trent_reforms_done_counter subtract = 1 }
				}
			}
		}
		clear_global_variable_list = council_of_trent_unvoters
		clear_global_variable_list = council_of_trent_voting_for_debate
		clear_global_variable_list = council_of_trent_voting_against_debate
	}
	council_of_trent_force_end_effect = yes
}

		limit = {

			is_situation_active = situation:council_of_trent

		remove_list_global_variable = { name = council_of_trent_unvoters target = scope:actor }

					has_voted_for = {

					}
				}
			}
			add_to_global_variable_list = { name = council_of_trent_voting_for_debate target = scope:actor }
		}
		else = {
			add_to_global_variable_list = { name = council_of_trent_voting_against_debate target = scope:actor }
		}
	}
}

reconcile_hre_league_leadership = {
	international_organization:foreign_league_hre ?= {
		if = {
			limit = {
				international_organization:hre.leader_country ?= {
					NOR = {
						is_member_of_international_organization = international_organization:foreign_league_balkan
						is_member_of_international_organization = international_organization:foreign_league_france
						is_member_of_international_organization = international_organization:foreign_league_iberia
					}
				}
				NOT = { leader_country = international_organization:hre.leader_country }
			}
			international_organization:hre.leader_country = {
				save_temporary_scope_as = new_hre_league_leader
			}
			scope:new_hre_league_leader = {
				if = {
					limit = { is_leader_of_international_organization = international_organization:italian_league_1 }
					international_organization:italian_league_1 = { save_temporary_scope_as = absorbed_italian_league }
				}
				else_if = {
					limit = { is_leader_of_international_organization = international_organization:italian_league_2 }
					international_organization:italian_league_2 = { save_temporary_scope_as = absorbed_italian_league }
				}
				else_if = {
					limit = { is_leader_of_international_organization = international_organization:italian_league_3 }
					international_organization:italian_league_3 = { save_temporary_scope_as = absorbed_italian_league }
				}
			}
			if = {
				limit = { exists = scope:absorbed_italian_league }
				scope:absorbed_italian_league = {
					every_international_organization_member = {
						add_to_temporary_list = absorbed_league_members
					}
				}
				every_in_list = {
					list = absorbed_league_members
					international_organization:foreign_league_hre = {
						add_country_to_international_organization = prev
					}
				}
			}
			if = {
				limit = {
					scope:new_hre_league_leader = {
						NOT = { is_member_of_international_organization = international_organization:foreign_league_hre }
					}
				}
				add_country_to_international_organization = scope:new_hre_league_leader
			}
			set_leader_country = scope:new_hre_league_leader
		}
	}
}

reconcile_single_italian_wars_league_land = {
	#this = a league international organization
	save_temporary_scope_as = iw_reconcile_league
	#Drop recorded land whose current owner is no longer a member of this league
	every_international_organization_owned_location = {
		limit = {
			OR = {
				has_owner = no
				owner ?= { NOT = { is_member_of_international_organization = scope:iw_reconcile_league } }
			}
		}
		scope:iw_reconcile_league = {
			remove_location_from_international_organization = prev
		}
	}
	#Add any Italy-region land a current member owns that the league has not recorded yet
	every_international_organization_member = {
		every_owned_location = {
			limit = {
				region = region:italy_region
				is_land = yes
				is_passable = yes
				NOT = { is_owned_by_international_organization = scope:iw_reconcile_league }
			}
			scope:iw_reconcile_league = {
				add_location_to_international_organization = prev
			}
		}
	}
	
}

						}

							attacker_leader = international_organization:hre.leader_country

				}
			}
			random_current_war = {

						AND = {
							attacker_leader = c:BOH
							defender_leader = international_organization:hre.leader_country
						}

						}

					}

			}
		}

							}
						}
					}
					every_attacker = {
						hussite_wars_add_country_to_side_effect = {
							LIST = hussite_wars_hussite_side_list
							STRENGTH = hussite_wars_hussite_strength_variable
						}
					}
					every_defender = {
						hussite_wars_add_country_to_side_effect = {
							LIST = hussite_wars_catholic_side_list
							STRENGTH = hussite_wars_catholic_strength_variable
						}
					}
				}
				else = {
					attacker_leader = {
						situation:hussite_wars = {
							set_variable = {
								name = hussite_wars_catholic_leader
								value = prev
							}
						}
					}
					every_defender = {
						hussite_wars_add_country_to_side_effect = {
							LIST = hussite_wars_hussite_side_list
							STRENGTH = hussite_wars_hussite_strength_variable
						}
					}
					every_attacker = {
						hussite_wars_add_country_to_side_effect = {
							LIST = hussite_wars_catholic_side_list
							STRENGTH = hussite_wars_catholic_strength_variable
						}
					}
				}
			}
		}
		else = {
			if = {
				limit = { country_exists = international_organization:hre.leader_country }
				set_variable = {
					name = hussite_wars_catholic_leader
					value = international_organization:hre.leader_country
				}
			}
			c:BOH ?= {
				hussite_wars_add_country_to_side_effect = {
					LIST = hussite_wars_hussite_side_list
					STRENGTH = hussite_wars_hussite_strength_variable
				}
				every_other_country = {
					limit = {
						OR = {
							AND = {
								is_subject_or_below_of = c:BOH
								subject_loyalty > 50
							}
							has_mutual_scripted_relation = {
								type = relation_type:alliance
								target = c:BOH
							}
						}
					}
					hussite_wars_add_country_to_side_effect = {
						LIST = hussite_wars_hussite_side_list
						STRENGTH = hussite_wars_hussite_strength_variable
					}
				}
			}
			international_organization:hre.leader_country ?= {
				hussite_wars_add_country_to_side_effect = {
					LIST = hussite_wars_catholic_side_list
					STRENGTH = hussite_wars_catholic_strength_variable
				}
				every_other_country = {
					limit = {
						NOT = { tag = BOH }
						NOT = {
							is_target_in_global_variable_list = {
								name = hussite_wars_hussite_side_list
								target = this
							}
						}
						OR = {
							AND = {
								is_subject_or_below_of = international_organization:hre.leader_country
								subject_loyalty > 50
							}
							has_mutual_scripted_relation = {
								type = relation_type:alliance
								target = international_organization:hre.leader_country
							}
						}
					}
					hussite_wars_add_country_to_side_effect = {
						LIST = hussite_wars_catholic_side_list
						STRENGTH = hussite_wars_catholic_strength_variable
					}
				}
			}
		}
	}

	hussite_wars_recompute_relative_strength_effect = yes
}

hussite_wars_recompute_relative_strength_effect = {
	situation:hussite_wars = {
		set_variable = {
			name = hussite_wars_hussite_relative_strength_variable
			value = {
				value = var:hussite_wars_hussite_strength_variable
				divide = {
					value = 1
					add = var:hussite_wars_hussite_strength_variable
					add = var:hussite_wars_catholic_strength_variable
				}
			}
		}
		set_variable = {
			name = hussite_wars_catholic_relative_strength_variable
			value = {
				value = var:hussite_wars_catholic_strength_variable
				divide = {
					value = 1
					add = var:hussite_wars_hussite_strength_variable
					add = var:hussite_wars_catholic_strength_variable
				}
			}
		}
	}
}

hussite_wars_add_war_joiner_to_side_effect = {
	save_temporary_scope_as = hw_joiner
	if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = hussite_wars_hussite_side_list
					target = scope:hw_joiner
				}
			}
			NOT = {
				is_target_in_global_variable_list = {
					name = hussite_wars_catholic_side_list
					target = scope:hw_joiner
				}
			}
		}
		if = {
			limit = {
				scope:war = {
					OR = {
						AND = {
							any_attacker = { tag = BOH }
							is_an_attacker = scope:hw_joiner
						}
						AND = {
							any_defender = { tag = BOH }
							is_a_defender = scope:hw_joiner
						}
					}
				}
			}
			hussite_wars_add_country_to_side_effect = {
				LIST = hussite_wars_hussite_side_list
				STRENGTH = hussite_wars_hussite_strength_variable
			}
		}
		else = {
			hussite_wars_add_country_to_side_effect = {
				LIST = hussite_wars_catholic_side_list
				STRENGTH = hussite_wars_catholic_strength_variable
			}
		}
		hussite_wars_recompute_relative_strength_effect = yes
	}
}

hussite_wars_clear_sides_effect = {
	clear_global_variable_list = hussite_wars_hussite_side_list
	clear_global_variable_list = hussite_wars_catholic_side_list
}
