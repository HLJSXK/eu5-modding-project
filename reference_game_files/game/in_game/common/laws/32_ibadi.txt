
ibadi_shurat_law = {
	law_category = religious
	potential = {
		religion = religion:ibadi
	}
	ibadi_institutionalize_the_shurat_policy = {
		country_modifier = {
			mobilize_the_shurat_price_cost_modifier = -0.33
			pure_tooltip_entry = enables_shurat_bureaucracy
		}

		allow = {
			trigger_if = {
				limit = {
					has_policy = ibadi_institutionalize_the_shurat_policy
				}
				has_policy = ibadi_institutionalize_the_shurat_policy
			}
			trigger_else = {
				AND = {
					"estate_power(estate_type:crown_estate)" > 0.25
					num_of_non_rural >= 1
				}
			}
		}

		estate_preferences = {
			crown_estate
			nobles_estate
		}
		
		years = 2
	}

	ibadi_maintain_the_status_of_imam_policy = {
		country_modifier = {
			pure_tooltip_entry = enables_ibadi_shari_imam_policy
			global_levy_size_modifier = 0.5
			pop_join_rebel_threshold = 0.1
		}
		years = 2
	}

	# If adding a new policy here, inspect mobilize_the_shurat_action
}

ibadi_imamate_law = {
	law_category = religious
	potential = {
		religion = religion:ibadi
	}
	ibadi_kitman_imam_policy = {
		country_modifier = {
			global_clergy_estate_power = 0.5
			global_max_control = -0.1
			tax_income_efficiency = 0.1
		}
		allow = {
			is_forced_into_kitman_state = yes
		}
		years = 0
	}
	ibadi_zuhur_imam_policy = {
		country_modifier = {
			global_monthly_prosperity = 0.001
			clergy_estate_target_satisfaction = medium_permanent_target_satisfaction
		}
		allow = {
			trigger_if = {
				limit = {
					has_policy = ibadi_zuhur_imam_policy
				}
				has_policy = ibadi_zuhur_imam_policy
			}
			trigger_else = {
				is_forced_into_kitman_state = no
				trigger_if = {
					limit = {
						has_policy = ibadi_daif_imam_policy
					}
					can_be_changed_to_via_generic_action = {
						action = return_to_zuhur_imamate
					}
				}
				NOT = { has_policy = ibadi_shari_imam_policy }
			}
		}
		years = 0
	}
	ibadi_daif_imam_policy = {
		country_modifier = {
			global_clergy_estate_power = 1
			land_morale_modifier = 0.15
			discipline = 0.05
		}
		country_modifier = {
			clergy_estate_target_satisfaction = tiny_permanent_target_satisfaction_penalty
			potential_trigger = {
				has_variable = weak_imam_penalty
			}
			display_trigger = {
				custom_tooltip = {
					text = has_penalty_for_weak_imam
					has_variable = weak_imam_penalty
					country_allows_a_weak_imam = no
				}
			}
			scale = {
				value = 0
				if = {
					limit = {
						has_variable = weak_imam_penalty
					}
					add = "months_difference(current_date|var:weak_imam_penalty)"
				}
				divide = 12
			}
		}
		allow = {
			trigger_if = {
				limit = {
					has_policy = ibadi_daif_imam_policy
				}
				has_policy = ibadi_daif_imam_policy
			}
			trigger_else = {
				is_forced_into_kitman_state = no
				can_be_changed_to_via_generic_action = {
					action = elect_weak_imam
				}
			}
		}
		years = 0
	}
	ibadi_shari_imam_policy = {
		country_modifier = {
			land_morale_modifier = 0.1 # Should be 2 * imam_al_shari_weak_modifier
		}
		allow = {
			trigger_if = {
				limit = {
					has_policy = ibadi_shari_imam_policy
				}
				has_policy = ibadi_shari_imam_policy
			}
			trigger_else = {
				is_forced_into_kitman_state = no
				has_policy = ibadi_maintain_the_status_of_imam_policy
				can_be_changed_to_via_generic_action = {
					action = mobilize_the_shurat_action
				}
			}
		}
		years = 0
	}
}
