independence_movement = {
	has_target = yes
	unique = no
	expel_members_who_are_targets_of_other_members = yes
	expel_members_who_are_attackers_at_war_with_other_members = yes
	has_leader_country = yes
	leader_type = country
	leader_change_method = none
	disband_if_no_leader = yes
	show_on_diplomatic_map = yes
	subject_limited = no #even subjects with limited diplomacy can create these
	show_strength_comparison_with_target = yes
	declare_war_on_target_casus_belli = cb_independence_war

	custom_name = independence_movement_name
	
	#member_color = rgb { 40 160 100 }
	leader_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_LEADER_COLOR
	target_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_TARGET_COLOR

	join_offensive_wars_always = {
		custom_tooltip = {
			text = independence_movement_join_war_always_tt
			scope:target ?= {
				this = root.international_organization_target
			}
			trigger_if = {
				limit = {
					exists = scope:casus_belli
				}
				scope:casus_belli = casus_belli:cb_independence_war
			}
		}
	}
	
	join_defensive_wars_always = {
		or = {
			custom_tooltip = {
				text = independence_movement_join_war_def_always_tt
				scope:target ?= {
					this = root.international_organization_target
				}
			}
			custom_tooltip = {
				text = independence_movement_join_war_def_always_tt_during
				scope:recipient = {
					is_at_war_with = root.international_organization_target
				}
			}
		}
	}

	create_visible_trigger = {
		is_subject = yes
		trigger_if = {
			limit = { exists = overlord }
			overlord = scope:target
		}
		trigger_else = { always = no }
	}

	create_enabled_trigger = {
		is_disloyal_subject = yes
	}

	invite_visible_trigger = {
		always = yes
	}

	can_declare_war = {
		always = yes
	}

	potential_target_trigger = {
		is_overlord = yes
		any_subject = {
			is_disloyal_subject = yes
			not = { is_allied_with = { target = root } }
			not = { is_at_war_with = root }
			not = { has_truce_with = root }
			not = { is_fighting_war_together_with = root }
		}
	}

	can_target_trigger = {
	}

	can_lead_trigger = {
		is_subject = yes
	}

	can_join_trigger = {
		trigger_if = {
			limit = {
				exists = scope:recipient
				scope:recipient = {
					international_organization_has_leader = yes
				}
			}
			opinion = {
				target = scope:recipient.leader_country
				value > 0
			}
		}
		is_rebel_country = no
		not = { is_allied_with = { target = scope:target } }
		not = { is_at_war_with = scope:target }
		not = { has_truce_with = scope:target }
		not = { is_fighting_war_together_with = scope:target }
		scope:target = {
			NOT = {
				any_overlord_or_above = {
					this = root
				}
			}
		}
		#either we can do our own diplomacy; or our overlord is the same as the leader's
		or = {
			has_limited_diplomacy = no
			is_subject_of = scope:target
		}
	}
	
	can_leave_trigger = {
		always = yes
	}
	
	auto_leave_trigger = {
		or = {
			is_allied_with = { target = scope:target }
			has_truce_with = scope:target
			AND = {
				is_subject_of = scope:target
				subject_loyalty > 80
			}
			AND = {
				has_limited_diplomacy = yes
				is_subject = yes
				NOT = { is_subject_of = scope:target }
			}
		}
	}
	
	auto_disband_trigger = {
		leader_country = {
			or = {
				subject_loyalty > 80
				not = { overlord ?= scope:target }
				is_allied_with = { target = scope:target }
				has_truce_with = scope:target
			}
		}
	}

	on_joined = {
		if = {
			limit = {
				scope:recipient = {
					total_members = 1
				}
			}
			add_casus_belli = {
				type = casus_belli:cb_independence_war
				province = scope:recipient.leader_country.capital.province
				target = scope:target
			}
		}
		scope:target = {
			random_current_war = {
				limit = {
					attacker_leader = scope:recipient.leader_country
					defender_leader = scope:target
				}
				save_scope_as = independence_war
			}
		}
		if = {
			limit = {
				exists = scope:independence_war
			}
			join_war_as_attacker = { war = scope:independence_war }
		}
	}

	on_left = {
	}
	
	monthly_effect = {
	}
	
	ai_desire_to_join = {
		#Base value
		add = {
			desc = "BASE_REASON"
			value = -50
		}

		#Relative strength of the movement
		save_temporary_value_as = {
			name = relative_strength_of_io
			value = {
				value = 0
				if = {
					limit = {
						exists = scope:recipient
					}
					scope:recipient = {
						every_international_organization_member = {
							add = country_strength
						}
					}
					add = "root.country_strength"
				}
				else = {
					add = country_strength
				}		
			}
		}
		save_temporary_value_as = {
			name = relative_strength_of_target
			value = {
				value = 0
				scope:target = {
					add = country_strength
					every_subject = {
						limit = {
							exists = scope:recipient 
							is_member_of_international_organization = scope:recipient 
						}
						subtract = country_strength
					}
					every_related_country = {
						type = alliance
						limit = { 
							OR = {
								NOT = { exists = scope:recipient }
								NOT = { is_member_of_international_organization = scope:recipient } 
							}						
						}
						add = country_strength
					}
				}
			}
		}
		save_temporary_value_as = {
			name = combined_relative_strength
			value = {
				value = scope:relative_strength_of_target
				add = scope:relative_strength_of_io
			}
		}
		if = {
			limit = {
				exists = scope:relative_strength_of_io
				exists = scope:relative_strength_of_target
				scope:relative_strength_of_target != 0
				scope:combined_relative_strength != 0
			}
			add = {
				#get a value in range of -50..0..25
				desc = "RELATIVE_STRENGTH_IO"
				value = 100
				multiply = {
					value = scope:relative_strength_of_io
					divide = scope:combined_relative_strength
				}
				subtract = 50
				max = 25
			}
		}
		else = {
			add = {
				#get a value in range of -50..0..25
				desc = "RELATIVE_STRENGTH_TO_TARGET"
				value = 100
				multiply = "root.relative_strength(scope:target)" #this is 0..1
				subtract = 50
				max = 25
			}
		}

		#"We have our own issues and won't be joining anyone"
		subtract = {
			desc = "DIPLOREASON_IN_DEBT"
			value = num_loans
			multiply = 20
		}
		subtract = {
			desc = "[war_exhaustion|e]"
			value = war_exhaustion
			multiply = 2
		}
		if = {
			limit = { is_during_bankruptcy = yes }
			subtract = {
				desc = "IS_BANKRUPT_TOOLTIP"
				value = 1000
			}
		}
		if = {
			limit = { at_war = yes }
			subtract = {
				desc = "IS_AT_WAR"
				value = 50
			}
		}

		#"Do we like the actor / the subjects in the movement?"
		if = {
			limit = {
				exists = scope:recipient
			}
			if = {
				limit = { is_subject_of = scope:target }
				#Subjects of the target country should less care about the opinion of the other subjects who want independence.. the enemy of my enemy kind of deal here
				if = {
					limit = {
						scope:recipient = {
							any_international_organization_member = {
								in_union_with = root
							}
						}
					}
					add = {
						desc = "SAME_UNION_AS_ORGANIZATION"
						value = 200
					}
				}
			}
			else = {
				save_temporary_value_as = {
					name = subjects_number
					value = {
						scope:recipient = {
							every_international_organization_member = {
								limit = {
									is_subject_of = scope:target
								}
								add = 1
							}
						}
					}
				}
				save_temporary_value_as = {
					name = average_subjects_opinion
					value = {
						if = {
							limit = { scope:subjects_number > 0 }
							scope:recipient = {
								every_international_organization_member = {
									limit = { is_subject_of = scope:target }
									add = "opinion(root)"
									if = {
										limit = { is_leader_of_international_organization = scope:recipient }
										multiply = 2	#Opinion of the leader country is weighted twice
									}
									if = {
										limit = { is_rival_of = root }
										subtract = 100
									}
									if = {
										limit = { is_enemy_of = root }
										subtract = 50
									}
								}
							}
							divide = scope:subjects_number
							multiply = 0.4
						}
						else = { value = 0 }
					}
				}
				save_temporary_value_as = {
					name = average_subject_trust
					value = {
						if = {
							limit = { scope:subjects_number > 0 }
							scope:recipient = {
								every_international_organization_member = {
									limit = { is_subject_of = scope:target }
									add = "trust(root)"
									subtract = 50
								}
							}
							divide = scope:subjects_number
						}
						else = { value = 0 }
					}
				}
				add = {
					desc = "DIPLOREASON_AVERAGE_OPINION_OF_ORGANIZATION"
					value = scope:average_subjects_opinion
				}
				add = {
					desc = "DIPLOREASON_AVERAGE_TRUST_OF_ORGANIZATION"
					value = scope:average_subject_trust
				}

				save_temporary_value_as = {
					name = non_subjects_number
					value = {
						value = scope:recipient.total_members
						subtract = scope:subjects_number
					}
				}
				#Do we *really* have to join though? - Let's see if other non-subject countries are already in there
				subtract = {
					desc = "ORGANIZATION_POWER"
					value = {
						scope:recipient = {
							every_international_organization_member = {
								add = country_economical_base
							}
						}
						subtract = scope:target.country_economical_base
						min = 0
						max = 300
					}
				}
			}
		}
		else_if = {
			limit = { exists = scope:actor }
			if = {
				limit = { is_rival_of = scope:actor }
				subtract = {
					desc = "RIVAL_OF_ACTOR"
					value = 100
				}
			}
			if = {
				limit = { is_enemy_of = scope:actor }
				subtract = {
					desc = "ENEMY_OF_ACTOR"
					value = 50
				}
			}
			add = {
				desc = "OPINION_OF_SUBJECT"
				scope:actor = {
					value = prev_opinion_of_this
					multiply = 0.4
				}
			}
			add = {
				desc = "TRUST_OF_SUBJECT"
				scope:actor = {
					value = prev_trust_of_this
					subtract = 50
				}
			}
			if = {
				limit = {
					in_union_with = scope:actor
					is_subject_of = scope:target
				}
				add = {
					desc = "SAME_UNION_AS_ACTOR"
					value = 200
				}
			}
		}

		#"Do we hate the target?"
		if = {
			limit = { is_rival_of = scope:target }
			add = {
				desc = "RIVAL_OF_TARGET"
				value = 100
			}
		}
		if = {
			limit = { is_enemy_of = scope:target }
			add = {
				desc = "ENEMY_OF_TARGET"
				value = 50
			}
		}
		add = {
			desc = "ANTAGONISM_OF_TARGET"
			scope:target = {
				value = prev_antagonism_towards_this
			}
		}
		subtract = {
			desc = "OPINION_OF_TARGET"
			scope:target = {
				value = prev_opinion_of_this
				multiply = 0.4
			}
		}		
		add = {
			desc = "TRUST_OF_TARGET"
			scope:target = {
				value = prev_trust_of_this
				subtract = 50
				multiply = -1
			}
		}

		#"Wait, can we even reach the target?" - only relevant for non-subjects
		if = {
			limit = { NOT = { is_subject_of = scope:target } }
			subtract = {
				desc = "DISTANCE_TO_TARGET"
				scope:target = {
					value = "border_distance_to(root)"
					multiply = 0.66
				}
				if = {
					limit = {
						scope:target = {
							"border_distance_to(root)" > 300
						}
					}
					add = 1000
				}
			}
		}
		else = {	#Subject reasons
			#"Are we disloyal enough to want to break free?"
			add = {
				desc = "game_concept_subject_loyalty"
				value = 50
				subtract = subject_loyalty
				multiply = 5
			}
		}

		#Content related reasons
		if = {
			limit = {
				exists = scope:recipient
			}
			if = {
				limit = {
					tag = CAS
					scope:recipient.leader_country = {
						has_variable = mll_sought_castile_support
					}
					opinion = {
						target = scope:recipient.leader_country
						value >= 150
					}
				}
				add = {
					desc = "MLL_PLEDGED_SUPPORT"
					value = 50
				}
			}
			if = {
				limit = {
					tag = NAP
					scope:recipient.leader_country = {
						has_variable = mil_sought_naples_support
					}
					opinion = {
						target = scope:recipient.leader_country
						value >= 125
					}
				}
				add = {
					desc = "MLL_PLEDGED_SUPPORT"
					value = 50
				}
			}
			if = {
				limit = {
					tag = GEN
					scope:recipient.leader_country = {
						has_variable = mil_sought_genoa_support
					}
					opinion = {
						target = scope:recipient.leader_country
						value >= 100
					}
				}
				add = {
					desc = "MLL_PLEDGED_SUPPORT"
					value = 50
				}
			}
			if = {
				limit = {
					tag = FRA
					scope:recipient.leader_country = {
						has_variable = mll_fra_guarantees_for_independence
					}
				}
				add = {
					desc = "MLL_PLEDGED_SUPPORT"
					value = 100
				}
			}
		}
	}
	
	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}
