italian_wars = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_italian_wars
	
	can_start = {
		NOT = { is_situation_active = situation:guelphs_and_ghibellines }
		current_year > 1450

		OR = {
			current_year > 1495
			any_great_power = {
				any_enemy_war_leader = { capital.region = region:italy_region }
			}
		}
	}
	
	can_end = {
		situation:italian_wars = { years_since_situation_start > 50 }
		OR = {
			international_organization:foreign_league_balkan ?= {
				total_locations_owned >= 200
			}
			international_organization:foreign_league_france ?= {
				total_locations_owned >= 200
			}
			international_organization:foreign_league_hre ?= {
				total_locations_owned >= 200
			}
			international_organization:foreign_league_iberia ?= {
				total_locations_owned >= 200
			}
			international_organization:italian_league_1 ?= {
				total_locations_owned >= 200
			}
			international_organization:italian_league_2 ?= {
				total_locations_owned >= 200
			}
			international_organization:italian_league_3 ?= {
				total_locations_owned >= 200
			}
			custom_tooltip = {
				text = no_league_io_owns_any_land_in_italy_tt
				AND = {
					years_since_situation_start > 1
					international_organization:foreign_league_balkan ?= {
						total_locations_owned < 1
					}
					international_organization:foreign_league_france ?= {
						total_locations_owned < 1
					}
					international_organization:foreign_league_hre ?= {
						total_locations_owned < 1
					}
					international_organization:foreign_league_iberia ?= {
						total_locations_owned < 1
					}
					international_organization:italian_league_1 ?= {
						total_locations_owned < 1
					}
					international_organization:italian_league_2 ?= {
						total_locations_owned < 1
					}
					international_organization:italian_league_3 ?= {
						total_locations_owned < 1
					}
				}
			}
			situation:italian_wars = { years_since_situation_start > 50 }
		}
	}
	visible = {
		OR = {
			has_presence_in = region:italy_region
			is_leader_of_international_organization = international_organization:foreign_league_balkan
			is_leader_of_international_organization = international_organization:foreign_league_france
			is_leader_of_international_organization = international_organization:foreign_league_hre
			is_leader_of_international_organization = international_organization:foreign_league_iberia

			international_organization:foreign_league_balkan ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}

			international_organization:foreign_league_france ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}
			international_organization:foreign_league_hre ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}
			international_organization:foreign_league_iberia ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}

			international_organization:italian_league_1 ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}
			international_organization:italian_league_2 ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}
			international_organization:italian_league_3 ?= {
				leader_country ?= {
					OR = {
						is_allied_with = { target = root }
						is_enemy_of   = root
						is_rival_of   = root
						is_subject_of = root
					}
				}
			}
		}
	}
	

	on_start = {
		#Calculate Strongest Balkan Country
		if = {
			limit = {
				region:balkan_region = {
					any_country_with_capital_in_geography = {
						basic_checks_for_calculating_foreign_league_leaders = yes
						religion.group = religion_group:christian
						owns = location:constantinople
						num_locations_owned_or_owned_by_subjects >= 100
					}
				}
			}
			region:balkan_region = {
				ordered_country_with_capital_in_geography = {
					limit = { 
						basic_checks_for_calculating_foreign_league_leaders = yes 
						owns = location:constantinople
						religion.group = religion_group:christian
						num_locations_owned_or_owned_by_subjects >= 100
					}
					order_by = military_strength
					max = 1
					check_range_bounds = no
					save_temporary_scope_as = target_balkan_leader
					create_international_organization = {
						type = international_organization_type:foreign_league_balkan
						add_country_to_international_organization = scope:target_balkan_leader
						set_leader_country = scope:target_balkan_leader
					}
					international_organization:foreign_league_balkan = {
						add_to_global_variable_list = {
							name = iw_foreign_leagues_list
							target = this 
						}
					}
					trigger_event_non_silently = italian_wars.1
				}
			}
		}

		#Calculate Strongest French Country
		region:france_region = {
			ordered_country_with_capital_in_geography = {
				limit = { 
					basic_checks_for_calculating_foreign_league_leaders = yes 
					is_member_of_no_foreign_or_italian_league = yes
					religion.group = religion_group:christian
				}
				order_by = military_strength
				max = 1  
				check_range_bounds = no
				save_temporary_scope_as = target_french_leader
				create_international_organization = {
					type = international_organization_type:foreign_league_france
					add_country_to_international_organization = scope:target_french_leader
					set_leader_country = scope:target_french_leader
				}
				international_organization:foreign_league_france = {
					add_to_global_variable_list = {
						name = iw_foreign_leagues_list
						target = this
					}
				}
				trigger_event_non_silently = italian_wars.1
				add_area_preference = foreign_league_italy_conquest
			}
		}

		#Calculate Strongest Iberian Country
		region:iberia_region = {
			ordered_country_with_capital_in_geography = {
				limit = { 
					basic_checks_for_calculating_foreign_league_leaders = yes 
					is_member_of_no_foreign_or_italian_league = yes
					religion.group = religion_group:christian
				}
				order_by = military_strength
				max = 1 
				check_range_bounds = no
				save_temporary_scope_as = target_iberian_leader
				create_international_organization = {
					type = international_organization_type:foreign_league_iberia
					add_country_to_international_organization = scope:target_iberian_leader
					set_leader_country = scope:target_iberian_leader
				}
				international_organization:foreign_league_iberia = {
					add_to_global_variable_list = {
						name = iw_foreign_leagues_list
						target = this
					}
				}
				trigger_event_non_silently = italian_wars.1
				add_area_preference = foreign_league_italy_conquest
			}
		}

		#Calculate Strongest HRE Leader Country (there should always only be 1)
		international_organization:hre.leader_country ?= {
			if = {
				limit = {
					NOR = {
						is_member_of_international_organization = international_organization:foreign_league_balkan
						is_member_of_international_organization = international_organization:foreign_league_france
						is_member_of_international_organization = international_organization:foreign_league_iberia
					}
				}
				save_temporary_scope_as = target_hre_leader
				create_international_organization = {
					type = international_organization_type:foreign_league_hre
					add_country_to_international_organization = scope:target_hre_leader
					set_leader_country = scope:target_hre_leader
				}
				international_organization:foreign_league_hre = {
					add_to_global_variable_list = {
						name = iw_foreign_leagues_list
						target = this
					}
				}
				trigger_event_non_silently = italian_wars.1
				add_area_preference = foreign_league_italy_conquest
			}
		}

		#Calculate Strongest Italian Defenders

		region:italy_region = {
			ordered_country_with_capital_in_geography = {
				limit = { 
					basic_checks_for_calculating_foreign_league_leaders = yes 
					is_member_of_no_foreign_or_italian_league = yes
					religion.group = religion_group:christian
				}
				order_by = military_strength 
				max = 1 
				check_range_bounds = no
				save_temporary_scope_as = target_ita1_leader
				create_international_organization = {
					type = international_organization_type:italian_league_1
					add_country_to_international_organization = scope:target_ita1_leader
					set_leader_country = scope:target_ita1_leader
				}
				international_organization:italian_league_1 = {
					add_to_global_variable_list = {
						name = iw_italian_leagues_list
						target = this 
					}
				}
				trigger_event_non_silently = italian_wars.1
			}
		}

		region:italy_region = {
			ordered_country_with_capital_in_geography = {
				limit = { 
					basic_checks_for_calculating_foreign_league_leaders = yes 
					is_member_of_no_foreign_or_italian_league = yes
					religion.group = religion_group:christian
				}
				order_by = military_strength
				max = 1 
				check_range_bounds = no
				save_temporary_scope_as = target_italian_2_leader
				create_international_organization = {
					type = international_organization_type:italian_league_2
					add_country_to_international_organization = scope:target_italian_2_leader
					set_leader_country = scope:target_italian_2_leader
				}
				international_organization:italian_league_2 = {
					add_to_global_variable_list = {
						name = iw_italian_leagues_list
						target = this 
					}
				}
				trigger_event_non_silently = italian_wars.1
			}
		}
			
		region:italy_region = {
			ordered_country_with_capital_in_geography = {
				limit = { 
					basic_checks_for_calculating_foreign_league_leaders = yes 
					is_member_of_no_foreign_or_italian_league = yes
					religion.group = religion_group:christian
				}
				order_by = military_strength
				max = 1 
				check_range_bounds = no
				save_temporary_scope_as = target_italian_3_leader
				create_international_organization = {
					type = international_organization_type:italian_league_3
					add_country_to_international_organization = scope:target_italian_3_leader
					set_leader_country = scope:target_italian_3_leader
				}
				international_organization:italian_league_3 = {
					add_to_global_variable_list = {
						name = iw_italian_leagues_list
						target = this 
					}
				}
				trigger_event_non_silently = italian_wars.1
			}
		}

		region:italy_region = {
			every_country_with_capital_in_geography = {
				limit = {
					is_subject = no
					is_member_of_no_foreign_or_italian_league = yes
				}
				trigger_event_non_silently = italian_wars.3
			}
		}
		sort_global_variable_list = {
			name = iw_italian_leagues_list
			order = {
				value = total_locations_owned
			}
		}
		sort_global_variable_list = {
			name = iw_foreign_leagues_list
			order = {
				value = total_locations_owned
			}
		}
		# Initialize tension variable
		situation:italian_wars = {
			set_variable = { name = iw_tension value = 0 }
		}
	}

	on_monthly = {
		sort_global_variable_list = {
			name = iw_italian_leagues_list
			order = {
				value = total_locations_owned
			}
		}
		sort_global_variable_list = {
			name = iw_foreign_leagues_list
			order = {
				value = total_locations_owned
			}
		}
		
		#Safety-net resync of each league's owned Italian locations against actual ownership (see reconcile_italian_wars_league_land)
		reconcile_italian_wars_league_land = yes

		# Passive tension drift
		iw_increase_tension_effect = { value = iw_tension_gain_low }
		# Flat bonus if any foreign league leader is at war (not per-leader, to avoid stacking)
		if = {
			limit = {
				any_in_global_list = {
					variable = iw_foreign_leagues_list
					leader_country ?= { at_war = yes }
				}
			}
			iw_increase_tension_effect = { value = iw_tension_gain_low }
		}
		# Flat bonus if any Italian league leader is at war
		if = {
			limit = {
				any_in_global_list = {
					variable = iw_italian_leagues_list
					leader_country ?= { at_war = yes }
				}
			}
			iw_increase_tension_effect = { value = iw_tension_gain_low }
		}
		if = {
			limit = {
				has_variable = allow_call_members_offensive_wars_variable
				current_year < {
					value = var:allow_call_members_offensive_wars_variable
					add = 10
				}
			}
			iw_increase_tension_effect = { value = iw_tension_loss_decay }
		}
		if = {
			limit = {
				has_variable = allow_call_members_offensive_wars_variable
				OR = {
					current_year >= {
						value = var:allow_call_members_offensive_wars_variable
						add = 10
					}
					var:iw_tension <= 0
				}
			}
			remove_variable = allow_call_members_offensive_wars_variable
		}
		# Fires italian_wars.11 the first month tension hits 100; gated by allow_call_members_offensive_wars_variable (10 years) so it fires once per PoNR cycle
		if = {
			limit = {
				var:iw_tension >= 100
				NOT = { has_variable = allow_call_members_offensive_wars_variable }
			}
			every_in_global_list = {
				variable = iw_italian_leagues_list
				leader_country ?= {
					trigger_event_non_silently = italian_wars.11
				}
			}
			every_in_global_list = {
				variable = iw_foreign_leagues_list
				leader_country ?= {
					trigger_event_non_silently = italian_wars.11
				}
			}
			set_variable = {
				name = allow_call_members_offensive_wars_variable
				value = current_year
			}
		}

		if = {
			limit = { var:iw_tension >= 50 }
			add_italian_wars_militarization_demands = yes	#See common\scripted_effects\situation_effects.txt
		}
		else = {
			remove_italian_wars_militarization_demands = yes
		}

		#Approximately 3.5% chance u get a relevant event per month	
		every_in_global_list = {
			variable = iw_italian_leagues_list
			every_international_organization_member = {
				random_list = {
					261 = {}
					1 = { trigger_event_non_silently = italian_wars.20 }
					1 = { trigger_event_non_silently = italian_wars.21 }
					1 = { trigger_event_non_silently = italian_wars.22 }
					1 = { trigger_event_non_silently = italian_wars.23 }
					1 = { trigger_event_non_silently = italian_wars.24 }
					1 = { trigger_event_non_silently = italian_wars.25 }
					1 = { trigger_event_non_silently = italian_wars.26 }
					1 = { trigger_event_non_silently = italian_wars.27 }
					1 = { trigger_event_non_silently = italian_wars.28 }
				}
			}
		}

		every_in_global_list = {
			variable = iw_foreign_leagues_list
			every_international_organization_member = {
				random_list = {
					261 = {}
					1 = { trigger_event_non_silently = italian_wars.20 }
					1 = { trigger_event_non_silently = italian_wars.21 }
					1 = { trigger_event_non_silently = italian_wars.22 }
					1 = { trigger_event_non_silently = italian_wars.23 }
					1 = { trigger_event_non_silently = italian_wars.24 }
					1 = { trigger_event_non_silently = italian_wars.25 }
					1 = { trigger_event_non_silently = italian_wars.26 }
					1 = { trigger_event_non_silently = italian_wars.27 }
					1 = { trigger_event_non_silently = italian_wars.28 }
				}
			}
		}
	}


	on_ended = {
		remove_italian_wars_militarization_demands = yes
		#Italian Leagues triumphant
		if = {
			limit = { international_organization:italian_league_1 ?= { total_locations_owned >= 200 } }
			iw_italian_league_triumph_events = { winner_io = italian_league_1 }
		}
		if = {
			limit = { international_organization:italian_league_2 ?= { total_locations_owned >= 200 } }
			iw_italian_league_triumph_events = { winner_io = italian_league_2 }
		}
		if = {
			limit = { international_organization:italian_league_3 ?= { total_locations_owned >= 200 } }
			iw_italian_league_triumph_events = { winner_io = italian_league_3 }
		}
		#Foreign League Triumph
		if = {
			limit = { international_organization:foreign_league_balkan ?= { total_locations_owned >= 200 } }
			iw_foreign_league_triumph_events = { winner_io = foreign_league_balkan }
		}
		if = {
			limit = { international_organization:foreign_league_france ?= { total_locations_owned >= 200 } }
			iw_foreign_league_triumph_events = { winner_io = foreign_league_france }
		}
		if = {
			limit = { international_organization:foreign_league_hre ?= { total_locations_owned >= 200 } }
			iw_foreign_league_triumph_events = { winner_io = foreign_league_hre }
		}
		if = {
			limit = { international_organization:foreign_league_iberia ?= { total_locations_owned >= 200 } }
			iw_foreign_league_triumph_events = { winner_io = foreign_league_iberia }
		}
		#Stalemate
		if = {
			limit = {
				NOR = {
					international_organization:italian_league_1 ?= { total_locations_owned >= 200 }
					international_organization:italian_league_2 ?= { total_locations_owned >= 200 }
					international_organization:italian_league_3 ?= { total_locations_owned >= 200 }
					international_organization:foreign_league_balkan ?= { total_locations_owned >= 200 }
					international_organization:foreign_league_france ?= { total_locations_owned >= 200 }
					international_organization:foreign_league_hre ?= { total_locations_owned >= 200 }
					international_organization:foreign_league_iberia ?= { total_locations_owned >= 200 }
				}
			}
			every_country = {
				limit = { presence_in_eligible_regions_or_member_of_any_league = yes }
				trigger_event_non_silently = italian_wars.400
			}
		}
		clear_global_variable_list = iw_foreign_leagues_list
		clear_global_variable_list = iw_italian_leagues_list
		destroy_all_italian_leagues = yes
	}

	tooltip = {
		#Balkans League
		if = { #Leader of the Balkans League
			limit = {
				has_owner = yes
				owner ?= { is_leader_of_international_organization = international_organization:foreign_league_balkan }
			}
			custom_tooltip = "LEADER_BALKANS_POWER_TOOLTIP"
		}
		else_if = { #Member of the Balkans League
			limit = {
				has_owner = yes
				owner ?= { is_member_of_international_organization = international_organization:foreign_league_balkan }
			}
			custom_tooltip = "MEMBER_BALKANS_POWER_TOOLTIP"
		}
		#French League
		if = { #Leader of the French League
			limit = {
				has_owner = yes
				owner ?= { is_leader_of_international_organization = international_organization:foreign_league_france }
			}
			custom_tooltip = "LEADER_FRENCH_POWER_TOOLTIP"
		}
		else_if = { #Member of the French League
			limit = {
				has_owner = yes
				owner ?= { is_member_of_international_organization = international_organization:foreign_league_france }
			}
			custom_tooltip = "MEMBER_FRENCH_POWER_TOOLTIP"
		}
		#Espanita League
		if = { #Leader of the Iberian League
			limit = {
				has_owner = yes
				owner ?= { is_leader_of_international_organization = international_organization:foreign_league_iberia }
			}
			custom_tooltip = "LEADER_IBERIAN_POWER_TOOLTIP"
		}
		else_if = { #Member of the Iberian League
			limit = {
				has_owner = yes
				owner ?= { is_member_of_international_organization = international_organization:foreign_league_iberia }
			}
			custom_tooltip = "MEMBER_IBERIAN_POWER_TOOLTIP"
		}
		#Emperor-led League
		if = { #Leader of the HRE League
			limit = {
				has_owner = yes
				owner ?= { is_leader_of_international_organization = international_organization:foreign_league_hre }
			}
			custom_tooltip = "LEADER_HRE_POWER_TOOLTIP"
		}
		else_if = { #Member of the HRE League
			limit = {
				has_owner = yes
				owner ?= { is_member_of_international_organization = international_organization:foreign_league_hre }
			}
			custom_tooltip = "MEMBER_HRE_POWER_TOOLTIP"
		}
		if = { #Sponsor of the 1st Italian League
			limit = {
				has_owner = yes
				owner ?= {
					has_special_status_in_international_organization = {
						type = special_status:italian_league_sponsor
						international_organization = international_organization:italian_league_1
					}
				}
			}
			custom_tooltip = "SPONSOR_FIRST_ITALIAN_LEAGUE"
		}
		if = { #Sponsor of the 2nd Italian League
			limit = {
				has_owner = yes
				owner ?= {
					has_special_status_in_international_organization = {
						type = special_status:italian_league_sponsor
						international_organization = international_organization:italian_league_2
					}
				}
			}
			custom_tooltip = "SPONSOR_SECOND_ITALIAN_LEAGUE"
		}
		if = { #Sponsor of the 3rd Italian League
			limit = {
				has_owner = yes
				owner ?= {
					has_special_status_in_international_organization = {
						type = special_status:italian_league_sponsor
						international_organization = international_organization:italian_league_3
					}
				}
			}
			custom_tooltip = "SPONSOR_THIRD_ITALIAN_LEAGUE"
		}
		#Italian League
		if = { #Leader of the 1st Italian League
