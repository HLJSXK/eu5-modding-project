call_imperial_circle_to_war = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	potential = {
		scope:actor = { is_emperor = yes }
		international_organization:hre = { circles_are_active = yes }
		NOT = { scope:actor = { has_cooldown = called_imperial_circle_to_war } }
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			international_organization:hre = {
				every_international_organization_member = {
					limit = { is_imperial_circle_leader = yes }
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_circle_leader"
		column = {
			data = name
		}
		visible = {
			is_imperial_circle_leader = yes
			NOT = { in_same_imperial_circle = scope:actor }
		}
		enabled = {
			scope:actor = {
				any_current_war = {
					defender_leader = scope:actor
					NOT = {
						attacker_leader = {
							is_member_of_international_organization = international_organization:hre
						}
					}
				}
			}
		}
	}

	effect = {
		scope:actor = {
			random_current_war = {
				limit = {
					defender_leader = scope:actor
					NOT = {
						attacker_leader = {
							is_member_of_international_organization = international_organization:hre
						}
					}
				}
				save_scope_as = imperial_war
			}
		}
		if = {
			limit = { exists = scope:imperial_war }
			scope:actor = {
				add_cooldown = {
					type = called_imperial_circle_to_war
					months = 60
				}
			}
			scope:recipient = {
				every_imperial_circle_country_is_member_of = {
					every_circle_member = {
						limit = { NOT = { this = scope:actor } }
						join_war_as_defender = { war = scope:imperial_war }
					}
				}
			}
		}
	}

	reject_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
		}
	}

	diplo_chance = {
		base = -20
		opinion = 0.3
		diplomatic_reputation = 2
		trust_in_actor = 1.0
	}

	accept = {
		add = {
			desc = "CIRCLE_WAR_CALL_DEFENSIVE_WAR"
			value = 100
		}
		add = hre_default_opinion_check
	}

	ai_will_do = {
		add = { value = 30 }
	}
}

appoint_imperial_circle_leader = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	potential = {
		scope:actor = { is_emperor = yes }
		international_organization:hre = {
			circles_are_active = yes
			international_organization_has_policy = policy:improved_imperial_authority_policy
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			international_organization:hre = {
				every_international_organization_member = {
					limit = {
						is_imperial_circle_leader = no
						any_imperial_circle_country_is_member_of = {
							NOT = { is_dormant_imperial_circle = yes }
							NOT = { has_circle_leader = yes }
						}
					}
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			is_member_of_international_organization = international_organization:hre
			is_imperial_circle_leader = no
		}
		enabled = {
			any_imperial_circle_country_is_member_of = {
				NOT = { is_dormant_imperial_circle = yes }
				NOT = { has_circle_leader = yes }
			}
		}
	}

	effect = {
		scope:recipient = {
			every_imperial_circle_country_is_member_of = {
				set_circle_leader = scope:recipient
			}
		}
	}

	ai_will_do = {
		add = { value = 40 }
	}
}

petition_imperial_diet = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	cooldown = { months = 120 }

	potential = {
		scope:actor = { is_imperial_circle_leader = yes }
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			international_organization:hre = {
				leader_country ?= {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_emperor"
		column = {
			data = name
		}
		visible = {
			international_organization:hre = { circles_are_active = yes }
			NOT = { is_leader_of_international_organization = international_organization:hre }
		}
		enabled = {
			international_organization:hre = { international_organization_has_leader = yes }
		}
	}

	effect = {
		scope:actor = {
			add_prestige = prestige_mild_bonus
		}
		scope:recipient = {
			change_imperial_authority = { value = 2 }
		}
	}

	reject_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
		}
	}

	diplo_chance = {
		base = 10
		opinion = 0.2
		diplomatic_reputation = 2
	}

	accept = {
		add = hre_default_opinion_check
		add = {
			desc = "CIRCLE_LOW_SATISFACTION_GRIEVANCE"
			if = {
				limit = {
					scope:actor = {
						any_imperial_circle_country_is_member_of = {
							imperial_circle_satisfaction < 50
						}
					}
				}
				value = 30
			}
		}
	}

	ai_will_do = {
		add = { value = 20 }
	}
}

challenge_circle_leadership = {
	type = diplomacy
	category = CATEGORY_HRE_ACTIONS

	cooldown = { months = 60 }

	potential = {
		scope:actor = {
			is_member_of_international_organization = international_organization:hre
			is_imperial_circle_leader = no
			any_imperial_circle_country_is_member_of = {
				has_circle_leader = yes
				NOT = { is_dormant_imperial_circle = yes }
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_imperial_circle_country_is_member_of = {
					circle_leader ?= {
						add_to_list = source
					}
				}
			}
		}
		target_flag = recipient
		name = "choose_circle_leader"
		column = {
			data = name
		}
		visible = {
			international_organization:hre = { circles_are_active = yes }
			NOT = { is_leader_of_international_organization = international_organization:hre }
		}
		enabled = {
			OR = {
				scope:actor = { prestige >= 50 }
				scope:actor = {
					any_imperial_circle_country_is_member_of = {
						imperial_circle_satisfaction < 15
					}
				}
			}
			NOT = { has_country_modifier = consolidating_circle_leadership }
		}
	}

	effect = {
		scope:actor = {
			trigger_event_non_silently = { id = imperial_circles.50 }
		}
	}

	ai_will_do = {
		add = { value = 10 }
		if = {
			limit = {
				scope:actor = {
					any_imperial_circle_country_is_member_of = {
						imperial_circle_satisfaction < 40
					}
				}
			}
			add = { value = 40 }
		}
		# Never challenge if no tier's win condition can fire (matches imperial_circles.50.a logic)
		if = {
			limit = {
				scope:actor = {
					NOT = {
						any_imperial_circle_country_is_member_of = {
							imperial_circle_satisfaction < 15
						}
					}
					OR = {
						NOT = {
							any_imperial_circle_country_is_member_of = {
								imperial_circle_satisfaction < 30
							}
						}
						prestige < scope:recipient.prestige
					}
					prestige < {
						value = scope:recipient.prestige
						add = 25
					}
				}
			}
			multiply = 0
		}
	}
}
