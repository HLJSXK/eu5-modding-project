	type = subject
	category = CATEGORY_SUBJECT_ACTIONS

	potential = {
	}

	allow = {
	}

	cooldown = { 
		type = enforce_culture
		years = 10
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			not = { culture = scope:actor.culture }
		}
        enabled = {
			subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
			liberty_desire < MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION
			dependency_length_days = {
				target = scope:actor
				value >= MIN_DAYS_AS_SUBJECT_FOR_ENFORCEMENT
			}
		}
	}

	effect = {
        enforce_culture_effect = {
            actor = scope:actor
            target = scope:recipient
        }
		scope:recipient = {
			add_liberty_desire = liberty_desire_ultimate_plus
		}
	}
}
