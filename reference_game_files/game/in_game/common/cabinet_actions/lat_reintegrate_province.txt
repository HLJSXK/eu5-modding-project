lat_reintegrate_province = {
    ability = mil
    allow_multiple = yes
    
    potential = {
        modifier:lat_access_to_latin_reintegration_cabinet = yes
    }

    allow = {
        custom_tooltip = {
			text = location_located_in_frankokratia_region
			any_province = {
                any_location_in_province = {
                    is_in_scripted_geography = scripted_geography:frankokratia_geography
                    integration_level = conquered
                }
            }
		}
    }
    
    select_trigger = {
        looking_for_a = province
        source = actor
        target_flag = target
        name = "integrate_province_select_province"
        none_available_msg_key = "integrate_province_no_provinces"
        column = {
            data = name
        }
        column = {
            data = integration
        }
        column = {
            data = population
        }
        visible = {
            NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = lat_reintegrate_province interaction_target = target } }
            custom_tooltip = {
			    text = location_located_in_frankokratia_region
                any_location_in_province = {
                    is_in_scripted_geography = scripted_geography:frankokratia_geography
                    integration_level = conquered
                }
            }
        }    
        map_color = {
            lerp = {
                min_color = define:NMapColors|MAP_COLOR_LOW
                max_color = define:NMapColors|MAP_COLOR_HIGH
                factor = integration_progress
            }
        }
    }

    is_finished = {
        scope:target = {
            any_location_in_province = {
                OR = {
                    integration_level = integrated
                    integration_level = core
                }
                count = all
            }
        }
    }

    province_modifier = {
        province_integration_speed = 1.5
    }

    progress = {
        scope:target = {
            value = province_average_integration
        }
    }

    map_marker = {
        show_for_owner = yes
    }
}
