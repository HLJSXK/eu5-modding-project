aptekarsky = {
	ability = adm
	potential = {
		modifier:unlock_prikazi_reform_cabinet_actions = yes
		has_variable = aptekarsky_prikaz_established
	}

	allow = {
		any_province = {
			any_location_in_province = {
				integration_level = core
				count = all
				has_any_disease_present = yes
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "aptekarsky_select_province"
		none_available_msg_key = "aptekarsky_no_provinces"
		column = {
			data = name
		}
		column = {
			data = development
		}
		column = {
			data = population
		}
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = develop_province interaction_target = target } }
		}
		enabled = {
			any_location_in_province = {
				integration_level = core
				count = all
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = {
					value = development
					divide = 100
				}
			}
		}
	}
	
	location_modifier = {
		local_disease_resistance = 0.05
		local_monthly_prosperity = 0.002
	}

}
