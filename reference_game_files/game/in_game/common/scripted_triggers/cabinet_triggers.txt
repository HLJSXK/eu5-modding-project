cabinet_already_performing_same_task_on_target = {
	custom_description = {
		text = cabinet_already_performing_same_task_on_target_text
		scope:actor = {
			any_cabinet ?= {
				has_cabinet_action = yes
				cabinet_action = cabinet_action:$cabinet_action$
				interaction_target:$interaction_target$ ?= root
			}
		}
	}
}

	scope:actor = {

			}
		}
	}
}
