rebel_demand_rev_chaos_1101_bonaparte = {	# disaster/revolutionary_chaos.txt — revolutionary_chaos.1101 option a
	trigger = { always = no }
	concession_effect = {
		pacify_rebel_pops = yes
		owner = {
			set_new_ruler = var:var_target_character
			if = {
				limit = { has_country_modifier = the_reign_of_terror }
				remove_country_modifier = the_reign_of_terror
			}
			if = {
				limit = { has_reform = government_reform:revolutionary_republic }
				remove_reform = government_reform:revolutionary_republic
			}
			if = {
				limit = { has_reform = government_reform:constitutional_state }
				remove_reform = government_reform:constitutional_state
			}
			add_reform = government_reform:revolutionary_autocracy
			if = {
				limit = { NOT = { succession_law = heir_selection:military_dictatorship } }
				change_heir_selection = heir_selection:military_dictatorship
			}
			trigger_event_silently = {
				id = revolutionary_chaos.1102
				days = { 1460 2190 }
			}
			trigger_event_silently = {
				id = revolutionary_chaos.1103
				days = { 365 1000 }
			}
			hidden_effect = {
				var:var_target_character = { set_variable = { name = bonaparte_flag value = yes } }
				if = {
					limit = { tag = FRA }
					var:var_target_character = { set_variable = bonaparte_dynasty }
				}
			}
		}
	}
}

# Situations

rebel_demand_hussite_wars_1 = {	# situations/hussite_wars.txt — hussite_wars.1 option a
	trigger = { always = no }
	concession_effect = {
		pacify_rebel_pops = yes
		owner = {
			if = {
				limit = { NOT = { religion = religion:hussite } }
				change_religion = religion:hussite
				change_religion_for_ruler_and_family = { country = this religion = religion:hussite }
			}
			else = { add_religious_influence = religious_influence_mild_bonus }
			if = {
				limit = {
					has_ruler = yes
					ruler = { has_character_modifier = enacted_placitum_regium }
				}
				ruler = { remove_character_modifier = enacted_placitum_regium }
			}
			upset_countries_of_religion = {
				religion = religion:catholic
				modifier = religious_disagreements_modifier
			}
			add_religious_influence = religious_influence_ultimate_bonus
			add_country_modifier = {
				modifier = in_defense_of_hussitism_modifier
				years = modifier_duration_years_short
				mode = replace
			}
		}
	}
}
