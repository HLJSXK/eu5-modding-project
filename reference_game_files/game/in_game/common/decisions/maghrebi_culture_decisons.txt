maghreb_ibadi_nation_establish_majlis_ammi_said = { # Establish the Majlis 'Ammi Sa'īd
	decision_category = government_decisions
	only_once = yes

	potential = {
		religion = religion:ibadi
	}

	allow = {
		scripted_geography:ibadi_heartland_of_the_maghreb = {
            any_location_in_scripted_geography = {
                owner ?= ROOT
                count > 28
            }
        }
		stability > 40
		has_ruler = yes
        is_subject = no
		at_war = no
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = maghreb_ibadi_nation_establish_majlis_ammi_said.a
		ai_chance = { value = 100 }
		effect = {
			unlock_government_reform_effect = { type = maghreb_ibadi_majlis_ammi_said }
			add_reform = government_reform:maghreb_ibadi_majlis_ammi_said
			add_prestige = prestige_extreme_bonus
			add_estate_satisfaction = {
				type = estate_type:clergy_estate
				value = estate_satisfaction_mild_bonus
			}
		}
	}
}

maghreb_masters_of_the_sahara = { # Bonus to Arid and desert locations for the rest of the game
	decision_category = government_decisions
	only_once = yes

	potential = {
		OR = {
			culture = { has_culture_group = culture_group:amazigh_group }
			culture = { has_culture_group = culture_group:maghrebi_group }
		}
		at_war = no
	}

	allow = {
		area:desert_area = {
			any_location_in_area = {
				owner ?= root
                count > 25
			}
		}
		area:desert_area = {
			any_location_in_area = {
				owner ?= root
				development >= 18.00
				count > 15
			}
		}
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = maghreb_masters_of_the_sahara.a
		ai_chance = { value = 100 }
		effect = {
			add_prestige = prestige_extreme_bonus
			custom_tooltip = {
				text = empower_the_desert_power_privilege_tt
				set_variable = { name = enable_desert_power_upgrade value = 1 }
			}
			custom_tooltip = empower_the_desert_power_privilege_tt_default_bonus
		}
	}
}
