byz_army_reform_decision = { #Alt-history. Byzantium never really reformed their army since they died
	decision_category = country_specific_decisions
	only_once = yes

	tags = { BYZ ROM }

	potential = {
		current_age_or_later = {
			age = age_1_traditions
		}
		has_or_had_tag = BYZ
	}

	allow = {
		current_age_or_later = {
			age = age_2_renaissance
		}
		has_embraced_institution = institution:professional_armies
		has_reform = government_reform:theme_system
		regular_army_size >= 10
		has_ruler = yes
		any_cabinet_character = {
			count > 0
		}
		NOT = { has_unlocked_government_reform_trigger = { type = modern_imperial_army } }
	}

	ai_will_do = {
		value = 100
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	illustration_scope = {
		event_illustration_poptype_effect = { foreground = pop_type:soldiers background = pop_type:soldiers }
		
		ruler_or_regent ?= {
			save_scope_as = target_character
		}
		ordered_cabinet_character = {
			order_by = total_abilities
			max = 1
			check_range_bounds = no
			save_scope_as = cabinet_character
		}

		unlock_government_reform_effect = { type = modern_imperial_army }
	}

	option = {
		name = byz_army_reform_decision.a
		ai_chance = { value = 100 }
		effect = {
			change_societal_value = {
				type = quality_vs_quantity
				value = societal_value_huge_move_to_left
			}

			if = {
				limit = { has_reform = government_reform:theme_system }
				remove_reform = government_reform:theme_system
			}
			add_reform = government_reform:modern_imperial_army

			add_gold = {
				value = monthly_income_trade_and_tax
				divide = {
					value = ruler_or_regent.mil
					divide = 10
				}
				multiply = -3
			}

			custom_tooltip = BYZ_THE_COST_WILL_SCALE_WITH_RULERS_MIL

			if = {
				limit = { has_dlc = "d008_fate_of_the_phoenix" }
				every_current_bureaucracy = {
					limit = { bureaucracy_type = bureaucracy_type:themata_bureaucracy }
					root = { remove_bureaucracy = prev }
				}
				add_bureaucracy = bureaucracy_type:magister_militum_bureaucracy
			}
		}
	}

	option = { #This is Byzantium at the end of the day
		name = byz_army_reform_decision.b
		ai_chance = { value = 0 }
		effect = {
			change_societal_value = {
				type = quality_vs_quantity
				value = societal_value_move_to_right
			}

			add_estate_satisfaction = {
				type = estate_type:nobles_estate
				value = estate_satisfaction_mild_bonus
			}

			custom_tooltip = {
				text = BYZ_ALL_THEMATA_PROVINCES_GAIN_NOBLE_LOYALTY
				every_owned_location = {
					limit = {
						has_building = building_type:thema_headquarters
						any_pop = {
							pop_type = pop_type:nobles
						}
					}
					every_pop = {
						limit = {
							owner = root
							pop_type = pop_type:nobles
						}
						add_pop_satisfaction = pop_satisfaction_ultimate_bonus
					}
				}
			}

			custom_tooltip = able_to_add_reform_later_tt
		}
	}
}
