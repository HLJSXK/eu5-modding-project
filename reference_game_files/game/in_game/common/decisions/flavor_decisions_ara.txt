ara_coronate_ruler = {
	decision_category = country_specific_decisions

	tags = { ARA }

	potential = {
		government_type = government_type:monarchy

		religion = religion:catholic
		has_fired_unique_event = flavor_ara.22
	}

	allow = {
		at_war = no
		has_regent = no
		owns = location:zaragoza

		political_influence >= 25

        ruler ?= {
            NOT = { has_character_modifier = ara_official_coronation }
        }
	}

	ai_will_do = {
		value = 100
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:nobles_estate }

		ruler ?= {
			save_scope_as = target_character
		}

		location:zaragoza = {
			save_scope_as = target_location
		}
	}

	option = {
		name = ara_coronate_ruler.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = political_influence_extreme_penalty
			ruler ?= {
                remove_character_modifier = ara_lack_of_coronation
                add_character_modifier = {
                    mode = add
                    modifier = ara_official_coronation
                    years = -1
                }
            }
		}
	}

	option = {
		name = ara_coronate_ruler.b
		ai_chance = { value = 25 }
		effect = {
			ruler ?= { remove_character_modifier = ara_lack_of_coronation }

			add_political_influence = political_influence_extreme_penalty

			add_legitimacy = legitimacy_severe_penalty

			set_variable = ara_abolished_self_coronation
		}
	}
}
