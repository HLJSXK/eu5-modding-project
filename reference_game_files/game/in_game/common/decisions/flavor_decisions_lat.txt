# Decisions relating to Frankokratia and the Latin Empire

ach_greek_populace_decision = {
    decision_category = country_specific_decisions
    only_once = yes

    tags = { ACH LAT }

    potential = {
        current_age_or_later = {
            age = age_2_renaissance
        }
    }

    allow = {
        at_war = no
        stability > 50
        NOT = { culture = { has_culture_group = culture_group:greek_group } }
    }

    ai_will_do = { value = 100 }

    illustration_tags = {
        10 = regular
        10 = interior
    }

    illustration_scope = {
        event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
        ruler ?= {
            save_scope_as = scope_ruler
        }
    }

    option = { # Maintain the current status quo.
        name = ach_greek_populace_decision.a
        ai_chance = { value = 30 }
        effect = {
            add_stability = stability_mild_bonus 
            add_legitimacy = legitimacy_mild_bonus  
            add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_weak_penalty }
            add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_weak_penalty }
            add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_weak_penalty }
        }
    }

    option = { # Favor the French elite
        name = ach_greek_populace_decision.b
        ai_chance = { value = 30 }
        effect = {
            add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_severe_bonus }
            add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_severe_bonus }
            add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_mild_penalty }
            add_country_modifier = { modifier = ach_favored_the_latin years = 10 mode = add_and_extend }  
        }
    }

    option = { # Strive for a compromise with the greek elite
        name = ach_greek_populace_decision.c
        ai_chance = { value = 40 }
        effect = {
            add_prestige = prestige_mild_penalty
            add_legitimacy = legitimacy_mild_penalty
            add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_mild_penalty }
            add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
            add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_severe_bonus }
            add_country_modifier = { modifier = ach_compromise_wth_the_greeks years = 15 mode = add_and_extend }  
            unlock_estate_privilege_effect = { type = lat_compromise_with_greek_populace }
            grant_estate_privilege = estate_privilege:lat_compromise_with_greek_populace
            culture:greek_culture = {
                change_cultural_view = {
                    target = root.culture
                    change = 1
                }
            }
        }
    }    
}

ach_pope_proposes_latin_restoration = { # Achaea Unique version as the Titular rulers
    decision_category = country_specific_decisions
    only_once = yes

    tags = { ACH }

    potential = {
        current_age_or_later = {
            age = age_2_renaissance
        }
        NOT = { country_exists = c:LAT }
    }

    allow = {
        has_reform = government_reform:ach_heirs_of_the_latin_empire
        at_war = no
        has_ruler = yes
        country_exists = c:PAP
        country_exists = c:BYZ
        c:PAP ?= {
            has_ruler = yes
        }
        NOT = { country_exists = c:LAT }
        religion = religion:catholic
        NOT = { owns = location:constantinople }
        NOT = { num_locations >= 70 }
        scripted_geography:frankokratia_geography = {
            any_location_in_scripted_geography = {
                owner ?= ROOT
                count > 20
            }
        }
    }

    ai_will_do = { value = 100 }

    illustration_tags = {
        10 = regular
        10 = interior
    }

    illustration_scope = {
        event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
        ruler ?= {
            save_scope_as = scope_ruler
        }
    }

    option = { # Accept the Pope offer to restore the Latin empire.
        name = ach_pope_proposes_latin_restoration.a
        ai_chance = { value = 90 }
        effect = {
            custom_tooltip = lat_aggressive_conversion_policies_tt
            unlock_government_reform_effect = { type = lat_papal_restoration_claimant }
            add_reform = government_reform:lat_papal_restoration_claimant
            add_country_modifier = { modifier = lat_support_from_the_papacy years = 25 mode = add }
            add_gold = {
                value = c:PAP.monthly_income_trade_and_tax
                multiply = 6
                min = 500
                max = 1200
            }
            set_variable = {
                name = lat_accepted_pope_latin_empire_proposal
                value = 1
                years = 100
            }
            add_opinion = { target = c:PAP modifier = ach_papal_latin_restoration_guarantee }
            c:PAP ?= {
                add_opinion = { target = ROOT modifier = ach_papal_latin_restoration_guarantee }
            }
        }
    }

    option = { # Reject his offer, we shall go our own path.
        name = ach_pope_proposes_latin_restoration.b
        ai_chance = { value = 10 }
        effect = {
            add_prestige = prestige_severe_bonus
            add_legitimacy = legitimacy_mild_penalty
            c:PAP = {
                add_opinion = { target = ROOT modifier = ach_rejected_papacy_offer }
            }
        }
    }   
}
