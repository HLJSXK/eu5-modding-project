mll_a_hub_of_mediterranean_trade = {
	decision_category = government_decisions
	only_once = yes

	potential = {
		has_or_had_tag = MLL
	}

	allow = {
		at_war = no
		is_subject = no
		government_type = government_type:monarchy
		own_entire_area = area:balearics_area
		area:balearics_area = {
			any_location_in_area = {
				owner ?= root
				development >= 25.00
				location_building_level = {
					building_type = building_type:marketplace
					value >= 12
				}
				count > 4
			}
		}
		area:balearics_area = {
			any_location_in_area = {
				owner ?= root
				OR = {
					location_rank = location_rank:city
					location_rank = location_rank:megalopolis
				}
				count > 2
			}
		}
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = mll_a_hub_of_mediterranean_trade.a
		effect = {
			add_prestige = prestige_extreme_bonus
			unlock_government_reform_effect = { type = mll_pearl_of_the_mediterranean }
			add_reform = government_reform:mll_pearl_of_the_mediterranean
		}
		
	}
}


mll_exchange_of_montpellier = {
	decision_category = government_decisions
	only_once = yes

	potential = {
		has_or_had_tag = MLL
		has_variable = mll_enable_sale_of_montpellier_decision
	}

	allow = {
		at_war = no
		is_subject = yes
		government_type = government_type:monarchy
		owns = location:montpellier
	}

	ai_will_do = {
		value = 0
	}

	option = { # Offer to sell it to the french court
		name = mll_exchange_of_montpellier.a
		effect = {
			add_gold = {
            	value = c:PAP.monthly_income_trade_and_tax
            	multiply = 4
            	min = 1000
            	max = 2000
        	}
			add_opinion = { target = c:FRA modifier = mll_exchange_of_montpellier_opinion }
        	reverse_add_opinion = { target = c:FRA modifier = mll_exchange_of_montpellier_opinion } 
			location:montpellier = {
				change_location_owner = c:FRA
				add_core = c:FRA
				remove_core = c:MLL
			}
		}
	}

	option = { # Exchange for french guarantees
		name = mll_exchange_of_montpellier.b
		effect = {
			add_opinion = { target = c:FRA modifier = mll_exchange_of_montpellier_huge_opinion }
        	reverse_add_opinion = { target = c:FRA modifier = mll_exchange_of_montpellier_huge_opinion } 
			location:montpellier = {
				change_location_owner = c:FRA
				add_core = c:FRA
				remove_core = c:MLL
			}
			custom_tooltip = {
				text = mll_french_support_for_montpellier_tt
				set_variable = {
            		name = mll_fra_guarantees_for_independence
            		value = 1
            		years = 50
       			}
			}			
		}
		
	}

	option = {
		name = mll_exchange_of_montpellier.c
		effect = {
			add_prestige = prestige_mild_bonus
		}
		
	}		
}
