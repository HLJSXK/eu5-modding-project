navarre_charles_ii_claim_to_france = { # Rightful Claim to France
	decision_category = country_specific_decisions
	only_once = yes

	potential = {
		has_variable = nav_enable_claim_the_french_throne_decision
		NOT = { is_subject_of = c:FRA }
		NOT = { is_subject_type = appanage }
	}

	allow = {
		NOT = { is_subject_of = c:FRA }
		NOT = { is_subject_type = appanage }
		has_ruler = yes
		character:nav_charles_ii_evreux = {
			is_alive = yes
		}
		at_war = no
		stability >= 50
		OR = {
			political_influence >= 50
			legitimacy >= 95
			prestige >= 20
			modifier:diplomatic_reputation > 10
		}
	}

	ai_will_do = {
		value = 100
	}

	option = { # Claim the French Throne
		name = navarre_charles_ii_claim_to_france.a
		ai_chance = { value = 100 }
		effect = {
			add_legitimacy = legitimacy_mild_penalty
			add_political_influence = political_influence_severe_penalty
			add_casus_belli = {
				target = c:FRA
				type = casus_belli:cb_claim_throne
			}
			remove_variable = nav_enable_claim_the_french_throne_decision
		}
	}	
}

navarre_charles_ii_claim_to_france_appanage_route = { # Rightful Claim to France
	decision_category = country_specific_decisions
	only_once = yes

	potential = {
		has_variable = nav_enable_claim_the_french_throne_decision
		is_subject_of = c:FRA
		is_subject_type = appanage
	}

	allow = {
		is_subject_of = c:FRA
		is_subject_type = appanage
		has_ruler = yes
		character:nav_charles_ii_evreux = {
			is_alive = yes
		}
		at_war = no
		stability >= 50
		OR = {
			political_influence >= 50
			legitimacy >= 95
			prestige >= 20
			modifier:diplomatic_reputation > 10
		}
	}

	ai_will_do = {
		value = 100
	}

	option = { # Appanage Route
		name = navarre_charles_ii_claim_to_france_appanage_route.a
		ai_chance = { value = 100 }
		effect = {
			custom_tooltip = nav_appanage_willingness_to_join_tt
			add_political_influence = political_influence_ultimate_penalty
			set_variable = {
            	name = nav_charles_rallies_loyal_appanages
            	value = 1
            	years = 5
        	}
			c:FRA ?= {
				every_subject = {
					limit = {
						is_subject_of = c:FRA
						is_subject_type = appanage
						NOT = { tag = NAV }
						opinion = {
 							target = c:NAV
 							value > 150
						}
					}
					set_variable = {
            			name = likely_supporter_of_charles_of_navarre
            			value = 1
            			years = 5
        			}
				}
			}
			add_casus_belli = {
				target = c:FRA
				type = casus_belli:cb_claim_throne
			}
			c:FRA = { cancel_subject = c:NAV }
			declare_war_with_cb = {
				target = c:FRA
				type = casus_belli:cb_claim_throne
			}
			c:FRA ?= {
				every_subject = {
					limit = {
						is_subject_of = c:FRA
						is_subject_type = appanage
						has_variable = likely_supporter_of_charles_of_navarre
					}
					trigger_event_silently = { id = flavor_nav.16 }	
				}
			}	
			remove_variable = nav_enable_claim_the_french_throne_decision
		}
	}	
}

nav_coercing_of_normandy = { # Charles II claim to Normandy
	decision_category = government_decisions
	only_once = yes

	potential = {
		has_variable = nav_may_seize_duchy_of_normandy
	}

	allow = {
		at_war = no
		c:NRM ?= {
			is_ai = yes
			is_subject_of = c:FRA
			opinion = {
 				target = c:NAV
 				value > 190
			}
		}
		OR = {
			political_influence >= 80
			AND = {
				political_influence >= 50
				c:FRA ?= {
					at_war = yes
				}
			}
		}
	}

	ai_will_do = {
		value = 100
	}

	option = { # Coerce control of Normandy!
		name = nav_coercing_of_normandy.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = political_influence_ultimate_penalty
			add_legitimacy = legitimacy_mild_penalty
			add_prestige = prestige_severe_bonus
			annex_country = { country = c:NRM }
			add_country_modifier = { modifier = nav_seized_normandy_politically years = 20 mode = add }
			if = {
				limit = {
					c:FRA ?= {
						at_war = yes
					}
				}
				add_political_influence = political_influence_mild_bonus
			}
		}
	}
}


nav_end_the_blood_feuds = { # Charles II claim to Normandy
	decision_category = government_decisions
	only_once = yes

	potential = {
		has_variable = nav_end_the_blood_feud_decisions
	}

	allow = {
		at_war = no
		OR = {
			current_year >= 1400
			has_country_modifier = nav_expansion_of_the_royal_hermandades
		}
		OR = {
			stability >= 75
			AND = {
				has_country_modifier = nav_expansion_of_the_royal_hermandades
				stability >= 40
			}
		}
		legitimacy >= 95
		political_influence >= 30
	}

	ai_will_do = {
		value = 100
	}

	option = { # It is time these feuds ended for good
		name = nav_end_the_blood_feuds.a
		ai_chance = { value = 100 }
		effect = {
			if = {
				limit = {
					has_country_modifier = nav_open_blood_feuds_in_navarre
				}
				remove_country_modifier = nav_open_blood_feuds_in_navarre
			}
			add_prestige = prestige_ultimate_bonus
			add_stability = stability_ultimate_bonus
			add_political_influence = political_influence_ultimate_bonus
			add_country_modifier = { modifier = nav_end_of_the_blood_feuds years = 50 mode = add }	
		}
	}
}
