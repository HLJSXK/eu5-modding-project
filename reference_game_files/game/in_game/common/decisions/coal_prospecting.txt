prospect_for_coal = {
	decision_category = economy_decisions
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_coal.dds"

	potential = {
		current_age_or_later = { age = age_6_revolutions }
		scripted_geography:potential_coal = {
			any_location_in_scripted_geography = {
				owner = root
				is_city = no
				NOT = { raw_material = goods:coal }
			}
		}
	}

	allow = {
		political_influence >= coal_prospecting_cost
		OR = {
			is_ai = no
			NOT = { any_owned_location = { raw_material = goods:coal } }
		}
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = prospect_for_coal.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = {
				value = coal_prospecting_cost
				multiply = -1
			}
			scripted_geography:potential_coal = {
				ordered_location_in_scripted_geography = {
					limit = {
						owner = root
						is_city = no
						NOT = { raw_material = goods:coal }
					}
					order_by = {
						value = relative_raw_material_price
						multiply = -1
					}
					max = 1
					save_scope_as = coal_prospect_location
					custom_tooltip = {
						text = prospect_for_coal_effect_tt
						change_max_raw_material_workers = {
						value = rgo_workers
						subtract = 1
						min = 0
						multiply = -1
					}
					change_raw_material = goods:coal
						}
				}
			}
		}
	}
}
