hab_privilegium_maius_decision = {
	decision_category = country_specific_decisions
	only_once = yes

	tags = { HAB STY TIR }

	potential = {
		current_age_or_later = {
			age = age_1_traditions
		}
		ruler = { dynasty = dynasty:habsburg_dynasty }
		OR = {
			custom_tooltip = {
				text = hab_privilegium_maius_decision.privilegium_not_granted_tt
				AND = {
					NOT = {
						has_variable = privilegium_maius_granted
					}
					has_variable = privilegium_maius_accepted
				}
			}
			custom_tooltip = {
				text = hab_privilegium_maius_decision.privilegium_not_decided_tt
				AND = {
					NOT = {
						has_variable = flavor_hab_1000_has_fired
					}
					current_year >= 1370
				}
			}
		}
	}

	allow = {
		current_age_or_later = {
			age = age_2_renaissance
		}
		has_ruler = yes
		is_leader_of_international_organization = international_organization:hre
		owns = location:vienna
		OR = {
			custom_tooltip = {
				text = hab_privilegium_maius_decision.privilegium_not_granted_tt
				AND = {
					NOT = {
						has_variable = privilegium_maius_granted
					}
					has_variable = privilegium_maius_accepted
				}
			}
			custom_tooltip = {
				text = hab_privilegium_maius_decision.privilegium_not_decided_tt
				AND = {
					NOT = {
						has_variable = flavor_hab_1000_has_fired
					}
					current_year >= 1370
				}
			}
		}
	}

	ai_will_do = {
		value = 100
	}

	illustration_tags = {
		10 = happy
		10 = interior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }

		ruler ?= {
			save_scope_as = target_ruler
		}
	}

	option = {
		name = hab_privilegium_maius_decision.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = privilegium_maius_granted
			unlock_government_reform_effect = { type = austrian_archduchy_reform }
			add_reform = government_reform:austrian_archduchy_reform
			hidden_effect = {
				ruler ?= {
					reset_ruler_title = {}
				}
				remove_variable = flavor_hab_1000_has_fired
			}
			if = {
				limit = {
					NOT = {
						has_variable = privilegium_maius_accepted
					}
				}
				set_variable = privilegium_maius_accepted
				add_reform = government_reform:libertas_affectandi
				change_heir_selection = heir_selection:austrian_cognatic_primogeniture
			}
		}
	}
}
