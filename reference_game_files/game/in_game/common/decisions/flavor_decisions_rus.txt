				has_building = building_type:castle
			}
		}
	}

	ai_will_do = {
		value = 100
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		location:moscow = { save_scope_as = target_location }
		root = { save_scope_as = target_root_country }
	}

	option = {
		name = mos_build_kremlin_decision.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = {
				name = kremlin
				value = 1
			}
			capital = {
				construct_building = {
					building_type = building_type:kremlin
				}
			}
			capital.market = {
				add_temporary_demand = {
					type = demand:mos_kremlin_construction
					years = 10
				}
			}
			custom_tooltip = {
				text = mos_build_kremlin_decision.a.custom1
				every_neighbor_country = {
					if = {
						limit = {
							NOT = { is_subject_of = root }
						}
						add_opinion = {
							target = root
							modifier = threatened_by_military_expansion
						}
					}
					else = {
						add_opinion = {
							target = root
							modifier = safe_haven_during_raids
						}
					}
				}
			}
			custom_tooltip = {
				text = mos_build_kremlin_decision.a.custom2
			}
			custom_tooltip = mos_build_kremlin_decision.tt1
			custom_tooltip = mos_build_kremlin_decision.tt2
		}
	}

	option = {
		name = mos_build_kremlin_decision.b
		ai_chance = { value = 0 }
		effect = {
			custom_tooltip = {
				text = mos_build_kremlin_decision.b.custom1
				every_neighbor_country = {
					if = {
						limit = {
							NOT = { is_subject_of = root }
						}
						add_trust_equilibrium = {
							target = root
							modifier = trust_peaceful_coexistence
						}
					}
					else = {
						add_opinion = {
							target = root
							modifier = lack_of_protection
						}
					}
				}
			}
			custom_tooltip = {
				text = mos_build_kremlin_decision.b.custom2
			}
			change_societal_value = {
				type = belligerent_vs_conciliatory
				value = societal_value_move_to_right
			}
			add_prestige = prestige_mild_bonus
		}
	}

	option = {
		name = mos_build_kremlin_decision.c
		ai_chance = { value = 0 }
		effect = {
			set_variable = {
				name = kremlin
				value = 1
			}
			capital = {
				construct_building = {
					building_type = building_type:kremlin
				}
			}
			if = {
				limit = {
					any_neighbor_country = {
						is_subject_of = root
						any_owned_location = {
							has_building = building_type:castle
						}
					}
				}
				random_neighbor_country = {
					limit = {
						is_subject_of = root
						any_owned_location = {
							has_building = building_type:castle
						}
					}

						limit = {
							has_building = building_type:castle
						}
						destroy_building = "building(building_type:castle|owner)"
					}

						type = demand:mos_kremlin_construction
						years = 10
					}
				}
			}
			custom_tooltip = mos_build_kremlin_decision.tt3
