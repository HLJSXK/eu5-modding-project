hafsid_caliphate_decision = {
	decision_category = country_specific_decisions
	potential = {
		has_variable = hafsid_caliphate_not_respected
		religion = religion:sunni
	}
	allow = {
		societal_value:mysticism_vs_jurisprudence >= 25
		stability >= 50
		custom_tooltip = {
			text = tun_have_more_gp_score_than_us
			NOT = {
				any_known_country = {
					religion = religion:sunni
					great_power_score >= ROOT.great_power_score
				}
			}
		}
	}

	only_once = yes

	ai_will_do = {
		value = 100
	}

	option = {
		ai_chance = { value = 100 }
		name = flavor_tun.3.a
		effect = {
			remove_variable = hafsid_caliphate_not_respected
			show_as_tooltip = { hafsid_caliphate_effect = yes }
			hidden_effect = { 
				trigger_event_non_silently = flavor_tun.3
			}
		}
	}
}
