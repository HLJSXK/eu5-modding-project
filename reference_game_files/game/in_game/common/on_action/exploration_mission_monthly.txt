#root = country, scope:target = area  scope:actor = character
on_exploration_success = {
	on_actions = {
	}

	effect = {
		if = {
			limit = {
				has_dlc = "d009_across_the_pillars"
				has_global_variable = battle_for_the_strait_winner_country
				this = global_var:battle_for_the_strait_winner_country
				is_situation_active = situation:golden_century
			}
			gc_change_colonial_dominance = { value = gc_bump_exploration }
		}
		if = {
			limit = {
				scope:target.sub_continent = sub_continent:south_america
				has_variable = mal_founding_expedition_start
				#this area belongs to south america, and the country has the mal_founding_expedition_start variable
			}
			scope:actor = {
				save_scope_as = target_character
			}
			trigger_event_non_silently = flavor_mal.25
		}
		if = {
			limit = { has_variable = mission_new_maps_variable }
			remove_variable = mission_new_maps_variable
			trigger_event_non_silently = generic_colonial_exploration_events.2
		}
		if = {
			limit = {
				NOT = { has_variable = num_of_explorations }
			}
			set_variable = {
				name = num_of_explorations
				value = 1
			}
		}
		if = {
			limit = { 
				has_country_modifier = POR_into_the_beyond 
				OR = {
					current_age = age_1_traditions
					current_age = age_2_renaissance
					current_age = age_3_discovery
				}
			}
			random_list = {
				10 = { trigger_event_non_silently = flavor_por.68 }
				90 = { }
			}
		}	
		else = {
			change_variable = {
				name = num_of_explorations
				add = 1
			}
		}
	}

	random_events =	{
		10 = exploration_success.1

		chance_to_happen = 5
	}
}
