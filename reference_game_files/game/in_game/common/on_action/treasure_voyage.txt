
# Country Scope - nation harboring the treasure voyage
# scope:expedition_initiator - country - the nation from which the expedition originates
chinese_expedition_event_pulse = {
	trigger = {
		is_current_treasure_fleet_host = yes
	}
	first_valid_on_action = {
		# Fires treasure_fleet_events.50 when the expedition has valid enemies among the host's neighbors.
		# Otherwise it fires the normal chi_expedition_rotation_pulse
		chi_expedition_war_support_pulse
		chi_expedition_rotation_pulse
	}
}

chi_expedition_war_support_pulse = {
	trigger = {
		scope:expedition_initiator ?= {
			has_variable_list = potential_enemy_list
		}
		any_neighbor_country = {
			NOT = { has_variable = chinese_war_support_declined_variable }
			scope:expedition_initiator = {
				is_target_in_variable_list = {
					name = potential_enemy_list
					target = PREV
				}
			}
		}
	}
	events = {
		treasure_fleet_events.50
	}
}

	}
}
