ask_for_extra_levies = {
	type = owncountry

	ai_tick = daily
	ai_tick_frequency = 180
	automation_tick = never

	potential = {
		scope:actor = {
			any_estate = {
				estate_tax_base > 0.01
			}
		}
	}

	allow = {
	}

	cooldown = {
		type = ask_for_extra_levies_cooldown
		years = 5
	}

	select_trigger = {
		looking_for_a = estate
		source = actor
		target_flag = target
		name = "mission_select_estate"
		column = {
			data = name
		}
		enabled = {
			estate_tax_base > 0.01
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = {
					any_current_war = { is_a_defender = scope:actor }
				}
				add_estate_satisfaction = {
					type = scope:target.estate_type
					value = -0.125
				}
			}
			else = {
				add_estate_satisfaction = {
					type = scope:target.estate_type
					value = -0.25
				}
			}
			add_country_modifier = {
				modifier = estate_emergency_extra_levies
				months = 12
				mode = add_and_extend
			}
		}
	}

	ai_prerequisite = {
		at_war = yes
		any_estate = {
			satisfaction > 0.35
		}
	}

	ai_will_do = {
		if = {
			limit = {
				exists = scope:target
				scope:target = { satisfaction > 0.35 }
				scope:actor = {
					at_war = yes
					lowest_war_score < 0
				}
			}

			subtract = {
				desc = "War score pressure"
				value = scope:actor.lowest_war_score
			}

			if = {
				limit = { scope:actor = { max_manpower < 5 } }
				add = { desc = "Critically low manpower" value = 20 }
			}

			add = {
				desc = "Estate satisfaction buffer"
				value = scope:target.satisfaction
				multiply = 30
			}

			if = {
				limit = { scope:target = { satisfaction < 0.35 } }
				multiply = 0
			}
		}
	}
}


extraordinary_taxes = {
	type = owncountry

	ai_tick = daily
	ai_tick_frequency = 180
	automation_tick = never

	potential = {
		scope:actor = {
			any_estate = {
				estate_tax_base > 0.01
			}
		}
	}

	allow = {
	}

	cooldown = {
		type = extraordinary_taxes_cooldown
		years = 5
	}

	# Catholic clergy is tax-free by religion definition (christian.txt: clergy_estate_max_tax = -1.0).
	# Block the clergy target for Catholic actors so the +5% emergency modifier cannot bypass that rule.
	select_trigger = {
		looking_for_a = estate
		source = actor
		target_flag = target
		name = "mission_select_estate"
		column = {
			data = name
		}
		enabled = {
			estate_tax_base > 0.01
			NOR = {
				AND = {
					estate_type = estate_type:clergy_estate
					scope:actor = { religion = religion:catholic }
				}
			}
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = {
					any_current_war = { is_a_defender = scope:actor }
				}
				add_estate_satisfaction = {
					type = scope:target.estate_type
					value = -0.125
				}
			}
			else = {
				add_estate_satisfaction = {
					type = scope:target.estate_type
					value = -0.25
				}
			}

			if = {
				limit = { scope:target = { estate_type = estate_type:nobles_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_nobles
					months = 12
					mode = add_and_extend
				}
			}
			else_if = {
				limit = { scope:target = { estate_type = estate_type:burghers_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_burghers
					months = 12
					mode = add_and_extend
				}
			}
			else_if = {
				limit = { scope:target = { estate_type = estate_type:clergy_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_clergy
					months = 12
					mode = add_and_extend
				}
			}
			else_if = {
				limit = { scope:target = { estate_type = estate_type:peasants_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_peasants
					months = 12
					mode = add_and_extend
				}
			}
			else_if = {
				limit = { scope:target = { estate_type = estate_type:tribes_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_tribes
					months = 12
					mode = add_and_extend
				}
			}
			else_if = {
				limit = { scope:target = { estate_type = estate_type:dhimmi_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_dhimmi
					months = 12
					mode = add_and_extend
				}
			}
			else_if = {
				limit = { scope:target = { estate_type = estate_type:cossacks_estate } }
				add_country_modifier = {
					modifier = estate_emergency_taxes_cossacks
					months = 12
					mode = add_and_extend
				}
			}
		}
	}

	ai_prerequisite = {
		OR = {
			monthly_balance < 0
			AND = {
				at_war = yes
				gold < 10
			}
		}
		any_estate = { satisfaction > 0.35 }
	}

	ai_will_do = {
		if = {
			limit = {
				exists = scope:target
				scope:target = { satisfaction > 0.35 }
				scope:actor = {
					OR = {
						monthly_balance < 0
						AND = { at_war = yes gold < 10 }
					}
				}
			}

			if = {
				limit = { scope:actor = { monthly_balance < 0 } }
				subtract = {
					desc = "Monthly deficit"
					value = scope:actor.monthly_balance
				}
			}

			if = {
				limit = { scope:actor = { at_war = yes lowest_war_score < 0 } }
				add = { desc = "At war and losing" value = 15 }
			}

			add = {
				desc = "Estate satisfaction buffer"
				value = scope:target.satisfaction
				multiply = 20
			}

			if = {
				limit = { scope:target = { satisfaction < 0.35 } }
				multiply = 0
			}
		}
	}
}

ask_burghers_for_loan = {
	type = owncountry

	# ai_tick = never — AI bailout still goes through take_estate_loan (real loans against
	# RemainingLoanCapacity), because the AI needs more headroom than the burgher pool
	# can typically spare. The grant is a player-only convenience.
	ai_tick = never
	automation_tick = never

	potential = {
		scope:actor = {
			country_has_estate = estate_type:burghers_estate
		}
		scope:target ?= {
			estate_type = estate_type:burghers_estate
		}
	}

	# Burghers must have enough gold to make the grant worthwhile. BURGHER_GRANT_MIN_TRANSFER
	# is also the C++ floor — the effect refuses to pay out anything smaller, so the gate keeps
	# the action from appearing actionable when it would no-op in execution.
	allow = {
		scope:actor = {
			any_estate = {
				estate_type = estate_type:burghers_estate
				estate_gold >= "define:NEconomy|BURGHER_GRANT_MIN_TRANSFER"
			}
		}
	}

	cooldown = {
		type = ask_burghers_for_loan_cooldown
		years = 5
	}

	select_trigger = {
		looking_for_a = estate
		source = actor
		target_flag = target
		name = "mission_select_estate"
		column = {
			data = name
		}
		enabled = {
			estate_type = estate_type:burghers_estate
		}
	}

	# Burghers pay from their own treasury (capped at BURGHER_GRANT_MAX_FRACTION of their pool in C++).
	# No loan, no interest, no impact on country loan capacity — the bankruptcy spiral the legacy
	# create_estate_loan caused is gone with it. Requested amount is the legacy 12x trickle for parity
	# with the old payout ceiling; the C++ cap kicks in when the burghers cannot afford that much.
	effect = {
		scope:actor = {
			transfer_gold_from_estate = {
				estate_type = estate_type:burghers_estate
				value = {
					value = country_estate_loan_size
					multiply = 12
				}
			}
			add_estate_satisfaction = {
				type = scope:target.estate_type
				value = -0.1
			}
		}
	}
}

call_emergency_parliament = {
	type = owncountry

	ai_tick = never
	automation_tick = never

	potential = {
		scope:actor = {
			has_parliament = yes
		}
	}

	allow = {
		scope:actor = {
			is_active_parliament = no
			months_since_last_parliament_called <= "define:NCountry|PARLIAMENT_COOLDOWN_MONTHS"
			at_war = yes
			NOT = {
				modifier:parliament_abolished = yes
				modifier:has_a_parliamentary_system = no
			}
		}
	}

	cooldown = {
		type = call_emergency_parliament_cooldown
		years = 5
	}

	effect = {
		scope:actor = {
			set_parliament_location = scope:actor.capital
			set_parliament_active = yes
			add_legitimacy = legitimacy_mild_penalty
		}
	}
}
