grand_tour_expedition = {

	unique = no
	dynamic_first_waypoint = yes
	should_end_at_last_waypoint = no
	travel_speed = 0.75
	fail_if_no_leader = yes
	show_start_message = no
	stalled_status_text = grand_tour_expedition_stalled_status

		gt_diligence = {
			icon = "gfx/interface/icons/expedition_types/gt_diligence.dds"
			min = 0
			max = 100

		}
		knowledge = {
			icon = "gfx/interface/icons/expedition_types/knowledge.dds"
			start = 0
			min = 0
			max = 10
		}
		connections = {
			icon = "gfx/interface/icons/expedition_types/connections.dds"
			start = 0
			min = 0
			max = 10
		}
		experience = {
			icon = "gfx/interface/icons/expedition_types/experience.dds"
			start = 0
			min = 0
			max = 10
		}
	}

	potential = {
		has_heir = yes
		capital = {
			continent = continent:europe
			NOT = { region = region:italy_region }
		}
	}

	can_start = {
		gold >= 250
		has_heir = yes
		heir = {
			custom_tooltip = {
				text = grand_tour_expedition_heir_completed_tt
				NOT = { has_variable = grand_tour_complete }
			}
			is_adult = yes
			is_female = no
			character_age < 35
			NOT = { in_cabinet = yes }
		}
		NOT = { has_cooldown = grand_tour_cooldown }
		OR = {
			has_embraced_institution = institution:scientific_revolution
			has_embraced_institution = institution:military_revolution
		}
	}

	leader = {
		is_heir_of_court_country = yes
		is_ruler = no
		is_adult = yes
		is_female = no
		is_expedition_leader = no
		has_exploration = no
		character_age < 35
	}

	ai_leader_source_list = {
		heir = {
			add_to_list = source
		}
	}

	on_start = {
		add_cooldown = {
			type = grand_tour_cooldown
			years = 5
		}
		scope:expedition = {
			if = {
				limit = { exists = expedition_leader }
				remove_commander = expedition_leader
			}
		}
		scope:expedition = {
			set_variable = {
				name = gt_diligence
				value = {
					add = root.heir.adm
					add = root.heir.dip
					add = root.heir.mil
					divide = 3
				}
			}
			add_new_waypoint = location:venice
		}

		trigger_event_non_silently = { id = grand_tour_events.1 }
	}

			limit = {
				has_heir = yes
				heir = { NOT = { has_variable = grand_tour_complete } }
			}
			if = {
				limit = {

						var:knowledge >= var:connections
						var:knowledge >= var:experience
						var:knowledge > 5
					}
				}
				if = {
					limit = { heir = { num_of_traits < 3 } }
					heir = { add_trait = trait:free_thinker }
				}
			}
			else_if = {
				limit = {
					scope:expedition = {
						var:connections >= var:knowledge
						var:connections >= var:experience
						var:connections > 5
					}
				}
				if = {
					limit = { heir = { num_of_traits < 3 } }
					heir = { add_trait = trait:well_connected }
				}
			}
			else_if = {
				limit = {
					scope:expedition = {
						var:experience > 5
					}
				}
				if = {
					limit = { heir = { num_of_traits < 3 } }
					heir = { add_trait = trait:worldly }
				}
			}
			heir = {
				set_variable = { name = grand_tour_complete value = 1 }
				add_character_modifier = {
					modifier = returned_from_grand_tour
				}

					value = scope:expedition.var:connections
					add = scope:expedition.var:experience
				}
				add_adm = {
					value = scope:expedition.var:experience
					add = scope:expedition.var:knowledge
				}

					value = scope:expedition.var:connections
