# -------------------------------------------------------------------------
# .20 — Our Circle Has Been Bought
# Fires to every non-leader member of a circle whose leader has just been
# bribed by the Emperor via bribe_imperial_circle. Triggered from the
# generic action's every_circle_member loop so each affected country gets
# a visible log entry (popup for human players, silent history line for AI)
# explaining who paid us, how much, and that our vote is now locked.
# Informational only — single OK option, no decision branch (the rewrite
# keeps the bribe deterministic).
# scope:actor    = the bribing Emperor
# scope:target   = our circle leader (the country named by the action)
# scope:received_bribe = the gold we received as our share of the bribe
# -------------------------------------------------------------------------
imperial_circles.20 = {
	type = country_event
	title = imperial_circles.20.title
	desc = imperial_circles.20.desc
	outcome = neutral
	category = international_organization_event
	image = "gfx/interface/illustrations/international_organization_types/hre.dds"

	immediate = {
		save_scope_value_as = {
			name = received_bribe
			value = {
				value = bribe_circle_leader_cost
				multiply = 0.25
			}
		}
	}

	option = {
		name = imperial_circles.20.a
	}
}

# -------------------------------------------------------------------------
# .50 — Challenge for Circle Leadership
# Fires to the challenger (scope:actor of challenge_circle_leadership).
# scope:recipient = incumbent circle leader
# Option A: press the challenge (outcome determined by prestige comparison)
# Option B: withdraw
# -------------------------------------------------------------------------
imperial_circles.50 = {
	type = country_event
	title = imperial_circles.50.title
	desc = imperial_circles.50.desc
	image = "gfx/interface/illustrations/international_organization_types/hre.dds"
	# TODO: stub to fix error log, set proper value later
	outcome = neutral

	immediate = {
		save_scope_as = challenger
		scope:recipient = { save_scope_as = incumbent }
	}

	option = {
		name = imperial_circles.50.a		# Press the challenge
		# Win path (tiered):
		#   Crisis  — any circle has satisfaction < 15: challenger wins regardless of prestige
		#   Discontent — any circle has satisfaction < 30 and challenger prestige >= incumbent
		#   Standard — challenger prestige >= incumbent + 25 (margin damps oscillation)
		if = {
			limit = {
				OR = {
					any_imperial_circle_country_is_member_of = {
						imperial_circle_satisfaction < 15
					}
					AND = {
						any_imperial_circle_country_is_member_of = {
							imperial_circle_satisfaction < 30
						}
						prestige >= scope:incumbent.prestige
					}
					prestige >= {
						value = scope:incumbent.prestige
						add = 25
					}
				}
			}
			every_imperial_circle_country_is_member_of = {
				set_circle_leader = scope:challenger
			}
			add_country_modifier = {
				modifier = consolidating_circle_leadership
				years = 2
			}
			scope:incumbent = {
				add_prestige = prestige_mild_penalty
				trigger_event_silently = { id = imperial_circles.51 }
			}
		}
		else = {
			add_prestige = prestige_mild_penalty
			scope:incumbent = {
				trigger_event_silently = { id = imperial_circles.52 }
			}
			trigger_event_silently = { id = imperial_circles.53 }
		}
	}

	option = {
		name = imperial_circles.50.b		# Withdraw the challenge
		add_prestige = prestige_weak_penalty
	}
}
