flavor_gra.1 = {
	type = country_event
	title = flavor_gra.1.title
	desc = flavor_gra.1.desc
	outcome = neutral
	trigger = {
		religion.group = religion_group:muslim
		NOT = {
			has_variable = received_granada_refugees
			country_exists = c:GRA
		}
		num_of_ports > 5
		country_exists = c:CAS
		location:granada = {
			owner = c:CAS
		}
		current_year < 1550
		
	
		capital = {
			OR = {
				area = area:levant_area
				region = region:egypt_region
				region = region:maghreb_region
				region = region:anatolia_region
				region = region:balkan_region
			}
		}
	}

	illustration_tags = {
        10 = exterior
    }

	immediate = {
		event_illustration_estate_background_effect = { background = estate_type:peasants_estate }
		set_variable = { name = received_granada_refugees value = 1 }
		
		region:iberia_region = {
			ordered_location_in_region = {
				limit = {
					exists = dominant_religion
					dominant_religion ?= religion:sunni
				}
				order_by = population
				
				max = 1
				province_definition = {
					save_scope_as = origin_province
				}
				
			}
		}
	}

	option = {
		name = flavor_gra.1.a
		add_migration = {
			owner = root
			from = scope:origin_province
			to = capital.province_definition
			religion = religion:sunni
			amount = 0.150
			months = 240
		}
		
		add_prestige = prestige_weak_bonus
		c:CAS = { add_opinion = { target = root  modifier = opinion_mildly_displeased } }
	}
	option = {
		name = flavor_gra.1.b
		add_prestige = prestige_severe_penalty
		c:CAS = { add_opinion = { target = root  modifier = opinion_pleased } }
	}

					limit = {

	title = flavor_gra.1300.title
	desc = flavor_gra.1300.desc
	outcome = neutral

	dynamic_historical_event = {
		tag = GRA
		from = 1337.1.1
		to = 1821.1.1
		monthly_chance = 25
	}

	fire_only_once = yes

	trigger = {
		is_situation_active = situation:rise_of_the_ottomans

		religion.group = religion_group:muslim

		situation:rise_of_the_ottomans.var:strongest_beylik_variable ?= {
			religion.group = religion_group:muslim
			owns = location:constantinople
		}
	}

	immediate = {
		ruler_or_regent ?= {
			save_scope_as = target_character
		}

		save_scope_as = from_country

		situation:rise_of_the_ottomans.var:strongest_beylik_variable = {
			save_scope_as = target_country
			ruler_or_regent ?= {
				save_scope_as = target_character2
			}
		}
	}

	option = {
		name = flavor_gra.1300.a
		
		scope:target_country = {
			trigger_event_non_silently = {
				id = flavor_gra.1301
			}
		}

		custom_tooltip = gra_demand_a_mercenary_company
	}

	option = {
		name = flavor_gra.1300.b
		
		add_prestige = prestige_mild_bonus
	}
}

#Ottomans Receive Delegate from Granada
flavor_gra.1301 = {
	type = country_event
	title = flavor_gra.1301.title
	desc = flavor_gra.1301.desc
	outcome = neutral

	major = yes
	major_trigger = {
		this = scope:from_country	
	}

	immediate = {
		if = {
			limit = {
				any_character = {
					NOT = { has_estate = estate_type:crown_estate }
					is_artist = no
					is_general = no
					is_admiral = no
					in_cabinet = no
					modifier:blocked_from_being_leader = no
					modifier:character_on_task_modifier = no
				}
			}
			ordered_character = {
				order_by = total_abilities
				limit = {
					NOT = { has_estate = estate_type:crown_estate }
					is_artist = no
					is_general = no
					is_admiral = no
					in_cabinet = no
					modifier:blocked_from_being_leader = no
					modifier:character_on_task_modifier = no
				}
				max = 1
				check_range_bounds = no
				save_scope_as = mercenary_general
			}
		}
		else = {
			create_character = {
				estate = estate_type:peasants_estate
				age = { 20 40 }
				adm = { 20 70 }
				dip = { 20 70 }
				mil = { 20 70 }
				save_scope_as = mercenary_general
			}
		}
	}

	option = {
		name = flavor_gra.1301.a
		
		scope:mercenary_general = {
			move_country = scope:from_country
		}

		scope:from_country = {
			create_mercenary = {
				name = gra_turkish_volunteers
				home = scope:from_country.capital
				categories = {
					army_heavy_infantry = 10
					army_heavy_cavalry = 2
				}
				cost_multiplier = 0.5
				leader = scope:mercenary_general
			}
		}

		add_opinion_mutual_effect = {
			modifier = gra_received_military_help
			target = scope:from_country
		}

		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_right
		}
	}

	option = {
		name = flavor_gra.1301.b
		
		transfer_yearly_gold = {
			target = scope:from_country
			value = 0.75
		}

		add_opinion_mutual_effect = {
			modifier = gra_received_gold
			target = scope:from_country
		}

		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_right
		}
	}

	option = {
		name = flavor_gra.1301.c
		
		add_opinion_mutual_effect = {
			modifier = gra_rejected_helping_us
			target = scope:from_country
		}

		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_left
		}
	}
}
