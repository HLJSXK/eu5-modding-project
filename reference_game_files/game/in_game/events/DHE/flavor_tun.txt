	save_scope_as = appointed_heir
	ROOT = {
		set_variable = {
			name = tunisian_successor
			value = PREV
		}
		hidden_effect = {
			set_as_designated_heir_effect = {
				target = PREV
			}
		}
	}
	custom_tooltip = TUN_THIS_will_be_appointed_heir_tt
	if = {
		limit = {
			is_ruler = yes
			owner ?= {
				is_hafsid_governor_trigger = yes
			}
		}
		custom_tooltip = TUN_unite_two_realms_tt
	}
	if = {
		limit = {
			this != scope:preferred_heir
		}
		custom_tooltip = TUN_civil_war_from_preferred_heir_tt
	}
	if = {
		limit = {
			this != scope:future_usurper
		}
		custom_tooltip = TUN_abu_hafs_will_try_to_start_a_civil_war_tt
	}
}
