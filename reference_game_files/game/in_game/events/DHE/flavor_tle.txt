namespace = flavor_tle

# War with Morocco: Warning about Fall of Tlemcen
flavor_tle.1 = {
	type = country_event
	title = flavor_tle.1.title
	desc = flavor_tle.1.desc
	outcome = neutral

	immediate = {
		event_illustration_estate_effect = {
			background = estate_type:burghers_estate
			foreground = estate_type:nobles_estate
		}
		c:MOR = {
			"war_with_country(c:TLE)" = {
				set_variable = is_starting_morocco_war
				save_scope_as = target_war
			}
			save_scope_as = mor_scope
			ruler_or_regent = {
				save_scope_as = mor_ruler
			}
		}
		c:TLE = {
			save_scope_as = tle_scope
			ruler_or_regent = {
				save_scope_as = tle_ruler
			}
		}
		location:tlemcen = {
			siege = {
				save_scope_as = target_siege
			}
			save_scope_as = target_location
		}
	}

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	option = {
		name = flavor_tle.1.a
		
		custom_tooltip = tle_morrocan_tlemcen_war

		if = {
			limit = {
				this = scope:tle_scope
			}
			custom_tooltip = tle_additional_warning
		}
	}
}

scripted_effect warn_and_setup_info_about_zayyanid_pretenders = {
	custom_tooltip = TLE_once_truce_ends_zayyanid_pretenders_tt
	scope:target_country = {
		set_variable = {
			name = zayyanid_pretenders
			value = root
		}
		trigger_event_non_silently = flavor_tle.3 
	}
}

# Fall of Tlemcen - MOR
flavor_tle.2 = {
	type = country_event
	title = flavor_tle.2.title
	desc = flavor_tle.2.desc
	outcome = positive

	immediate = {
		event_illustration_estate_effect = {
			background = estate_type:burghers_estate
			foreground = estate_type:nobles_estate
		}
		c:TLE = {
			save_scope_as = target_country
			ruler_or_regent ?= {
				save_scope_as = tle_ruler
			}
		}
		save_scope_as = from_country
		ruler_or_regent ?= {
			save_scope_as = mor_ruler
		}
		"war_with_country(c:TLE)" = { 
			save_scope_as = target_war
		}
		dynasty:zayyanid_dynasty = {
			save_scope_as = target_dynasty
		}
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	major = yes
	major_trigger = {
		has_presence_in = region:maghreb_region	
		NAND = { # special event for this scenario to make sure Tlemcen knows
			this = c:TLE
			has_local_variable = vassalized_tlemcen 
		}
	}

	option = {
		name = flavor_tle.2.a

		historical_option = yes

		ai_chance = { factor = 50 }
		
		if = {
			limit = {
				scope:target_war = {
					is_in_war = root
					is_in_war = scope:target_country
				}
			}
			set_local_variable = vassalized_tlemcen
			white_peace = scope:target_war
			scope:target_country = {
				make_subject_of = {
					target = ROOT
					type = subject_type:fiefdom
				}
			}

			drop_antagonism_bomb = {
				modifier = antagonism_peace_treaty_demands
				target = scope:target_country.capital
				value = 50
			}
			
			change_societal_value = {
				type = belligerent_vs_conciliatory
				value = societal_value_move_to_left
			}
			warn_and_setup_info_about_zayyanid_pretenders = yes
		}
	}

	option = {
		name = flavor_tle.2.b

		ai_chance = { factor = 50 }
		
		add_army_tradition = army_tradition_severe_bonus
	}
}

		capital.province = {
			add_province_modifier = {
				modifier = tle_zayyanid_tribe_support
				years = 5
			}
		}
		add_country_modifier = {
			modifier = tle_support_of_the_tribes
			years = 5
		}
		add_country_modifier = { modifier = parliament_approved_extra_levies years = 3 mode = add_and_extend }  
	}
}

		event_illustration_estate_effect = {

		}
		
		c:MOR = { 
			save_scope_as = target_country
		}
		save_scope_as = from_country
		dynasty:zayyanid_dynasty = {
			save_scope_as = target_dynasty
		}
		capital = {
			save_scope_as = target_location
		}
	}

	option = {
		name = flavor_tle.7.a
		
		scope:target_country = {
			trigger_event_non_silently = flavor_tle.4
		}
	}

	after = {
		trigger_event_silently = {
			years = 20
			id = flavor_tle.11
		}
	}

# Growth of Oran
flavor_tle.13 = {
	type = country_event
	title = flavor_tle.13.title
	desc = flavor_tle.13.desc
	outcome = positive

	fire_only_once = yes
	dynamic_historical_event = {
		tag = TLE
		from = 1337.1.1
		to = 1821.1.1
		monthly_chance = 5
	}

	trigger = {
		owns = location:oran
		owns = location:tlemcen
		location:oran = {
			any_buildings_in_location = {
				building_category = naval_category
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = {
			background = estate_type:burghers_estate
			foreground = estate_type:burghers_estate
		}

		location:oran = {
			save_scope_as = target_location
		}

		location:tlemcen = {
			save_scope_as = target_location2
		}

		save_scope_as = target_country

		"estate(estate_type:burghers_estate)" = {
			save_scope_as = target_estate
		}
	}

	option = {
		name = flavor_tle.13.a
		
		ai_chance = {
			factor = 75
		}

		change_societal_value = {
			type = land_vs_naval
			value = societal_value_move_to_right
		}

		scope:target_location = {
			add_location_modifier = {
				mode = replace
				years = -1
				modifier = tle_wehran_privileges
			}
		}
	}

	option = {
		name = flavor_tle.13.b

		ai_chance = {
			factor = 25
		}
		
		location:tlemcen = {
			change_development = development_severe_bonus
		}

		add_estate_satisfaction = {
			type = scope:target_estate.estate_type
			value = estate_satisfaction_severe_penalty
		}
	}
}
