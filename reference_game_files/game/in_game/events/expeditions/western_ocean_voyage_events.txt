namespace = western_ocean_voyage_events

# western_ocean_voyage_events.1 — Departure (on_start)
western_ocean_voyage_events.1 = {
	type = country_event
	title = western_ocean_voyage_events.1.title
	desc = western_ocean_voyage_events.1.desc
	historical_info = western_ocean_voyage_events.1.historical_info
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
		}
		capital ?= { save_scope_as = target_location }
	}

	# Columbus-specific departure
	option = {
		name = western_ocean_voyage_events.1.columbus
		trigger = {
			exists = character:christopher_columbus
			scope:target_leader = character:christopher_columbus
		}
		add_prestige = prestige_weak_bonus
	}

	# Generic departure
	option = {
		name = western_ocean_voyage_events.1.a
		trigger = {
			OR = {
				NOT = { exists = character:christopher_columbus }
				NOT = { scope:target_leader = character:christopher_columbus }
			}
		}
		add_prestige = prestige_weak_bonus
	}
}

# western_ocean_voyage_events.4 — Land! (on_arrive_to_waypoint, Caribbean landfall)
western_ocean_voyage_events.4 = {
	type = country_event
	title = western_ocean_voyage_events.4.title
	desc = western_ocean_voyage_events.4.desc
	historical_info = western_ocean_voyage_events.4.historical_info
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_wilderness_landing }

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		set_variable = { name = landfall_set value = 2 }
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
		}
		stall_expedition = scope:expedition
		scope:location = {
			save_scope_as = target_location
			random_neighbor_location = {
				limit = { is_land = yes }
				save_scope_as = adjacent_land_loc
			}
		}
	}

	# Columbus plants the flag — we have found the Indies
	option = {
		name = western_ocean_voyage_events.4.columbus
		trigger = {
			exists = character:christopher_columbus
			scope:target_leader = character:christopher_columbus
		}
		add_prestige = prestige_mild_bonus
		if = {
			limit = { exists = scope:adjacent_land_loc }
			discover_area = scope:adjacent_land_loc.area
		}
		trigger_event_non_silently = western_ocean_voyage_events.6
	}

	# Generic — claim the land for the crown
	option = {
		name = western_ocean_voyage_events.4.a
		trigger = {
			OR = {
				NOT = { exists = character:christopher_columbus }
				NOT = { scope:target_leader = character:christopher_columbus }
			}
		}
		add_prestige = prestige_mild_bonus
		if = {
			limit = { exists = scope:adjacent_land_loc }
			discover_area = scope:adjacent_land_loc.area
		}
		trigger_event_non_silently = western_ocean_voyage_events.6
	}

	# Something is different — not the Indies
	option = {
		name = western_ocean_voyage_events.4.b
		add_prestige = prestige_mild_bonus
		add_legitimacy = legitimacy_mild_bonus
		if = {
			limit = { exists = scope:adjacent_land_loc }
			discover_area = scope:adjacent_land_loc.area
		}
		trigger_event_non_silently = western_ocean_voyage_events.6
	}

	after = {
		resume_expedition = scope:expedition
	}
}
